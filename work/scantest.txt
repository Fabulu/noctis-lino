	( emit SCAN line )

( SC byte - update [SCfnv] with the ASCII byte in C. )

"SC byte"
	D = [SCfnv]; D # C; D '* 16777619; [SCfnv] = D;
	end;

( SC dec - feed the decimal ASCII of A (signed) into the fnv stream. )

"SC dec"
	? A >= 0 -> SC dec pos;
	A +-; C = 45; => SC byte;
    "SC dec pos"
	? A = 0 -> SC dec zero;
	[SCix] = 0;
	[SCtmp] = A;
    "SC dec loop"
	A = [SCtmp]; A '/ 10; ? A = 0 -> SC dec emit;
	B = A; B '* 10; B +-;
	C = [SCtmp]; C + B;	( C = value - (value/10)*10 = digit )
	E = scbuf; E + [SCix]; [E] = C;
	[SCix]+;
	A = [SCtmp]; A '/ 10; [SCtmp] = A;
	-> SC dec loop;
    "SC dec zero"
	E = scbuf; E + 0; [E] = 0;
	[SCix] = 0;
	-> SC dec feed;
    "SC dec emit"
	( store the most significant digit too )
	E = scbuf; E + [SCix]; C = [SCtmp]; [E] = C;
    "SC dec feed"
	( emit scbuf[SCix] down to [0] )
	E = scbuf; E + [SCix]; C = [E]; C + 48; => SC byte;
	? [SCix] = 0 -> SC dec done;
	A = [SCix]; A - 1; [SCix] = A;
	-> SC dec feed;
    "SC dec done"
	end;
