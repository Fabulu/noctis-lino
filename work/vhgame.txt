"VHG UTC timestamp"
	( NIV+'s checkpoint epoch: UTC seconds since 1984-01-01, retaining its
	  every-four-years leap rule.  This signed value remains valid through
	  the date range supported by the port. )
	[Timer Command] = READ UTC TIME; isocall;
	=> VHG fpu clean; => SU fp init;
	A = [Year]; A - 1984; [VHGutcdays] = A; A '* 365; [VHGutcsecs] = A;
	A = [VHGutcdays]; A '/ 4; A + [VHGutcsecs]; [VHGutcdays] = A;
	[VHGutcmonthbase] = 0; A = [Month]; ? A <= 1 -> VHG UTC month ready;
	[VHGutcmonthbase] = 31; ? A = 2 -> VHG UTC month ready;
	[VHGutcmonthbase] = 59; ? A = 3 -> VHG UTC month ready;
	[VHGutcmonthbase] = 90; ? A = 4 -> VHG UTC month ready;
	[VHGutcmonthbase] = 120; ? A = 5 -> VHG UTC month ready;
	[VHGutcmonthbase] = 151; ? A = 6 -> VHG UTC month ready;
	[VHGutcmonthbase] = 181; ? A = 7 -> VHG UTC month ready;
	[VHGutcmonthbase] = 212; ? A = 8 -> VHG UTC month ready;
	[VHGutcmonthbase] = 243; ? A = 9 -> VHG UTC month ready;
	[VHGutcmonthbase] = 273; ? A = 10 -> VHG UTC month ready;
	[VHGutcmonthbase] = 304; ? A = 11 -> VHG UTC month ready;
	[VHGutcmonthbase] = 334;
    "VHG UTC month ready"
	A = [VHGutcdays]; A + [VHGutcmonthbase]; [VHGutcdays] = A;
	A = [Month]; ? A <= 2 -> VHG UTC leap ready;
	A = [Year]; A '% 4; ? A != 0 -> VHG UTC leap ready; [VHGutcdays]+;
    "VHG UTC leap ready"
	A = [Day]; A - 1; A + [VHGutcdays]; A '* 86400; [VHGutcsecs] = A;
	A = [Hour]; A '* 3600; A + [VHGutcsecs]; [VHGutcsecs] = A;
	A = [Minute]; A '* 60; A + [VHGutcsecs]; A + [Second]; [VHGutcsecs] = A;
	end;

"VHG offline restore"
	=> VHG UTC timestamp;
	A = [VHGutcsecs]; A - [VHGsavedsecs]; ? A <= 0 -> VHG offline done;
	[VHGelapsed] = A;

	( A pending compressed rescue advances at the fixed 18.206-Hz gameplay
	  rate.  Ten or more closed seconds complete the fly-by and preserve the
	  source minimum of three delivered units. )
	A = [VHGrescueactive]; ? A = 0 -> VHG offline collector;
	A = [VHGelapsed]; ? A >= 10 -> VHG offline rescue finish;
	A '* 18; A + [VHGrescuetick]; ? A >= 180 -> VHG offline rescue finish;
	[VHGrescuetick] = A; ? A < 45 -> VHG offline collector;
	A = [MgCharge]; ? A >= 3 -> VHG offline collector; [MgCharge] = 3;
	-> VHG offline collector;
    "VHG offline rescue finish"
	[VHGrescueactive] = 0; [VHGrescuetick] = 0;
	A = [MgCharge]; ? A >= 3 -> VHG offline collector; [MgCharge] = 3;

    "VHG offline collector"
	A = [VHGcollector]; ? A = 0 -> VHG offline done;
	A = [VHGelapsed]; A '/ 30; ? A <= 0 -> VHG offline done;
	[VHGofflinegain] = A; A + [MgCharge]; ? A <= 120 -> VHG offline charge ready; A = 120;
    "VHG offline charge ready"
	[MgCharge] = A; ? A >= 120 -> VHG offline collector full;
	A = [VHGutcsecs]; => BrtlSrand; A = 5000; => BrtlRandom; A + 15000; [MgPwr] = A;
	-> VHG offline done;
    "VHG offline collector full"
	[MgPwr] = 20000;
    "VHG offline done"
	end;

"VHG restore window"
	( Version-7 checkpoints retain the player's last valid iGUI dimensions.
	  Resize through iGUI so every chrome/work-area bound is updated together. )
	A = [VHSVok]; ? A = 0 -> VHG restore window done;
	A = [Display Width]; ? A != [VHSVwindoww] -> VHG restore window apply;
	A = [Display Height]; ? A = [VHSVwindowh] -> VHG restore window done;
    "VHG restore window apply"
	[New Display Width] = [VHSVwindoww]; [New Display Height] = [VHSVwindowh];
	=> Resize Display;
    "VHG restore window done"
	end;
