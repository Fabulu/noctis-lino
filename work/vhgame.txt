"VHG UTC timestamp"
	( NIV+'s checkpoint epoch: UTC seconds since 1984-01-01, retaining its
	  every-four-years leap rule.  This signed value remains valid through
	  the date range supported by the port. )
	[Timer Command] = READ UTC TIME; isocall;
	=> VHG fpu clean; => SU fp init;
	A = [Year]; A - 1984; [VHGutcdays] = A; A '* 365; [VHGutcsecs] = A;
	A = [VHGutcdays]; A '/ 4; A + [VHGutcsecs]; [VHGutcdays] = A;
	[VHGutcmonthbase] = 0; A = [Month]; ? A <= 1 -> VHG UTC month ready;
	[VHGutcmonthbase] = 31; ? A = 2 -> VHG UTC month ready;
	[VHGutcmonthbase] = 59; ? A = 3 -> VHG UTC month ready;
	[VHGutcmonthbase] = 90; ? A = 4 -> VHG UTC month ready;
	[VHGutcmonthbase] = 120; ? A = 5 -> VHG UTC month ready;
	[VHGutcmonthbase] = 151; ? A = 6 -> VHG UTC month ready;
	[VHGutcmonthbase] = 181; ? A = 7 -> VHG UTC month ready;
	[VHGutcmonthbase] = 212; ? A = 8 -> VHG UTC month ready;
	[VHGutcmonthbase] = 243; ? A = 9 -> VHG UTC month ready;

	=> VHG UTC timestamp;

	[VHGelapsed] = A;

	  the distance-dependent delay simply counts down; more than twenty seconds
	  after arrival ends the event and supplies the source's offline four units. )
	A = [VHGrescueactive]; ? A = 0 -> VHG offline collector;

"VHG timing step"

	( The exact Noctis skip-to-grid timer remains authoritative in original
	  mode.  The 60-Hz presenter needs a different overrun rule: wait when
	  early, but rebase immediately when late.  TK step would skip to the
	  *next* future point after iGUI's retrace reached this one, producing
	  the observed 30-Hz double wait. )
	A = [VHGfast]; ? A != 0 -> VHG timing fast;
	=> TK step;
	-> VHG timing done;
    "VHG timing fast"
	A = [TKcpms]; A '* 17; [TKbase] = A;
	A = [TKcpms]; A '* 198864; [TKsubper] = A;
	=> TK advance;
	[Timer Command] = READ COUNTS; isocall; [TKnow] = [Counts]; => TK is late;
	A = [TKlate]; ? A = 0 -> VHG timing fast wait;
	( Overrun: make now the new origin; do not discard another interval. )
	[TKdeadline] = [TKnow]; [TKacc] = 0; -> VHG timing done;
    "VHG timing fast wait"
	=> TK wait;
    "VHG timing done"
	end;
