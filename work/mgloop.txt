      ( *** mgloop - the flight dynamics: vimana + approach integrators,
	                lithium consumes, and the landing roundtrip ***

	A LIBRARY.  Needs fpabi/fpctl/fpx87/fpconv (linked before it) and
	mgmain (the driver, for the corpus reader and dump).  nsident is
	not required here.

	These are the per-frame numerical kernels of the 22-phase do-while
	(NOCTIS.CPP:2268-4485).  Everything display- or mouse-bound is out
	of scope for headless grading and is documented in mg_grade.py.

	dzat / ap_target / dxx / l_dsd / coefficients are all DOUBLE; pwr
	is a 16-bit signed int (chop-then-wrap on every store); charge is
	a signed byte.  The low-pass filter
	    cur += (req - cur) * rt
	is the vimana/approach drive (NOCTIS.CPP:3217-3363).  The landing
	roundtrip is the DELIBERATE double->float->double loss
	(NOCTIS-1.CPP:3411-3413) reproduced via F32Narrow.

	Double constants arrive as their binary64 bit halves because
	L.in.oleum has no decimal float literal: the values were emitted by
	mg_spec's dhalves() and are not hand-transcribed.                     )

      ( MG i16 C - wrap C (a chopped long) into the signed 16-bit range,
	matching the DOS `int pwr` store. )
"MG i16 C"
	A = C; A & 0FFFFh; C = A;
	A = C; ? A '< 8000h -> MG i16 done;
	A = C; A - 65536; C = A;
    "MG i16 done"
	end;

      ( MG widen f32 FA from C - C holds a binary32 bit pattern; FA <- the
	value widened to double.  fld dword / fstp qword. )
"MG widen f32 FA from C"
	[FS0] = C;
	{
	    D9 87 <dFS0 mtp bytesperunit>	(fld  dword [edi+FS0*4])
	    DD 9F <dFA0 mtp bytesperunit>	(fstp qword [edi+FA0*4])
	}
	end;

      ( MG vimana emit - dump one frame: dzat(3d) pwr cur ap_reached status )
"MG vimana emit"
	[FA0] = [MgDzatX0]; [FA1] = [MgDzatX1]; => MGD d FA;
	[FA0] = [MgDzatY0]; [FA1] = [MgDzatY1]; => MGD d FA;
	[FA0] = [MgDzatZ0]; [FA1] = [MgDzatZ1]; => MGD d FA;
	C = [MgPwr]; => MGD i;
	[FA0] = [MgCur0];  [FA1] = [MgCur1];  => MGD d FA;
	C = [MgApreached]; => MGD i;
	C = [MgStatus];    => MGD i;
	end;

( ==================================================================== )
( MG vimana - opcode 1                                                 )
( ==================================================================== )

"MG vimana"
	( read inputs )
	=> MG CR next; [MgNframes] = C;
	=> MG CR next; [MgDzatX0] = C; => MG CR next; [MgDzatX1] = C;
	=> MG CR next; [MgDzatY0] = C; => MG CR next; [MgDzatY1] = C;
	=> MG CR next; [MgDzatZ0] = C; => MG CR next; [MgDzatZ1] = C;
	=> MG CR next; [MgApX0] = C;    => MG CR next; [MgApX1] = C;
	=> MG CR next; [MgApY0] = C;    => MG CR next; [MgApY1] = C;
	=> MG CR next; [MgApZ0] = C;    => MG CR next; [MgApZ1] = C;
	=> MG CR next; [MgInitD0] = C;  => MG CR next; [MgInitD1] = C;
	=> MG CR next; [MgCur0] = C;    => MG CR next; [MgCur1] = C;
	=> MG CR next; [MgTmp] = C;	( ap_target_ray float bits )
	=> MG CR next; [MgAnti] = C;
	=> MG CR next; [MgAptgt] = C;
	=> MG CR next; [MgPwr] = C;
	[MgApreached] = 0; [MgStspeed] = 1; [MgStatus] = 0;
	( materialise ray as a double into MgT0/1 for ras )
	C = [MgTmp]; => MG widen f32 FA from C;
	[MgT00] = [FA0]; [MgT01] = [FA1];

    "MG vimana loop"
	A = [MgNframes]; ? A <= 0 -> MG vimana done;
	[MgNframes]-;

	( if drive gated off, emit a static frame )
	A = [MgAptgt]; ? A = 0 -> MG vimana static;
	A = [MgStspeed]; ? A = 0 -> MG vimana static;
	-> MG vimana drive;
    "MG vimana static"
	[MgStatus] = 0;
	=> MG vimana emit;
	-> MG vimana loop;

    "MG vimana drive"
	( dxx = dzat_x - ap_x ; dyy ; dzz )
	[FA0]=[MgDzatX0];[FA1]=[MgDzatX1]; [FB0]=[MgApX0];[FB1]=[MgApX1]; => FSub;
	[MgDxx0]=[FA0];[MgDxx1]=[FA1];
	[FA0]=[MgDzatY0];[FA1]=[MgDzatY1]; [FB0]=[MgApY0];[FB1]=[MgApY1]; => FSub;
	[MgDyy0]=[FA0];[MgDyy1]=[FA1];
	[FA0]=[MgDzatZ0];[FA1]=[MgDzatZ1]; [FB0]=[MgApZ0];[FB1]=[MgApZ1]; => FSub;
	[MgDzz0]=[FA0];[MgDzz1]=[FA1];

	( l_dsd = sqrt(dxx^2+dyy^2+dzz^2) )
	[FA0]=[MgDxx0];[FA1]=[MgDxx1]; [FB0]=[MgDxx0];[FB1]=[MgDxx1]; => FMul;
	[MgAcc0]=[FA0];[MgAcc1]=[FA1];
	[FA0]=[MgDyy0];[FA1]=[MgDyy1]; [FB0]=[MgDyy0];[FB1]=[MgDyy1]; => FMul;
	[FB0]=[MgAcc0];[FB1]=[MgAcc1]; => FAdd;
	[MgAcc0]=[FA0];[MgAcc1]=[FA1];
	[FA0]=[MgDzz0];[FA1]=[MgDzz1]; [FB0]=[MgDzz0];[FB1]=[MgDzz1]; => FMul;
	[FB0]=[MgAcc0];[FB1]=[MgAcc1]; => FAdd;
	=> FSqrt;
	[MgLdsd0]=[FA0];[MgLdsd1]=[FA1];

	( ras = ap_targetted==-1 ? 25000 : anti_rad ? 44*ray : 1.5*ray )
	A = [MgAptgt]; A + 1; ? A = 0 -> MG vimana ras apm1;
	( ap_targetted is 1 in the corpus: compute 44*ray or 1.5*ray )
	A = [MgAnti]; ? A = 0 -> MG vimana ras 15;
	[FA0]=[MgK440];[FA1]=[MgK441]; [FB0]=[MgT00];[FB1]=[MgT01]; => FMul;
	-> MG vimana ras save;
    "MG vimana ras 15"
	[FA0]=[MgK1p50];[FA1]=[MgK1p51]; [FB0]=[MgT00];[FB1]=[MgT01]; => FMul;
	-> MG vimana ras save;
    "MG vimana ras apm1"
	[FA0]=[MgK250000];[FA1]=[MgK250001];
    "MG vimana ras save"
	[MgRas0]=[FA0];[MgRas1]=[FA1];

	( thresholds )
	[FA0]=[MgK0p99990];[FA1]=[MgK0p99991]; [FB0]=[MgInitD0];[FB1]=[MgInitD1]; => FMul;
	[MgT10]=[FA0];[MgT11]=[FA1];			( t_charge = 0.9999*init_d )
	[FA0]=[MgK75000];[FA1]=[MgK75001]; [FB0]=[MgRas0];[FB1]=[MgRas1]; => FAdd;
	[MgReq0]=[FA0];[MgReq1]=[FA1];			( borrow Req as t_park )
	[FA0]=[MgK150000];[FA1]=[MgK150001]; [FB0]=[MgRas0];[FB1]=[MgRas1]; => FAdd;
	[MgRt0]=[FA0];[MgRt1]=[FA1];			( borrow Rt as t_link )
	[FA0]=[MgK0p9990];[FA1]=[MgK0p9991]; [FB0]=[MgInitD0];[FB1]=[MgInitD1]; => FMul;
	[MgAcc0]=[FA0];[MgAcc1]=[FA1];			( borrow Acc as t_drive )

	( if l_dsd < ras -> CALIBRATED, terminal )
	[FA0]=[MgLdsd0];[FA1]=[MgLdsd1]; [FB0]=[MgRas0];[FB1]=[MgRas1]; => FCmp;
	A = [FI]; ? A < 0 -> MG vimana calibrated;

	( elif l_dsd > t_charge -> CHARGING )
	[FA0]=[MgLdsd0];[FA1]=[MgLdsd1]; [FB0]=[MgT10];[FB1]=[MgT11]; => FCmp;
	A = [FI]; ? A > 0 -> MG vimana charging;
	( elif l_dsd < t_park -> PARKING )
	[FA0]=[MgLdsd0];[FA1]=[MgLdsd1]; [FB0]=[MgReq0];[FB1]=[MgReq1]; => FCmp;
	A = [FI]; ? A < 0 -> MG vimana parking;
	( elif l_dsd < t_link -> LINKING )
	[FA0]=[MgLdsd0];[FA1]=[MgLdsd1]; [FB0]=[MgRt0];[FB1]=[MgRt1]; => FCmp;
	A = [FI]; ? A < 0 -> MG vimana linking;
	( elif l_dsd < t_drive -> DRIVING )
	[FA0]=[MgLdsd0];[FA1]=[MgLdsd1]; [FB0]=[MgAcc0];[FB1]=[MgAcc1]; => FCmp;
	A = [FI]; ? A < 0 -> MG vimana driving;
	-> MG vimana ignition;

    "MG vimana calibrated"
	[MgStatus] = 6; [MgApreached] = 1; [MgStspeed] = 0;
	=> MG vimana emit;
	-> MG vimana loop;

    "MG vimana charging"
	[FA0]=[MgKreqC10];[FA1]=[MgKreqC11]; [FB0]=[MgLdsd0];[FB1]=[MgLdsd1]; => FMul;
	[MgReq0]=[FA0];[MgReq1]=[FA1];
	[MgRt0]=[MgKrtCharg0];[MgRt1]=[MgKrtCharg1]; [MgStatus]=1;
	-> MG vimana integrate;
    "MG vimana parking"
	[FA0]=[MgKreqC20];[FA1]=[MgKreqC21]; [FB0]=[MgLdsd0];[FB1]=[MgLdsd1]; => FMul;
	[MgReq0]=[FA0];[MgReq1]=[FA1];
	[MgRt0]=[MgKrtPark0];[MgRt1]=[MgKrtPark1]; [MgStatus]=2;
	-> MG vimana integrate;
    "MG vimana linking"
	[FA0]=[MgKreqC20];[FA1]=[MgKreqC21]; [FB0]=[MgLdsd0];[FB1]=[MgLdsd1]; => FMul;
	[MgReq0]=[FA0];[MgReq1]=[FA1];
	[MgRt0]=[MgKrtLink0];[MgRt1]=[MgKrtLink1]; [MgStatus]=3;
	-> MG vimana integrate;
    "MG vimana driving"
	[FA0]=[MgKreqC30];[FA1]=[MgKreqC31]; [FB0]=[MgLdsd0];[FB1]=[MgLdsd1]; => FMul;
	[MgReq0]=[FA0];[MgReq1]=[FA1];
	[MgRt0]=[MgKrtDrive0];[MgRt1]=[MgKrtDrive1]; [MgStatus]=4;
	-> MG vimana integrate;
    "MG vimana ignition"
	[FA0]=[MgKreqC40];[FA1]=[MgKreqC41]; [FB0]=[MgLdsd0];[FB1]=[MgLdsd1]; => FMul;
	[MgReq0]=[FA0];[MgReq1]=[FA1];
	[MgRt0]=[MgKrtIgn0];[MgRt1]=[MgKrtIgn1]; [MgStatus]=5;
	( -> fall through to integrate )

    "MG vimana integrate"
	( cur += (req-cur)*rt )
	[FA0]=[MgReq0];[FA1]=[MgReq1]; [FB0]=[MgCur0];[FB1]=[MgCur1]; => FSub;
	[FB0]=[MgRt0];[FB1]=[MgRt1]; => FMul;
	[FB0]=[MgCur0];[FB1]=[MgCur1]; => FAdd;
	[MgCur0]=[FA0];[MgCur1]=[FA1];
	( if cur < 10 -> cur = 10 )
	[FB0]=[MgK100];[FB1]=[MgK101]; => FCmp;
	A = [FI]; ? A >= 0 -> MG vimana cur ok;
	[FA0]=[MgK100];[FA1]=[MgK101]; [MgCur0]=[FA0];[MgCur1]=[FA1];
    "MG vimana cur ok"
	( dzat_x -= dxx/cur )
	[FA0]=[MgDxx0];[FA1]=[MgDxx1]; [FB0]=[MgCur0];[FB1]=[MgCur1]; => FQuo;
	[MgT00]=[FA0];[MgT01]=[FA1];
	[FA0]=[MgDzatX0];[FA1]=[MgDzatX1]; [FB0]=[MgT00];[FB1]=[MgT01]; => FSub;
	[MgDzatX0]=[FA0];[MgDzatX1]=[FA1];
	[FA0]=[MgDyy0];[FA1]=[MgDyy1]; [FB0]=[MgCur0];[FB1]=[MgCur1]; => FQuo;
	[MgT00]=[FA0];[MgT01]=[FA1];
	[FA0]=[MgDzatY0];[FA1]=[MgDzatY1]; [FB0]=[MgT00];[FB1]=[MgT01]; => FSub;
	[MgDzatY0]=[FA0];[MgDzatY1]=[FA1];
	[FA0]=[MgDzz0];[FA1]=[MgDzz1]; [FB0]=[MgCur0];[FB1]=[MgCur1]; => FQuo;
	[MgT00]=[FA0];[MgT01]=[FA1];
	[FA0]=[MgDzatZ0];[FA1]=[MgDzatZ1]; [FB0]=[MgT00];[FB1]=[MgT01]; => FSub;
	[MgDzatZ0]=[FA0];[MgDzatZ1]=[FA1];
	( pwr -= l_dsd*1e-5, i16 )
	[FA0]=[MgLdsd0];[FA1]=[MgLdsd1]; [FB0]=[MgK1em50];[FB1]=[MgK1em51]; => FMul;
	[MgT00]=[FA0];[MgT01]=[FA1];
	[FI]=[MgPwr]; => IntToF;
	[FB0]=[MgT00];[FB1]=[MgT01]; => FSub;
	=> FToIntChop; C = [FI]; => MG i16 C; [MgPwr] = C;
	=> MG vimana emit;
	-> MG vimana loop;

    "MG vimana done"
	end;

( ==================================================================== )
( MG approach - opcode 2                                               )
( ==================================================================== )

"MG approach"
	=> MG CR next; [MgNframes] = C;
	=> MG CR next; [MgDzatX0] = C; => MG CR next; [MgDzatX1] = C;
	=> MG CR next; [MgDzatY0] = C; => MG CR next; [MgDzatY1] = C;
	=> MG CR next; [MgDzatZ0] = C; => MG CR next; [MgDzatZ1] = C;
	=> MG CR next; [MgPlX0] = C;    => MG CR next; [MgPlX1] = C;
	=> MG CR next; [MgPlY0] = C;    => MG CR next; [MgPlY1] = C;
	=> MG CR next; [MgPlZ0] = C;    => MG CR next; [MgPlZ1] = C;
	=> MG CR next; [MgInitD0] = C;  => MG CR next; [MgInitD1] = C;
	=> MG CR next; [MgCur0] = C;    => MG CR next; [MgCur1] = C;
	=> MG CR next; [MgPray] = C;
	=> MG CR next; [MgPwr] = C;
	=> MG CR next; [MgIpreaching] = C;
	[MgIpreached] = 0; A = [MgIpreaching]; ? A = 0 -> MG approach ipr1; [MgIpreached] = 1;
    "MG approach ipr1"
	[MgStatus] = 0;

    "MG approach loop"
	A = [MgNframes]; ? A <= 0 -> MG approach done;
	[MgNframes]-;
	A = [MgIpreaching]; ? A = 0 -> MG approach static;
	-> MG approach drive;
    "MG approach static"
	[MgStatus] = 0;
	[FA0]=[MgDzatX0];[FA1]=[MgDzatX1]; => MGD d FA;
	[FA0]=[MgDzatY0];[FA1]=[MgDzatY1]; => MGD d FA;
	[FA0]=[MgDzatZ0];[FA1]=[MgDzatZ1]; => MGD d FA;
	C=[MgPwr]; => MGD i;
	[FA0]=[MgCur0];[FA1]=[MgCur1]; => MGD d FA;
	C=[MgIpreached]; => MGD i;
	C=[MgStatus]; => MGD i;
	-> MG approach loop;

    "MG approach drive"
	[FA0]=[MgDzatX0];[FA1]=[MgDzatX1]; [FB0]=[MgPlX0];[FB1]=[MgPlX1]; => FSub;
	[MgDxx0]=[FA0];[MgDxx1]=[FA1];
	[FA0]=[MgDzatY0];[FA1]=[MgDzatY1]; [FB0]=[MgPlY0];[FB1]=[MgPlY1]; => FSub;
	[MgDyy0]=[FA0];[MgDyy1]=[FA1];
	[FA0]=[MgDzatZ0];[FA1]=[MgDzatZ1]; [FB0]=[MgPlZ0];[FB1]=[MgPlZ1]; => FSub;
	[MgDzz0]=[FA0];[MgDzz1]=[FA1];
	[FA0]=[MgDxx0];[FA1]=[MgDxx1]; [FB0]=[MgDxx0];[FB1]=[MgDxx1]; => FMul;
	[MgAcc0]=[FA0];[MgAcc1]=[FA1];
	[FA0]=[MgDyy0];[FA1]=[MgDyy1]; [FB0]=[MgDyy0];[FB1]=[MgDyy1]; => FMul;
	[FB0]=[MgAcc0];[FB1]=[MgAcc1]; => FAdd;
	[MgAcc0]=[FA0];[MgAcc1]=[FA1];
	[FA0]=[MgDzz0];[FA1]=[MgDzz1]; [FB0]=[MgDzz0];[FB1]=[MgDzz1]; => FMul;
	[FB0]=[MgAcc0];[FB1]=[MgAcc1]; => FAdd;
	=> FSqrt;
	[MgLdsd0]=[FA0];[MgLdsd1]=[FA1];

	( thresholds for the approach cascade )
	[FA0]=[MgK0p999990];[FA1]=[MgK0p999991]; [FB0]=[MgInitD0];[FB1]=[MgInitD1]; => FMul;
	[MgT10]=[FA0];[MgT11]=[FA1];			( 0.99999*init_d )
	( if l_dsd > 0.99999*init_d -> WARMING )
	[FA0]=[MgLdsd0];[FA1]=[MgLdsd1]; [FB0]=[MgT10];[FB1]=[MgT11]; => FCmp;
	A = [FI]; ? A > 0 -> MG approach warming;
	( elif l_dsd<25 and init_d>500 -> REFINING )
	[FA0]=[MgLdsd0];[FA1]=[MgLdsd1]; [FB0]=[MgK250];[FB1]=[MgK251]; => FCmp;
	A = [FI]; ? A >= 0 -> MG approach notref;
	[FA0]=[MgK5000];[FA1]=[MgK5001]; [FB0]=[MgInitD0];[FB1]=[MgInitD1]; => FCmp;
	A = [FI]; ? A <= 0 -> MG approach notref;
	-> MG approach refining;
    "MG approach notref"
	( elif l_dsd<100 and init_d>500 -> BREAKING )
	[FA0]=[MgLdsd0];[FA1]=[MgLdsd1]; [FB0]=[MgK1000];[FB1]=[MgK1001]; => FCmp;
	A = [FI]; ? A >= 0 -> MG approach notbrk;
	[FA0]=[MgK5000];[FA1]=[MgK5001]; [FB0]=[MgInitD0];[FB1]=[MgInitD1]; => FCmp;
	A = [FI]; ? A <= 0 -> MG approach notbrk;
