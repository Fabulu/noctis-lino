"VHTR load raw"
	[VHTRloadpos] = 0; [VHTRloadok] = 1;
    "VHTR load raw chunk"
	A = [VHTRloadtotal]; A - [VHTRloadpos];
	? A <= 0 -> VHTR load raw done;
	? A <= 32768 -> VHTR load raw size ready; A = 32768;
    "VHTR load raw size ready"
	[VHTRloadchunk] = A;
	[File Name] = [VHTRloadname]; [File Position] = [VHTRloadpos];
	[File Command] = READ; [Block Pointer] = stage; [Block Size] = [VHTRloadchunk];
	isocall;
	A = [Block Size]; ? A = [VHTRloadchunk] -> VHTR load raw unpack;
	[VHTRloadok] = 0; end;
    "VHTR load raw unpack"
	[VHTRloadi] = 0;
    "VHTR load raw byte"
	A = [VHTRloadi]; C = A; C > 2; D = stage; D + C; C = [D];
	A = [VHTRloadi]; A & 3; A '* 8; C > A; C & 255;
	A = nw; A + [VHTRloadbase]; A + [VHTRloadpos]; A + [VHTRloadi]; [A] = C;
	[VHTRloadi]+; A = [VHTRloadi]; ? A < [VHTRloadchunk] -> VHTR load raw byte;
	A = [VHTRloadpos]; A + [VHTRloadchunk]; [VHTRloadpos] = A;
	-> VHTR load raw chunk;
    "VHTR load raw done"
	end;
