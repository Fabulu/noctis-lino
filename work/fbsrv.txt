      ( *** fbsrv - the servo battery: CRITICAL 1, demonstrated ***

	The claim being made about "TK servo" is a claim about behaviour
	OVER TIME, and the defect it replaces only appears eight minutes
	into a run.  Waiting eight minutes is not a test.  So the servo
	is split into a SAMPLER, which reads the clock, and an ESTIMATOR,
	which does not - and the estimator is driven here with exact
	synthetic windows, in milliseconds, across the counter wrap.

	Nothing test-only lives in fbtick.  This programme calls the same
	"TK servo" and "TK servo apply" the shell calls.

	SEVEN BATTERIES.

	B1 WINDOW LENGTHS.  Six window lengths, three consecutive firings
	   each, with the exact synthetic counts a rate of 8999 cpms would
	   produce.  Expected: 500 ms refused as too short (why 3), 4000 /
	   14061 / 60000 accepted at 8999, 470000 exact but refused as too
	   long, 500000 refused as too long - which is the whole point,
	   because at 500000 ms the count product has ALIASED past 2^32
	   and the old servo would have computed 409 cpms from it and
	   applied a permanent clamp.

	B2 THE OLD SERVO, REPLAYED.  The shipped servo divided
	   ([Counts] - [Counts at run start]) by the milliseconds since
	   run start.  Six elapsed times; the count product is taken mod
	   2^32 exactly as the hardware counter does.  Expected: 8999 at
	   60/300/450 s, then the collapse.  Then the +-1% clamp is
	   applied twenty times to show that the clamp does not save it -
	   it converts a one-shot collapse into a permanent ratchet.

	B3 ROUNDING.  ms = 14061 and cnt = 8999*14061 + 7031 is
	   constructed so that truncation gives 8999 and rounding gives
	   9000.  One added term; a systematic ~0.2 s/hour.

	B4 THE CLAMP FLOOR.  cpms/100 truncates to zero below cpms 100,
	   so the +-1% band collapses to a single point and NO sample can
	   ever move it again: an absorbing state.  Driven from cpms 99
	   with a correct 8999 sample, five times.

	B5 RE-BASING.  The full "TK servo" is fired twice in a row with
	   the anchors set once.  If the anchors are re-based, firing 2
	   sees a window of a few milliseconds and refuses it; if they are
	   not, firing 2 sees the SAME window again and accepts it.  That
	   is the difference between a bracket that stays bounded and one
	   that grows back toward the wrap.

	B6 MIDNIGHT.  "TK wall fold" is driven with synthetic raw values
	   across midnight - 23:59:59.900 then 00:00:00.100 - instead of
	   waiting for midnight.  Then the estimator is fed the negative
	   window an UNFOLDED clock would produce, to show the SIGNED band
	   refuses it where the old unsigned '< 500 accepted it as 4.2e9.

	B7 THE RING SWEEP.  65,536 anchors at an odd stride, nine window
	   lengths each, checking that the unsigned difference is exact
	   across the wrap.  The subtraction was never the bug; this
	   proves it.

	Output: work/fb-out.bin, FBDUMP v2, four records.               )

"libraries"

	fbmem;
	fbtick;

"directors"

	program name = { fbsrv };
	unit = 32;

"constants"

	SERVON	= 1;		( fire on every call, this is a probe )
	SVRATE	= 8999;		( the rate every synthetic window encodes )
	SVNW	= 6;		( window lengths in B1 )
	SVNF	= 3;		( consecutive firings per length )
	SVNO	= 6;		( elapsed times in B2 )
	SVRATCH	= 20;		( clamp applications in B2 )
	SVNANCH	= 65536;
	SVSTR	= 65537;
	SVNRING	= 9;

	TSVW	= 19;		( probe-local tags )
	TSVO	= 20;
	TSVR	= 21;

"variables"

	SVi	= 0;
	SVf	= 0;
	SVp	= 0;
	SVcnt	= 0;
	SVms	= 0;
	SVel	= 0;
	SVnaive	= 0;
	SVrat	= 0;
	SVstep	= 0;
	SVank	= 0;
	SVexp	= 0;
	SVgot	= 0;
	SVringn	= 0;
	SVringf	= 0;
	SVfold0	= 0;
	SVfold1	= 0;
	SVfold2	= 0;
	SVfoldd	= 0;
	SVmidwhy = 0;
	SVmidcp	= 0;
	SVrb0	= 0;
	SVrb1	= 0;
	SVrbw0	= 0;
	SVrbw1	= 0;
	SVrbc0	= 0;
	SVrbc1	= 0;
	SVround	= 0;
	SVtrunc	= 0;
	SVfl	= 0;
	SVfl1	= 0;
	SVfl5	= 0;
	SVtmp	= 0;

"workspace"

	svw	= 144;		( SVNW x SVNF x 8 )
	svo	= 24;		( SVNO x 4 )
	svs	= 64;		( scalars )

"programme"

	=> SV run;
	end;

"SV run"
	=> TK seed;
	=> SV b1;
	=> SV b2;
	=> SV b3;
	=> SV b4;
	=> SV b5;
	=> SV b6;
	=> SV b7;
	=> SV dump;
	end;

( ==================================================================== )
( B1 - window lengths, three consecutive firings each                   )
( ==================================================================== )

"SV b1"
	A = svw; [SVp] = A;
	[SVi] = 0;
    "SV b1 case"
	=> SV window;			( [SVms] <- the length )
	[TKcpms] = SVRATE; => TK period;
	[TKsrvlogn] = 0;
	[SVf] = 0;
    "SV b1 fire"
	( the counts a true SVRATE would deliver over [SVms], taken mod
	  2^32 exactly as a 32-bit hardware counter does )
	A = [SVms]; A '* SVRATE; [SVcnt] = A;
	[TKsrvcnt] = [SVcnt];
	[TKsrvms]  = [SVms];
	=> TK servo apply;
	A = [SVp];
	[A plus 0] = [SVms];
	[A plus 1] = [SVf];
	[A plus 2] = [SVcnt];
	[A plus 3] = [TKsrvnew];
	[A plus 4] = [TKcpms];
	[A plus 5] = [TKsrvwhy];
	[A plus 6] = [TKbase];
	[A plus 7] = [TKsrvlogn];
	A + 8; [SVp] = A;
	[SVf]+;
	A = [SVf]; ? A '< SVNF -> SV b1 fire;
	[SVi]+;
	A = [SVi]; ? A '< SVNW -> SV b1 case;
	end;

"SV window"
	A = [SVi];
	? A = 0 -> SV w0;
	? A = 1 -> SV w1;
	? A = 2 -> SV w2;
	? A = 3 -> SV w3;
	? A = 4 -> SV w4;
	[SVms] = 500000; end;
    "SV w0"
	[SVms] = 500; end;
    "SV w1"
	[SVms] = 4000; end;
    "SV w2"
	[SVms] = 14061; end;
    "SV w3"
	[SVms] = 60000; end;
    "SV w4"
	[SVms] = 470000; end;

( ==================================================================== )
( B2 - the OLD servo, replayed, and the ratchet the clamp produces      )
( ==================================================================== )

      ( cpms_naive = ((elapsed * 8999) mod 2^32) / elapsed.
	Then, starting from 8999, apply the +-1% clamp to that sample
	SVRATCH times and record where it lands: the clamp does not
	reject a nonsense sample, it walks towards it forever. )

"SV b2"
	A = svo; [SVp] = A;
	[SVi] = 0;
    "SV b2 case"
	=> SV elapsed;
	A = [SVel]; A '* SVRATE; [SVcnt] = A;
	A = [SVcnt]; A '/ [SVel]; [SVnaive] = A;

	[SVrat] = SVRATE;
	B = 0; B + SVRATCH;
    "SV b2 ratchet"
	D = [SVrat]; D '/ 100; ? D != 0 -> SV b2 step;
	D = 1;
    "SV b2 step"
	C = [SVrat]; C - D;		( low bound )
	E = [SVrat]; E + D;		( high bound )
	A = [SVnaive];
	? A '>= C -> SV b2 lo;
	A = C;
    "SV b2 lo"
	? A '<= E -> SV b2 hi;
	A = E;
    "SV b2 hi"
	[SVrat] = A;
	B ^ SV b2 ratchet;

	A = [SVp];
	[A plus 0] = [SVel];
	[A plus 1] = [SVcnt];
	[A plus 2] = [SVnaive];
	[A plus 3] = [SVrat];
	A + 4; [SVp] = A;
	[SVi]+;
	A = [SVi]; ? A '< SVNO -> SV b2 case;
	end;

"SV elapsed"
	A = [SVi];
	? A = 0 -> SV e0;
	? A = 1 -> SV e1;
	? A = 2 -> SV e2;
	? A = 3 -> SV e3;
	? A = 4 -> SV e4;
	[SVel] = 1200000; end;
    "SV e0"
	[SVel] = 60000; end;
    "SV e1"
	[SVel] = 300000; end;
    "SV e2"
	[SVel] = 450000; end;
    "SV e3"
	[SVel] = 600000; end;
    "SV e4"
	[SVel] = 900000; end;

( ==================================================================== )
( B3 - rounding: one added term, constructed to differ by exactly 1     )
( ==================================================================== )

"SV b3"
	A = SVRATE; A '* 14061; A + 7031; [SVcnt] = A;
	A = [SVcnt]; A '/ 14061; [SVtrunc] = A;
	[TKcpms] = SVRATE; => TK period;
	[TKsrvcnt] = [SVcnt]; [TKsrvms] = 14061;
	=> TK servo apply;
	[SVround] = [TKsrvnew];
	end;

( ==================================================================== )
( B4 - the clamp floor: cpms/100 truncating to zero is an absorbing     )
(      state, and it is reached by the very collapse B2 shows           )
( ==================================================================== )

      ( NOTE, found by this battery rather than assumed: "TK servo apply"
	clobbers A, B, C and D - the last thing it does is compute a log
	pointer into B.  A caller must NOT carry a loop count in a
	register across it.  The first version of this battery did, and
	the loop ran 59 times instead of 4, which is how the servo-log
	overflow counter earned its keep on its first run. )

"SV b4"
	[TKcpms] = 99; => TK period;
	[TKsrvlogn] = 0;
	A = 14061; A '* SVRATE; [SVcnt] = A;
	[TKsrvcnt] = [SVcnt]; [TKsrvms] = 14061;
	=> TK servo apply;
	[SVfl1] = [TKcpms];
	[SVtmp] = 0;
    "SV b4 more"
	[TKsrvcnt] = [SVcnt]; [TKsrvms] = 14061;
	=> TK servo apply;
	[SVtmp]+;
	A = [SVtmp]; ? A '< 4 -> SV b4 more;
	[SVfl5] = [TKcpms];
	( what the floorless form would do: step = 99/100 = 0, so the
	  band is [99,99] and no sample can ever move it )
	A = 99; A '/ 100; [SVfl] = A;
	end;

( ==================================================================== )
( B5 - re-basing: fire the WHOLE servo twice with the anchors set once  )
( ==================================================================== )

      ( Anchor the servo 5,000 ms and 5,000*8999 counts in the past, then
	fire twice with no real time in between.

	  re-based    : firing 1 sees ~5,000 ms and accepts; firing 2 sees
	                a few milliseconds and refuses it (why 3)
	  not re-based: firing 2 sees ~5,000 ms AGAIN and accepts it, and
	                a run of firings walks the bracket back toward the
	                counter wrap )

"SV b5"
	[TKcpms] = SVRATE; => TK period;
	[TKsrvlogn] = 0;
	[Timer Command] = READ COUNTS; isocall;
	A = [Counts]; C = 5000; C '* SVRATE; A - C; [TKsrv0c] = A;
	=> TK read wall;
	A = [TKtmp]; A - 5000; [TKsrv0w] = A;
	[TKsrvn] = 0;
	=> TK servo;
	[SVrbw0] = [TKsrvms];
	[SVrb0]  = [TKsrvwhy];
	[SVrbc0] = [TKsrvcnt];
	[TKsrvn] = 0;
	=> TK servo;
	[SVrbw1] = [TKsrvms];
	[SVrb1]  = [TKsrvwhy];
	[SVrbc1] = [TKsrvcnt];
	end;

      ( BOTH anchors are recorded, and that is not decoration.  The
	COUNTS anchor and the WALL anchor are re-based on separate lines,
	so a sabotage that deletes only one of them leaves the other
	working - and if the battery records only the millisecond window
	it sees nothing at all.  It was a blind spot on the first run of
	this battery, and closing it is what the counts columns are for:
	with the counts anchor stale, firing 2 reports a five-second
	count against a zero-millisecond window. )

( ==================================================================== )
( B6 - midnight, folded once instead of special-cased per consumer      )
( ==================================================================== )

"SV b6"
	( a clean state for the fold )
	[TKwallprev] = 0; [TKwallday] = 0;
	[TKwraw] = 86399900; => TK wall fold; [SVfold0] = [TKtmp];
	[TKwraw] = 100;      => TK wall fold; [SVfold1] = [TKtmp];
	[TKwraw] = 300;      => TK wall fold; [SVfold2] = [TKtmp];
	A = [SVfold1]; A - [SVfold0]; [SVfoldd] = A;	( must be +200 )

	( and the window an UNFOLDED clock would hand the estimator at
	  midnight: 100 - 86,399,900 = -86,399,800.  The old unsigned
	  test read that as 4.2e9 and ACCEPTED it. )
	[TKcpms] = SVRATE; => TK period;
	[TKsrvcnt] = 1000000;
	A = 100; A - 86399900; [TKsrvms] = A;
	=> TK servo apply;
	[SVmidwhy] = [TKsrvwhy];
	[SVmidcp]  = [TKcpms];
	end;

( ==================================================================== )
( B7 - the ring sweep: unsigned subtraction is exact across the wrap    )
( ==================================================================== )

"SV b7"
	[SVringn] = 0; [SVringf] = 0;
	[SVank] = 0; [SVi] = 0;
    "SV b7 anchor"
	[SVf] = 0;
    "SV b7 len"
	=> SV ring len;			( [SVms] <- one of nine lengths )
	A = [SVms]; A '* SVRATE; [SVexp] = A;
	( now = anchor + expected, mod 2^32; delta = now - anchor )
	A = [SVank]; A + [SVexp];
	A - [SVank]; [SVgot] = A;
	[SVringn]+;
	A = [SVgot]; ? A = [SVexp] -> SV b7 ok;
	[SVringf]+;
    "SV b7 ok"
	[SVf]+;
	A = [SVf]; ? A '< SVNRING -> SV b7 len;
	[SVank] + SVSTR;
	[SVi]+;
	A = [SVi]; ? A '< SVNANCH -> SV b7 anchor;
	end;

"SV ring len"
	A = [SVf];
	? A = 0 -> SV rl0;
	? A = 1 -> SV rl1;
	? A = 2 -> SV rl2;
	? A = 3 -> SV rl3;
	? A = 4 -> SV rl4;
	? A = 5 -> SV rl5;
	? A = 6 -> SV rl6;
	? A = 7 -> SV rl7;
	[SVms] = 470000; end;
    "SV rl0"
	[SVms] = 1; end;
    "SV rl1"
	[SVms] = 500; end;
    "SV rl2"
	[SVms] = 4000; end;
    "SV rl3"
	[SVms] = 14061; end;
    "SV rl4"
	[SVms] = 60000; end;
    "SV rl5"
	[SVms] = 100000; end;
    "SV rl6"
	[SVms] = 200000; end;
    "SV rl7"
	[SVms] = 400000; end;
