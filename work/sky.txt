      ( *** sky - NIV+ planetary sky production library ***

	Source: NOCTIS-1.CPP:1674-1736, :2736-3139 and the caller-side
	horizon pass at :3683-3697.  The caller fills RSBG before GRSK create.
	GRSK create deliberately stops before the horizon pass; GRSK horizon is
	the separate source-order operation used by planetary_main and by J.

	All caller-visible state is in the exclusive GRSK namespace.  The two
	random streams and the delivered smoothers/palette code remain owned by
	SU and PAL.  GRSKquadwords models the historical global solely so it can
	be saved and restored around both painters.  SU's delivered smoothers
	already carry the exact 16,200-qword bounds and the 41-byte readable tail.

	Float globals and locals below are binary32 bit images.  Helpers widen
	them through fp/fpconv, perform an expression in the delivered x87
	binary64 register ABI, and narrow only at C float assignment boundaries.
	This library contains no corpus, file, dump, fixture, or display code. )

      ( Every semantic byte write performed directly by this library
	passes through this gate.  Bounds are derived solely from the requested
	relative address, before the effective pointer is formed; address mutants
	therefore cannot forge the ledger by changing a neighbouring counter.
	Delivered SU smoothers remain separately witnessed whole-buffer kernels:
	lssmooth writes 64,480 bytes, ssmooth 63,360, psmooth 63,520. )
"GRSK write sbg"
	[GRSKwrites]+;
	A=[GRSKwaddr]; ? A < 0 -> GR sky write sbg oob; ? A '>= GRSKSTB -> GR sky write sbg oob;
	C=[GRSKwmin]; ? C != 0FFFFFFFFh -> GR sky write sbg min; [GRSKwmin]=A; -> GR sky write sbg max;
    "GR sky write sbg min" ? A '>= C -> GR sky write sbg max; [GRSKwmin]=A;
    "GR sky write sbg max"
	C=[GRSKwmax]; ? C != 0FFFFFFFFh -> GR sky write sbg max compare; [GRSKwmax]=A; -> GR sky write sbg store;
    "GR sky write sbg max compare" ? A '<= C -> GR sky write sbg store; [GRSKwmax]=A;
    "GR sky write sbg store"
	A=nw; A+RSBG; A+[GRSKwaddr]; [A]=[GRSKwbyte]; end;
    "GR sky write sbg oob" [GRSKoob]+; end;

"GRSK nebular sky"
	[GRSKqsave] = [GRSKquadwords]; [GRSKquadwords] = GRSKQW;
	C = 10000; => SU rnd; [GRSKax] = C;
	[GRSKcnt] = GRSKSTB; [GRSKi] = 0;
    "GR sky nebular noise"
	A = [GRSKax]; A + [GRSKcnt]; A & 0FFFFh;
	D = A; D & 8000h; ? D = 0 -> GR sky nebular positive;
	A - 65536;
    "GR sky nebular positive"
	B = A; A * B;
	D = A; D & 0FFFFh;
	A > 16; A & 0FFFFh; D + A; D & 0FFFFh; [GRSKax] = D;
	D & 3Fh; [GRSKbl] = D;
	[GRSKwaddr]=[GRSKi]; [GRSKwbyte]=[GRSKbl]; => GRSK write sbg;
	[GRSKi]+; A = [GRSKcnt]; A - 1; [GRSKcnt] = A;
	? A != 0 -> GR sky nebular noise;
	[SUpbase] = RSBG; => SU lssmooth;
	[GRSKwrites]+64480;
	C = 2; => SU rnd; A = C; ? A = 0 -> GR sky nebular no ss;
	[SUpbase] = RSBG; => SU ssmooth;
	[GRSKwrites]+63360;
    "GR sky nebular no ss"
	C = 3; => SU rnd; A = C; ? A = 0 -> GR sky nebular no ps;
	[SUpbase] = RSBG; => SU psmooth grays;
	[GRSKwrites]+63520;
    "GR sky nebular no ps"

	end;

	[FT0]=[FA0]; [FT1]=[FA1];
	[FI] = [GRSKr]; => IntToF; [FB0]=[FA0]; [FB1]=[FA1];

	? A '>= GRSKSTB -> GR sky cloud xnext;

	( b = 1.4142 / sqrt((x+r)^2+(y+r)^2), a float assignment )
	A = [GRSKxi]; A + [GRSKr]; [FI] = A; => IntToF;
	[FB0]=[FA0]; [FB1]=[FA1]; => FMul; [FT0]=[FA0]; [FT1]=[FA1];

	[FB0]=[FA0]; [FB1]=[FA1];
	[FA0]=2DE00D1Bh; [FA1]=3FF6A090h; => FQuo;
	=> FStoreF32; [GRSKx]=[FS0];
	=> GRSK k64; => GRSK x mul k; => GRSK store o; [GRSKx]=[GRSKo];

	[GRSKy]=GRSKF63; => GRSK compare;
	A = [FI]; ? A <= 0 -> GR sky cloud store;
	[FS0]=GRSKF63; => FLoadF32; => GRSK store o; [GRSKx]=[GRSKo];

	[GRSKx]=GRSKF0; => GRSK x assign; A=[GRSKptr]; [A]=[GRSKo];
    "GR sky clamp fb"
	A=GRSKfb; A+[GRSKi]; [GRSKptr]=A; [GRSKx]=[A]; [GRSKy]=GRSKF0; => GRSK compare; A=[FI]; ? A >= 0 -> GR sky clamp next;
	[GRSKx]=GRSKF0; => GRSK x assign; A=[GRSKptr]; [A]=[GRSKo];

	[GRSKi]+; A=[GRSKi]; ? A '< 4 -> GR sky clamp colours;

	A=[GRSKptype]; ? A=3 -> GR sky saturate; ? A=5 -> GR sky saturate; -> GR sky scaled;
    "GR sky saturate"
	[GRSKi]=0;
    "GR sky sat loop"
	A=GRSKfr; A+[GRSKi]; [GRSKptr]=A; [GRSKx]=[A]; => GRSK saturate x; => GRSK store o; A=[GRSKptr]; [A]=[GRSKo];
	A=GRSKfg; A+[GRSKi]; [GRSKptr]=A; [GRSKx]=[A]; => GRSK saturate x; => GRSK store o; A=[GRSKptr]; [A]=[GRSKo];
	A=GRSKfb; A+[GRSKi]; [GRSKptr]=A; [GRSKx]=[A]; => GRSK saturate x; => GRSK store o; A=[GRSKptr]; [A]=[GRSKo];
	[GRSKi]+; A=[GRSKi]; ? A '< 4 -> GR sky sat loop;

    "GR sky scaled"
	[GRSKi]=0;
    "GR sky scale loop"
	A=GRSKfr; A+[GRSKi]; [GRSKptr]=A; [GRSKz]=[A]; => GRSK scale x; => GRSK store o; A=[GRSKptr]; [A]=[GRSKo];
	A=GRSKfg; A+[GRSKi]; [GRSKptr]=A; [GRSKz]=[A]; => GRSK scale x; => GRSK store o; A=[GRSKptr]; [A]=[GRSKo];
	A=GRSKfb; A+[GRSKi]; [GRSKptr]=A; [GRSKz]=[A]; => GRSK scale x; => GRSK store o; A=[GRSKptr]; [A]=[GRSKo];
	[GRSKi]+; A=[GRSKi]; ? A=1 -> GR sky scale skip1; ? A '< 4 -> GR sky scale loop; -> GR sky scale slot1;
    "GR sky scale skip1" [GRSKi]=2; -> GR sky scale loop;
	"GR sky scale slot1"
	( slot 1 source assignment has the extra sb in the same expression. )
	[GRSKz]=[GRSKfr plus 1]; => GRSK scale sky x; => GRSK store o; [GRSKfr plus 1]=[GRSKo];
	[GRSKz]=[GRSKfg plus 1]; => GRSK scale sky x; => GRSK store o; [GRSKfg plus 1]=[GRSKo];
	[GRSKz]=[GRSKfb plus 1]; => GRSK scale sky x; => GRSK store o; [GRSKfb plus 1]=[GRSKo];

	[SHdstb]=srfpal6;
	A=[GRSKatmosphere]; ? A != 0 -> GR sky pal atmosphere;
	[FBSHfirst]=64; [FBSHn]=64; [SHsr]=0; [SHsg]=0; [SHsb]=0; [SHer]=42C80000h; [SHeg]=42DC0000h; [SHeb]=42F00000h; => PAL shade;
	-> GR sky pal normal;
    "GR sky pal atmosphere"
	A=[GRSKnightzone]; ? A=0 -> GR sky pal day;
	[FBSHfirst]=64; [FBSHn]=64; [SHsr]=0; [SHsg]=0; [SHsb]=0; [SHer]=42700000h; [SHeg]=42780000h; [SHeb]=42800000h; => PAL shade;
	A=[GRSKptype]; ? A != 3 -> GR sky pal nightdim;
	[FBSHfirst]=0; [FBSHn]=64; [SHsr]=0; [SHsg]=0; [SHsb]=0; [SHer]=42800000h; [SHeg]=42780000h; [SHeb]=42700000h; => PAL shade;
	[FBSHfirst]=128; [FBSHn]=64; [SHsr]=0; [SHsg]=0; [SHsb]=0; [SHer]=42800000h; [SHeg]=42800000h; [SHeb]=42800000h; => PAL shade;
	[FBSHfirst]=192; [FBSHn]=64; [SHsr]=41000000h; [SHsg]=41400000h; [SHsb]=41800000h; [SHer]=42600000h; [SHeg]=42700000h; [SHeb]=42800000h; => PAL shade;
	-> GR sky pal done;
    "GR sky pal nightdim"
	( land/horizon/vegetation filters are reduced before normal ramps )
	[GRSKi]=0;
    "GR sky night dim loop"
	A=GRSKfr; A+[GRSKi]; [GRSKptr]=A; [GRSKx]=[A]; => GRSK k033; => GRSK x mul k; => GRSK store o; A=[GRSKptr]; [A]=[GRSKo];
	A=GRSKfg; A+[GRSKi]; [GRSKptr]=A; [GRSKx]=[A]; => GRSK k044; => GRSK x mul k; => GRSK store o; A=[GRSKptr]; [A]=[GRSKo];
	A=GRSKfb; A+[GRSKi]; [GRSKptr]=A; [GRSKx]=[A]; => GRSK k055; => GRSK x mul k; => GRSK store o; A=[GRSKptr]; [A]=[GRSKo];
	[GRSKi]+; A=[GRSKi]; ? A=1 -> GR sky night skip1; ? A '< 4 -> GR sky night dim loop; -> GR sky pal normal;
    "GR sky night skip1" [GRSKi]=2; -> GR sky night dim loop;
    "GR sky pal day"
	[FBSHfirst]=64; [FBSHn]=64; [SHsr]=0; [SHsg]=0; [SHsb]=0; [SHer]=[GRSKfr plus 1]; [SHeg]=[GRSKfg plus 1]; [SHeb]=[GRSKfb plus 1]; => PAL shade;
    "GR sky pal normal"
	[FBSHfirst]=0; [FBSHn]=44; [SHsr]=0; [SHsg]=0; [SHsb]=0; [SHer]=[GRSKfr plus 0]; [SHeg]=[GRSKfg plus 0]; [SHeb]=[GRSKfb plus 0]; => PAL shade;
	[FBSHfirst]=44; [FBSHn]=20; [SHsr]=[GRSKfr plus 0]; [SHsg]=[GRSKfg plus 0]; [SHsb]=[GRSKfb plus 0]; [SHer]=[GRSKfr plus 1]; [SHeg]=[GRSKfg plus 1]; [SHeb]=[GRSKfb plus 1]; => PAL shade;
	[FBSHfirst]=128; [FBSHn]=10; [SHsr]=0; [SHsg]=0; [SHsb]=0; [SHer]=[GRSKfr plus 0]; [SHeg]=[GRSKfg plus 0]; [SHeb]=[GRSKfb plus 0]; => PAL shade;
	[FBSHfirst]=138; [FBSHn]=44; [SHsr]=[GRSKfr plus 0]; [SHsg]=[GRSKfg plus 0]; [SHsb]=[GRSKfb plus 0]; [SHer]=[GRSKfr plus 2]; [SHeg]=[GRSKfg plus 2]; [SHeb]=[GRSKfb plus 2]; => PAL shade;
	[FBSHfirst]=182; [FBSHn]=10; [SHsr]=[GRSKfr plus 2]; [SHsg]=[GRSKfg plus 2]; [SHsb]=[GRSKfb plus 2]; [SHer]=[GRSKfr plus 1]; [SHeg]=[GRSKfg plus 1]; [SHeb]=[GRSKfb plus 1]; => PAL shade;
	[FBSHfirst]=192; [FBSHn]=10; [SHsr]=0; [SHsg]=0; [SHsb]=0; [SHer]=[GRSKfr plus 0]; [SHeg]=[GRSKfg plus 0]; [SHeb]=[GRSKfb plus 0]; => PAL shade;
	[FBSHfirst]=202; [FBSHn]=44; [SHsr]=[GRSKfr plus 0]; [SHsg]=[GRSKfg plus 0]; [SHsb]=[GRSKfb plus 0]; [SHer]=[GRSKfr plus 3]; [SHeg]=[GRSKfg plus 3]; [SHeb]=[GRSKfb plus 3]; => PAL shade;
	[FBSHfirst]=246; [FBSHn]=10; [SHsr]=[GRSKfr plus 3]; [SHsg]=[GRSKfg plus 3]; [SHsb]=[GRSKfb plus 3]; [SHer]=[GRSKfr plus 1]; [SHeg]=[GRSKfg plus 1]; [SHeb]=[GRSKfb plus 1]; => PAL shade;

"GRSK thermo"

	[GRSKx]=[GRSKdsd1]; => GRSK k033; => GRSK x mul k; [FT0]=[FA0]; [FT1]=[FA1];
	[GRSKx]=GRSKF90; => GRSK fa x; [FB0]=[FT0]; [FB1]=[FT1]; => FSub; => GRSK store o; [GRSKtemp]=[GRSKo];
