      ( *** pgproj - the two projection nuclei, the near-plane clip, the
	    four 2-D clip stages, getcoords and facing ***

	THE TWO NUCLEI ARE KEPT SEPARATE ON PURPOSE.

	  poly3d   folds dpp INTO the rotation (opt_pcosbeta = cos*dpp,
		   opt_psinalfa = sin*dpp) and then divides by  1/Zc .
	  polymap  rotates with a UNIT matrix (opt_tcos*, opt_tsin*) and
		   divides by  dpp/Zc .

	Algebraically the same; numerically not, and they do not share a
	rounding schedule.  Factoring them into one routine would be the
	single most tempting and most wrong simplification in this wave.

	THE FST ASYMMETRY IS REPRODUCED BY CONSTRUCTION, NOT BY CHOICE.
	At "fst zz / fmul opt_psinbeta" the value that continues into the
	multiply is the WIDE one still in st(0); the value the later
	"fld zz" reloads is the NARROW one in memory.  This port keeps
	the wide value in FA and calls PGF narrow on the slot, so both
	happen.  The four sites are zz/xx/z2/yy, rzf[si], zk, and
	video_x0/video_y0.  There is no flag to pick a side because the
	original does not pick a side.

	THE NEAR-PLANE TEST TREATS UNORDERED AS "BEHIND".  "fcomp uneg /
	fstsw / sahf / jb" takes CF, which is C0, and a NaN sets
	C3:C2:C0 = 111.  FCmp returns 2 for unordered; 2 and -1 both map
	to rwf = 0.

	THE EQUALITY BRANCHES ARE UNREACHABLE FOR ORDERED INPUT, AND THAT
	IS WORTH SAYING OUT LOUD.  Every "fcomp ... / jne interpolate /
	jmp copy" pair in the near-plane walk and in the four 2-D stages
	compares the OUTSIDE vertex against an INSIDE one.  Outside means
	strictly beyond the bound and inside means not beyond it, so the
	two can never be equal.  The copy branch is reachable only
	through a NaN, which is also the only way "fcomp" reports
	unordered.  This port implements the copy branch as written.

	ONE THING THIS PORT DOES NOT REPRODUCE, AND SAYS SO.
	TDPOLYGS.H:906 reads  "jne if25 / jmp else35" - a jump into the
	NEXT clip stage's code, which would then run stage 3's copy on
	video_x1/video_x2 and exit through STOP3 with vr3 unset.  It is
	the else25 label that was meant; else25 is unreachable because
	if25 ends in "jmp JMPR21".  Since the branch is taken only on an
	unordered compare, the divergence is unobservable for every
	ordered input.  It is recorded as an OPEN ITEM, not silently
	normalised.                                                      )

      ( PJ rotate : for each of [PJnrv] vertices, rotate into rxf/ryf/rzf
	and set rwf[] and doflag.

	[PJmode] 0 - poly3d: dpp is FUSED into the matrix, so rxf and ryf
		     come out pre-scaled and the projector divides 1/Zc.
	[PJmode] 1 - polymap: a UNIT matrix, and the projector divides
		     dpp/Zc.

	With alfa = beta = 0 the two matrices are
	    poly3d   rxf = dpp*(zz*0 + xx*1),  ryf = dpp*(yy*1 - z2*0)
	    polymap  rxf =      zz*0 + xx*1 ,  ryf =      yy*1 - z2*0
	but the CODE is written out in full, because a port that special-
	cases the identity cannot be run at any other camera angle and
	cannot be checked against one. )

"PJ rotate"
	[PJvr] = 0;
"PJ rotate selected"
	[PJdoflag] = 0;
    "PJ rot v"
	( zz = z[i] - cam_z ; the WIDE value continues, the NARROW one
	  is what "fld zz" reloads two instructions later. )
	A = [PJvr]; A + FSINZ; [PGFi] = A; => PGF a;
	[PGFi] = FSCAMZ; => PGF sub;
	[PGFi] = FSZZ; => PGF narrow;

	A = [PJvr]; A + FSINX; [PGFi] = A; => PGF a;
	[PGFi] = FSCAMX; => PGF sub;
	[PGFi] = FSXX; => PGF narrow;

	A = [PJvr]; A + FSINY; [PGFi] = A; => PGF a;
	[PGFi] = FSCAMY; => PGF sub;
	[PGFi] = FSYY; => PGF narrow;

	( rxf = zz*sinbeta + xx*cosbeta, with the p- or t- table )
	[PGFi] = FSZZ; => PGF a;
	A = FSPSB; C = [PJmode]; ? C = 0 -> PJ rot b1; A = FSTSB;
    "PJ rot b1"
	[PGFi] = A; => PGF mul;
	[PGFi] = FSW0; => PGF sa;
	[PGFi] = FSXX; => PGF a;
	A = FSPCB; C = [PJmode]; ? C = 0 -> PJ rot b2; A = FSTCB;
    "PJ rot b2"
	[PGFi] = A; => PGF mul;
	[PGFi] = FSW0; => PGF add;
	A = [PJvr]; A + FSRXF; [PGFi] = A; => PGF narrow;

	( z2 = zz*cosbeta - xx*sinbeta, ALWAYS the t- table )
	[PGFi] = FSZZ; => PGF a; [PGFi] = FSTCB; => PGF mul;
	[PGFi] = FSW0; => PGF sa;
	[PGFi] = FSXX; => PGF a; [PGFi] = FSTSB; => PGF mul;
	[PGFi] = FSW0; => PGF rsub;
	[PGFi] = FSZ2; => PGF narrow;

	( rzf = z2*cosalfa + yy*sinalfa, t- table in BOTH nuclei )
	[PGFi] = FSZ2; => PGF a; [PGFi] = FSTCA; => PGF mul;
	[PGFi] = FSW0; => PGF sa;
	[PGFi] = FSYY; => PGF a; [PGFi] = FSTSA; => PGF mul;
	[PGFi] = FSW0; => PGF add;
	( "fst rzf[si] / fcomp uneg" - the STORE is narrow, the COMPARE
	  is on the wide value still in st(0). )
	[PGFi] = FSW1; => PGF sa;
	A = [PJvr]; A + FSRZF; [PGFi] = A; => PGF narrow;
	[PGFi] = FSW1; => PGF a;
	[PGFi] = FSUNEG; => PGF b;
	=> FCmp;
	C = 0;
	A = [FI]; ? A = 2 -> PJ rot behind;	( unordered = behind )
	A = [FI]; ? A < 0 -> PJ rot behind;
	C = 1;
	[PJdoflag]+;
    "PJ rot behind"
	A = rwf; A + [PJvr]; [A] = C;

	( ryf = yy*cosalfa - z2*sinalfa, p- or t- table )
	[PGFi] = FSYY; => PGF a;
	A = FSPCA; C = [PJmode]; ? C = 0 -> PJ rot a1; A = FSTCA;
    "PJ rot a1"
	[PGFi] = A; => PGF mul;
	[PGFi] = FSW0; => PGF sa;
	[PGFi] = FSZ2; => PGF a;
	A = FSPSA; C = [PJmode]; ? C = 0 -> PJ rot a2; A = FSTSA;
    "PJ rot a2"
	[PGFi] = A; => PGF mul;
	[PGFi] = FSW0; => PGF rsub;
	A = [PJvr]; A + FSRYF; [PGFi] = A; => PGF narrow;

	[PJvr]+;
	A = [PJvr]; ? A '< [PJnrv] -> PJ rot v;
	end;

( ==================================================================== )
( the near-plane walk - Sutherland-Hodgman against z == uneg            )
( ==================================================================== )

      ( PJ zemit : emit ONE interpolated vertex from the outside vertex
	[PJbx] towards the visible vertex [PJvv], into ultima_*[[PJdi]].

	    zk = (uneg - Z_o) / (Z_v - Z_o)
	    X  = X_o + zk*(X_v - X_o)
	    Y  = Y_o + zk*(Y_v - Y_o)
	    Z  = uneg, copied as a BIT PATTERN

	zk is a "float" and is narrowed; the multiply that consumes it
	uses the value still in st(0), which is wide.  Both, again.

	THE GUARD IS EXACT FLOAT EQUALITY, NOT AN EPSILON.  When the
	denominator is one ULP the quotient is around 1e7 and the emitted
	vertex lands arbitrarily far off screen, which then changes WHICH
	2-D clip cases fire - so topology itself diverges and no envelope
	is coherent.  Those cases are counted and reported, never graded.
	An epsilon here is sabotage ZKEPS. )

"PJ zemit"
	A = [PJbx]; A + FSRZF; [PGFi] = A; => PGF a;
	A = [PJvv]; A + FSRZF; [PGFi] = A; => PGF b;
	=> FCmp;
	A = [FI]; ? A = 0 -> PJ ze copy;

	( zk = (uneg - Z_v) / (Z_o - Z_v)  -  read the assembly, not the
	  comment: "fld uneg / fsub rzf[vv] / fld rzf[vv] / fsubr
	  rzf[bx] / fdivp".  st1 = uneg - Z_vv, st0 = Z_bx - Z_vv. )
	[PGFi] = FSUNEG; => PGF a;
	A = [PJvv]; A + FSRZF; [PGFi] = A; => PGF sub;
	[PGFi] = FSW0; => PGF sa;
	A = [PJvv]; A + FSRZF; [PGFi] = A; => PGF a;
	A = [PJbx]; A + FSRZF; [PGFi] = A; => PGF rsub;
	[PGFi] = FSW1; => PGF sa;
	[PGFi] = FSW0; => PGF a;
	[PGFi] = FSW1; => PGF quo;
	[PGFi] = FSZK; => PGF narrow;

	( X = zk*(X_bx - X_vv) + X_vv, with zk still WIDE in st(0) )
	[PGFi] = FSW0; => PGF a; [PGFi] = FSW1; => PGF quo;
	[PGFi] = FSW2; => PGF sa;
	A = [PJbx]; A + FSRXF; [PGFi] = A; => PGF a;
	A = [PJvv]; A + FSRXF; [PGFi] = A; => PGF sub;
	[PGFi] = FSW2; => PGF mul;
	A = [PJvv]; A + FSRXF; [PGFi] = A; => PGF add;
	A = [PJdi]; A + FSUX; [PGFi] = A; => PGF narrow;

	( Y likewise, and this one reloads zk from MEMORY: "fld zk". )
	[PGFi] = FSZK; => PGF a;
	[PGFi] = FSW2; => PGF sa;
	A = [PJbx]; A + FSRYF; [PGFi] = A; => PGF a;
	A = [PJvv]; A + FSRYF; [PGFi] = A; => PGF sub;
	[PGFi] = FSW2; => PGF mul;
	A = [PJvv]; A + FSRYF; [PGFi] = A; => PGF add;
	A = [PJdi]; A + FSUY; [PGFi] = A; => PGF narrow;
	-> PJ ze z;
    "PJ ze copy"
	( the degenerate guard: copy the outside vertex's x and y as BIT
	  PATTERNS.  Reachable only through an unordered compare. )
	A = [PJbx]; A + FSRXF; [PGFi] = A;
	A = [PJdi]; A + FSUX;  [PGFj] = A; => PGF mov;
	A = [PJbx]; A + FSRYF; [PGFi] = A;
	A = [PJdi]; A + FSUY;  [PGFj] = A; => PGF mov;
    "PJ ze z"
	[PGFi] = FSUNEG; A = [PJdi]; A + FSUZ; [PGFj] = A; => PGF mov;
	end;

      ( PJ zclip : the whole "Conversione punti alle spalle
	dell'osservatore" loop.  Produces ultima_*[0..PJvr2-1]. )

"PJ zclip"
	[PJvr] = 0; [PJdi] = 0;
	A = [PJnrv]; A-; [PJdx] = A;
    "PJ zc for"
	A = rwf; A + [PJvr]; A = [A];
	? A != 0 -> PJ zc bypass;

	A = [PJvr]; A-;
	? A >= 0 -> PJ zc pv;
	A = [PJdx];
    "PJ zc pv"
	[PJpv] = A;
	A = [PJvr]; A+;
	? A '<= [PJdx] -> PJ zc nv;
	A = 0;
    "PJ zc nv"
	[PJnv] = A;

	( if BOTH neighbours are also behind, this vertex emits nothing )
	A = rwf; A + [PJpv]; A = [A];
	? A != 0 -> PJ zc jmpr;
	A = rwf; A + [PJnv]; A = [A];
	? A != 0 -> PJ zc jmpr;
	-> PJ zc stop;
    "PJ zc jmpr"
	[PJbx] = [PJvr];
	A = rwf; A + [PJpv]; C = [A];
	A = rwf; A + [PJnv]; A = [A]; C + A;
	? C != 2 -> PJ zc else11;

	( both neighbours visible: TWO emitted vertices )
	A = [PJpv]; [PJvv] = A; => PJ zemit;
	A = [PJdi]; A+; [PJdi] = A;
	A = [PJnv]; [PJvv] = A; => PJ zemit;
	A = [PJdi]; A+; [PJdi] = A;
	-> PJ zc stop;
    "PJ zc else11"
	( exactly one visible: ONE emitted vertex )
	A = rwf; A + [PJpv]; A = [A];
	C = [PJnv];
	? A = 0 -> PJ zc vset;
	C = [PJpv];
    "PJ zc vset"
	[PJvv] = C;
	=> PJ zemit;
	A = [PJdi]; A+; [PJdi] = A;
	-> PJ zc stop;
    "PJ zc bypass"
	A = [PJvr]; A + FSRXF; [PGFi] = A;
	A = [PJdi]; A + FSUX;  [PGFj] = A; => PGF mov;
	A = [PJvr]; A + FSRYF; [PGFi] = A;
	A = [PJdi]; A + FSUY;  [PGFj] = A; => PGF mov;
	A = [PJvr]; A + FSRZF; [PGFi] = A;
	A = [PJdi]; A + FSUZ;  [PGFj] = A; => PGF mov;
	A = [PJdi]; A+; [PJdi] = A;
    "PJ zc stop"
	[PJvr]+;
	A = [PJvr]; ? A '< [PJnrv] -> PJ zc for;
	A = [PJdi]; [PJvr2] = A;
	A = [PJdi]; A + A; [PJvr22] = A;
	end;

      ( PG trace : one scanline per iteration from [BXminy] to [BXmaxy].

	The row setup computes the shared factor (ipart[i] - x_centro + 1)
	ONCE and uses it three times.  _z stays WIDE for the reciprocal and
	NARROW in memory; _x and _y are narrowed and reloaded for u and v. )

"PG trace"
	A = PGSCRT; A + PGDOFF; [PGdi] = A; [PGval] = [SPtinta]; => PG store;
	A = PGSCRE; A + PGDOFF; [PGdi] = A; [PGval] = [SPescr];  => PG store;
	[SPi] = [BXminy];
    "PG tr row"
	( Fixed-slot form of the same exact row schedule.  Every arithmetic operator
	  still performs one PC64 operation followed by its binary64 spill, while ~:
	  remains the original float-variable narrowing point.  Avoiding the PGFi
	  stores, address loops, and tiny PGF calls matters across thousands of rows. )

	( the shared factor, WIDE )
	A = [SPi]; C = ipart; C + A; [FI] = [C]; [FA0] := [FI];
	[FA0] -: [fw plus 38];
	[FA0] +: [fw plus 36];
	[fw plus 502] = [FA0]; [fw plus 503] = [FA1];

	( _z = oz + hz*(i - y_centro) + vz*factor )
	[FA0] *: [fw plus 10];
	[fw plus 496] = [FA0]; [fw plus 497] = [FA1];
	[FI] = [SPi]; [FA0] := [FI];
	[FA0] -: [fw plus 40];
	[FA0] *: [fw plus 4];
	[FA0] +: [fw plus 496];
	[FA0] +: [fw plus 16];
	[fw plus 498] = [FA0]; [fw plus 499] = [FA1];
	~: [FA0]; [fw plus 30] = [FA0]; [fw plus 31] = [FA1];

	( k4 = 1/wide_z, then narrowed )
	[FA0] = [fw plus 36]; [FA1] = [fw plus 37];
	[FA0] /: [fw plus 498];
	~: [FA0]; [fw plus 24] = [FA0]; [fw plus 25] = [FA1];

	( _x = ox + hx*(i - y_centro) + vx*factor )
	[FA0] = [fw plus 502]; [FA1] = [fw plus 503];
	[FA0] *: [fw plus 6];
	[fw plus 496] = [FA0]; [fw plus 497] = [FA1];
	[FI] = [SPi]; [FA0] := [FI];
	[FA0] -: [fw plus 40];
	[FA0] *: [fw];
	[FA0] +: [fw plus 496];
	[FA0] +: [fw plus 12];
	~: [FA0]; [fw plus 26] = [FA0]; [fw plus 27] = [FA1];

	( _y = oy + hy*(i - y_centro) + vy*factor )
	[FA0] = [fw plus 502]; [FA1] = [fw plus 503];
	[FA0] *: [fw plus 8];
	[fw plus 496] = [FA0]; [fw plus 497] = [FA1];
	[FI] = [SPi]; [FA0] := [FI];
	[FA0] -: [fw plus 40];
	[FA0] *: [fw plus 2];
	[FA0] +: [fw plus 496];
	[FA0] +: [fw plus 14];
	~: [FA0]; [fw plus 28] = [FA0]; [fw plus 29] = [FA1];

	( u and v from the reloaded narrow x and y )
	[FA0] = [fw plus 26]; [FA1] = [fw plus 27];
	[FA0] *: [fw plus 32];
	[FA0] *: [fw plus 24];
	[FI] =: [FA0]; [SPu] = [FI];
	[FA0] = [fw plus 28]; [FA1] = [fw plus 29];
	[FA0] *: [fw plus 34];
	[FA0] *: [fw plus 24];
	[FI] =: [FA0]; [SPv] = [FI];

	A = [SPi]; C = fpart; C + A; D = [C];
	A = [SPi]; C = ipart; C + A; E = [C];
	D - E; [SPsec] = D;
	C = [SPi]; => PG riga; C + E; C & 65535; [SPdi] = C;
	=> PG scanline;

	( row_end: halfscan writes rows i and i+1, then both paths increment i )
	A = [SPhalf]; A & 1;
	? A = 0 -> PG tr single;
	[SPi]+;
	A = [BXmaxy]; ? A < [SPi] -> PG tr end;
	=> PG halfscan;
    "PG tr single"
	[SPi]+;
	A = [BXmaxy]; ? A < [SPi] -> PG tr end;
	-> PG tr row;
    "PG tr end"
	end;

"PG polymap"
	( Preprojected terrain has already supplied the three exact rotated and
	  projected vertices, proved them all in front of the near plane, and
	  computed their screen box.  Enter with the true triangle counts: the
	  generic path's slot-3 copies and zload are observably dead here. )
	[PJgate] = 0;
	A = [PJpreproject]; ? A != 0 -> PG pm preprojected terrain;
	( nv == 3 duplicates the LAST vertex into slot 3, and it does so
	  in the CALLER's arrays - polymap really does write x[3].  The
	  caller zeroes slot 3 before the call; polymap then overwrites
	  it from slot 2, and the order matters. )
	A = [PJnrv]; ? A != 3 -> PG pm r;
-> PG pm duplicate input generic;
    "PG pm duplicate input generic"
	A = FSINX; A + 2; [PGFi] = A; A = FSINX; A + 3; [PGFj] = A; => PGF mov;
	A = FSINY; A + 2; [PGFi] = A; A = FSINY; A + 3; [PGFj] = A; => PGF mov;
	A = FSINZ; A + 2; [PGFi] = A; A = FSINZ; A + 3; [PGFj] = A; => PGF mov;
    "PG pm r"
	[PJmode] = 1;
	[PJdx] = [PJnrv];
	A = [PJdx]; ? A != 3 -> PG pm rotate four;
	( A triangle's fourth input is the bitwise duplicate of vertex 2 above.
	  Rotate the three unique inputs, then duplicate the resulting float slots
	  and visibility byte.  This leaves the four-entry near-plane walk,
	  closing edge, projection and texture basis exactly as before. )
	[PJnrv] = 3;
	=> PJ rotate;
-> PG pm duplicate rotated generic;
    "PG pm duplicate rotated generic"
	A = FSRXF; A + 2; [PGFi] = A; A = FSRXF; A + 3; [PGFj] = A; => PGF mov;
	A = FSRYF; A + 2; [PGFi] = A; A = FSRYF; A + 3; [PGFj] = A; => PGF mov;
	A = FSRZF; A + 2; [PGFi] = A; A = FSRZF; A + 3; [PGFj] = A; => PGF mov;
	A = rwf; A + 2; C = [A]; A = rwf; A + 3; [A] = C;
	A = [PJdoflag]; A + C; [PJdoflag] = A;
	"PG pm duplicate rotated done"
	A = 4; [PJvr] = A; [PJnrv] = A;
	-> PG pm rotated;
    "PG pm rotate four"
	[PJnrv] = 4;
	=> PJ rotate;
    "PG pm preloaded"
	A = 4; [PJvr] = A; [PJnrv] = A;
    "PG pm rotated"
	[PJgate] = 1;
	A = [PJdoflag]; ? A = 0 -> PG pm out;

	A = [PJdoflag];
	? A != 4 -> PG pm clip;
	=> PJ zload;

    "PG pm clip"

	[PJgate] = 2;

    "PG pm preprojected terrain"
	[PJmode] = 1; [PJdx] = 3; [PJnrv] = 3; [PJvr] = 3;

    "PG pm projected"
	[PJpreproject] = 0;

	[PJgate] = 7;
	A = [PJminx]; ? A > PGUBX -> PG pm out;
	A = [PJmaxx]; ? A < PGLBX -> PG pm out;
	A = [BXminy]; ? A > PGUBY -> PG pm out;
	A = [BXmaxy]; ? A < PGLBY -> PG pm out;
	A = [BXminy]; ? A >= PGLBY -> PG pm s1;

    "PG pm s1"
	A = [BXmaxy]; ? A <= PGUBY -> PG pm s2;
	[BXmaxy] = PGUBY;
    "PG pm s2"
	[PJgate] = 8;
	A = [BXminy]; ? A > [BXmaxy] -> PG pm out;

	( The texture basis depends only on the original rotated vertices.  Build
	  it after projection's conservative screen rejection so wholly off-screen
	  polygons do not pay for nine gradient crosses and antialias expansion.
	  Preprojected terrain can go one step further: construct its exact integer
	  edges first and skip the basis and row setup when every span is empty.  The
	  two visible scratch bytes that trace always writes are still preserved. )
	A = [SPterrain]; ? A = 0 -> PG pm basis;
	A = [SPhalf]; ? A = 0 -> PG pm terrain edges;
	[SPterrain] = 0; -> PG pm basis;
    "PG pm terrain edges"
	[EWvr22] = [PJvr22];
	[EWminy] = [BXminy];

	[SPi] = [BXminy];
    "PG pm terrain span"
	A = [SPi]; C = fpart; C + A; D = [C];
	C = ipart; C + A; E = [C];
	D - E; [SPsec] = D; ? D > 0 -> PG pm basis;
	[SPi]+; A = [SPi]; ? A '<= [BXmaxy] -> PG pm terrain span;
	[SPsrc] = 1;
	A = PGSCRT; A + PGDOFF; [PGdi] = A; [PGval] = [SPtinta]; => PG store;

	[PJnrv] = 4;
