      ( *** grmain - the Wave 7b driver, namespace GR ***

	Tests round_hill (the float painter).  Wipes p_surfacemap, calls
	round_hill(cx=100,cz=100,r=50,h=30,hmax=0,allowcanyons=1), then
	writes the 40000-byte map + 4 counter units to gr-out.bin.

	NOTE: use constants (RPSM, SADPR) directly as workspace addresses.
	`A = nw` from this compilation unit gives the wrong value; the
	fbmem routines that use `A = nw` work because they're in fbmem's
	own compilation unit.						)

"libraries"

	fp/fpabi;
	fp/fpctl;
	fp/fpx87;
	fp/fpconv;
	fbmem;
	brtl;
	mul64frag;
	surng;
	suseed;
	grnd;

"directors"
	program name = { grmain };
	unit = 32;

"variables"
	groutname	= { gr-out.bin };
	GRpki	= 0;
	GRpkb	= 0;
	GRtmpa	= 0;

"programme"

	=> FEnter;
	=> SU ledger reset;
	=> GR float init;

	( wipe p_surfacemap via MEM put byte )
	[GRptr] = 0;
"GR wipe sm"
	C = 0; [MBval] = C;
	A = RPSM; A + [GRptr]; [MBptr] = A;
	=> MEM put byte;
	[GRptr]+;
	? [GRptr] '< 40000 -> GR wipe sm;

	( round_hill: cx=100, cz=100, r=50, h=30.0, hmax=0, allowcanyons=1 )
	[SUcxi] = 100;
	[SUcyi] = 100;
	[SUia] = 50;
	[GRfht] = 41F00000h;	( 30.0f )
	[GRfhmt] = 0;		( 0.0f )
	[GRfcanyon] = 1;
	=> GR round hill;

	( copy p_surfacemap to SADPR, one byte per unit )
	[GRptr] = 0;
"GR cp map"
	A = RPSM; A + [GRptr]; [MBptr] = A;
	=> MEM get byte;
	A = SADPR; A + [GRptr];
	C = [MBval]; C & 0FFh; [A] = C;
	[GRptr]+;
	? [GRptr] '< 40000 -> GR cp map;

	( append draw counts at SADPR + 40000 )
	A = SADPR; A + 40000;
	[A] = [SUfn]; A+;
	[A] = [SUbn]; A+;
	[A] = [SUfh]; A+;
	[A] = [SUbh];

	( write 40004 units to file )
	[File Name] = groutname;
	[File Position] = 0;
	[File Command] = WRITE;
	[Block Pointer] = SADPR;
	[Block Size] = 40004;
	[Block Size] * BYTESPERUNIT;
	isocall;

	[File Name] = groutname;
	[File Position] = 0;
	[File Command] = SET SIZE;
	[File Size] = 40004;
	[File Size] * BYTESPERUNIT;
	isocall;

	=> FLeave;
	end;
