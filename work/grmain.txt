      ( *** grmain - Wave 7b multi-painter lino driver ***

	Exercises: GR prologue + GR rockyground + GR plains noise.
	These cover the integer painters: rockyground (fill + smoothterrain
	+ level adjust), plains noise (the ADD-vs-ADD fast_random path).
	Output: 40000-byte smap (one byte per unit) + 4 counter units.

	Build: powershell -NoProfile -ExecutionPolicy Bypass
	       -File C:\programmieren\linoleum\lino_build.ps1
	       -Src C:\programmieren\linoleum\work\grmain.txt	)

"libraries"

	fp/fpabi;
	fp/fpctl;
	fp/fpx87;
	fp/fpconv;

	fbmem;
	brtl;
	mul64frag;

	surng;
	suseed;
	grnd;

"directors"

	program name = { grmain };
	unit = 32;

"variables"

	groutname	= { gr-out.bin };

"programme"

	=> FEnter;
	=> SU ledger reset;
	=> GR float init;

	( wipe p_surfacemap )
	[GRptr] = 0;
"GR wipe sm"
	C = 0; [MBval] = C;
	A = RPSM; A + [GRptr]; [MBptr] = A;
	=> MEM put byte;
	[GRptr]+;
	? [GRptr] '< 40000 -> GR wipe sm;

	( wipe objectschart )
	[GRptr] = 0;
"GR wipe ob"
	C = 0; [MBval] = C;
	A = ROBJ; A + [GRptr]; [MBptr] = A;
	=> MEM put byte;
	[GRptr]+;
	? [GRptr] '< 40000 -> GR wipe ob;

	( test case: gseed=123456, type=1, sctype=0, albedo=20, lat=30 )
	[GRgseed]  = 123456;
	[GRiptype] = 1;
	[GRsctype] = 0;
	[GRalbedo] = 20;
	[GRlat]    = 30;

	( 1. prologue — 12 brtl draws for type 1 )
	=> GR prologue;

	( 2. re-seed for painters )
	A = [GRgseed];
	=> SU fast srand;
	A = [GRgseed]; A & 0FFFFh;
	=> SU srand;

	( 3. rockyground(25, 4, 0) )
	[GRrough] = 25;
	[GRround] = 4;
	[GRlevel] = 0;
	=> GR rockyground;

	( 4. plains noise — p_surfacemap[ptr] += fast_random(3) )
	=> GR plains noise;

	( copy p_surfacemap to SADPR )
	[GRptr] = 0;
"GR cp map"
	A = RPSM; A + [GRptr]; [MBptr] = A;
	=> MEM get byte;
	A = SADPR; A + [GRptr];
	C = [MBval]; C & 0FFh; [A] = C;
	[GRptr]+;
	? [GRptr] '< 40000 -> GR cp map;

	( append draw counts at SADPR + 40000 )
	A = SADPR; A + 40000;
	[A] = [SUfn]; A+;
	[A] = [SUbn]; A+;
	[A] = [SUfh]; A+;
	[A] = [SUbh];

	( write 40004 units = 160016 bytes )
	[File Name] = groutname;
	[File Position] = 0;
	[File Command] = WRITE;
	[Block Pointer] = SADPR;
	[Block Size] = 40004;
	[Block Size] * BYTESPERUNIT;
	isocall;

	[File Name] = groutname;
	[File Position] = 0;
	[File Command] = SET SIZE;
	[File Size] = 40004;
	[File Size] * BYTESPERUNIT;
	isocall;

	=> FLeave;
	end;
