( battery 4 - THE CANARY THAT CAN FAIL                                  )
( ==================================================================== )

      ( FBDUMP kind 6 v1 was 18 units in which both the "expected" and
	the "actual" field were 0xA5A5A5A5, written by construction on
	BOTH sides.  The record carried no information about whether
	either canary mechanism worked: a clean run and a stubbed check
	produced a BIT-IDENTICAL dump, so "lino CANARY == the reference"
	passed for a build with no canary in it.

	v2 is 4 units per pad, 11 pads, 44 units, and NO LITERALS.  Every
	unit is either read back out of the workspace or produced by the
	walker, and the grader knows all four independently from the
	layout:

	  0  clean read : nw[padbase + slot] after poisoning
	                  -> PGUARD if slot < 8, else PALLOW
	  1  dirty read : the same address after storing WITNESS(i)
	                  -> WITNESS(i)
	  2  fired      : the PAD index + 1 the walker reported
	                  -> i + 1
	  3  at         : the nw offset of the first violation
	                  -> padbase(i) + slot

	  WITNESS(i) = 0xC0DE0000 + i     region-specific, so copy-pasting
	                                  one answer fails
	  slot(i)    = (7i + 1) mod 12    sweeps the pad

	Four failure modes are separated: no poison (unit 0), a constant
	"actual" (unit 1), a stubbed comparison (unit 2), the wrong
	address or the wrong pad set (unit 3).

	WHY mod 12 AND NOT mod 16.  Units +12..+15 of every SUB zone are
	the region's segment offsets 0..3 and are therefore an ALLOWANCE,
	not a guard - a violation probe there cannot fire BY DESIGN, and a
	probe that expects it to fire would be asserting the guard model
	is wrong.  slot = (7i+1) mod 12 over i = 0..10 yields
	{1,8,3,10,5,0,7,2,9,4,11}: eleven distinct offsets out of the
	twelve guarded ones, no two pads alike, and none of them an
	allowance unit.  The +1 is not cosmetic: with slot(0) = 0 and
	padbase(0) = 0 the "at" field would be 0 in both the clean and the
	stubbed case. )

"FB canary battery"
	( the clean pass FIRST: a canary that always fires cannot pass )
	=> MEM poison pads;
	=> MEM check pads;
	[FBclnf] = [MCfired];
	[FBclnn] = [MCn];
	[FBclne] = [MCexp];

	A = canary; [FBcp] = A;
	[FBi] = 0;
    "FB cb pad"
	=> MEM poison pads;
	A = [FBi]; A '* 7; A + 1; A '% 12; [FBslot] = A;
	[MEMi] = [FBi]; => MEM pad base;
	A = [MEMtmp]; A + [FBslot]; [FBoff] = A;
	A = nw; A + [FBoff]; [FBap] = A;

	A = [FBap]; C = [A];
	D = [FBcp]; [D plus 0] = C;

	( WITNESS(i) = 0xC0DE0000 + i + (clean read AND 15).
	  The last term is what makes the DIRTY READ a read.  With a plain
	  0xC0DE0000 + i the store value and the expected value are the
	  same literal on both sides, so replacing the read-back with that
	  literal produces a BIT-IDENTICAL dump - which is exactly the
	  defect kind 6 v1 had, reproduced inside its own replacement.
	  Measured: with the plain literal, sabotage S-CAN-CONSTACTUAL
	  moved nothing at all.  Folding in four bits of the poison the
	  walker itself wrote makes the witness role-dependent - 5 in a
	  TAIL zone, 10 in a SUB zone - and a literal can no longer stand
	  in for it.  The grader still knows the value from the layout
	  alone, because it knows which zone the slot falls in. )
	C = 3235774464; C + [FBi];
	D = [FBcp]; D = [D plus 0]; D & 15; C + D;
	A = [FBap]; [A] = C;

	=> MEM check pads;

	A = [FBap]; C = [A];
	D = [FBcp];
	[D plus 1] = C;
	[D plus 2] = [MCfired];
	[D plus 3] = [MCat];

	A = [FBcp]; A + 4; [FBcp] = A;
	[FBi]+;
	A = [FBi]; ? A '< NPAD -> FB cb pad;
	end;

	  cirrus naive  = ROBJ + ((py + px) > 1)           32-bit
	         masked = SOBJ + (((((py+px) AND 65535) > 1) + 4) AND 65535)

	deltas therefore differ - 65,536 for spot, 32,768 for cirrus.  A
	single "mask the final index" helper would halve cirrus's error

	catchable sabotage rather than a no-op.

	lie between base - 8 and base + size - 1, the union of the region
	and its SUB zone, because a masked offset of 0..3 lands on the
	segment origin, which is SUB+4..+7.                              )

      ( digimap2 is loaded from disk in the game (load_digimap2,
	NOCTIS-0.CPP:6094).  There is no font file here, so the fixture
	pins it arithmetically - one line, reproducible by any
	implementation:

	    digimap2[j] = (j * 2654435761) mod 2^32,  j = 0..2339

	2,340 = 65 glyphs x 36 rows, which is what (digit-32)*36 + 35
	reaches for digit = 96.  Knuth's multiplicative constant is used
	because roughly half its bits are set at every j, so both branches
	of the "digimap2[n+d] AND pp[m]" test are exercised on every row.
	pp[m] is 1 << m, NOCTIS.CPP:581-588. )

"FB font build"
	A = digimap2; C = 0; B = 0; B + 2340;
    "FB fnt loop"
	D = C; D '* 2654435761;
	[A] = D; A+; C+;
	B ^ FB fnt loop;
	end;

"FB psm clear"

    "FB psm loop"

	end;

      ( FB digit at : NOCTIS.CPP:614-628, the texture-building half.
	The polymap call that follows it in the original is a renderer and
	is out of Wave 5's scope; everything that writes memory is here.

	    if (digit>32 && digit<=96) {
	        txtr = p_surfacemap;
	        d = (digit - 32) * 36;
	        for (n = 0; n < 36; n++) {
	            i = 256 * n - 5;
	            txtr[i-1] = 0;
	            for (m = 0; m < 32; m++) {
	                if (digimap2[n+d] & pp[m]) txtr[i] = pixel_color;
	                else                       txtr[i] = 0;
	                i++; }
	            if (shader) pixel_color--; }
	        txtr[256 * 36 - 6] = 0;

	THE LOOP STARTS AT n = 0.  niv-lr changed it to n = 1 after
	Valgrind complained (noctis.cpp:643-646, "I just blindly changed
	this loop to start at one.  It probably breaks things...") and it
	does: it silently drops the top scanline of every cockpit glyph.
	At n = 0 the writes at i-1 = -6 and i = -5..-1 land six units
	BELOW p_surfacemap, in pad 5's SUB zone, which is exactly why that
	zone is an allowance and not a guard. )

"FB digit at"
	A = [DGdigit]; ? A '<= 32 -> FB dg done;
	A = [DGdigit]; ? A '> 96 -> FB dg done;
	C = [DGcolor]; C & 255; C '% 64; [DGpix] = C;
	C = [DGcolor]; C & 255; C > 6; C < 6; [DGmapb] = C;
	A = [DGdigit]; A - 32; A '* 36; [DGd] = A;
	[DGn] = 0;
    "FB dg n"
	A = [DGn]; A '* 256; A - 5; [DGi] = A;
	( txtr[i-1] = 0 - "evita aliasing a fine scanline" )
	A = [DGi]; A - 1; [DGval] = 0; => FB txtr put;
	A = [DGn]; A + [DGd]; A + digimap2; [DGrow] = [A];
	[DGm] = 0;
    "FB dg m"
	C = [DGrow]; B = 1; D = [DGm]; B < D; C & B;
	? C != 0 -> FB dg on;
	[DGval] = 0; -> FB dg st;
    "FB dg on"
	[DGval] = [DGpix];
    "FB dg st"
	A = [DGi]; => FB txtr put;
	[DGi]+;
	[DGm]+;
	A = [DGm]; ? A '< 32 -> FB dg m;
	A = [DGshader]; ? A = 0 -> FB dg nosh;
	[DGpix]-;
    "FB dg nosh"
	[DGn]+;
	A = [DGn]; ? A '< 36 -> FB dg n;
	( txtr[256*36 - 6] = 0 - "evita aliasing a fine matrice" )
	A = 9210; [DGval] = 0; => FB txtr put;
    "FB dg done"
	end;

      ( FB txtr put : txtr[A] = [DGval], txtr = p_surfacemap.

	    mov cx, QUADWORDS

	    add dword ptr [di], mask*0x01010101

( ==================================================================== )

      ( FB expand : adaptor -> fb through pal, unrolled four ways.

	IT IS READ-ONLY ON THE INDEX PAGE.  LINOBUF section 5.4 directed
	that Noctis's background colour cycle be FUSED in here, for a
	measured 22%.  BUFFERMODEL section 7 overrules it: the loop is
	real (NOCTIS.CPP:3779-3784) but it cycles s_background - a
