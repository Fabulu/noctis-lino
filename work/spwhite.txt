      ( *** spwhite - whiteglobe() and whitesun(): one body, three
	    parameterised differences ***

	NOCTIS-0.CPP:3298-3389 and :3535-3608.  The two functions are the
	same forty lines twice, and the port writes them once with three
	parameters, because the interesting question is exactly which
	three things differ:

	  STEP      whiteglobe walks x and y by 2 and accumulates ya by
	            2.4; whitesun walks by 1 and accumulates ya by 1.2.
	            The 1.2 : 1 ratio is the 320x200-on-4:3 pixel aspect
	            and it is the same 1.2 in both, applied at the two
	            different steps.
	  SHAPE     whiteglobe stores a 2x2 block - word at [di+4] and
	            word at [di+0x144] - and whitesun stores one byte.
	  xsun_onscreen  whitesun writes it BEFORE the rx and ry reject
	            tests, so a sun that is off screen still leaves its
	            projected column behind for the surface renderer to
	            read.  Moving that write below the tests is a real
	            defect with no visible effect in this file's page and
	            a visible one two subsystems away, which is why it is
	            emitted as a number.

	niv-lr parameterised globe() for arbitrary resolution and left
	glowing_globe, white_globe and white_sun hard-coded, so its four
	sphere renderers disagree about the clip rectangle.  That is a
	KNOWN WRONG ANSWER in LR.  This file follows vanilla: the window
	here is "xx > 9 && xx < 313 && yy > 9 && yy < 190", which is not
	the 6/191/6/311 of globe() and not the 10/190/9/310 of
	glowinglobe(), and the three disagreeing rectangles are all
	correct.

	THE CENTRES ARE NOT CAST.  "center_x = rx + x_centro_f + 0.5" is a
	double and stays a double; the truncation happens later and
	separately at "(int)xx" and "(int)yy".  So this is the one sphere
	renderer where the +0.5 is doing the rounding, and dropping it
	moves the whole disc by half a pixel.  globe() chops with no
	+0.5 and project3d rounds with fistp: three different roundings
	across four functions.

	pix IS A SIGNED CHAR AND IT OVERFLOWS.  "pix += target[pixptr]"
	is a char += unsigned char, so at a destination byte of 200 and a
	pix of 0x3F the sum is 263, which wraps to 7 - and 7 is NOT
	greater than 0x3F, so the clamp does not fire and the pixel goes
	DARKER.  Typing pix unsigned gives 63 and the wrong picture.  The
	corpus pre-fills a page above 0x3F on purpose so that this fires.

	fgm_factor REACHES NEGATIVE VALUES.  NOCTIS.CPP:2557 passes
	-0.15.  Then fgm is negative, shade_ext = mag - fgm is LARGER than
	mag, and "if (zz > fgmsq)" compares against a positive square, so
	the inner disc still exists but the ramp is gentler.  Clamping
	fgm_factor to zero is a plausible-looking edit that changes every
	pixel of the corona.                                              )

"variables"

	WHmag	= 0;		( binary32 pattern, mag_factor  )
	WHfgm	= 0;		( binary32 pattern, fgm_factor  )
	WHshape	= 0;		( 1 = one byte, 2 = a 2x2 block )
	WHsun	= 0;		( 1 = whitesun's early write    )
	WHdstreg = 0;

	WHok	= 0;
	WHxsun	= 0;		( xsun_onscreen, as a binary64 pattern pair )
	WHxsunl	= 0;
	WHxsunh	= 0;
	WHcxl	= 0;	WHcxh	= 0;
	WHcyl	= 0;	WHcyh	= 0;

	WHiy	= 0;
	WHix	= 0;
	WHptr	= 0;
	WHpix	= 0;
	WHtex	= 0;
	WHn	= 0;		( pixels stored          )
	WHclamp	= 0;		( stores that hit the 0x3F clamp )
	WHwrap	= 0;		( stores where the signed char wrapped )
	WHrows	= 0;
	WHcols	= 0;

	WHi	= 0;

"programme"

( ==================================================================== )

      ( SP white : the whole routine.  xx/yy/zz arrive in SFXX/SFYY/SFZZ
	already reduced by dzat, exactly as in spglobe. )

"SP white"
	[WHok] = 0; [WHn] = 0; [WHclamp] = 0; [WHwrap] = 0;
	[WHrows] = 0; [WHcols] = 0;
	[WHxsunl] = 0; [WHxsunh] = 0;

	( the rotation nucleus - identical to globe's, and identical to
	  project3d's, which is what makes the centre a cross-check
	  rather than a restatement )
	[PGFi] = SFXX; => PGF a; [PGFi] = FSPCB; => PGF mul;
	[PGFi] = SFT0; => PGF sa;
	[PGFi] = SFZZ; => PGF a; [PGFi] = FSPSB; => PGF mul;
	[PGFi] = SFT0; => PGF add; [PGFi] = SFRX; => PGF sa;

	[PGFi] = SFZZ; => PGF a; [PGFi] = FSTCB; => PGF mul;
	[PGFi] = SFT0; => PGF sa;
	[PGFi] = SFXX; => PGF a; [PGFi] = FSTSB; => PGF mul;
	[PGFi] = SFT0; => PGF rsub; [PGFi] = SFZ2; => PGF sa;

	[PGFi] = SFZ2; => PGF a; [PGFi] = FSTCA; => PGF mul;
	[PGFi] = SFT0; => PGF sa;
	[PGFi] = SFYY; => PGF a; [PGFi] = FSTSA; => PGF mul;
	[PGFi] = SFT0; => PGF add; [PGFi] = SFRZ; => PGF sa;

	[PGFi] = SFYY; => PGF a; [PGFi] = FSPCA; => PGF mul;
	[PGFi] = SFT0; => PGF sa;
	[PGFi] = SFZ2; => PGF a; [PGFi] = FSPSA; => PGF mul;
	[PGFi] = SFT0; => PGF rsub; [PGFi] = SFRY; => PGF sa;

	[PGFt] = D001L; [PGFu] = D001H; => PGF ldf64;
	[FB0] = [FA0]; [FB1] = [FA1];
	[PGFi] = SFRZ; => PGF a;
	=> FCmp;
	A = [FI]; ? A < 0 -> SP wh out;

	[PGFt] = [WHmag]; => PGF ldf32;
	[PGFi] = SFRZ; => PGF quo;
	=> F32Narrow;
	[PGFi] = SFMAG; => PGF sa;

	( 2.99 ceiling, then a 0.01 FLOOR - not the 0.001 of the other two )
	[PGFt] = D299L; [PGFu] = D299H; => PGF ldf64;
	[FB0] = [FA0]; [FB1] = [FA1];
	[PGFi] = SFMAG; => PGF a; => FCmp;
	A = [FI]; ? A != 1 -> SP wh lo;
	[PGFt] = F299; => PGF ldf32; [PGFi] = SFMAG; => PGF sa;
    "SP wh lo"
	[PGFt] = D01L; [PGFu] = D01H; => PGF ldf64;
	[FB0] = [FA0]; [FB1] = [FA1];
	[PGFi] = SFMAG; => PGF a; => FCmp;
	A = [FI]; ? A >= 0 -> SP wh rxy;
	[PGFt] = F01; => PGF ldf32; [PGFi] = SFMAG; => PGF sa;

    "SP wh rxy"
	[PGFi] = SFRX; => PGF a; [PGFi] = SFRZ; => PGF quo;
	[PGFi] = SFRX; => PGF sa;
	[PGFi] = SFRY; => PGF a; [PGFi] = SFRZ; => PGF quo;
	[PGFi] = SFRY; => PGF sa;

	( whitesun writes xsun_onscreen HERE, before either reject test )
	A = [WHsun]; ? A = 0 -> SP wh rej;
	[PGFt] = F32XC; => PGF ldf32; [FB0] = [FA0]; [FB1] = [FA1];
	[PGFi] = SFRX; => PGF a; => FAdd;
	[WHxsunl] = [FA0]; [WHxsunh] = [FA1];

    "SP wh rej"
	[PGFt] = D460L; [PGFu] = D460H; => PGF ldf64;
	[FB0] = [FA0]; [FB1] = [FA1];
	[PGFi] = SFRX; => PGF a; => FAbs; => FCmp;
	A = [FI]; ? A = 1 -> SP wh out;
	[PGFt] = D400L; [PGFu] = D400H; => PGF ldf64;
	[FB0] = [FA0]; [FB1] = [FA1];
	[PGFi] = SFRY; => PGF a; => FAbs; => FCmp;
	A = [FI]; ? A = 1 -> SP wh out;

	( center_x = rx + x_centro_f + 0.5 - a DOUBLE, never cast.
	  x_centro_f is a float and is widened exactly; the two adds are
	  left to right, which is what the C expression says. )
	[PGFt] = F32XC; => PGF ldf32; [PGFi] = SFT1; => PGF sa;
	[PGFt] = D05L; [PGFu] = D05H; => PGF ldf64; [PGFi] = SFT2; => PGF sa;
	[PGFi] = SFRX; => PGF a;
	[PGFi] = SFT1; => PGF add;
	[PGFi] = SFT2; => PGF add;
	[PGFi] = SFCX; => PGF sa;

	[PGFt] = F32YC; => PGF ldf32; [PGFi] = SFT1; => PGF sa;
	[PGFi] = SFRY; => PGF a;
	[PGFi] = SFT1; => PGF add;
	[PGFi] = SFT2; => PGF add;
	[PGFi] = SFCY; => PGF sa;

	( mag = mag_factor*100 + 1.5.  The 1.5 goes into the OPERAND
	  register with "SP fb64"; loading it with PGF ldf64 would
	  overwrite the product and compute 1.5 + 1.5. )
	[PGFt] = D100L; [PGFu] = D100H; => SP fb64;
	[PGFi] = SFMAG; => PGF a; => FMul;
	[PGFt] = D15L; [PGFu] = D15H; => SP fb64;
	=> FAdd;
	[PGFi] = SFMG; => PGF sa;

	( fgm = fgm_factor * mag ;  shade_ext = mag - fgm ;
	  if (shade_ext < 1) shade_ext = 1 ;  ise = 0x3F / shade_ext )
	[PGFt] = [WHfgm]; => PGF ldf32;
	[PGFi] = SFMG; => PGF mul;
	[PGFi] = SFFGM; => PGF sa;

	[PGFi] = SFMG; => PGF a; [PGFi] = SFFGM; => PGF sub;
	[PGFi] = SFSHE; => PGF sa;
	[PGFt] = D1L; [PGFu] = D1H; => PGF ldf64; [FB0] = [FA0]; [FB1] = [FA1];
	[PGFi] = SFSHE; => PGF a; => FCmp;
	A = [FI]; ? A >= 0 -> SP wh ise;
	[PGFt] = D1L; [PGFu] = D1H; => PGF ldf64; [PGFi] = SFSHE; => PGF sa;
    "SP wh ise"
	[PGFt] = D63L; [PGFu] = D63H; => PGF ldf64;
	[PGFi] = SFSHE; => PGF quo;
	[PGFi] = SFISE; => PGF sa;

	[PGFi] = SFMG; => PGF a; [PGFi] = SFMG; => PGF mul;
	[PGFi] = SFMSQ; => PGF sa;
	[PGFi] = SFFGM; => PGF a; [PGFi] = SFFGM; => PGF mul;
	[PGFi] = SFFSQ; => PGF sa;

	( ya = -mag * 1.2 )
	[PGFt] = D12L; [PGFu] = D12H; => PGF ldf64; [FB0] = [FA0]; [FB1] = [FA1];
	[PGFi] = SFMG; => PGF a; => FMul; => FNeg;
	[PGFi] = SFYA; => PGF sa;

	( yb = center_y + mag ;  yy = center_y - mag )
	[PGFi] = SFCY; => PGF a; [PGFi] = SFMG; => PGF add; [PGFi] = SFYB; => PGF sa;
	[PGFi] = SFCY; => PGF a; [PGFi] = SFMG; => PGF sub; [PGFi] = SFYP; => PGF sa;

	( the two steps, chosen once )
	A = [WHshape]; ? A = 1 -> SP wh s1;
	[PGFt] = D2L;  [PGFu] = D2H;  => PGF ldf64; [PGFi] = SFV0; => PGF sa;
	[PGFt] = D24L; [PGFu] = D24H; => PGF ldf64; [PGFi] = SFV1; => PGF sa;
	-> SP wh yloop;
    "SP wh s1"
	[PGFt] = D1L;  [PGFu] = D1H;  => PGF ldf64; [PGFi] = SFV0; => PGF sa;
	[PGFt] = D12L; [PGFu] = D12H; => PGF ldf64; [PGFi] = SFV1; => PGF sa;

    "SP wh yloop"
	[PGFi] = SFYB; => PGF a; [FB0] = [FA0]; [FB1] = [FA1];
	[PGFi] = SFYP; => PGF a; => FCmp;
	A = [FI]; ? A >= 0 -> SP wh ydone;
	[WHrows]+;

	[PGFi] = SFMG; => PGF a; => FNeg; [PGFi] = SFXA; => PGF sa;
	[PGFi] = SFCX; => PGF a; [PGFi] = SFMG; => PGF add; [PGFi] = SFXB; => PGF sa;
	[PGFi] = SFCX; => PGF a; [PGFi] = SFMG; => PGF sub; [PGFi] = SFXP; => PGF sa;

    "SP wh xloop"
	[PGFi] = SFXB; => PGF a; [FB0] = [FA0]; [FB1] = [FA1];
	[PGFi] = SFXP; => PGF a; => FCmp;
	A = [FI]; ? A >= 0 -> SP wh xdone;
	[WHcols]+;

	( if (xx > 9 && xx < 313 && yy > 9 && yy < 190) )
	[PGFt] = D9L; [PGFu] = D9H; => PGF ldf64; [FB0] = [FA0]; [FB1] = [FA1];
	[PGFi] = SFXP; => PGF a; => FCmp;
	A = [FI]; ? A != 1 -> SP wh xnext;
	[PGFt] = D313L; [PGFu] = D313H; => PGF ldf64; [FB0] = [FA0]; [FB1] = [FA1];
	[PGFi] = SFXP; => PGF a; => FCmp;
	A = [FI]; ? A >= 0 -> SP wh xnext;
	[PGFt] = D9L; [PGFu] = D9H; => PGF ldf64; [FB0] = [FA0]; [FB1] = [FA1];
	[PGFi] = SFYP; => PGF a; => FCmp;
	A = [FI]; ? A != 1 -> SP wh xnext;
	[PGFt] = D190L; [PGFu] = D190H; => PGF ldf64; [FB0] = [FA0]; [FB1] = [FA1];
	[PGFi] = SFYP; => PGF a; => FCmp;
	A = [FI]; ? A >= 0 -> SP wh xnext;

	( zz = xa*xa + ya*ya )
	[PGFi] = SFXA; => PGF a; [PGFi] = SFXA; => PGF mul; [PGFi] = SFT1; => PGF sa;
	[PGFi] = SFYA; => PGF a; [PGFi] = SFYA; => PGF mul;
	[PGFi] = SFT1; => PGF add; [PGFi] = SFZW; => PGF sa;

	[PGFi] = SFMSQ; => PGF a; [FB0] = [FA0]; [FB1] = [FA1];
	[PGFi] = SFZW; => PGF a; => FCmp;
	A = [FI]; ? A >= 0 -> SP wh xnext;

	( pix = 0x3F  or  0x3F - (sqrt(zz)-fgm)*ise )
	[PGFi] = SFFSQ; => PGF a; [FB0] = [FA0]; [FB1] = [FA1];
	[PGFi] = SFZW; => PGF a; => FCmp;
	A = [FI]; ? A != 1 -> SP wh full;
	[PGFi] = SFZW; => PGF a; => FSqrt;
	[PGFi] = SFFGM; => PGF sub;
	[PGFi] = SFISE; => PGF mul;
	[PGFi] = SFT2; => PGF sa;
	[PGFt] = D63L; [PGFu] = D63H; => PGF ldf64;
	[PGFi] = SFT2; => PGF sub;
	=> FToIntChop;
	C = [FI]; => SP sx8;
	[WHpix] = C;
	-> SP wh ptr;
    "SP wh full"
	[WHpix] = 63;

    "SP wh ptr"
	( pixptr = riga[(int)yy] + (int)xx - both casts CHOP )
	[PGFi] = SFYP; => PGF a; => FToIntChop; [WHiy] = [FI];
	[PGFi] = SFXP; => PGF a; => FToIntChop; [WHix] = [FI];
	A = [WHiy]; A + A; A & 65535; [GBt] = A;
	=> SP riga at;
	C + [WHix]; C & 65535; [WHptr] = C;

	( pix += target[pixptr] - a CHAR add, and it wraps )
	[SPreg] = [WHdstreg];
	A = [WHptr]; A + 4; [SPoff] = A;
	=> SP get;
	[WHtex] = [SPval];
	A = [WHpix]; A + [WHtex]; [GBt] = A;
	C = [GBt]; => SP sx8;
	[WHpix] = C;
	A = [GBt]; ? A = [WHpix] -> SP wh nowrap;
	[WHwrap]+;
    "SP wh nowrap"

	A = [WHpix]; ? A <= 63 -> SP wh keep;
	[WHpix] = 63;
	[WHclamp]+;
    "SP wh keep"
	C = [WHpix]; C & 255; [SPval] = C;

	( the store shape - the whole difference between the two )
	A = [WHptr]; A + 4; [SPoff] = A; => SP put; [WHn]+;
	A = [WHshape]; ? A = 1 -> SP wh xnext;
	A = [WHptr]; A + 5; [SPoff] = A; => SP put; [WHn]+;
	A = [WHptr]; A + 324; [SPoff] = A; => SP put; [WHn]+;
	A = [WHptr]; A + 325; [SPoff] = A; => SP put; [WHn]+;

    "SP wh xnext"
	[PGFi] = SFXA; => PGF a; [PGFi] = SFV0; => PGF add; [PGFi] = SFXA; => PGF sa;
	[PGFi] = SFXP; => PGF a; [PGFi] = SFV0; => PGF add; [PGFi] = SFXP; => PGF sa;
	-> SP wh xloop;

    "SP wh xdone"
	[PGFi] = SFYA; => PGF a; [PGFi] = SFV1; => PGF add; [PGFi] = SFYA; => PGF sa;
	[PGFi] = SFYP; => PGF a; [PGFi] = SFV0; => PGF add; [PGFi] = SFYP; => PGF sa;
	-> SP wh yloop;

    "SP wh ydone"
	[PGFi] = SFCX; => PGF a; [WHcxl] = [FA0]; [WHcxh] = [FA1];
	[PGFi] = SFCY; => PGF a; [WHcyl] = [FA0]; [WHcyh] = [FA1];
	[WHok] = 1;
    "SP wh out"
	end;

"SP white emit"
	A = sphdr;
	[A plus 0]  = [WHok];
	[A plus 1]  = [WHn];
	[A plus 2]  = [WHclamp];
	[A plus 3]  = [WHwrap];
	[A plus 4]  = [WHrows];
	[A plus 5]  = [WHcols];
	[A plus 6]  = [WHcxl];
	[A plus 7]  = [WHcxh];
	[A plus 8]  = [WHcyl];
	[A plus 9]  = [WHcyh];
	[A plus 10] = [WHxsunl];
	[A plus 11] = [WHxsunh];
	[A plus 12] = [WHshape];
	[A plus 13] = [WHsun];
	[A plus 14] = [WHmag];
	[A plus 15] = [WHfgm];
	[SKind] = K3B; [STag] = 3;
	[SW] = 16; [SH] = 1;
	[SBptr] = sphdr; [SBcnt] = 16;
	=> SDMP rec;
	end;
