      ( *** vhflare - Stardrifter and surface lens flares ***

	Alogena calls lens_flares_for unconditionally at clean startup when
	ilightv=1 and elight=0.  This is its primary 90-spoke path: project the
	fixture point, derive k=-step/rz for step=-50000, and draw through the
	original flares=1 low-six-bit additive rule.  Its condition is zero;
	planetary suns use the center-pixel condition instead. )

"constants"

	VHFF1 = 1065353216;		( 1.0f )
	VHFF15 = 1069547520;		( 1.5f )
	VHFF3 = 1077936128;		( 3.0f )
	VHFF23 = 1059760811;		( 2/3f )
	VHFM10 = 3240099840;		( -10.0f )
	VHFP10 = 1092616192;		( +10.0f )

"variables"

	VHFcx = 0; VHFcy = 0; VHFk = 0; VHFl = 0; VHFu = 0; VHFang = 0;
	VHFadd = 2; VHFpx = 0; VHFpy = 0; VHFpz = 0;
	VHFdist = 0; VHFdist0 = 0; VHFdist1 = 0; VHFray = 0; VHFstep = 0;
	VHFok = 0;
	VHFghost = 0; VHFgdx = 0; VHFgdy = 0; VHFgx = 0; VHFgy = 0; VHFgr = 0;
	VHFdx = 0; VHFdy = 0;
	VHFx0 = 0; VHFy0 = 0; VHFx1 = 0; VHFy1 = 0;
	VHFldx = 0; VHFldy = 0; VHFlsx = 0; VHFlsy = 0;
	VHFerr = 0; VHFe2 = 0;

"programme"

"VH halogen flare"
	[VHFadd] = 2;
	[PGFi] = FSINX; [PGFt] = VHFM10; => PGF setf32;
	[PGFi] = FSINY; [PGFt] = VHFP10; => PGF setf32;
	[PGFi] = FSINZ; [PGFt] = VHFM10; => PGF setf32;
	=> PG getcoords;
	A = rwf; A = [A]; ? A = 0 -> VHF done;
	A = [GCx]; A+; [VHFcx] = A;
	A = [GCy]; A+; [VHFcy] = A;
	A = [VHFcx]; ? A < 10 -> VHF done; ? A > 310 -> VHF done;
	A = [VHFcy]; ? A < 10 -> VHF done; ? A > 190 -> VHF done;
	[FI] = 50000; => IntToF; [PGFi] = FSW0; => PGF sa;
	[PGFi] = FSW0; => PGF a; [PGFi] = FSRZF; => PGF quo;
	=> F32Narrow; => PGF stf32; [VHFk] = [PGFt];
	-> VHF draw;

"VH rescue flare"
	( other_vehicle_at() passes a negative 500000-unit step, added=3, and
	  condition=1 for each of the helper Stardrifter's four hull fixtures. )
	[VHFadd] = 3;
	[FI] = [VHFpx]; => IntToF; => F32Narrow; => FStoreF32;
	[PGFi] = FSINX; [PGFt] = [FS0]; => PGF setf32;
	[FI] = [VHFpy]; => IntToF; => F32Narrow; => FStoreF32;
	[PGFi] = FSINY; [PGFt] = [FS0]; => PGF setf32;
	[FI] = [VHFpz]; => IntToF; => F32Narrow; => FStoreF32;
	[PGFi] = FSINZ; [PGFt] = [FS0]; => PGF setf32;
	=> PG getcoords;
	A = rwf; A = [A]; ? A = 0 -> VHF done;
	A = [GCx]; A+; [VHFcx] = A;
	A = [GCy]; A+; [VHFcy] = A;
	A = [VHFcx]; ? A <= 10 -> VHF done; ? A >= 320 -> VHF done;
	A = [VHFcy]; ? A <= 10 -> VHF done; ? A >= 190 -> VHF done;
	A '* 320; A + [VHFcx]; [SPoff] = A; [SPreg] = RGADP; => SP get;
	A = [SPval]; ? A < 64 -> VHF done;
	[FI] = 500000; => IntToF; [PGFi] = FSW0; => PGF sa;
	[PGFi] = FSW0; => PGF a; [PGFi] = FSRZF; => PGF quo;
	=> F32Narrow; => PGF stf32; [VHFk] = [PGFt];
	-> VHF draw;

"VH space flare"
	( NOCTIS.CPP:3639-3701 with xshift=3, yshift=0 and condition=0.
	  The selected star is already relative in SFXX/SFYY/SFZZ; preserve
	  the source's double rotation and strict relative-screen bounds. )
	=> VHF space added; [VHFok] = 0;
	[PGFi] = SFXX; => PGF a; [PGFi] = FSPCB; => PGF mul;
	[PGFi] = SFT0; => PGF sa;
	[PGFi] = SFZZ; => PGF a; [PGFi] = FSPSB; => PGF mul;
	[PGFi] = SFT0; => PGF add; [PGFi] = SFRX; => PGF sa;
	[PGFi] = SFZZ; => PGF a; [PGFi] = FSTCB; => PGF mul;
	[PGFi] = SFT0; => PGF sa;
	[PGFi] = SFXX; => PGF a; [PGFi] = FSTSB; => PGF mul;
	[PGFi] = SFT0; => PGF rsub; [PGFi] = SFZ2; => PGF sa;
	[PGFi] = SFZ2; => PGF a; [PGFi] = FSTCA; => PGF mul;
	[PGFi] = SFT0; => PGF sa;
	[PGFi] = SFYY; => PGF a; [PGFi] = FSTSA; => PGF mul;
	[PGFi] = SFT0; => PGF add; [PGFi] = SFRZ; => PGF sa;
	[PGFi] = SFYY; => PGF a; [PGFi] = FSPCA; => PGF mul;
	[PGFi] = SFT0; => PGF sa;
	[PGFi] = SFZ2; => PGF a; [PGFi] = FSPSA; => PGF mul;
	[PGFi] = SFT0; => PGF rsub; [PGFi] = SFRY; => PGF sa;
	[FI] = 1; => IntToF; [FB0] = [FA0]; [FB1] = [FA1];
	[PGFi] = SFRZ; => PGF a; => FCmp;
	A = [FI]; ? A <= 0 -> VHF done;
	[PGFi] = SFRX; => PGF a; [PGFi] = SFRZ; => PGF quo; => FToIntChop;
	A = [FI]; A + 3; C = 0; C - 150; ? A <= C -> VHF done;
	? A >= 160 -> VHF done; A + 160; [VHFcx] = A;
	[PGFi] = SFRY; => PGF a; [PGFi] = SFRZ; => PGF quo; => FToIntChop;
	A = [FI]; C = 0; C - 90; ? A <= C -> VHF done;
	? A >= 90 -> VHF done; A + 100; [VHFcy] = A; [VHFok] = 1;
	-> VHF draw;

"VH surface flare"
	( Surface suns pass source condition 2: only the completed sky band
	  64..127 at the projected centre permits the complete flare. )
	A = [VHFcx]; ? A <= 10 -> VHF done; ? A >= 320 -> VHF done;
	A = [VHFcy]; ? A <= 10 -> VHF done; ? A >= 190 -> VHF done;
	A '* 320; A + [VHFcx]; [SPoff] = A; [SPreg] = RGADP; => SP get;
	A = [SPval]; ? A < 64 -> VHF done; ? A > 127 -> VHF done;

	( Source range: d >= 10*ray and d < 1000*ray.  Retain the binary32
	  expression boundary used by the float star radius and distance. )
	[FS0] = [VHFray]; => FLoadF32; [FB0] = [FA0]; [FB1] = [FA1];
	[FI] = 10; => IntToF; => FMul; => F32Narrow; [VHFstep] = [FS0];
	[FS0] = [VHFstep]; => FLoadF32; [FB0] = [FA0]; [FB1] = [FA1];
	[FS0] = [VHFdist]; => FLoadF32; => FCmp;
	A = [FI]; ? A < 0 -> VHF done;
	[FS0] = [VHFray]; => FLoadF32; [FB0] = [FA0]; [FB1] = [FA1];
	[FI] = 1000; => IntToF; => FMul; => F32Narrow; [VHFstep] = [FS0];
	[FS0] = [VHFstep]; => FLoadF32; [FB0] = [FA0]; [FB1] = [FA1];
	[FS0] = [VHFdist]; => FLoadF32; => FCmp;
	A = [FI]; ? A >= 0 -> VHF done;

	( step=(10*ray)/distance is a float argument; k=10/step. )
	[FS0] = [VHFray]; => FLoadF32; [FB0] = [FA0]; [FB1] = [FA1];
	[FI] = 10; => IntToF; => FMul; => F32Narrow; [VHFstep] = [FS0];
	[FS0] = [VHFdist]; => FLoadF32; [FB0] = [FA0]; [FB1] = [FA1];
	[FS0] = [VHFstep]; => FLoadF32; => FQuo; => F32Narrow; [VHFstep] = [FS0];
	[FS0] = [VHFstep]; => FLoadF32; [FB0] = [FA0]; [FB1] = [FA1];
	[FI] = 10; => IntToF; => FQuo; => F32Narrow; [VHFk] = [FS0];
	=> VHF surface added;

    "VHF draw"
	[VHFl] = VHFF1; [VHFu] = VHFF15; [VHFang] = 0;

    "VHF spoke"
	[VHVangle] = [VHFang]; => VHV sincos;
	[PGFi] = FSW0; [PGFt] = [VHVcos]; => PGF setf32;
	[PGFi] = FSW1; [PGFt] = [VHFk]; => PGF setf32;
	[PGFi] = FSW2; [PGFt] = [VHFl]; => PGF setf32;
	[PGFi] = FSW0; => PGF a; [PGFi] = FSW1; => PGF mul; [PGFi] = FSW2; => PGF mul;
	=> PGF int; [VHFdx] = [FI];
	[PGFi] = FSW0; [PGFt] = [VHVsin]; => PGF setf32;
	[PGFi] = FSW0; => PGF a; [PGFi] = FSW1; => PGF mul; [PGFi] = FSW2; => PGF mul;
	=> PGF int; [VHFdy] = [FI];

	A = [VHFcx]; A - [VHFdx]; [VHFx0] = A;
	A = [VHFcy]; A - [VHFdy]; [VHFy0] = A;
	A = [VHFcx]; A + [VHFdx]; [VHFx1] = A;
	A = [VHFcy]; A + [VHFdy]; [VHFy1] = A;
	=> VHF clamp; => VHF additive line;
	A = [VHFghost]; ? A = 0 -> VHF spoke scale;
	A = [VHFang]; A % 8; ? A != 0 -> VHF spoke scale;
	=> VHF ghost reflections;

	"VHF spoke scale"
	[PGFi] = FSW0; [PGFt] = [VHFl]; => PGF setf32;
	[PGFi] = FSW1; [PGFt] = [VHFu]; => PGF setf32;
	[PGFi] = FSW0; => PGF a; [PGFi] = FSW1; => PGF mul;
	=> F32Narrow; => PGF stf32; [VHFl] = [PGFt];
	A = [VHFl]; ? A > VHFF3 -> VHF flip; ? A < VHFF1 -> VHF flip;
	-> VHF scale ready;
    "VHF flip"
	A = [VHFu]; ? A = VHFF15 -> VHF scale down;
	[VHFu] = VHFF15; -> VHF scale ready;
    "VHF scale down"
	[VHFu] = VHFF23;
    "VHF scale ready"
	A = [VHFang]; A + [VHFadd]; [VHFang] = A; ? A < 180 -> VHF spoke;
    "VHF done"
	end;

"VHF surface added"
	( int added = 1 + 0.002 * dsd; dsd is the source float. )
	[FS0] = [VHFdist]; => FLoadF32;
	[FB0] = D2F1A9FCh; [FB1] = 3F60624Dh; => FMul;
	[FT0] = [FA0]; [FT1] = [FA1]; [FI] = 1; => IntToF;
	[FB0] = [FT0]; [FB1] = [FT1]; => FAdd; => FToIntChop;
	[VHFadd] = [FI];
	end;

"VHF space added"
	( int added = 1 + 0.001 * distance; Stardrifter distances remain double. )
	[FA0] = [VHFdist0]; [FA1] = [VHFdist1];
	[FB0] = D2F1A9FCh; [FB1] = 3F50624Dh; => FMul;
	[FT0] = [FA0]; [FT1] = [FA1]; [FI] = 1; => IntToF;
	[FB0] = [FT0]; [FB1] = [FT1]; => FAdd; => FToIntChop;
	[VHFadd] = [FI];
	end;

"VHF ghost reflections"
	( lens_flares_for's three secondary reflections on every eighth spoke.
	  The menu's ALWAYS ON mode forces this path even with the visor HUD open. )
	A = [VHFdx]; A / 10; [VHFgdx] = A;
	A = [VHFdy]; A / 10; [VHFgdy] = A;
	A = [VHFcx]; A - 160; C = 0; C - A; C / 10; [VHFgx] = C;
	A = [VHFcy]; A - 100; C = 0; C - A; C / 10; [VHFgy] = C;
	[VHFgr] = 0;
    "VHF ghost reflection"
	A = 160; A + [VHFgx]; A - [VHFgdx]; [VHFx0] = A;
	A = 100; A + [VHFgy]; A - [VHFgdy]; [VHFy0] = A;
	A = 160; A + [VHFgx]; A + [VHFgdx]; [VHFx1] = A;
	A = 100; A + [VHFgy]; A + [VHFgdy]; [VHFy1] = A;
	=> VHF clamp; => VHF additive line;
	A = [VHFgdx]; A '* 4; [VHFgdx] = A;
	A = [VHFgdy]; A '* 4; [VHFgdy] = A;
	A = [VHFgx]; A '* 3; [VHFgx] = A;
	A = [VHFgy]; A '* 3; [VHFgy] = A;
	[VHFgr]+; A = [VHFgr]; ? A < 3 -> VHF ghost reflection;
	end;

"VHF clamp"
	A = [VHFx0]; ? A >= 10 -> VHF cx0u; [VHFx0] = 10;
    "VHF cx0u"
	A = [VHFx0]; ? A <= 310 -> VHF cy0l; [VHFx0] = 310;
    "VHF cy0l"
	A = [VHFy0]; ? A >= 10 -> VHF cy0u; [VHFy0] = 10;
    "VHF cy0u"
	A = [VHFy0]; ? A <= 190 -> VHF cx1l; [VHFy0] = 190;
    "VHF cx1l"
	A = [VHFx1]; ? A >= 10 -> VHF cx1u; [VHFx1] = 10;
    "VHF cx1u"
	A = [VHFx1]; ? A <= 310 -> VHF cy1l; [VHFx1] = 310;
    "VHF cy1l"
	A = [VHFy1]; ? A >= 10 -> VHF cy1u; [VHFy1] = 10;
    "VHF cy1u"
	A = [VHFy1]; ? A <= 190 -> VHF clamp done; [VHFy1] = 190;
    "VHF clamp done"
	end;

"VHF additive line"
	A = [VHFx1]; A - [VHFx0]; [VHFlsx] = 1; ? A >= 0 -> VHF dx positive;
	C = 0; C - A; A = C; C = 0; C - 1; [VHFlsx] = C;
    "VHF dx positive"
	[VHFldx] = A;
	A = [VHFy1]; A - [VHFy0]; [VHFlsy] = 1; ? A >= 0 -> VHF dy positive;
	C = 0; C - 1; [VHFlsy] = C; C = 0; C - A; A = C;
    "VHF dy positive"
	C = 0; C - A; [VHFldy] = C;
	A = [VHFldx]; A + [VHFldy]; [VHFerr] = A;

    "VHF line loop"
	A = [VHFy0]; A '* 320; A + [VHFx0]; [SPoff] = A; [SPreg] = RGADP; => SP get;
	C = [SPval]; D = C; D & 192; C & 63; C + 8;
	? C <= 62 -> VHF pixel ready; C = 62;
    "VHF pixel ready"
	C | D; [SPval] = C; => SP put;
	A = [VHFx0]; ? A != [VHFx1] -> VHF line step;
	A = [VHFy0]; ? A = [VHFy1] -> VHF line done;
    "VHF line step"
	A = [VHFerr]; A + A; [VHFe2] = A;
	A = [VHFe2]; ? A < [VHFldy] -> VHF skip x;
	A = [VHFerr]; A + [VHFldy]; [VHFerr] = A;
	A = [VHFx0]; A + [VHFlsx]; [VHFx0] = A;
    "VHF skip x"
	A = [VHFe2]; ? A > [VHFldx] -> VHF skip y;
	A = [VHFerr]; A + [VHFldx]; [VHFerr] = A;
	A = [VHFy0]; A + [VHFlsy]; [VHFy0] = A;
    "VHF skip y"
	-> VHF line loop;
    "VHF line done"
	end;
