      ( *** vhflare - Stardrifter and surface lens flares ***

	Alogena calls lens_flares_for unconditionally at clean startup when
	ilightv=1 and elight=0.  This is its primary 90-spoke path: project the
	fixture point, derive k=-step/rz for step=-50000, and draw through the
	original flares=1 low-six-bit additive rule.  Its condition is zero;
	planetary suns use the center-pixel condition instead. )

"constants"

	VHFF15 = 1069547520;		( 1.5f )
	VHFM10 = 3240099840;		( -10.0f )
	VHFP10 = 1092616192;		( +10.0f )

"variables"

	VHFcx = 0; VHFcy = 0; VHFk = 0; VHFang = 0;
	VHFk0 = 0; VHFk1 = 0; VHFl0 = 0; VHFl1 = 0; VHFu0 = 0; VHFu1 = 0;
	VHFtrigready = 0; VHFtrigi = 0; VHFtriga0 = 0; VHFtriga1 = 0;
	VHFadd = 2; VHFpx = 0; VHFpy = 0; VHFpz = 0;
	VHFdist = 0; VHFdist0 = 0; VHFdist1 = 0; VHFray = 0; VHFstep = 0;
	VHFok = 0;
	VHFghost = 0; VHFgdx = 0; VHFgdy = 0; VHFgr = 0;
	VHFgfx = 0; VHFgfy = 0;
	VHFdx = 0; VHFdy = 0;
	VHFx0 = 0; VHFy0 = 0; VHFx1 = 0; VHFy1 = 0;
	VHFclipbound = 0; VHFlineleft = 0; VHFlinetop = 0;
	VHFlinea = 0; VHFlineb = 0; VHFlineL = 0;
	VHFlinegx = 0; VHFlinegy = 0; VHFlineend = 0;
	VHFlineax = 0; VHFlineay = 0; VHFlinep = 0; VHFlinepf = 0;

"workspace"

	VHFsintab = 361; VHFcostab = 361;

"programme"

"VH halogen flare"
	[VHFadd] = 2;
	[PGFi] = FSINX; [PGFt] = VHFM10; => PGF setf32;
	[PGFi] = FSINY; [PGFt] = VHFP10; => PGF setf32;
	[PGFi] = FSINZ; [PGFt] = VHFM10; => PGF setf32;
	=> PG getcoords;
	A = rwf; A = [A]; ? A = 0 -> VHF done;
	A = [GCx]; A+; [VHFcx] = A;
	A = [GCy]; A+; [VHFcy] = A;
	A = [VHFcx]; ? A < 10 -> VHF done; ? A > 310 -> VHF done;
	A = [VHFcy]; ? A < 10 -> VHF done; ? A > 190 -> VHF done;
	[FI] = 50000; => IntToF; [PGFi] = FSW0; => PGF sa;
	[PGFi] = FSW0; => PGF a; [PGFi] = FSRZF; => PGF quo;
	=> F32Narrow; => PGF stf32; [VHFk] = [PGFt];
	-> VHF draw;

"VH rescue flare"
	( other_vehicle_at() passes a negative 500000-unit step, added=3, and
	  condition=1 for each of the helper Stardrifter's four hull fixtures. )
	[VHFadd] = 3;
	[FI] = [VHFpx]; => IntToF; => F32Narrow; => FStoreF32;
	[PGFi] = FSINX; [PGFt] = [FS0]; => PGF setf32;
	[FI] = [VHFpy]; => IntToF; => F32Narrow; => FStoreF32;
	[PGFi] = FSINY; [PGFt] = [FS0]; => PGF setf32;
	[FI] = [VHFpz]; => IntToF; => F32Narrow; => FStoreF32;
	[PGFi] = FSINZ; [PGFt] = [FS0]; => PGF setf32;
	=> PG getcoords;
	A = rwf; A = [A]; ? A = 0 -> VHF done;
	A = [GCx]; A+; [VHFcx] = A;
	A = [GCy]; A+; [VHFcy] = A;
	A = [VHFcx]; ? A <= 10 -> VHF done; ? A >= 320 -> VHF done;
	A = [VHFcy]; ? A <= 10 -> VHF done; ? A >= 190 -> VHF done;
	A '* 320; A + [VHFcx]; [SPoff] = A; [SPreg] = RGADP; => SP get;
	A = [SPval]; ? A < 64 -> VHF done;
	[FI] = 500000; => IntToF; [PGFi] = FSW0; => PGF sa;
	[PGFi] = FSW0; => PGF a; [PGFi] = FSRZF; => PGF quo;
	=> F32Narrow; => PGF stf32; [VHFk] = [PGFt];
	-> VHF draw;

"VH space flare"
	( NOCTIS.CPP:3639-3701 with xshift=3, yshift=0 and condition=0.
	  The selected star is already relative in SFXX/SFYY/SFZZ; preserve
	  the source's double rotation and strict relative-screen bounds. )
	=> VHF space added; [VHFok] = 0;
	[PGFi] = SFXX; => PGF a; [PGFi] = FSPCB; => PGF mul;
	[PGFi] = SFT0; => PGF sa;
	[PGFi] = SFZZ; => PGF a; [PGFi] = FSPSB; => PGF mul;
	[PGFi] = SFT0; => PGF add; [PGFi] = SFRX; => PGF sa;
	[PGFi] = SFZZ; => PGF a; [PGFi] = FSTCB; => PGF mul;
	[PGFi] = SFT0; => PGF sa;
	[PGFi] = SFXX; => PGF a; [PGFi] = FSTSB; => PGF mul;
	[PGFi] = SFT0; => PGF rsub; [PGFi] = SFZ2; => PGF sa;
	[PGFi] = SFZ2; => PGF a; [PGFi] = FSTCA; => PGF mul;
	[PGFi] = SFT0; => PGF sa;
	[PGFi] = SFYY; => PGF a; [PGFi] = FSTSA; => PGF mul;
	[PGFi] = SFT0; => PGF add; [PGFi] = SFRZ; => PGF sa;
	[PGFi] = SFYY; => PGF a; [PGFi] = FSPCA; => PGF mul;
	[PGFi] = SFT0; => PGF sa;
	[PGFi] = SFZ2; => PGF a; [PGFi] = FSPSA; => PGF mul;
	[PGFi] = SFT0; => PGF rsub; [PGFi] = SFRY; => PGF sa;
	[FI] = 1; => IntToF; [FB0] = [FA0]; [FB1] = [FA1];
	[PGFi] = SFRZ; => PGF a; => FCmp;
	A = [FI]; ? A <= 0 -> VHF done;
	[PGFi] = SFRX; => PGF a; [PGFi] = SFRZ; => PGF quo; => FToIntChop;
	A = [FI]; A + 3; C = 0; C - 150; ? A <= C -> VHF done;
	? A >= 160 -> VHF done; A + 160; [VHFcx] = A;
	[PGFi] = SFRY; => PGF a; [PGFi] = SFRZ; => PGF quo; => FToIntChop;
	A = [FI]; C = 0; C - 90; ? A <= C -> VHF done;
	? A >= 90 -> VHF done; A + 100; [VHFcy] = A; [VHFok] = 1;
	-> VHF draw;

"VH surface flare"
	( Surface suns pass source condition 2: only the completed sky band
	  64..127 at the projected centre permits the complete flare. )
	A = [VHFcx]; ? A <= 10 -> VHF done; ? A >= 320 -> VHF done;
	A = [VHFcy]; ? A <= 10 -> VHF done; ? A >= 190 -> VHF done;
	A '* 320; A + [VHFcx]; [SPoff] = A; [SPreg] = RGADP; => SP get;
	A = [SPval]; ? A < 64 -> VHF done; ? A > 127 -> VHF done;

	( Source range: d >= 10*ray and d < 1000*ray.  Retain the binary32
	  expression boundary used by the float star radius and distance. )
	[FS0] = [VHFray]; => FLoadF32; [FB0] = [FA0]; [FB1] = [FA1];
	[FI] = 10; => IntToF; => FMul; => F32Narrow; [VHFstep] = [FS0];
	[FS0] = [VHFstep]; => FLoadF32; [FB0] = [FA0]; [FB1] = [FA1];
	[FS0] = [VHFdist]; => FLoadF32; => FCmp;
	A = [FI]; ? A < 0 -> VHF done;
	[FS0] = [VHFray]; => FLoadF32; [FB0] = [FA0]; [FB1] = [FA1];
	[FI] = 1000; => IntToF; => FMul; => F32Narrow; [VHFstep] = [FS0];
	[FS0] = [VHFstep]; => FLoadF32; [FB0] = [FA0]; [FB1] = [FA1];
	[FS0] = [VHFdist]; => FLoadF32; => FCmp;
	A = [FI]; ? A >= 0 -> VHF done;

	( step=(10*ray)/distance is a float argument; k=10/step. )
	[FS0] = [VHFray]; => FLoadF32; [FB0] = [FA0]; [FB1] = [FA1];
	[FI] = 10; => IntToF; => FMul; => F32Narrow; [VHFstep] = [FS0];
	[FS0] = [VHFdist]; => FLoadF32; [FB0] = [FA0]; [FB1] = [FA1];
	[FS0] = [VHFstep]; => FLoadF32; => FQuo; => F32Narrow; [VHFstep] = [FS0];
	[FS0] = [VHFstep]; => FLoadF32; [FB0] = [FA0]; [FB1] = [FA1];
	[FI] = 10; => IntToF; => FQuo; => F32Narrow; [VHFk] = [FS0];
	=> VHF surface added;

    "VHF draw"
	( The step callers currently publish k as binary32.  Widen that exact
	  value, then keep k/l/u in the source's binary64 local format. )
	[FS0] = [VHFk]; => FLoadF32; [VHFk0] = [FA0]; [VHFk1] = [FA1];
	[FI] = 1; => IntToF; [VHFl0] = [FA0]; [VHFl1] = [FA1];
	[PGFt] = VHFF15; => PGF ldf32; [VHFu0] = [FA0]; [VHFu1] = [FA1];
	[VHFang] = 0; => VHF trig init;

    "VHF spoke"
	=> VHF sincos;
	[PGFt] = [VHVcos]; => PGF ldf32;
	[FB0] = [VHFk0]; [FB1] = [VHFk1]; => FMul;
	[FB0] = [VHFl0]; [FB1] = [VHFl1]; => FMul; => FToIntChop; [VHFdx] = [FI];
	[PGFt] = [VHVsin]; => PGF ldf32;
	[FB0] = [VHFk0]; [FB1] = [VHFk1]; => FMul;
	[FB0] = [VHFl0]; [FB1] = [VHFl1]; => FMul; => FToIntChop; [VHFdy] = [FI];

	( fline receives coordinates relative to the source's 158,100 centre. )
	A = [VHFcx]; A - 160; A - [VHFdx]; [VHFx0] = A;
	A = [VHFcy]; A - 100; A - [VHFdy]; [VHFy0] = A;
	A = [VHFcx]; A - 160; A + [VHFdx]; [VHFx1] = A;
	A = [VHFcy]; A - 100; A + [VHFdy]; [VHFy1] = A;
	=> VHF source fline;
	A = [VHFghost]; ? A = 0 -> VHF spoke scale;
	A = [VHFang]; A % 8; ? A != 0 -> VHF spoke scale;
	=> VHF ghost reflections;

	"VHF spoke scale"
	[FA0] = [VHFl0]; [FA1] = [VHFl1];
	[FB0] = [VHFu0]; [FB1] = [VHFu1]; => FMul;
	[VHFl0] = [FA0]; [VHFl1] = [FA1];
	[FI] = 3; => IntToF; [FB0] = [FA0]; [FB1] = [FA1];
	[FA0] = [VHFl0]; [FA1] = [VHFl1]; => FCmp;
	A = [FI]; ? A > 0 -> VHF flip;
	[FI] = 1; => IntToF; [FB0] = [FA0]; [FB1] = [FA1];
	[FA0] = [VHFl0]; [FA1] = [VHFl1]; => FCmp;
	A = [FI]; ? A < 0 -> VHF flip;
	-> VHF scale ready;
    "VHF flip"
	[FI] = 1; => IntToF; [FB0] = [VHFu0]; [FB1] = [VHFu1]; => FQuo;
	[VHFu0] = [FA0]; [VHFu1] = [FA1];
    "VHF scale ready"
	A = [VHFang]; A + [VHFadd]; [VHFang] = A; ? A < 180 -> VHF spoke;
    "VHF done"
	end;

"VHF trig init"
	A = [VHFtrigready]; ? A != 0 -> VHF trig init done;
	[VHFtrigi] = 0; [VHFtriga0] = 0; [VHFtriga1] = 0;
    "VHF trig build"
	[PGFt] = [VHFtriga0]; [PGFu] = [VHFtriga1]; => PGF ldf64;
	[PGFi] = FSW0; => PGF sa; => FCos; => F32Narrow; => FStoreF32;
	A = VHFcostab; A + [VHFtrigi]; [A] = [FS0];
	[PGFi] = FSW0; => PGF a; => FSin; => F32Narrow; => FStoreF32;
	A = VHFsintab; A + [VHFtrigi]; [A] = [FS0];
	[PGFi] = FSW0; => PGF a; [FB0] = VHVDEGLO; [FB1] = VHVDEGHI; => FAdd;
	[VHFtriga0] = [FA0]; [VHFtriga1] = [FA1];
	[VHFtrigi]+; A = [VHFtrigi]; ? A <= 360 -> VHF trig build;
	[VHFtrigready] = 1;
    "VHF trig init done"
	end;

"VHF sincos"
	A = VHFcostab; A + [VHFang]; [VHVcos] = [A];
	A = VHFsintab; A + [VHFang]; [VHVsin] = [A];
	end;

"VHF surface added"
	( int added = 1 + 0.002 * dsd; dsd is the source float. )
	[FS0] = [VHFdist]; => FLoadF32;
	[FB0] = D2F1A9FCh; [FB1] = 3F60624Dh; => FMul;
	[FT0] = [FA0]; [FT1] = [FA1]; [FI] = 1; => IntToF;
	[FB0] = [FT0]; [FB1] = [FT1]; => FAdd; => FToIntChop;
	[VHFadd] = [FI];
	end;

"VHF space added"
	( int added = 1 + 0.001 * distance; Stardrifter distances remain double. )
	[FA0] = [VHFdist0]; [FA1] = [VHFdist1];
	[FB0] = D2F1A9FCh; [FB1] = 3F50624Dh; => FMul;
	[FT0] = [FA0]; [FT1] = [FA1]; [FI] = 1; => IntToF;
	[FB0] = [FT0]; [FB1] = [FT1]; => FAdd; => FToIntChop;
	[VHFadd] = [FI];
	end;

"VHF ghost reflections"
	( lens_flares_for's three secondary reflections on every eighth spoke.
	  The menu's ALWAYS ON mode forces this path even with the visor HUD open. )
	A = [VHFdx]; A / 10; [VHFgdx] = A;
	A = [VHFdy]; A / 10; [VHFgdy] = A;
	A = [VHFcx]; A - 160; [FI] = A; => IntToF;
	[FB0] = 9999999Ah; [FB1] = BFB99999h; => FMul; => F32Narrow; [VHFgfx] = [FS0];
	A = [VHFcy]; A - 100; [FI] = A; => IntToF;
	[FB0] = 9999999Ah; [FB1] = BFB99999h; => FMul; => F32Narrow; [VHFgfy] = [FS0];
	[VHFgr] = 0;
    "VHF ghost reflection"
	[FS0] = [VHFgfx]; => FLoadF32; [FT0] = [FA0]; [FT1] = [FA1];
	[FI] = [VHFgdx]; => IntToF; [FB0] = [FA0]; [FB1] = [FA1];
	[FA0] = [FT0]; [FA1] = [FT1]; => FSub; => FToIntChop; [VHFx0] = [FI];
	[FA0] = [FT0]; [FA1] = [FT1]; => FAdd; => FToIntChop; [VHFx1] = [FI];
	[FS0] = [VHFgfy]; => FLoadF32; [FT0] = [FA0]; [FT1] = [FA1];
	[FI] = [VHFgdy]; => IntToF; [FB0] = [FA0]; [FB1] = [FA1];
	[FA0] = [FT0]; [FA1] = [FT1]; => FSub; => FToIntChop; [VHFy0] = [FI];
	[FA0] = [FT0]; [FA1] = [FT1]; => FAdd; => FToIntChop; [VHFy1] = [FI];
	=> VHF source fline;
	A = [VHFgdx]; A '* 4; [VHFgdx] = A;
	A = [VHFgdy]; A '* 4; [VHFgdy] = A;
	[FS0] = [VHFgfx]; => FLoadF32; [FI] = 3; => IntToF;
	[FB0] = [FA0]; [FB1] = [FA1]; [FS0] = [VHFgfx]; => FLoadF32; => FMul;
	=> F32Narrow; [VHFgfx] = [FS0];
	[FS0] = [VHFgfy]; => FLoadF32; [FI] = 3; => IntToF;
	[FB0] = [FA0]; [FB1] = [FA1]; [FS0] = [VHFgfy]; => FLoadF32; => FMul;
	=> F32Narrow; [VHFgfy] = [FS0];
	[VHFgr]+; A = [VHFgr]; ? A < 3 -> VHF ghost reflection;
	end;

( Source fline() clipping order and Stick(..., flares=1) fixed-point
  sampling.  The original non-vertical Stick path deliberately samples
  every other fixed-point position and writes at di+4; its vertical special
  case writes every row without that displacement. )
"VHF source fline"
	C = 0; C - 150; [VHFlineleft] = C;
	C = 0; C - 90; [VHFlinetop] = C;
	A = [VHFx0]; ? A >= [VHFlineleft] -> VHF fl x1 left;
	A = [VHFx0]; A - [VHFx1]; ? A = 0 -> VHF fl x1 left;
	[VHFclipbound] = [VHFlineleft]; => VHF clip x0;
    "VHF fl x1 left"
	A = [VHFx1]; ? A >= [VHFlineleft] -> VHF fl y0 top;
	A = [VHFx1]; A - [VHFx0]; ? A = 0 -> VHF fl y0 top;
	[VHFclipbound] = [VHFlineleft]; => VHF clip x1;
    "VHF fl y0 top"
	A = [VHFy0]; ? A >= [VHFlinetop] -> VHF fl y1 top;
	A = [VHFy0]; A - [VHFy1]; ? A = 0 -> VHF fl y1 top;
	[VHFclipbound] = [VHFlinetop]; => VHF clip y0;
    "VHF fl y1 top"
	A = [VHFy1]; ? A >= [VHFlinetop] -> VHF fl x0 right;
	A = [VHFy1]; A - [VHFy0]; ? A = 0 -> VHF fl x0 right;
	[VHFclipbound] = [VHFlinetop]; => VHF clip y1;
    "VHF fl x0 right"
	A = [VHFx0]; ? A <= 160 -> VHF fl x1 right;
	A = [VHFx0]; A - [VHFx1]; ? A = 0 -> VHF fl x1 right;
	[VHFclipbound] = 160; => VHF clip x0;
    "VHF fl x1 right"
	A = [VHFx1]; ? A <= 160 -> VHF fl y0 bottom;
	A = [VHFx1]; A - [VHFx0]; ? A = 0 -> VHF fl y0 bottom;
	[VHFclipbound] = 160; => VHF clip x1;
    "VHF fl y0 bottom"
	A = [VHFy0]; ? A <= 90 -> VHF fl y1 bottom;
	A = [VHFy0]; A - [VHFy1]; ? A = 0 -> VHF fl y1 bottom;
	[VHFclipbound] = 90; => VHF clip y0;
    "VHF fl y1 bottom"
	A = [VHFy1]; ? A <= 90 -> VHF fl reject point;
	A = [VHFy1]; A - [VHFy0]; ? A = 0 -> VHF fl reject point;
	[VHFclipbound] = 90; => VHF clip y1;
    "VHF fl reject point"
	A = [VHFx0]; ? A != [VHFx1] -> VHF fl bounds;
	A = [VHFy0]; ? A = [VHFy1] -> VHF source fline done;
    "VHF fl bounds"
	A = [VHFy0]; ? A < [VHFlinetop] -> VHF source fline done; ? A > 90 -> VHF source fline done;
	A = [VHFy1]; ? A < [VHFlinetop] -> VHF source fline done; ? A > 90 -> VHF source fline done;
	A = [VHFx0]; ? A < [VHFlineleft] -> VHF source fline done; ? A > 160 -> VHF source fline done;
	A = [VHFx1]; ? A < [VHFlineleft] -> VHF source fline done; ? A > 160 -> VHF source fline done;
	A = [VHFx0]; A + 158; [VHFx0] = A;
	A = [VHFx1]; A + 158; [VHFx1] = A;
	A = [VHFy0]; A + 100; [VHFy0] = A;
	A = [VHFy1]; A + 100; [VHFy1] = A;
	=> VHF source stick;
    "VHF source fline done"
	end;

"VHF source stick"
	A = [VHFx0]; ? A != [VHFx1] -> VHF stick nonvertical;
	A = [VHFy1]; ? A < [VHFy0] -> VHF stick vertical reverse;
	A = [VHFy0]; A '* 320; A + [VHFx0]; A + RADPT; A + nw; [VHFlinep] = A;
	A = [VHFy1]; A+; A '* 320; A + RADPT; A + nw; [VHFlinepf] = A;
	-> VHF stick vertical loop;
    "VHF stick vertical reverse"
	A = [VHFy1]; A '* 320; A + [VHFx0]; A + RADPT; A + nw; [VHFlinep] = A;
	A = [VHFy0]; A+; A '* 320; A + RADPT; A + nw; [VHFlinepf] = A;
    "VHF stick vertical loop"
	A = [VHFlinep]; ? A >= [VHFlinepf] -> VHF source stick done; D = A; => VHF brighten;
	A = [VHFlinep]; A + 320; [VHFlinep] = A; -> VHF stick vertical loop;

    "VHF stick nonvertical"
	A = [VHFx1]; ? A >= [VHFx0] -> VHF stick ordered;
	C = [VHFx0]; [VHFx0] = [VHFx1]; [VHFx1] = C;
	C = [VHFy0]; [VHFy0] = [VHFy1]; [VHFy1] = C;
    "VHF stick ordered"
	A = [VHFx1]; A - [VHFx0]; [VHFlinea] = A; [VHFlineL] = A;
	A = [VHFy1]; A - [VHFy0]; ? A >= 0 -> VHF stick b positive;
	C = 0; C - A; A = C; C = 0; C - 1; [VHFlineay] = C; -> VHF stick b ready;
    "VHF stick b positive"
	[VHFlineay] = 1;
    "VHF stick b ready"
	[VHFlineb] = A; C = [VHFlineL]; ? A < C -> VHF stick length ready; [VHFlineL] = A;
    "VHF stick length ready"
	[VHFlineL]+;
	A = [VHFx0]; A '* 65536; [VHFlinegx] = A;
	A = [VHFy0]; A '* 65536; [VHFlinegy] = A;
	A = [VHFx1]; A '* 65536; [VHFlineend] = A;
	A = [VHFlinea]; A '* 65536; A / [VHFlineL]; A + A; [VHFlineax] = A;
	A = [VHFlineb]; A '* 65536; A / [VHFlineL]; A + A;
	C = [VHFlineay]; ? C > 0 -> VHF stick ystep ready; C = 0; C - A; A = C;
    "VHF stick ystep ready"
	[VHFlineay] = A;
    "VHF stick loop"
	( Native Stick addresses seg_adapted:[di+4] after discarding the far
	  pointer's four-byte allocation offset.  Against the logical page those
	  two offsets cancel, so the packed framebuffer index is simply di. )
	A = [VHFlinegy]; A '/ 65536; A '* 320; C = [VHFlinegx]; C '/ 65536; A + C;
	A + RADPT; A + nw; D = A; => VHF brighten nonvertical;
	A = [VHFlinegx]; A + [VHFlineax]; [VHFlinegx] = A;
	A = [VHFlinegy]; A + [VHFlineay]; [VHFlinegy] = A;
	A = [VHFlinegx]; ? A < [VHFlineend] -> VHF stick loop;
    "VHF source stick done"
	end;

"VHF brighten"
	A = [D]; A & 255; C = A; C & 63; C + 8;
	? C <= 62 -> VHF brighten ready; C = 62;
    "VHF brighten ready"
	A & 192; C | A; [D] = C;
	end;

"VHF brighten nonvertical"
	( Stick's general flare-1 assembly saturates only when the low-byte add
	  carries.  A source level of 55 therefore becomes 63; levels 56..63
	  become 62.  Its vertical special case above clamps all of them to 62. )
	A = [D]; A & 255; C = A; C & 63; ? C > 55 -> VHF brighten nonvertical sat;
	A + 8; [D] = A; end;
    "VHF brighten nonvertical sat"
	A & 192; A | 62; [D] = A;
	end;

( fline's float kk and diff temporaries narrow at their declarations and the
  resulting floating expression is converted back to long with C truncation. )
"VHF clip x0"
