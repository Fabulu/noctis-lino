      ( *** fpconv - float to int, int to float, and the deliberate
	    narrowings ***

	THE ONE RULE IN THIS FILE.
	The original has TWO float-to-int behaviours and they are not
	interchangeable:

	  CAST SITES.  A C cast truncates.  Borland implements that in
	  __ftol, which flips the rounding control to chop, does the
	  fistp, and flips it back.  These are the SEED sites -
	  nearstar_identity, global_surface_seed, seedval.  One ULP here
	  is a DIFFERENT PLANET.  FToIntChop reproduces the flip.

	  HAND-WRITTEN SITES.  37 of them (36 fistp and 1 fist) in the
	  inline assembly of the projector and the texture mapper.  The
	  rounding control is left at 00 there, so they round to nearest
	  even.  FToIntNear is those.  A quarter pixel is invisible.

	THE FAIL-SAFE, stated as a rule for generated code:
	NEVER emit a bare =, and never call FToIntNear by default.  Emit
	FToIntChop unless the site registry names the site as one of the
	37.  The asymmetry is deliberate - a wrong chop site is CRITICAL
	because it is a seed, a wrong near site is COSMETIC because it is
	a pixel.  Default to the direction whose failure is cheap.

	INT16.  Borland large model makes int 16 bits.  The 32-bit fistp
	runs first and the low 16 bits are then taken and sign extended,
	because that is what the original's register usage does.  Doing
	it with fistp m16int instead would be WRONG: m16int traps as an
	invalid operation on overflow and, masked, stores 8000h for every
	one of them.  The wrap is not an edge case - star classes 1, 3, 4
	and 9 overflow by up to 6.7x as the NORMAL path.

	FCWCSAV is private to this file on purpose.  Sharing FCWSAV with
	FEnter/FLeave would mean a FToIntChop between them silently
	replaced the ambient word FLeave is supposed to put back.       )

"variables"

	FCWCSAV	= 0;		( private: __ftol's own save slot )
	FCWCHOP	= 0;		( private: the chop word it builds )
	CVS = 0; CVE = 0; CVMH = 0; CVML = 0;
	CVSHIFT = 0; CVROUND = 0; CVSTICKY = 0;
	CVMAGH = 0; CVMAGL = 0; CVEXP = 0;
	CVFRAC = 0; CVMODE = 0; CVTMP = 0;

"programme"

( ==================================================================== )

      ( FToIntChop - FI <- (long)FA.  C cast semantics, __ftol. )

"FToIntChop"
	---->;
	=> FToIntChop body;
	<----;
	end;
    "FToIntChop body"
	[CVMODE] = 3;
	=> CV F64 to int;
	end;

      ( FToIntNear - FI <- fistp(FA) under the AMBIENT control word.
	Under FEnter's 133Fh that is round to nearest even, which is the
	37 hand-written sites, for free. )

"FToIntNear"
	---->;
	=> FToIntNear body;
	<----;
	end;
    "FToIntNear body"
	A = [FCW]; A > 10; A & 3; [CVMODE] = A;
	=> CV F64 to int;
	end;

      ( FToInt16Chop / FToInt16Near - the same, then Borland's 16-bit
	int.  Low 16 bits, sign extended back into the unit. )

"FToInt16Chop"
	---->;
	=> FToInt16Chop body;
	<----;
	end;
    "FToInt16Chop body"
	=> FToIntChop;
	=> FNarrow16;
	end;

"FToInt16Near"
	---->;
	=> FToInt16Near body;
	<----;
	end;
    "FToInt16Near body"
	=> FToIntNear;
	=> FNarrow16;
	end;

"FNarrow16"
	---->;
	=> FNarrow16 body;
	<----;
	end;
    "FNarrow16 body"
	A = [FI]; A & 0FFFFh; [FI] = A;
      ? [FI] '< 8000h -> FN16 done;
	A = [FI]; A - 65536; [FI] = A;
    "FN16 done"
	end;

      ( IntToF - FA <- (double)FI.  fild of a dword is exact for every
	int32, at every precision control, so this one cannot round.  Integers
	through 2^24 are also exact in binary32: use Lino's native conversion
	there, widen its bit image exactly, and retain the integer path outside
	that interval. )

"IntToF"
	---->;
	=> IntToF body;
	<----;
	end;
    "IntToF body"
	A = [FI]; ? A > 16777216 -> IntToF exact;
	C = 0; C - 16777216; ? A < C -> IntToF exact;
	[CVTMP] = [FS0]; [FS0] ,= [FI]; => CV F32 to F64;
	[FS0] = [CVTMP];
	end;
    "IntToF exact"
	=> CV int to F64;
	end;

      ( F32Narrow - FA <- (double)(float)FA.

	This is not a workaround, it is BEHAVIOUR TO REPRODUCE.
	NOCTIS-1.CPP:3411-3413 saves dzat_x/y/z into float locals and
	restores them at :4633 and :5025.  Those coordinates sit around
	3.8e6, where a binary32 ULP is 0.25, so landing quantises the
	ship's position to 24 bits observably, in the original.  You can
	only narrow FROM something wider, which is why the ship's state
	is carried as a double and narrowed here rather than being kept
	in lino's native 24-bit floats throughout.

	fstp m32fp rounds with the live RC, so under FEnter this is
	round-to-nearest-even to binary32 - exactly a C store to float. )

"F32Narrow"
	---->;
	=> F32Narrow body;
	<----;
	end;
    "F32Narrow body"
	=> FStoreF32;
	=> FLoadF32;
	end;

      ( FStoreF32 - FS0 <- (float)FA, as a bit image, without widening
	back.  This is what writing a float to a file has to do. )

"FStoreF32"
	---->;
	=> FStoreF32 body;
	<----;
	end;
    "FStoreF32 body"
	=> CV F64 to F32;
	end;

      ( FLoadF32 - FA <- (double)the binary32 in FS0.  Always exact. )

"FLoadF32"
	---->;
	=> FLoadF32 body;
	<----;
	end;
    "FLoadF32 body"
	=> CV F32 to F64;
	end;

( ==================================================================== )
( Portable bit-exact conversion machinery.  These routines operate on )
( IEEE images with integer Lino only; no target CPU escape is needed. )
( ==================================================================== )

"CV unpack F64"
	A = [FA1]; A > 31; [CVS] = A;
	A = [FA1]; A > 20; A & 7FFh; [CVE] = A;
	A = [FA1]; A & 0FFFFFh; A < 11;
	B = [FA0]; B > 21; A | B; [CVMH] = A;
	A = [FA0]; A < 11; [CVML] = A;
      ? [CVE] = 0 -> CVUF denorm;
	A = [CVMH]; A | 80000000h; [CVMH] = A;
	end;
    "CVUF denorm"
	A = [CVMH]; B = [CVML]; A | B;
      ? A = 0 -> CVUF zero;
	[CVE] = 1;
    "CVUF norm"
	A = [CVMH]; A & 80000000h;
      ? A != 0 -> CVUF done;
	A = [CVMH]; A < 1; B = [CVML]; B > 31; A | B; [CVMH] = A;
	A = [CVML]; A < 1; [CVML] = A;
	[CVE]-;
	-> CVUF norm;
    "CVUF zero"
	[CVMH] = 0; [CVML] = 0;
    "CVUF done"
	end;

"CV F64 to int"
	=> CV unpack F64;
	A = [CVMH]; B = [CVML]; A | B;
      ? A = 0 -> CVI zero;
	A = [CVE]; A - 1023; [CVEXP] = A;
	      ? A >= 0 -> CVI exponent ok;
	      ? A = 0FFFFFFFFh -> CVI exponent ok;
	( Below one half, nearest and chop produce zero.  Directed modes
	  produce one only when they point away from zero. )
	A = [CVMODE];
	      ? A = 1 -> CVI tiny down;
	      ? A = 2 -> CVI tiny up;
	-> CVI zero;
    "CVI tiny down"
	      ? [CVS] = 0 -> CVI zero;
	[FI] = 0FFFFFFFFh;
	end;
    "CVI tiny up"
	      ? [CVS] != 0 -> CVI zero;
	[FI] = 1;
	end;
    "CVI exponent ok"
	A = [CVEXP];
	      ? A > 31 -> CVI invalid;
	A = 63; A - [CVEXP]; [CVSHIFT] = A;
	A = [CVMH]; [CVMAGH] = A;
	A = [CVML]; [CVMAGL] = A;
	[CVROUND] = 0; [CVSTICKY] = 0;
    "CVI shift"
	A = [CVSTICKY]; B = [CVROUND]; A | B; [CVSTICKY] = A;
	A = [CVMAGL]; A & 1; [CVROUND] = A;
	A = [CVMAGL]; A > 1; B = [CVMAGH]; B < 31; A | B; [CVMAGL] = A;
	A = [CVMAGH]; A > 1; [CVMAGH] = A;
	[CVSHIFT]-;
      ? [CVSHIFT] != 0 -> CVI shift;

	A = [CVMODE];
      ? A = 3 -> CVI signed;
      ? A = 0 -> CVI nearest;
	A = [CVROUND]; B = [CVSTICKY]; A | B;
      ? A = 0 -> CVI signed;
	A = [CVMODE];
      ? A = 1 -> CVI down;
      ( mode 2, round toward +infinity )
      ? [CVS] != 0 -> CVI signed;
	-> CVI increment;
    "CVI down"
      ? [CVS] = 0 -> CVI signed;
	-> CVI increment;
    "CVI nearest"
      ? [CVROUND] = 0 -> CVI signed;
      ? [CVSTICKY] != 0 -> CVI increment;
	A = [CVMAGL]; A & 1;
      ? A = 0 -> CVI signed;
    "CVI increment"
	A = [CVMAGL]; A + 1; [CVMAGL] = A;
      ? A != 0 -> CVI signed;
	[CVMAGH]+;
    "CVI signed"
      ? [CVMAGH] != 0 -> CVI invalid;
      ? [CVS] != 0 -> CVI negative;
	A = [CVMAGL];
      ? A '> 7FFFFFFFh -> CVI invalid;
	[FI] = A;
	end;
    "CVI negative"
	A = [CVMAGL];
      ? A '> 80000000h -> CVI invalid;
	A = 0; A - [CVMAGL]; [FI] = A;
	end;
    "CVI zero"
	[FI] = 0;
	end;
    "CVI invalid"
	[FI] = 80000000h;
	end;

"CV int to F64"
	[CVS] = 0;
	A = [FI];
      ? A != 0 -> CVIF nonzero;
	[FA0] = 0; [FA1] = 0;
	end;
    "CVIF nonzero"
	A = [FI]; A & 80000000h;
      ? A = 0 -> CVIF positive;
	[CVS] = 1;
	A = 0; A - [FI]; [CVMH] = A;
	-> CVIF normalize;
    "CVIF positive"
	A = [FI]; [CVMH] = A;
    "CVIF normalize"
	[CVEXP] = 31;
    "CVIF shift"
	A = [CVMH]; A & 80000000h;
      ? A != 0 -> CVIF pack;
	A = [CVMH]; A < 1; [CVMH] = A;
	[CVEXP]-;
	-> CVIF shift;
    "CVIF pack"
	A = [CVMH]; A < 21; [FA0] = A;
	A = [CVMH]; A > 11; A & 0FFFFFh;
	B = [CVEXP]; B + 1023; B < 20; A | B;
	B = [CVS]; B < 31; A | B; [FA1] = A;
	end;

"CV F32 to F64"
	A = [FS0]; A > 31; [CVS] = A;
	A = [FS0]; A > 23; A & 0FFh; [CVEXP] = A;
	A = [FS0]; A & 7FFFFFh; [CVFRAC] = A;
      ? [CVEXP] = 0 -> CV32 denorm;
	-> CV32 pack;
    "CV32 denorm"
      ? [CVFRAC] = 0 -> CV32 zero;
	[CVEXP] = 1;
    "CV32 norm"
	A = [CVFRAC]; A & 800000h;
      ? A != 0 -> CV32 implicit;
	A = [CVFRAC]; A < 1; [CVFRAC] = A;
	[CVEXP]-;
	-> CV32 norm;
    "CV32 implicit"
	A = [CVFRAC]; A & 7FFFFFh; [CVFRAC] = A;
    "CV32 pack"
	A = [CVFRAC]; A < 29; [FA0] = A;
	A = [CVFRAC]; A > 3;
	B = [CVEXP]; B + 896; B < 20; A | B;
	B = [CVS]; B < 31; A | B; [FA1] = A;
	end;
    "CV32 zero"
	[FA0] = 0;
	A = [CVS]; A < 31; [FA1] = A;
	end;

"CV F64 to F32"
	=> CV unpack F64;
	A = [CVMH]; B = [CVML]; A | B;
      ? A = 0 -> CV64 zero;
	A = [CVE]; A - 896; [CVEXP] = A;
      ? A <= 0 -> CV64 underflow;
      ? A >= 255 -> CV64 overflow;
	A = [CVMH]; A > 8; A & 7FFFFFh; [CVFRAC] = A;
	A = [CVMH]; A > 7; A & 1; [CVROUND] = A;
	A = [CVMH]; A & 7Fh; B = [CVML]; A | B; [CVSTICKY] = A;
	A = [FCW]; A > 10; A & 3; [CVMODE] = A;
      ? A = 3 -> CV64 pack;
      ? A = 0 -> CV64 nearest;
	A = [CVROUND]; B = [CVSTICKY]; A | B;
      ? A = 0 -> CV64 pack;
	A = [CVMODE];
      ? A = 1 -> CV64 down;
      ? [CVS] != 0 -> CV64 pack;
	-> CV64 increment;
    "CV64 down"
      ? [CVS] = 0 -> CV64 pack;
	-> CV64 increment;
    "CV64 nearest"
      ? [CVROUND] = 0 -> CV64 pack;
      ? [CVSTICKY] != 0 -> CV64 increment;
	A = [CVFRAC]; A & 1;
      ? A = 0 -> CV64 pack;
    "CV64 increment"
	A = [CVFRAC]; A + 1; [CVFRAC] = A;
      ? A != 800000h -> CV64 pack;
	[CVFRAC] = 0;
	[CVEXP]+;
      ? [CVEXP] >= 255 -> CV64 overflow;
    "CV64 pack"
	A = [CVFRAC];
	B = [CVEXP]; B < 23; A | B;
	B = [CVS]; B < 31; A | B; [FS0] = A;
	end;
    "CV64 zero"
	A = [CVS]; A < 31; [FS0] = A;
	end;
    "CV64 underflow"
	( Noctis never stores a binary32 subnormal; keep the sign on zero. )
	A = [CVS]; A < 31; [FS0] = A;
	end;
    "CV64 overflow"
	A = 7F800000h; B = [CVS]; B < 31; A | B; [FS0] = A;
	end;
