      ( *** fpstarsoft - the oracle again, with the FPU switched off ***

	fpstar links fpx87 and gets 4194/4194 because Intel's FPU is
	right.  This programme links fpsoft, which contains no x87
	instruction at all: every mantissa is two units of integer, every
	rounding decision is an explicit comparison, and the arithmetic
	is done at 64 bits of mantissa by hand.

	The claim under test is that the SAME SCHEDULE run on completely
	different machinery produces the SAME BITS.  That is circular with
	respect to the schedule - both sides are told the same sequence of
	mnemonics - and not circular with respect to the arithmetic, which
	is what is actually being checked.

	TWO BATTERIES, and the second is the interesting one:

	  0  control word 133Fh   PC=64, what the original ran at
	  1  control word 103Fh   PC=24, which crushes the x87 backend
	     from 4194 to 66

	Both must score 4194 and both must be BIT IDENTICAL to each other,
	because the precision control cannot reach code that never touches
	the FPU.  If battery 1 differs from battery 0 by a single value,
	something in fpsoft is quietly using the x87 after all, and the
	portability argument for having a soft backend collapses.

	REQUIRES THE EXTENDED TOOLCHAIN:
	   -Compiler C:\programmieren\linoleum\main\lib\gen\compiler114m.exe
	   -Cpu i386m
	for the *%' 32x32->64 multiply that Mul128 is built from.

	Reads  fpstarin.bin
	Writes fpstarsoftout.bin                                        )

"libraries"

	fpabi;
	fpctl;
	fpsoft;
	fpchainssoft;

"directors"

	program name = { fpstarsoft };
	unit = 32;

"constants"

	NSTAR	= 4194;
	NBAT	= 2;
	HDR	= 8;

"variables"

	fpqinput  = { fpstarin.bin };
	fpqresult = { fpstarsoftout.bin };

	fpqiptr	= 0;
	fpqoptr	= 0;
	fpqns	= 0;
	fpqbat	= 0;
	fpqK	= 100000;
	fpqbad	= 0;		( how many cases left the soft stack dirty )

"workspace"

	fpqhead	= HDR;
	fpqins	= 12582;
	fpqouts	= 16776;	( NSTAR multiplied 2 multiplied NBAT )

"programme"

	[File Name]	= fpqinput;
	[File Position]	= 0;
	[File Command]	= READ;
	[Block Pointer]	= fpqins;
	[Block Size]	= 12582;
	[Block Size]	* BYTES PER UNIT;
	isocall;

	[FCW] = FCWEXT;
	=> FEnter;

	[FA0] = 0; [FA1] = 0;
	{
	    D9 E8				(fld1)
	    DD 9F <dFA0 mtp bytesperunit>	(fstp qword [edi+FA0*4])
	}
	A = [FA0]; [fpqhead plus 0] = A;
	A = [FA1]; [fpqhead plus 1] = A;

	A = fpqouts; [fpqoptr] = A;
	[fpqbat] = 0;
	[fpqbad] = 0;

    "fpqbattery"

	[FCW] = FCWEXT;
      ? [fpqbat] = 0 -> fpqcw done;
	[FCW] = FCWSGL;			( PC=24 - must change nothing )
    "fpqcw done"
	=> FLoadCW;

	A = fpqins; [fpqiptr] = A;
	[fpqns] = 0;

    "fpqonestar"

	E = [fpqiptr];
	A = [E];		[FJ0] = A;
	A = [E plus 1];		[FJ1] = A;
	A = [E plus 2];		[FJ2] = A;

	=> fpqsoftid;

	E = [fpqoptr];
	A = [FA0];		[E] = A;
	A = [FA1];		[E plus 1] = A;
	A = [fpqoptr]; A + 2;	[fpqoptr] = A;

	A = [fpqiptr]; A + 3;	[fpqiptr] = A;
	[fpqns]+;
      ? [fpqns] < NSTAR -> fpqonestar;

	[fpqbat]+;
      ? [fpqbat] < NBAT -> fpqbattery;

	=> FCWRead;
	A = [FI]; [fpqhead plus 2] = A;
	A = [fpqbad];  [fpqhead plus 3] = A;
	A = [XREJ];    [fpqhead plus 4] = A;
	[fpqhead plus 5] = NBAT;
	[fpqhead plus 6] = NSTAR;
	[fpqhead plus 7] = 0DEFACEDh;

	=> FLeave;

	[File Name]	= fpqresult;
	[File Position]	= 0;
	[File Command]	= WRITE;
	[Block Pointer]	= fpqhead;
	[Block Size]	= HDR;
	[Block Size]	* BYTES PER UNIT;
	isocall;

	[File Name]	= fpqresult;
	[File Position]	= HDR;
	[File Position] * BYTES PER UNIT;
	[File Command]	= WRITE;
	[Block Pointer]	= fpqouts;
	[Block Size]	= 16776;
	[Block Size]	* BYTES PER UNIT;
	isocall;

	[File Name]	= fpqresult;
	[File Position]	= 0;
	[File Command]	= SET SIZE;
	[File Size]	= 16776;
	[File Size]	+ HDR;
	[File Size]	* BYTES PER UNIT;
	isocall;

	end;

      ( fpqsoftid - NOCTIS-0.CPP:4078, transcribed mnemonic for mnemonic
	against the soft stack.  Compare it against the fpsched entry for
	NsIdentity: same nine steps, same order, one store.

	    fild x / fidiv K / fild y / fmulp / fidiv K
	                     / fild z / fmulp / fidiv K / fstp id

	The stack depth is checked after every star, not once at the end,
	because a chain that leaks a register leaks it 4194 times and the
	symptom of that is a wrong answer somewhere in the middle rather
	than a crash at the end. )

"fpqsoftid"

	( NsIdentity here is GENERATED by tools/genfp.py --backend soft
	  from the SAME fpsched.txt entry that produced the x87 bytes in
	  fpchains.txt.  One schedule, two emitters, two completely
	  different machines.  Nothing about the sequence is retyped, so
	  the two cannot disagree about the schedule - only about the
	  arithmetic, which is the thing being tested. )

	=> NsIdentity;

      ? [XDEPTH] = 0 -> fpqbalanced;
	[fpqbad]+;
    "fpqbalanced"
	end;
