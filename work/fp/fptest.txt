      ( *** fptest - the generic vector driver ***

	Reads fpvec.bin, runs ONE schedule over every case through the
	linked backend, writes fpout.bin.  Which backend is linked is a
	build-time choice and this file does not know or care: it calls
	FAdd, FQuo, FToIntChop and the rest, and those names resolve to
	fpx87 here and would resolve to fpnative in a render build.

	EVERY CASE EXERCISES THE CONVERSIONS, not only the operation.  The
	result of the schedule is put through FToIntChop, FToIntNear,
	FToInt16Chop and FCmp and all four go in the output record.  That
	is deliberate: the conversions are where the two float-to-int
	behaviours of the original live, the chop one is a SEED and the
	near one is a PIXEL, and a test that only checked the arithmetic
	would leave the more dangerous half unmeasured.

	Schedules other than the transcendentals are graded byte-for-byte against
	fpref.exe, a gcc-built program executing the same x87 operations.  Schedules
	8-10 use fptransgrade.py instead: a 180-digit Decimal oracle grades the
	mathematical result within one final binary64 ULP and derives conversions,
	flags, CW and TOP independently.  fpref still runs for those schedules, but
	x87 FSIN/FCOS agreement is characterization because their documented 66-bit
	pi reduction loses accuracy on large inputs.

	Schedule ids:
	   1 FAdd    2 FSub   3 FMul   4 FQuo   5 FSqrt of |a|
	   6 FNeg    7 FAbs   8 FSin   9 FCos  10 FAtan2
	  11 FCmp only        12 IntToF        13 F32Narrow
	  20 NsIdentity chain 21 Prod4 chain   22 Prod4Spilled chain     )

"libraries"

	fpabi;
	fpctl;
	fpx87;
	fpconv;
	fpchains;

"directors"

	program name = { fptest };
	unit = 32;

"constants"

	NCASE	= 4096;
	CASEU	= 16;
	OUTU	= 8;
	HDR	= 8;
	INUNITS	= 65544;		( HDR plus NCASE multiplied CASEU )
	OUTUNITS = 32776;		( HDR plus NCASE multiplied OUTU  )

"variables"

	fptvecname = { fpvec.bin };
	fptoutname = { fpout.bin };

	fptsched = 0;
	fptcase	= 0;
	fptiptr	= 0;
	fptoptr	= 0;
	fptic	= 0;
	fptin	= 0;
	fpti16	= 0;
	fptcmp	= 0;
	fptflags = 0;
	fptncase = 0;

"workspace"

	fptin buf  = INUNITS;
	fptout buf = OUTUNITS;

"programme"

	( ---------------- read the vectors ---------------- )

	[File Name]	= fptvecname;
	[File Position]	= 0;
	[File Command]	= READ;
	[Block Pointer]	= fptin buf;
	[Block Size]	= INUNITS;
	[Block Size]	* BYTES PER UNIT;
	isocall;

	( Check the magic and the shape BEFORE running anything.  A short
	  read silently corrects [Block Size], so trusting the header is
	  the only way to notice. )

	A = [fptin buf plus 0];
      ? A != 46505643h -> fpt badmagic;
	A = [fptin buf plus 1];
      ? A != 1 -> fpt badmagic;
	A = [fptin buf plus 3];
      ? A != CASEU -> fpt badmagic;
	A = [fptin buf plus 2]; [fptncase] = A;
      ? A != NCASE -> fpt badmagic;

	A = [fptin buf plus 4]; [fptsched] = A;
	A = [fptin buf plus 5]; [FCW] = A;
	=> FEnter;

	A = fptin buf; A + HDR;   [fptiptr] = A;
	A = fptout buf; A + HDR;  [fptoptr] = A;
	[fptcase] = 0;

    "fpt one"

	E = [fptiptr];
	A = [E plus 0];  [FA0] = A;
	A = [E plus 1];  [FA1] = A;
	A = [E plus 2];  [FB0] = A;
	A = [E plus 3];  [FB1] = A;
	A = [E plus 4];  [FC0] = A;
	A = [E plus 5];  [FC1] = A;
	A = [E plus 6];  [FD0] = A;
	A = [E plus 7];  [FD1] = A;
	A = [E plus 8];  [FJ0] = A;
	A = [E plus 9];  [FJ1] = A;
	A = [E plus 10]; [FJ2] = A;
	A = [E plus 11]; [FJ3] = A;

	=> fpt run;

	( ---- the conversions, on the result of every case ---- )

	=> FToIntChop;
	A = [FI]; [fptic] = A;

	=> FToIntNear;
	A = [FI]; [fptin] = A;

	A = [fptic]; [FI] = A;
	=> FNarrow16;
	A = [FI]; [fpti16] = A;

	=> FCmp;
	A = [FI]; [fptcmp] = A;

	( ---- flags ---- )

	[fptflags] = 0;
	A = [FA1]; A > 20; A & 7FFh;
      ? A = 0    -> fpt reject;
      ? A = 7FFh -> fpt reject;
	-> fpt flagged;
    "fpt reject"
	A = [fptflags]; A | 1; [fptflags] = A;
    "fpt flagged"

	=> FStackOK;
      ? [FI] = 0 -> fpt balanced;
	A = [fptflags]; A | 2; [fptflags] = A;
    "fpt balanced"

	( ---- store ---- )

	E = [fptoptr];
	A = [FA0];	[E plus 0] = A;
	A = [FA1];	[E plus 1] = A;
	A = [fptic];	[E plus 2] = A;
	A = [fptin];	[E plus 3] = A;
	A = [fpti16];	[E plus 4] = A;
	A = [fptcmp];	[E plus 5] = A;
	A = [fptflags];	[E plus 6] = A;
	[E plus 7] = 5A5A5A5Ah;

	A = [fptiptr]; A + CASEU; [fptiptr] = A;
	A = [fptoptr]; A + OUTU;  [fptoptr] = A;
	[fptcase]+;
      ? [fptcase] < NCASE -> fpt one;

	( ---------------- header ---------------- )

	[fptout buf plus 0] = 46504F54h;	( 'FPOT' )
	[fptout buf plus 1] = 1;
	A = [fptncase];	[fptout buf plus 2] = A;
	[fptout buf plus 3] = OUTU;
	[fptout buf plus 4] = 1;		( backend id 1 = X87 )
	=> FCWRead;
	A = [FI];	[fptout buf plus 5] = A;
	=> FSWRead;
	A = [FI];	[fptout buf plus 6] = A;
	[fptout buf plus 7] = 0DEFACEDh;

	=> FLeave;

	[File Name]	= fptoutname;
	[File Position]	= 0;
	[File Command]	= WRITE;
	[Block Pointer]	= fptout buf;
	[Block Size]	= OUTUNITS;
	[Block Size]	* BYTES PER UNIT;
	isocall;

	[File Name]	= fptoutname;
	[File Position]	= 0;
	[File Command]	= SET SIZE;
	[File Size]	= OUTUNITS;
	[File Size]	* BYTES PER UNIT;
	isocall;

	end;

    "fpt badmagic"

	( Write a one-unit file that is obviously not a result file, so a
	  grader that forgot to check the magic still cannot mistake it
	  for data. )

	[fptout buf plus 0] = 0BADBAD0h;
	[File Name]	= fptoutname;
	[File Position]	= 0;
	[File Command]	= WRITE;
	[Block Pointer]	= fptout buf;
	[Block Size]	= 1;
	[Block Size]	* BYTES PER UNIT;
	isocall;
	[File Name]	= fptoutname;
	[File Position]	= 0;
	[File Command]	= SET SIZE;
	[File Size]	= 1;
	[File Size]	* BYTES PER UNIT;
	isocall;
	end;

( ==================================================================== )

      ( fpt run - dispatch.  FA holds the first f64 operand on entry and
	the result on exit, which is the ABI every backend agrees on. )

"fpt run"
      ? [fptsched] = 1  -> fptr add;
      ? [fptsched] = 2  -> fptr sub;
      ? [fptsched] = 3  -> fptr mul;
      ? [fptsched] = 4  -> fptr quo;
      ? [fptsched] = 5  -> fptr sqrt;
      ? [fptsched] = 6  -> fptr neg;
      ? [fptsched] = 7  -> fptr abs;
      ? [fptsched] = 8  -> fptr sin;
      ? [fptsched] = 9  -> fptr cos;
      ? [fptsched] = 10 -> fptr atan;
      ? [fptsched] = 11 -> fptr none;
      ? [fptsched] = 12 -> fptr itof;
      ? [fptsched] = 13 -> fptr narrow;
      ? [fptsched] = 20 -> fptr nsid;
      ? [fptsched] = 21 -> fptr prod4;
      ? [fptsched] = 22 -> fptr prod4s;
	end;
    "fptr add"	=> FAdd;	end;
    "fptr sub"	=> FSub;	end;
    "fptr mul"	=> FMul;	end;
    "fptr quo"	=> FQuo;	end;
    "fptr sqrt"	=> FAbs; => FSqrt;	end;
    "fptr neg"	=> FNeg;	end;
    "fptr abs"	=> FAbs;	end;
    "fptr sin"	=> FSin;	end;
    "fptr cos"	=> FCos;	end;
    "fptr atan"	=> FAtan2;	end;
    "fptr none"	end;
    "fptr itof"	A = [FJ0]; [FI] = A; => IntToF;	end;
    "fptr narrow" => F32Narrow;	end;
    "fptr nsid"	=> NsIdentity;	end;
    "fptr prod4" => Prod4;	end;
    "fptr prod4s" => Prod4Spilled;	end;
