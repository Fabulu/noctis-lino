      ( *** fpctl - portable production FP environment boundary ***

        Production arithmetic is ordinary Lino integer code.  The few remaining
        Lino floating instructions are deliberate binary32 operations and run
        under the fixed nearest/extended environment installed by the platform
        runtime before application code starts.  No target-machine byte belongs
        in this shipping library.

        The application owns that environment for its process lifetime, so
        FEnter/FLeave retain the historical call surface without pretending to
        save or restore an ambient host word.  A request for any word other than
        FCWEXT fails instead of silently claiming that portable source changed
        hardware state.  Historical x87 characterization uses fpctlx87.txt in a
        private test sandbox. )

"programme"

"FReset"
        [FCW] = FCWEXT;
        [FCWSAV] = FCWEXT;
        [FCWTMP] = FCWEXT;
        [FSW] = 0;
        end;

"FEnter"
      ? [FCW] = FCWEXT -> FPENV enter;
        fail;
    "FPENV enter"
        [FCWSAV] = FCWEXT;
        end;

"FLeave"
        [FCW] = FCWEXT;
        end;

"FLoadCW"
      ? [FCW] = FCWEXT -> FPENV loaded;
        fail;
    "FPENV loaded"
        end;

"FCWRead"
        [FI] = 033Fh;
        end;

"FSWRead"
        [FI] = 0;
        end;

"FStackOK"
        [FI] = 0;
        end;
