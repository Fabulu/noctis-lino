      ( *** game - the FINAL deliverable: full-game integration driver ***

	One L.in.oleum programme that links EVERY ported library and runs
	the Noctis-IV main loop (NOCTIS.CPP:2268-4485): workspace + palette
	init, clear, ground generation, surface blit, LIVE flight dynamics
	(ap_drive_mode vimana cascade + sector chop + nearstar_identity,
	driven from per-frame game state - not the corpus), present (expand +
	RETRACE), input, ESC-to-quit.  The window opens, a planetary surface
	is generated by grnd and drawn, the ship flies (position evolves,
	lithium drains, status transitions through the 5-window cascade),
	and ESC exits.

	WHAT LINKS.  The pure libraries of all eight waves compose here:
	  fp/fpabi fpctl fpx87 fpconv fpchains   - the float engine (W3)
	  fbmem fbpal fbtick fbshell             - 320x200 display + present
	  pgfp pgmem pgrast pgtex pgproj         - 3D polymap pipeline (W6)
	  spmem spscale spmap spglobe spglow spbg spwhite spdark spncc - spheres
	  mul64frag brtl                         - integer support (W7)
	  suseed surng subuf susm supaint supal sucase - surface() (W7a)
	  grnd                                   - ground painters (W7b)
	  mgmain nsident mgloop mgnav mgin       - flight/nav/input (W8)

	WHAT COLLIDES.  L.in.oleum is case-insensitive with no linker
	namespacing, so cross-wave identifier clashes must be prefixed by
	hand.  The collisions found and fixed for this link (symcollide.py
	lists them; renames.py applies them):
	  - supaint SP{ i j k t u save b2 } -> SUP*  (SP = surface paint,
	    collided 3-way with spmem's sphere SP and pgtex's span SP)
	  - sp-family SP{ i j k } -> SPM*  (yields to pgtex's span SPi,
	    6 files vs ~30 pgbrk callers)
	  - spncc PVn -> PVcnt, spscale SCt -> SCtmp  (contained renames)
	  - spmem SFTMP -> SFTMPR  (constant slot 303, case-collided with
	    suseed's SFtmp variable)
	  - surng SUbn -> SUbnc  (case-collided with fbtick's SUBN const)

	WHAT IS NOT LINKED.  walk / walkgrid / fragpage (W7b) and
	  svsave / svstarmap / clconsole (W8) are STANDALONE GRADERS: each
	  declares its own 'program name' and corpus harness, and walk +
	  walkgrid share an identical W* namespace that cannot coexist.
	  Their functionality is proven by BUILDING them (walk.exe,
	  fragpage.exe, svsave.exe, clconsole.exe) - not by listing them
	  here.  The ground render path the game uses is grnd + the pg
	  polymap pipeline, which is exactly the wiring fragpage proved.

	       C:\programmieren\linoleum\lino_build.ps1
	       -Src C:\programmieren\linoleum\work\game.txt
	Run:   powershell -ExecutionPolicy Bypass -File

	       -Out C:\programmieren\linoleum\work\game-out.bin         )

( ==================================================================== )
( GM run - the 22-phase main loop, NOCTIS.CPP:2268-4485                   )
(                                                                        )
( The do-while body, driven from LIVE per-frame game state.  Each phase   )
( calls a routine PROVEN by its own wave's grader (fbmain / grmain /     )
( spmain / pgmain / mgmain), so the composition is what is new - the     )
( kernels are the byte-exact ones.                                       )
(                                                                        )
(   P0    workspace + palette + tick init                                )
(   P1    clear the visible page                                         )
(   P2    ground: generate the planetary surface (grnd, once)            )
(   P3    blit the surface texture into the visible page (once)          )
(   Pflight seed live flight state, snapshot frame-0 dzat/pwr            )
(   ---- EVOLUTION: a tight isocall-free loop (the mgmain regime) ----   )
(   PF1   flight dynamics: ONE ap_drive_mode frame via MG vimana loop    )
(         (5-window coefficient cascade; updates dzat,pwr,cur,reached,   )
(          stspeed,status - the graded per-frame kernel, no corpus)      )
(   PF2   nav: sector chop + nearstar_identity via NsIdentityD           )
(   ---- the loop runs GMNFRAMES, then hands off to render + input ----  )
(   PF3   HUD: pwr/status bar onto adaptor                               )
(   PF4   present: expand adaptor->fb through pal + RETRACE              )
(   PF5   telemetry sentinel (frame-0 vs live dzat/pwr)                  )
(   PF6   ESC loop: present + GET CONSOLE INPUT + KEY ESCAPE             )
(                                                                        )
( Why the evolution is isocall-free: the win32 stub re-mounts the x87    )
( control word AND leaves operands on the x87 stack across every isocall )
( (RETRACE / GET CONSOLE INPUT / timer), so an FP op immediately after   )
( an isocall faults.  The mgmain grader never interleaves them; the game )
( keeps the same discipline - all GMNFRAMES flight frames run pure-FP,   )
( then the isocall-bound render/input phase runs with no FP needs.       )
( ==================================================================== )

"GM run"
	=> FEnter;

	( ---- P0 ---- )
	=> MEM build rtab;
	=> MEM build ztab;
	=> MEM wrap reset;
	=> PAL float init;
	=> FB palette pinned;
	=> TK seed;

	( ---- P2 / P1 / P3: ground once, then page ---- )
	=> GM ground;
	=> GM clear page;
	=> GM blit surface;

	( ---- seed live flight state, snapshot frame 0 ---- )
	=> GM flight init;
	[GMix0] = [MgDzatX0]; [GMix1] = [MgDzatX1]; [GMipwr] = [MgPwr];

	[Display Origin] = fb;
	[Display Live Region] = WHOLE DISPLAY;
	[GMframe] = 0; [GMsent] = 0; [GMesc] = 0;

    "GM loop"
	( The vimana cascade and the NsIdentity chain are pure x87; the win32
	  stub re-mounts the x87 control word AND leaves junk on the x87 stack
	  across every isocall (RETRACE / GET CONSOLE INPUT / tick), so an FP
	  op immediately after an isocall faults.  The evolution therefore
	  runs as a TIGHT isocall-free loop - the same regime the mgmain
	  grader runs in - and only the render/ESC phase mixes in isocalls. )
	=> FLoadCW;
	( ---- PF1: one graded ap_drive_mode frame ---- )
	=> GM flight step;
	( ---- PF2: navigation from live dzat ---- )
	=> GM sector step;
	=> GM identity step;
	[GMframe]+;
	A = [GMframe]; ? A '< GMNFRAMES -> GM loop;

	( ---- state has now evolved GMNFRAMES frames; render + hand off ---- )
	=> GM draw hud;
	=> FB present;
	=> GM write sentinel;
    "GM esc loop"
	( ---- present + input continuation; no FP here, so isocalls are safe )
	=> FB present;
	=> GM input;
	A = [GMesc]; ? A = 0 -> GM esc loop;

	[Display Live Region] = VOID REGION;
	=> FLeave;
	end;

( ==================================================================== )
( PFlight init - seed the integrator state the graded kernels read.       )
(   The Mg* variables ARE the live game state; vimana updates them in     )
(   place each frame.                                                    )
( ==================================================================== )

"GM flight init"
	[MgDzatX0] = 0;       [MgDzatX1] = GMDZHI;
	[MgDzatY0] = 0;       [MgDzatY1] = 0;
	[MgDzatZ0] = 0;       [MgDzatZ1] = 0;
	[MgApX0] = 0;         [MgApX1] = 0;
	[MgApY0] = 0;         [MgApY1] = 0;
	[MgApZ0] = 0;         [MgApZ1] = 0;
	[MgInitD0] = 0;       [MgInitD1] = GMINITHI;
	[MgCur0] = 0;         [MgCur1] = GMCURHI;
	[MgTmp] = GMRAYF;
	C = [MgTmp]; => MG widen f32 FA from C;
	[MgT00] = [FA0]; [MgT01] = [FA1];
	[MgAnti] = 0;  [MgAptgt] = 1;  [MgPwr] = GMSTARTPWR;
	[MgApreached] = 0; [MgStspeed] = 1; [MgStatus] = 0;
	[FI] = 100000; => IntToF; [MgskA] = [FA0]; [MgskB] = [FA1];
	end;

( ==================================================================== )
( PF1 - one ap_drive_mode frame, via the graded MG vimana loop.           )
(   MGD reset bounds the per-frame emit so a long session cannot walk the )
(   dump off the end of mgdump.                                          )
( ==================================================================== )

"GM flight step"
	=> MGD reset;
	[MgNframes] = 1;
	=> MG vimana loop;
	end;

( ==================================================================== )
( PF2a - sector chop, absolute X axis:                                    )
(   sect_x = (long)(dzat_x / 100000)   [__ftol chops toward zero]         )
( ==================================================================== )

"GM sector step"
	[FA0]=[MgDzatX0];[FA1]=[MgDzatX1]; [FB0]=[MgskA];[FB1]=[MgskB]; => FQuo;
	=> FToIntChop;
	[GMsecxi] = [FI];
	end;

( ==================================================================== )
( PF2b - nearstar_identity of the current sector via the graded chain.    )
(   NsIdentityD(x,y,z) = ((((x/100000)*y)/100000)*z)/100000, the 5-op     )
(   left-associated x87 chain Wave 4 pinned 4113/4113 on STARMAP.BIN.     )
( ==================================================================== )

"GM identity step"
	[FJ0] = [GMsecxi]; [FJ1] = 0; [FJ2] = 0;
	=> NsIdentityD;
	[GMnsid0] = [nsid0]; [GMnsid1] = [nsid1];
	end;

( ==================================================================== )
( PF3 - HUD: a pwr/status bar across the top-left of adaptor so the frame  )
(   is observably evolving as vimana integrates.  bar = (pwr-15000)/1000   )
(   pixels (clamped 0..40), lit in a status-derived palette index.         )
( ==================================================================== )

"GM draw hud"
	A = [MgPwr]; A - 15000; A '/ 1000; [GMbar] = A;
	A = [GMbar]; ? A '> 40 -> GM hud clamp;
	-> GM hud barpos;
    "GM hud clamp"
	[GMbar] = 40;
    "GM hud barpos"
	[GMptr] = 0;
    "GM hud clr"
	A = nw; A + RADPR; A + [GMptr]; C = 0; [A] = C;
	[GMptr]+; ? [GMptr] '< 40 -> GM hud clr;
	[GMptr] = 0;
    "GM hud lit"
	A = [GMptr]; ? A '>= [GMbar] -> GM hud done;
	C = [MgStatus]; C & 255; C '* 8; C + 64; C & 255;
	B = nw; B + RADPR; B + [GMptr]; [B] = C;
	[GMptr]+; -> GM hud lit;
    "GM hud done"
	end;

( ==================================================================== )
( PF5 - input.  ESC (the held-key table) or an ESC ASCII byte exits; any  )
(   other key is latched into GMkey for the (stubbed) console dispatch.   )
( ==================================================================== )

"GM input"
	[Console Command] = GET CONSOLE INPUT; isocall;
	A = [Console Input]; ? A = 0 -> GM in noascii;
	[GMkey] = A;
	A - 27; ? A = 0 -> GM in esc;
    "GM in noascii"
	A = [KEY ESCAPE]; ? A = OFF -> GM in nokey;
    "GM in esc"
	[GMesc] = 1;
    "GM in nokey"
	end;

( ==================================================================== )
( Telemetry sentinel - 16 units to game-out.bin so w7arun.ps1 sees a      )
(   stable artifact.  Captures frame-0 dzat_x/pwr next to the LIVE        )
(   dzat_x/pwr, so initial != final proves the integrator advanced state. )
( ==================================================================== )

"GM write sentinel"
	A = gmtele;
	[A plus 0]  = [GMix0];     [A plus 1]  = [GMix1];
	[A plus 2]  = [MgDzatX0];  [A plus 3]  = [MgDzatX1];
	[A plus 4]  = [GMipwr];    [A plus 5]  = [MgPwr];
	[A plus 6]  = [MgStatus];  [A plus 7]  = [GMframe];
	[A plus 8]  = [GMsecxi];   [A plus 9]  = [GMnsid0]; [A plus 10] = [GMnsid1];
	[A plus 11] = [MgCur0];    [A plus 12] = [MgCur1];
	[A plus 13] = [MgApreached]; [A plus 14] = [MgStspeed]; [A plus 15] = 0;
	[File Name]     = gmoutname;
	[File Position] = 0;
	[File Command]  = WRITE;
	[Block Pointer] = gmtele;
	[Block Size]    = 16;
	[Block Size] * BYTESPERUNIT;
	isocall;
	[File Name]     = gmoutname;
	[File Position] = 0;
	[File Command]  = SET SIZE;
	[File Size]     = 16;
	[File Size] * BYTESPERUNIT;
	isocall;
	end;

( ==================================================================== )
( P2 - ground: the grmain-proven surface() generation.                    )
(   GR prologue (12 brtl draws) + reseed + rockyground + plains noise,    )
(   leaving a 200x200 planetary surface in p_surfacemap.  grmain grades   )
(   this exact sequence three-way byte-exact, so the bytes on screen are  )
(   the bytes the oracle signed off on.                                   )
( ==================================================================== )

"GM ground"
	=> SU ledger reset;
	=> GR float init;

	( wipe p_surfacemap, the texture the painters write )
	[GMptr] = 0;
    "GM gnd wsm"
	A = nw; A + RPSM; A + [GMptr]; C = 0; [A] = C;
	[GMptr]+;
	? [GMptr] '< 40000 -> GM gnd wsm;
	( wipe objectschart )
	[GMptr] = 0;
    "GM gnd wob"
	A = nw; A + ROBJ; A + [GMptr]; C = 0; [A] = C;
	[GMptr]+;
	? [GMptr] '< 40000 -> GM gnd wob;

	( the pinned grmain case: seed 123456, type 1 rocky, albedo 20, lat 30 )
	[GRgseed]  = 123456;
	[GRiptype] = 1;
	[GRsctype] = 0;
	[GRalbedo] = 20;
	[GRlat]    = 30;
	=> GR prologue;

	( re-seed both generators from gseed, exactly as grmain does )
	A = [GRgseed]; => SU fast srand;
	A = [GRgseed]; A & 0FFFFh; => SU srand;

	[GRrough] = 25;
	[GRround] = 4;
	[GRlevel] = 0;
	=> GR rockyground;
	=> GR plains noise;
	end;

( ==================================================================== )
( P1 - clear the visible page (adaptor) over the 320x200 window.          )
( ==================================================================== )

"GM clear page"
	[GMptr] = 0;
    "GM cp loop"
	A = nw; A + RADPR; A + [GMptr]; C = 0; [A] = C;
	[GMptr]+;
	? [GMptr] '< 64000 -> GM cp loop;
	end;

( ==================================================================== )
( P3 - blit the 200x200 surface texture into the top of the visible page. )
(   p_surfacemap[0..39999] -> adaptor[0..39999]; the present path expands )
(   adaptor through the palette LUT to the framebuffer.                  )
( ==================================================================== )

"GM blit surface"
	[GMptr] = 0;
    "GM bs loop"
	A = nw; A + RPSM; A + [GMptr]; C = [A]; C & 255;
	B = nw; B + RADPR; B + [GMptr]; [B] = C;
	[GMptr]+;
	? [GMptr] '< 40000 -> GM bs loop;
	end;
