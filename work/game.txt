      ( *** game - the FINAL deliverable: full-game integration driver ***

	One L.in.oleum programme that links EVERY ported library and runs
	the Noctis-IV main loop (NOCTIS.CPP:2268-4485): workspace + palette
	init, ground generation, LIVE flight dynamics (ap_drive_mode vimana

	  fp/fpabi fpctl fpx87 fpconv fpchains   - the float engine (W3)

	    suseed's SFtmp variable)

( ==================================================================== )
( GM fpu clean - the win32 stub leaves operands on the x87 stack across   )
(   every isocall; FLoadCW only restores the control word, not the stack, )
(   so the next fld overflows.  fninit empties the stack and zeroes the   )
(   status word; the following fldcw re-installs [FCW]=133Fh (PC=64,      )
(   RC=nearest) that FEnter established once at startup.  Call this       )
(   before every FP-heavy phase that follows an isocall.                  )
( ==================================================================== )

"GM fpu clean"
	=> FReset;
	end;

( ==================================================================== )
( GM run - the 22-phase main loop, NOCTIS.CPP:2268-4485                   )
(                                                                        )
( The do-while body, driven from LIVE per-frame game state.  Each phase   )
( calls a routine PROVEN by its own wave's grader (fbmain / grmain /     )
( spmain / pgmain / mgmain), so the composition is what is new - the     )
( kernels are the byte-exact ones.                                       )
(                                                                        )
(   P0     workspace + palette + tick init                               )
(   P2     ground: generate the planetary surface (grnd, once)           )
(   Pfl    seed live flight state, snapshot frame-0 dzat/pwr             )
(   ---- PER-FRAME continuation (one iteration = one game frame) ----    )
(   PF1    flight dynamics: ONE ap_drive_mode frame via MG vimana loop   )
(          (5-window coefficient cascade; updates dzat,pwr,cur,reached,  )
(           stspeed,status - the graded per-frame kernel, no corpus)     )
(   PF2    nav: sector chop + nearstar_identity via NsIdentityD          )
(   P3     3D render: clear, polymap the live surface from the live      )
(          camera (viewtile pans with dzat), pcopy ADAPTED->ADAPTOR, HUD )
(   PF4    present: expand ADAPTOR->fb through pal + RETRACE             )
(   PF5    telemetry sentinel (written once, early)                      )
(   PF6    input: GET CONSOLE INPUT + KEY ESCAPE (every frame -> ESC is  )
(          responsive within one frame)                                  )
(                                                                        )
( Why GM fpu clean leads every iteration: the win32 stub re-mounts the   )
( x87 control word AND leaves operands on the x87 stack across every     )
( isocall (RETRACE / GET CONSOLE INPUT), so an FP op after an isocall    )
( faults.  fninit + fldcw at the top re-empties the stack and restores   )
( [FCW]=133Fh, so this frame's flight + polymap FP runs clean; the       )
( present/input isocalls are at the bottom, after all FP.                )
( ==================================================================== )

"GM run"
	=> FEnter;

	( ---- P0 ---- )
	=> MEM build rtab;
	=> MEM build ztab;
	=> MEM wrap reset;
	=> PAL float init;
	=> FB palette pinned;
	=> TK seed;

	( ---- P2: ground - generate the planetary surface once ---- )
	=> GM fpu clean;
	=> GM ground;

	( ---- seed live flight state, snapshot frame 0 ---- )
	=> GM fpu clean;
	=> GM flight init;
	[GMix0] = [MgDzatX0]; [GMix1] = [MgDzatX1]; [GMipwr] = [MgPwr];

	[Display Origin] = fb;
	[Display Live Region] = WHOLE DISPLAY;
	[GMframe] = 0; [GMsent] = 0; [GMesc] = 0;

    "GM loop"
	( PER-FRAME interleave: one FP frame, then one render+present+input.
	  GM fpu clean at the top re-empties the x87 stack and re-installs
	  [FCW]=133Fh after the PREVIOUS iteration's isocalls (present/input),
	  so this iteration's FP (flight + polymap) runs clean.  Input is
	  drained every frame, so ESC is responsive within one frame. )
	=> GM fpu clean;
	( ---- PF1: one graded ap_drive_mode frame ---- )
	=> GM flight step;
	( ---- PF2: navigation from live dzat ---- )
	=> GM sector step;
	=> GM identity step;
	[GMframe]+;
	( ---- render the 3D ground view from the live camera + live surface ---- )
	=> GM clear page;
	=> GM render ground;
	=> GM copy page;
	=> GM draw hud;
	=> FB present;
	( ---- telemetry sentinel: written once, early, then the loop continues )
	A = [GMsent]; ? A != 0 -> GM after sent;
	A = [GMframe]; ? A '< GMSENTFRAME -> GM after sent;
	=> GM write sentinel;
	[GMsent] = 1;
    "GM after sent"
	( ---- PF5/PF6: input + the ESC-to-quit continuation (every frame) ---- )
	=> GM input;
	A = [GMesc]; ? A = 0 -> GM loop;

	[Display Live Region] = VOID REGION;
	=> FLeave;
	end;

( ==================================================================== )
( PFlight init - seed the integrator state the graded kernels read.       )
(   The Mg* variables ARE the live game state; vimana updates them in     )
(   place each frame.                                                    )

"GM flight init"
	[MgDzatX0] = 0;       [MgDzatX1] = GMDZHI;
	[MgDzatY0] = 0;       [MgDzatY1] = 0;
	[MgDzatZ0] = 0;       [MgDzatZ1] = 0;
	[MgApX0] = 0;         [MgApX1] = 0;
	[MgApY0] = 0;         [MgApY1] = 0;
	[MgApZ0] = 0;         [MgApZ1] = 0;
	[MgInitD0] = 0;       [MgInitD1] = GMINITHI;
	[MgCur0] = 0;         [MgCur1] = GMCURHI;
	[MgTmp] = GMRAYF;
	C = [MgTmp]; => MG widen f32 FA from C;
	[MgT00] = [FA0]; [MgT01] = [FA1];
	[MgAnti] = 0;  [MgAptgt] = 1;  [MgPwr] = GMSTARTPWR;
	[MgApreached] = 0; [MgStspeed] = 1; [MgStatus] = 0;
	[FI] = 100000; => IntToF; [MgskA] = [FA0]; [MgskB] = [FA1];
	end;

( ==================================================================== )
( PF3 - HUD: a pwr/status bar across the top-left of adaptor so the frame  )
(   is observably evolving as vimana integrates.  bar = (pwr-15000)/1000   )
(   pixels (clamped 0..40), lit in a status-derived palette index.         )
( ==================================================================== )

"GM draw hud"
	A = [MgPwr]; A - 15000; A '/ 1000; [GMbar] = A;
	A = [GMbar]; ? A '> 40 -> GM hud clamp;
	-> GM hud barpos;
    "GM hud clamp"
	[GMbar] = 40;
    "GM hud barpos"
	[GMptr] = 0;
    "GM hud clr"
	A = nw; A + RADPR; A + [GMptr]; C = 0; [A] = C;
	[GMptr]+; ? [GMptr] '< 40 -> GM hud clr;
	[GMptr] = 0;
    "GM hud lit"
	A = [GMptr]; ? A '>= [GMbar] -> GM hud done;
	C = [MgStatus]; C & 255; C '* 8; C + 64; C & 255;
	B = nw; B + RADPR; B + [GMptr]; [B] = C;
	[GMptr]+; -> GM hud lit;
    "GM hud done"
	end;

( ==================================================================== )
( PF5 - input.  ESC (the held-key table) or an ESC ASCII byte exits; any  )

"GM input"

	A - 27; ? A = 0 -> GM in esc;
    "GM in noascii"
	A = [KEY ESCAPE]; ? A = OFF -> GM in nokey;
    "GM in esc"
	[GMesc] = 1;
    "GM in nokey"
	end;

( P2 - ground: the grmain-proven surface() generation.                    )

	=> GR float init;

	[GMptr] = 0;

	A = nw; A + ROBJ; A + [GMptr]; C = 0; [A] = C;

	? [GMptr] '< 40000 -> GM gnd wob;

( ==================================================================== )
( P-render - the 3D ground view via PG polymap (the fragpage-proven        )
(   wiring), driven from the LIVE camera and LIVE p_surfacemap.            )
(                                                                         )
(   camera: cam_x pans with dzat_x (viewtile = chop(dzat_x)>>14, wrapped  )
(           into 4..195 so a 9-tile grid stays inside the 200x200 map);   )
(           cam_y = +GMCAMY (above the surface - grnd generates LOW      )
(           terrain, surf 6..13 -> vy ~ -26000, so the eye must sit at    )
(           less-negative y or every tile lands behind the camera and is  )
(           depth-culled); cam_z = surface center (tile 100).             )
(   view:   alfa = -90 deg top-down (fragpage's exact, bit-pinned angle   )
(           constants), so the projection is the graded one.              )
(   grid:   viewtile +/- GMHALF x 100 +/- GMHALF, two triangles per tile, )
(           shade c1 = surf[h1] & 31, TEXTURED via PGtexf = 1 + SPsrc = 1 )
(           (the live float chain derives per-pixel (u,v) from the        )
(           projected position; TEX1 is the procedural formula).          )
( ==================================================================== )

"GM render ground"
	=> GM fpu clean;
	=> PGF constants;

	( live camera X from dzat: viewtile = GMHALF + (chop(dzat_x)>>14) mod
	  (200-2*GMHALF), so the GMHALF-wide grid always stays inside 0..199. )
	[FA0]=[MgDzatX0];[FA1]=[MgDzatX1]; => FToIntChop;
	A = [FI]; A > 14; [GMtile] = A;
	A = 200; A - GMHALF; A - GMHALF; [GMtmp2] = A;	( 200 - 2*GMHALF )
	A = [GMtile]; A '% [GMtmp2]; A + GMHALF; [GMtile] = A;
	( cam_x = viewtile<<14 + 8192 )
	A = [GMtile]; A < 14; A + 8192; [GMcamx] = A;
	[FI] = [GMcamx];  => IntToF; [PGFi] = FSCAMX; => PGF sa;
	( eye height: grnd generates LOW terrain (surf 6..13 -> vy ~ -26000), so
	  the camera must sit ABOVE it (less negative y) or every tile is behind
	  the camera and depth-culled.  +GMCAMY puts the eye well above the surface. )
	A = 0; A + GMCAMY; [FI] = A; => IntToF; [PGFi] = FSCAMY; => PGF sa;
	[FI] = GMCAMZ;    => IntToF; [PGFi] = FSCAMZ; => PGF sa;

	( top-down view: fragpage's graded angle constants )
	[PGFi] = FSPCA; [PGFt] = GMCOSA0; => PGF setf32;
	[PGFi] = FSPSA; [PGFt] = GMSINA0; => PGF setf32;
	[PGFi] = FSTCA; [PGFt] = GMCOSA0; => PGF setf32;
	[PGFi] = FSTSA; [PGFt] = GMSINZ;  => PGF setf32;

	( textured path: SPsrc=1 makes the span engine derive (u,v) per pixel from
	  the projected position (the live float chain - PJ vectors sets FSK1/2/3),
	  and PGtexf=1 is the procedural TEX1 formula, so each pixel's shade varies
	  across the tile face instead of the flat TEX2 fill. )
	[PGtexf] = 1;  [SPsrc] = 1;  [SPflar] = 0;  [SPcull] = 0;  [SPhalf] = 0;

	( grid X: viewtile-GMHALF .. viewtile+GMHALF )
	A = [GMtile]; A - GMHALF; [GMxlo] = A;
	A = [GMtile]; A + GMHALF; [GMxhi] = A;
	[GMx] = [GMxlo];
    "GM rg x"
	[GMz] = GMZLO;
    "GM rg z"
	=> GM fragment tile;
	[GMz]+; A = [GMz]; ? A '<= GMZHI -> GM rg z;
	[GMx]+; A = [GMx]; ? A '<= [GMxhi] -> GM rg x;
	end;

( ==================================================================== )
( GM fragment tile - read the four LIVE corner surf bytes, set the per-tile )
(   shade, load the two triangles' vertices into FSINX/Y/Z, call PG polymap.)
(   Vertex vy = -(surf<<11); vx/vz = tile_coord<<14.  fragpage:153-198.    )
( ==================================================================== )

"GM fragment tile"
	A = [GMz]; A '* 200; A + [GMx]; [GMh1] = A;

	A = RPSM; A + [GMh1];          [MBptr] = A; => MEM get byte; [GMs1] = [MBval];
	A = RPSM; A + [GMh1]; A + 1;   [MBptr] = A; => MEM get byte; [GMs2] = [MBval];
	A = RPSM; A + [GMh1]; A + 200; [MBptr] = A; => MEM get byte; [GMs4] = [MBval];
	A = RPSM; A + [GMh1]; A + 201; [MBptr] = A; => MEM get byte; [GMs3] = [MBval];

	A = [GMs1]; A & 31; [GMc1] = A;
	[SPtinta] = [GMc1]; [SPescr] = 0;

	( triangle 1: v0=(x,s1,z) v1=(x+1,s2,z) v2=(x,s4,z+1) )
	[GMvi] = 0;
	A = [GMx];          A < 14; [GMvv] = A; [GMvslot] = FSINX; => GM vload;
	A = [GMz];          A < 14; [GMvv] = A; [GMvslot] = FSINZ; => GM vload;
	A = [GMs1]; A & 255; C = A; C < GMSHIFT; A = 0; A - C; [GMvv] = A; [GMvslot] = FSINY; => GM vload;
	[GMvi]+;
	A = [GMx]; A + 1;   A < 14; [GMvv] = A; [GMvslot] = FSINX; => GM vload;
	A = [GMz];          A < 14; [GMvv] = A; [GMvslot] = FSINZ; => GM vload;
	A = [GMs2]; A & 255; C = A; C < GMSHIFT; A = 0; A - C; [GMvv] = A; [GMvslot] = FSINY; => GM vload;
	[GMvi]+;
	A = [GMx];          A < 14; [GMvv] = A; [GMvslot] = FSINX; => GM vload;
	A = [GMz]; A + 1;   A < 14; [GMvv] = A; [GMvslot] = FSINZ; => GM vload;
	A = [GMs4]; A & 255; C = A; C < GMSHIFT; A = 0; A - C; [GMvv] = A; [GMvslot] = FSINY; => GM vload;
	[PJnrv] = 3; => PG polymap;

	( triangle 2: v0=(x+1,s2,z) v1=(x+1,s3,z+1) v2=(x,s4,z+1) )
	[GMvi] = 0;
	A = [GMx]; A + 1;   A < 14; [GMvv] = A; [GMvslot] = FSINX; => GM vload;
	A = [GMz];          A < 14; [GMvv] = A; [GMvslot] = FSINZ; => GM vload;
	A = [GMs2]; A & 255; C = A; C < GMSHIFT; A = 0; A - C; [GMvv] = A; [GMvslot] = FSINY; => GM vload;
	[GMvi]+;
	A = [GMx]; A + 1;   A < 14; [GMvv] = A; [GMvslot] = FSINX; => GM vload;
	A = [GMz]; A + 1;   A < 14; [GMvv] = A; [GMvslot] = FSINZ; => GM vload;
	A = [GMs3]; A & 255; C = A; C < GMSHIFT; A = 0; A - C; [GMvv] = A; [GMvslot] = FSINY; => GM vload;
	[GMvi]+;
	A = [GMx];          A < 14; [GMvv] = A; [GMvslot] = FSINX; => GM vload;
	A = [GMz]; A + 1;   A < 14; [GMvv] = A; [GMvslot] = FSINZ; => GM vload;
	A = [GMs4]; A & 255; C = A; C < GMSHIFT; A = 0; A - C; [GMvv] = A; [GMvslot] = FSINY; => GM vload;
	[PJnrv] = 3; => PG polymap;
	end;

( GM vload - store the integer GMvv as a binary64 into pg slot (GMvslot+GMvi)
  via IntToF + PGF sa.  Integer vertices are exact in float32, so binary64
  storage loses nothing. )

"GM vload"
	[FI] = [GMvv];
	=> IntToF;
	A = [GMvslot]; A + [GMvi]; [PGFi] = A;
	=> PGF sa;
	end;
