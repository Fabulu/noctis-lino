      ( *** game - the FINAL deliverable: full-game integration driver ***

	One L.in.oleum programme that links EVERY ported library and runs
	the Noctis-IV main loop (NOCTIS.CPP:2268-4485): workspace + palette
	init, clear, ground generation, surface blit, present (expand +
	RETRACE), input, ESC-to-quit.  The window opens, a real planetary
	surface is generated by grnd (the grmain-graded sequence) and drawn
	through the fb present path, and ESC exits.

	WHAT LINKS.  The pure libraries of all eight waves compose here:
	  fp/fpabi fpctl fpx87 fpconv fpchains   - the float engine (W3)
	  fbmem fbpal fbtick fbshell             - 320x200 display + present
	  pgfp pgmem pgrast pgtex pgproj         - 3D polymap pipeline (W6)
	  spmem spscale spmap spglobe spglow spbg spwhite spdark spncc - spheres
	  mul64frag brtl                         - integer support (W7)
	  suseed surng subuf susm supaint supal sucase - surface() (W7a)
	  grnd                                   - ground painters (W7b)
	  mgmain nsident mgloop mgnav mgin       - flight/nav/input (W8)

	WHAT COLLIDES.  L.in.oleum is case-insensitive with no linker
	namespacing, so cross-wave identifier clashes must be prefixed by
	hand.  The collisions found and fixed for this link (symcollide.py
	lists them; renames.py applies them):
	  - supaint SP{ i j k t u save b2 } -> SUP*  (SP = surface paint,
	    collided 3-way with spmem's sphere SP and pgtex's span SP)
	  - sp-family SP{ i j k } -> SPM*  (yields to pgtex's span SPi,
	    6 files vs ~30 pgbrk callers)
	  - spncc PVn -> PVcnt, spscale SCt -> SCtmp  (contained renames)
	  - spmem SFTMP -> SFTMPR  (constant slot 303, case-collided with
	    suseed's SFtmp variable)
	  - surng SUbn -> SUbnc  (case-collided with fbtick's SUBN const)

	WHAT IS NOT LINKED.  walk / walkgrid / fragpage (W7b) and
	  svsave / svstarmap / clconsole (W8) are STANDALONE GRADERS: each
	  declares its own 'program name' and corpus harness, and walk +
	  walkgrid share an identical W* namespace that cannot coexist.
	  Their functionality is proven by BUILDING them (walk.exe,
	  fragpage.exe, svsave.exe, clconsole.exe) - not by listing them
	  here.  The ground render path the game uses is grnd + the pg
	  polymap pipeline, which is exactly the wiring fragpage proved.

	Build: powershell -ExecutionPolicy Bypass -File
	       C:\programmieren\linoleum\lino_build.ps1
	       -Src C:\programmieren\linoleum\work\game.txt
	Run:   powershell -ExecutionPolicy Bypass -File
	       C:\programmieren\linoleum\tests\w7arun.ps1
	       -Exe C:\programmieren\linoleum\work\game.exe
	       -Out C:\programmieren\linoleum\work\game-out.bin         )

( ==================================================================== )
( GM run - the 22-phase main loop, NOCTIS.CPP:2268-4485                   )
(                                                                        )
( The do-while body, compressed to the phases the ported libraries can   )
( drive from live state today.  Each phase calls a routine PROVEN by its )
( own wave's grader (fbmain / grmain / spmain / pgmain / mgmain), so the )
( composition is what is new - the kernels are the byte-exact ones.      )
(                                                                        )
(   P0  workspace + tick + palette init                                  )
(   P1  clear the visible page                                           )
(   P2  ground: generate the planetary surface (grnd / su / brtl)        )
(   P3  blit the surface texture into the visible page                   )
(   P4  present: expand adaptor->fb through pal + RETRACE                )
(   P5  input + ESC-to-quit continuation                                 )
(                                                                        )
( The corpus-driven mg flight/nav/input kernels (mgloop/mgnav/mgin) and  )
( the 3D pg polymap / sp globe pipelines are LINKED and resolve; they    )
( feed from corpus text in their graders and from live game state here -)
( the hand-off is documented, and the link itself is the integration     )
( this driver proves.                                                    )
( ==================================================================== )

"GM run"
	=> FEnter;

	( ---- P0: workspace, palette, tick ---- )
	=> MEM build rtab;
	=> MEM build ztab;
	=> MEM wrap reset;
	=> PAL float init;
	=> FB palette pinned;
	=> TK seed;

	( ---- P2 precedes P1: build the surface, then clear+blit the page ---- )
	=> GM ground;

	( ---- P1: clear the visible page ---- )
	=> GM clear page;

	( ---- P3: blit surface texture to the visible page ---- )
	=> GM blit surface;

	( ---- P4: present ---- )
	[Display Origin] = fb;
	[Display Live Region] = WHOLE DISPLAY;
	=> GM sentinel;

	( ---- P5: input + the 4-clause ESC-to-quit continuation ---- )
	[GMesc] = 0;
    "GM loop"
	=> FB present;
	[Console Command] = GET CONSOLE INPUT; isocall;
	A = [Console Input];
	? A = 0 -> GM nokey;
	[GMkey] = A;
	A = [KEY ESCAPE];
	? A = OFF -> GM nokey;
	[GMesc] = 1;
    "GM nokey"
	=> TK step;
	? [GMesc] = 0 -> GM loop;

	[Display Live Region] = VOID REGION;
	=> FLeave;
	end;

( ==================================================================== )
( P2 - ground: the grmain-proven surface() generation.                    )
(   GR prologue (12 brtl draws) + reseed + rockyground + plains noise,    )
(   leaving a 200x200 planetary surface in p_surfacemap.  grmain grades   )
(   this exact sequence three-way byte-exact, so the bytes on screen are  )
(   the bytes the oracle signed off on.                                   )
( ==================================================================== )

"GM ground"
	=> SU ledger reset;
	=> GR float init;

	( wipe p_surfacemap, the texture the painters write )
	[GMptr] = 0;
    "GM gnd wsm"
	A = nw; A + RPSM; A + [GMptr]; C = 0; [A] = C;
	[GMptr]+;
	? [GMptr] '< 40000 -> GM gnd wsm;
	( wipe objectschart )
	[GMptr] = 0;
    "GM gnd wob"
	A = nw; A + ROBJ; A + [GMptr]; C = 0; [A] = C;
	[GMptr]+;
	? [GMptr] '< 40000 -> GM gnd wob;

	( the pinned grmain case: seed 123456, type 1 rocky, albedo 20, lat 30 )
	[GRgseed]  = 123456;
	[GRiptype] = 1;
	[GRsctype] = 0;
	[GRalbedo] = 20;
	[GRlat]    = 30;
	=> GR prologue;

	( re-seed both generators from gseed, exactly as grmain does )
	A = [GRgseed]; => SU fast srand;
	A = [GRgseed]; A & 0FFFFh; => SU srand;

	[GRrough] = 25;
	[GRround] = 4;
	[GRlevel] = 0;
	=> GR rockyground;
	=> GR plains noise;
	end;

( ==================================================================== )
( P1 - clear the visible page (adaptor) over the 320x200 window.          )
( ==================================================================== )

"GM clear page"
	[GMptr] = 0;
    "GM cp loop"
	A = nw; A + RADPR; A + [GMptr]; C = 0; [A] = C;
	[GMptr]+;
	? [GMptr] '< 64000 -> GM cp loop;
	end;

( ==================================================================== )
( P3 - blit the 200x200 surface texture into the top of the visible page. )
(   p_surfacemap[0..39999] -> adaptor[0..39999]; the present path expands )
(   adaptor through the palette LUT to the framebuffer.                  )
( ==================================================================== )

"GM blit surface"
	[GMptr] = 0;
    "GM bs loop"
	A = nw; A + RPSM; A + [GMptr]; C = [A]; C & 255;
	B = nw; B + RADPR; B + [GMptr]; [B] = C;
	[GMptr]+;
	? [GMptr] '< 40000 -> GM bs loop;
	end;

( ==================================================================== )
( P4 helper - present one frame and drop a 64-unit sentinel straight out  )
( of the framebuffer so the poll-and-kill runner (tests/w7arun.ps1) sees  )
( a stable artifact and reports RAN-OK.                                  )
( ==================================================================== )

"GM sentinel"
	=> FB present;
	[File Name]     = gmoutname;
	[File Position] = 0;
	[File Command]  = WRITE;
	[Block Pointer] = fb;
	[Block Size]    = 64;
	[Block Size] * BYTESPERUNIT;
	isocall;
	[File Name]     = gmoutname;
	[File Position] = 0;
	[File Command]  = SET SIZE;
	[File Size]     = 64;
	[File Size] * BYTESPERUNIT;
	isocall;
	end;
