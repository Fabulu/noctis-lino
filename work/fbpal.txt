      ( *** fbpal - the 256-colour palette, emulated in software ***

	Normative source: docs-notes/LINOBUF.md section 5.3.
	Original: NOCTIS-0.CPP:166 (range8088), :179 (tavola_colori),
	:1151 (shade).  Every line below was written against those three
	functions, not against a paraphrase of them.

	THE FOUR STRUCTURES, and why the port needs all four.

	  pal6      the master, Noctis's tmppal.  768 six-bit components.
	  curpal6   what has actually been UPLOADED to the DAC.  Noctis
	            has no such array because the DAC is the array; we
	            need it because the DAC is now software.  niv-lr
	            calls it currpal.
	  pal       the 256-entry 00RRGGBB LUT, rebuilt from curpal6.
	  range8088 the fixed 64-entry greyscale ramp, NOCTIS-0.CPP:166.
	            range8088[3k+0..2] = k, for k = 0..63.  Built here
	            rather than typed out, because "value = index/3" is
	            checkable and 192 typed numbers are not.

	FOUR TRAPS, all reproduced deliberately.

	1. tavola_colori(tmppal + 3*first, first, n, ...) is a SELF-COPY:
	   NOCTIS.CPP:3777 and NOCTIS-0.CPP:5193 pass a pointer INTO
	   tmppal as the "new palette".  With pal6 and curpal6 separate,
	   the copy step must be SKIPPED, not redirected at curpal6.
	   [PVself] = 1 says so.  Copying from curpal6 instead would
	   silently change the sky, because curpal6 is one upload stale.

	2. filtro_rosso/verde/blu are signed char and temp is a 16-bit
	   unsigned.  The product is taken mod 65536 and the divide is
	   unsigned, so a filter above 127 goes NEGATIVE, becomes a huge
	   unsigned, and clamps to 63 - the "> 63" test does not catch it
	   because it never sees a negative.  Reproduced exactly, and
	   [PVrange] records whether any call was outside 0..127 rather
	   than trusting that none was.

	3. shade() TRUNCATES.  The store is a C float-to-unsigned-char
	   conversion, which is a cast, which chops.  lino's =, rounds to
	   NEAREST and would be wrong.  This calls FToIntChop from the
	   Wave 3 float engine (work/fp/fpconv), which reproduces
	   Borland's __ftol, and it does NOT rebuild it.  The running
	   value is narrowed to binary32 after every += because start_r
	   is a float VARIABLE, and a C store to a float rounds.

	4. Six bits to eight is v * 4, so 63 maps to 252.  Chosen because
	   the game's own snapshot() writer scales the DAC by 4, which
	   makes this LUT bit-identical to the palette inside the BMPs
	   that are oracle route 1.  DOSBox writes (v<<2)|(v>>4); PLTE>>2
	   recovers the six-bit DAC on both sides.  Made ONCE, here.

	THE UPLOAD ALWAYS STARTS AT COLOUR ZERO.  tavola_colori's asm
	tail sets CX = first*3 + n*3 and SI = tmppal, i.e. it outs from
	component 0 up to (first+n)*3.  Consequence the port must keep:
	updating band 64-127 uploads colours 0-127 and leaves 128-255
	stale.  That is behaviour, not an accident.                      )

"constants"

	PALN	= 256;
	PALC6	= 768;
	PALRN	= 192;		( range8088, 64 x 3 )

"variables"

	( ---- tavola_colori parameters ---- )
	PVsrc	= 0;		( workspace pointer to the source components )
	PVself	= 0;		( 1 = source aliases pal6, skip the copy )
	PVfirst	= 0;
	PVn	= 0;
	PVfr	= 0;
	PVfg	= 0;
	PVfb	= 0;
	PVrange	= 0;		( sticky: a filter was outside 0..127 )

	( ---- upload / LUT parameters ---- )
	PUn	= 0;		( upload colours 0 .. [PUn]-1 )
	PLfirst	= 0;
	PLn	= 0;

	( ---- shade parameters, binary32 bit patterns ---- )
	SHfirst	= 0;
	SHn	= 0;
	SHsr	= 0;		( start  r,g,b )
	SHsg	= 0;
	SHsb	= 0;
	SHer	= 0;		( end    r,g,b )
	SHeg	= 0;
	SHeb	= 0;
	SHk	= 0;
	SHdstb	= 0;		( shade's DESTINATION BUFFER - the original's
				  first parameter.  NOCTIS-0.CPP:1151 is
				  shade (unsigned char far *palette_buffer,
				         unsigned first_color, unsigned n,
				         float start_r, ..., float finish_b)
				  and 14 of its 21 call sites pass
				  surface_palette, not tmppal: 3 in
				  NOCTIS.CPP:3774-3776 (tmppal), 4 in
				  NOCTIS-0.CPP:5180-5183 (tmppal), 14 in
				  NOCTIS-1.CPP:3050-3086 (surface_palette).
				  A shade that can only write pal6 cannot
				  express two thirds of the game's calls, and
				  srfpal6/retpal6 - the buffers whose whole
				  purpose is to stop fades COMPOUNDING - are
				  then dead weight no test touches.

				  "PAL zero" sets the default to pal6, so the
				  7 tmppal sites need no change.

				  THREE DETAILS THE PARAMETERISATION MUST NOT
				  LOSE.  (1) tavola_colori stays hard-coded to
				  pal6: NOCTIS-0.CPP:179-241 always writes
				  tmppal regardless of its nuova_tavolozza
				  argument, which is the SOURCE.  [PVsrc]
				  already expresses that; do not "fix" it
				  symmetrically.  (2) surface_palette and
				  return_palette are char (signed) while
				  tmppal is unsigned char, but shade takes
				  unsigned char far * so stores are unsigned,
				  and every read is via tavola_colori's
				  unsigned char * parameter - so NO sign
				  extension ever occurs and "MEM sx8" must NOT
				  be applied to loads from them.  (3) shade
				  never writes return_palette in the game
				  (only memcpy at NOCTIS-1.CPP:3430 and
				  tavola_colori at :3434 and NOCTIS.CPP:3647);
				  the parameter is kept general anyway. )
	SHdst	= 0;
	SHcurp	= 0;		( pointer to the running value in use )
	SHdelp	= 0;		( pointer to its delta )
	SHb	= 0;
	SHz32	= 0;		( 0.0f  as a bit pattern )
	SH6432	= 0;		( 64.0f as a bit pattern )

	( ---- float glue scratch ---- )
	PFbits	= 0;
	PFsa0	= 0;
	PFsa1	= 0;
	PFnum	= 0;
	PFden	= 0;

	PALtmp	= 0;

"workspace"

	pal6		= 768;

	srfpal6		= 768;		( surface_palette, fade source )
	pal		= 256;

"programme"

( ==================================================================== )
( float glue - four routines, so nothing else has to know the ABI       )
( ==================================================================== )

      ( PAL fa from f32 : FA <- (double)the binary32 in [PFbits]. )

"PAL fa from f32"
	A = [PFbits]; [FS0] = A;
	=> FLoadF32;
	end;

      ( PAL fb from f32 : FB <- (double)the binary32 in [PFbits],
	leaving FA untouched.  FLoadF32 lands in FA, so FA is saved
	across it - getting this wrong silently destroys the accumulator
	in the middle of a shade. )

"PAL fb from f32"
	A = [FA0]; [PFsa0] = A;
	A = [FA1]; [PFsa1] = A;
	A = [PFbits]; [FS0] = A;
	=> FLoadF32;
	A = [FA0]; [FB0] = A;
	A = [FA1]; [FB1] = A;
	A = [PFsa0]; [FA0] = A;
	A = [PFsa1]; [FA1] = A;
	end;

      ( PAL fa to fb : FB <- FA, no FPU involved. )

"PAL fa to fb"
	A = [FA0]; [FB0] = A;
	A = [FA1]; [FB1] = A;
	end;

      ( PAL f32 make : [PFbits] <- (float)((double)[PFnum] / (double)[PFden]).
	The one place integers become floats.  [PFden] = 1 gives a plain
	integer-valued float. )

"PAL f32 make"
	[FI] = [PFden]; => IntToF; => PAL fa to fb;
	[FI] = [PFnum]; => IntToF;
	=> FQuo;
	=> FStoreF32;
	[PFbits] = [FS0];
	end;

      ( PAL float init : the two constants shade compares against.
	Built through the engine rather than typed as hex, so a broken
	float path shows up here instead of hiding inside a clamp. )

"PAL float init"
	[PFnum] = 0;  [PFden] = 1; => PAL f32 make; [SHz32]  = [PFbits];
	[PFnum] = 64; [PFden] = 1; => PAL f32 make; [SH6432] = [PFbits];
	end;

( the buffers                                                           )
( ==================================================================== )

"PAL zero"
	A = pal6; D = curpal6; E = retpal6; B = 0; B + PALC6;
    "PAL z6"
	[A] = 0; [D] = 0; [E] = 0; A+; D+; E+;
	B ^ PAL z6;
	A = srfpal6; B = 0; B + PALC6;
    "PAL z6b"
	[A] = 0; A+;
	B ^ PAL z6b;
	A = pal; B = 0; B + PALN;
    "PAL zl"
	[A] = 0; A+;
	B ^ PAL zl;
	[PVrange] = 0;
	( shade's destination defaults to pal6 - Noctis's tmppal - so the
	  seven tmppal call sites need no change at all. )
	A = pal6; [SHdstb] = A;
	end;

      ( range8088[3k+0] = range8088[3k+1] = range8088[3k+2] = k. )

"PAL range"
	A = range8088; C = 0; B = 0; B + 64;
    "PAL rn loop"
	[A plus 0] = C; [A plus 1] = C; [A plus 2] = C;
	A + 3; C+;
	B ^ PAL rn loop;
	end;

( ==================================================================== )
( tavola_colori - copy, filter in place, upload from ZERO               )
( ==================================================================== )

"PAL tavola"
	( --- the filters are signed chars widened to 16-bit unsigned --- )
	=> PAL filt check;

	( --- step 1: copy, unless the source aliases the destination --- )
	? [PVself] != 0 -> PAL tv filter;
	D = [PVsrc];
	A = [PVfirst]; A * 3; A + pal6;
	B = [PVn]; B * 3;
	? B = 0 -> PAL tv filter;
    "PAL tv copy"
	C = [D]; C & 255; [A] = C; A+; D+;
	B ^ PAL tv copy;

	( --- step 2: filter in place --- )
    "PAL tv filter"
	A = [PVfirst]; A * 3; A + pal6;
	B = 0; B + [PVn];
	? B = 0 -> PAL tv up;
    "PAL tv loop"
	C = [A]; C & 255; C '* [PVfr]; C & 65535; C '/ 63;
	? C '< 64 -> PAL tv r; C = 63;
    "PAL tv r"
	[A] = C; A+;
	C = [A]; C & 255; C '* [PVfg]; C & 65535; C '/ 63;
	? C '< 64 -> PAL tv g; C = 63;
    "PAL tv g"
	[A] = C; A+;
	C = [A]; C & 255; C '* [PVfb]; C & 65535; C '/ 63;
	? C '< 64 -> PAL tv b; C = 63;
    "PAL tv b"
	[A] = C; A+;
	B ^ PAL tv loop;

	( --- step 3: upload colours 0 .. first+n-1 --- )
    "PAL tv up"
	A = [PVfirst]; A + [PVn]; [PUn] = A;
	=> PAL upload;
	end;

      ( PAL filt check : sign-extend each filter from char, widen to the
	16-bit unsigned the original's arithmetic actually uses, and
	record whether it was inside the 0..127 the callers assume.
	NOCTIS-1.CPP:3934 passes random(64)+64 = 64..127, which fits -
	but the assumption is asserted here, not trusted. )

"PAL filt check"
	C = [PVfr]; => MEM sx8;
	? C >= 0 -> PAL fc r;
	A = [PVrange]; A | 1; [PVrange] = A;
    "PAL fc r"
	? C <= 127 -> PAL fc r2;
	A = [PVrange]; A | 1; [PVrange] = A;
    "PAL fc r2"
	C & 65535; [PVfr] = C;

	C = [PVfg]; => MEM sx8;
	? C >= 0 -> PAL fc g;
	A = [PVrange]; A | 2; [PVrange] = A;
    "PAL fc g"
	? C <= 127 -> PAL fc g2;
	A = [PVrange]; A | 2; [PVrange] = A;
    "PAL fc g2"
	C & 65535; [PVfg] = C;

	C = [PVfb]; => MEM sx8;
	? C >= 0 -> PAL fc b;
	A = [PVrange]; A | 4; [PVrange] = A;
    "PAL fc b"
	? C <= 127 -> PAL fc b2;
	A = [PVrange]; A | 4; [PVrange] = A;
    "PAL fc b2"
	C & 65535; [PVfb] = C;
	end;

      ( PAL upload : the first 3*[PUn] components of pal6 go to
	curpal6, and that
	span of the LUT.  ALWAYS from zero.  This is the DAC write. )

"PAL upload"
	A = pal6; D = curpal6;
	B = [PUn]; B * 3;
	? B = 0 -> PAL up lut;
    "PAL up loop"
	C = [A]; [D] = C; A+; D+;
	B ^ PAL up loop;
    "PAL up lut"
	[PLfirst] = 0; [PLn] = [PUn];
	=> PAL lut;
	end;

( ==================================================================== )
( the LUT - the ONE place six bits become eight                         )
( ==================================================================== )

"PAL lut"
	B = 0; B + [PLn];
	? B = 0 -> PAL lut done;
	D = [PLfirst]; D * 3; D + curpal6;
	E = [PLfirst]; E + pal;
    "PAL lut loop"
	C = [D];	  C & 63; C * 4; C < 16;
	A = [D plus 1];   A & 63; A * 4; A < 8;   C + A;
	A = [D plus 2];   A & 63; A * 4;          C + A;
	[E] = C;
	D + 3; E+;
	B ^ PAL lut loop;
    "PAL lut done"
	end;

( ==================================================================== )
( the background colour cycle - NOCTIS.CPP:3779-3784, on s_background   )
( ==================================================================== )

      ( PAL cycle : verbatim,

	    if (!farstar) { for (ir=0; ir<64800; ir++) {
	            ig = (s_background[ir]+1)%64;
	            ib = (s_background[ir]>>6)<<6;
	            s_background[ir] = ig+ib; } }

	It cycles the low six bits of s_background - a 64,800-byte
	OFFSCREEN star/sky map - and it lives inside the sky-palette
	redefinition path.  It is NOT the visible page and it does NOT run
	once per present.

	LINOBUF section 5.4 directed that this be FUSED into the present's
	expand loop for a measured 22%.  That is the right arithmetic
	applied to the wrong buffer at the wrong rate: fused, it corrupts
	every visible unit on every present.  BUFFERMODEL section 7
	overrules it and sabotage S12 reinstates the fusion.

	The shipped Wave 5 shell CONTAINED that fusion, and it survived
	only because the shell rebuilt the page immediately before the
	dump, so the cycled page was never recorded.  Both halves of that
	are fixed: the expand is read-only, and the dump is taken without
	a rebuild. )

"PAL cycle"
	D = nw; D + RSBG; B = 0; B + ZSBG;
    "PAL cy loop"
	C = [D]; C & 255;
	A = C; A > 6; A < 6;		( ib = (v>>6)<<6, from the ORIGINAL v )
	C + 1; C '% 64;			( ig = (v+1)%64, and 255+1 -> 0 )
	C + A;
	[D] = C; D+;
	B ^ PAL cy loop;
	end;

( ==================================================================== )
( shade - NOCTIS-0.CPP:1151, truncating, binary32 accumulators          )
( ==================================================================== )

      ( PAL shade : writes pal6[3*[SHfirst] ..] for [SHn] colours,
	ramping (SHsr,SHsg,SHsb) towards (SHer,SHeg,SHeb).
	All six are binary32 BIT PATTERNS, so any float the original
	could pass can be expressed. )

"PAL shade"
	( k = 1.00 / (float)n, stored to a float )
	[PFnum] = 1; [PFden] = [SHn]; => PAL f32 make; [SHk] = [PFbits];

	[shcur plus 0] = [SHsr];
	[PFbits] = [SHsr]; => PAL fa from f32;
	[PFbits] = [SHer]; => PAL fb from f32;
	=> PAL delta;  [shdel plus 0] = [PFbits];

	[shcur plus 1] = [SHsg];
	[PFbits] = [SHsg]; => PAL fa from f32;
	[PFbits] = [SHeg]; => PAL fb from f32;
	=> PAL delta;  [shdel plus 1] = [PFbits];

	[shcur plus 2] = [SHsb];
	[PFbits] = [SHsb]; => PAL fa from f32;
	[PFbits] = [SHeb]; => PAL fb from f32;
	=> PAL delta;  [shdel plus 2] = [PFbits];

	( first_color *= 3, into the DESTINATION BUFFER.  This one line is
	  the whole of defect 6: it used to read "A + pal6". )
	A = [SHfirst]; A * 3; A + [SHdstb]; [SHdst] = A;

	B = 0; B + [SHn];
	? B = 0 -> PAL sh done;
    "PAL sh loop"
	A = shcur;     [SHcurp] = A;
	A = shdel;     [SHdelp] = A;
	=> PAL shade comp;
	A = [SHdst]; A+; [SHdst] = A;
	A = shcur; A+; [SHcurp] = A;
	A = shdel; A+; [SHdelp] = A;
	=> PAL shade comp;
	A = [SHdst]; A+; [SHdst] = A;
	A = shcur; A + 2; [SHcurp] = A;
	A = shdel; A + 2; [SHdelp] = A;
	=> PAL shade comp;
	A = [SHdst]; A+; [SHdst] = A;
	B ^ PAL sh loop;
    "PAL sh done"
	end;

      ( PAL delta : [PFbits] <- (float)((FA - FB) * k), where FA holds
	start and FB holds finish.  The original computes
	(finish - start) * k, so the subtraction is reversed here and
	FSub is FA - FB: FB is loaded with finish and FA with start, then
	the two are swapped.  Written out rather than clever. )

"PAL delta"
	A = [FA0]; [PFsa0] = A;		( start, low  )
	A = [FA1]; [PFsa1] = A;		( start, high )
	A = [FB0]; [FA0] = A;		( FA <- finish )
	A = [FB1]; [FA1] = A;
	A = [PFsa0]; [FB0] = A;		( FB <- start  )
	A = [PFsa1]; [FB1] = A;
	=> FSub;			( FA = finish - start )
	[PFbits] = [SHk]; => PAL fb from f32;
	=> FMul;			( FA = (finish-start) * k )
	=> F32Narrow;			( stored to a float )
	=> FStoreF32;
	[PFbits] = [FS0];
	end;

      ( PAL shade comp : one component.  The clamp is the ORIGINAL's,
	including its inverted else branch, and the in-range store is a
	CHOP.  Then the running value advances and is narrowed. )

"PAL shade comp"
	A = [SHcurp]; [PFbits] = [A];
	=> PAL fa from f32;

	[PFbits] = [SHz32]; => PAL fb from f32;
	=> FCmp;
	A = [FI]; ? A < 0 -> PAL sc under;

	[PFbits] = [SH6432]; => PAL fb from f32;
	=> FCmp;
	A = [FI]; ? A < 0 -> PAL sc inrange;

	( >= 64, so the else branch, and start > 0 is true )
	[SHb] = 63; -> PAL sc store;

    "PAL sc under"
	( < 0, so the else branch, and start > 0 is false )
	[SHb] = 0; -> PAL sc store;

    "PAL sc inrange"
	=> FToIntChop;
	[SHb] = [FI];

    "PAL sc store"
	A = [SHdst]; C = [SHb]; C & 255; [A] = C;

	( start += delta, then narrowed because start is a float )
	A = [SHcurp]; [PFbits] = [A];
	=> PAL fa from f32;
	A = [SHdelp]; [PFbits] = [A];
	=> PAL fb from f32;
	=> FAdd;
	=> F32Narrow;
	=> FStoreF32;
	A = [SHcurp]; C = [FS0]; [A] = C;
	end;
