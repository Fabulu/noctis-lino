( rockyground -- NOCTIS-1.CPP:1545-1559                                  )

      ( GR rockyground -- fills p_surfacemap with random(roughness), runs
	smoothterrain(rounding), then adjusts by level.

	Input:  [GRrough], [GRround], [GRlevel]
	Output: p_surfacemap[0..39999] modified				)

"GR rockyground"

	( phase 1: fill with random(roughness) )
	[GRptr] = 0;
"GR rg fill"
	C = [GRrough];
	=> SU rnd;
	C & 0FFh;
	[MBval] = C;
	A = RPSM; A + [GRptr]; [MBptr] = A;
	=> MEM put byte;
	[GRptr]+;
	? [GRptr] '< 40000 -> GR rg fill;

	( phase 2: smooth )
	C = [GRround];
	=> GR smoothterrain;

	( phase 3: level adjust )
	( Every level applies the source value >= abs(level) cutoff before the
	  signed adjustment. This matters for positive levels too: zero-valued
	  cells stay at zero rather than being lifted to one. )
	A = [GRlevel];
	? A >= 0 -> GR rg absok;
	B = 0; B - A; A = B;
"GR rg absok"
	[GRabs] = A;

	[GRptr] = 0;
"GR rg level"
	A = RPSM; A + [GRptr]; [MBptr] = A;
	=> MEM get byte;
	A = [MBval];
	? A '>= [GRabs] -> GR rg add;
	C = 0;
	[MBval] = C;
	A = RPSM; A + [GRptr]; [MBptr] = A;
	=> MEM put byte;
	-> GR rg next;

"GR rg add"
	A = [MBval]; A + [GRlevel];
	? A '<= 127 -> GR rg store;
	A = 127;
"GR rg store"
	C = A; C & 0FFh;
	[MBval] = C;
	A = RPSM; A + [GRptr]; [MBptr] = A;
	=> MEM put byte;

"GR rg next"
	[GRptr]+;
	? [GRptr] '< 40000 -> GR rg level;
	end;

( smoothterrain -- NOCTIS-1.CPP:1530-1543                                )
( ==================================================================== )

      ( GR smoothterrain -- 2x2 box average, `rounding` passes.

	  while (rounding) {
	      for (ptr = 0; ptr < 39799; ptr++) {
		  n  = p_surfacemap[ptr];
		  n += p_surfacemap[ptr + 1];
		  n += p_surfacemap[ptr + 200];
		  n += p_surfacemap[ptr + 201];
		  p_surfacemap[ptr] = n >> 2;
	      }
	      rounding--;
	  }

	Input: C = passes.  No output regs.				)

"GR smoothterrain"
	[GRsmp] = C;

"GR sm outer"
	? [GRsmp] '<= 0 -> GR sm end;
	[GRptr] = 0;
"GR sm loop"
	? [GRptr] '>= 39799 -> GR sm nextpass;

	A = RPSM; A + [GRptr]; [MBptr] = A;
	=> MEM get byte;
	A = [MBval]; [GRn] = A;

	A = RPSM; A + [GRptr]; A + 1; [MBptr] = A;
	=> MEM get byte;
	A = [GRn]; A + [MBval]; [GRn] = A;

	A = 200; A + [GRptr]; A + RPSM; [MBptr] = A;
	=> MEM get byte;
	A = [GRn]; A + [MBval]; [GRn] = A;

	A = 201; A + [GRptr]; A + RPSM; [MBptr] = A;
	=> MEM get byte;
	A = [GRn]; A + [MBval]; [GRn] = A;

	A = [GRn]; A > 2; A & 0FFh;
	[MBval] = A;
	A = RPSM; A + [GRptr]; [MBptr] = A;
	=> MEM put byte;

	[GRptr]+;
	-> GR sm loop;

"GR sm nextpass"
	[GRsmp]-;
	-> GR sm outer;
"GR sm end"
	end;

( round_hill -- NOCTIS-1.CPP:1494-1528                                  )
( ==================================================================== )
( Input: [SUcxi] = cx, [SUcyi] = cz, [SUia] = r (unsigned int)        )
(        [GRfht] = h (float32 bits), [GRfhmt] = hmax (float32 bits)   )
(        [GRfcanyon] = allowcanyons (0 or 1)                          )
(                                                                     )
( The x87 ML fragments keep intermediates on the stack and only store  )
( to float32 where the C code stores, matching Borland's codegen.     )

"GR round hill"
	[GRprofileid]+;

	( v = (float)r / M_PI_2 -- fild r; fdiv M_PI_2; fstp dword v )
	A = [GRKPI2L]; [XU0] = A; A = [GRKPI2H]; [XU1] = A; => XFromF64;
	A = [XS]; [YS] = A; A = [XE]; [YE] = A;
	A = [XMH]; [YMH] = A; A = [XML]; [YML] = A;
	[XIN] = [SUia]; => XFromInt;
	=> XQuoCore; => XToF32;
	[GRfv] = [FS0];

	( Borland compares the int16 x/z locals against cx+r/cz+r as
	  uint16 because source r is unsigned.  A wrapped negative start
	  therefore skips the whole loop instead of clipping at zero. )
	A = [SUia]; A & 0FFFFh; [SUia] = A;	( uint16 r )
	B = [SUcxi];			( cx )
	[GRfrlo] = B; [GRfrlo] - A; C = [GRfrlo]; C & 0FFFFh; [GRfrlo] = C;
	C = B; C + A; C & 0FFFFh;
	[GRfrhi] = C;			( cx + r )
	[GRfrx] = [GRfrlo];

"GR rh xloop"
	A = [GRfrx];
	? A '>= [GRfrhi] -> GR rh done;

	( z loop bounds )
	A = [SUia];
	B = [SUcyi];
	D = B; D - A; D & 0FFFFh; [GRfrz] = D;

"GR rh zloop"
	A = [GRfrx];
	? A < 0 -> GR rh znext;
	? A >= 200 -> GR rh znext;
	A = [GRfrz];
	? A < 0 -> GR rh znext;
	? A >= 200 -> GR rh znext;

	( The binary32 dx/dz squares are exact small integers.  Use their integer
	  sum as the radial cache key, then execute the original qword-spilled
	  sqrt/cos profile only once per unique distance in this hill call. )
	B = [GRfrx]; B - [SUcxi]; [GRdxi] = B;
	B = [GRfrz]; B - [SUcyi]; [GRdzi] = B;
	A = [GRdxi]; A '* [GRdxi]; B = [GRdzi]; B '* [GRdzi]; A + B;
	[GRfd2] = A;
	A = GRsstamp; A + [GRfd2]; A = [A];
	? A = [GRprofileid] -> GR rh profile cached;

	( d = sqrt((double)d2).  The original float squares and sum are exact for
	  this bounded integer domain, followed by the same binary64 helper spill. )
	[FI] = [GRfd2]; => IntToF;
	[GRfs0] = [FA0]; [GRfs1] = [FA1];
	[FA0] = [GRfs0]; [FA1] = [GRfs1]; => FSqrt;
	=> FStoreF32; [GRfd] = [FS0];

	( y = cos(d / v) * h -- cos(double) has the same qword argument spill )
	[FS0] = [GRfd]; => FLoadF32;
	[GRfs0] = [FA0]; [GRfs1] = [FA1];
	[FS0] = [GRfv]; => FLoadF32;
	[FB0] = [FA0]; [FB1] = [FA1];
	[FA0] = [GRfs0]; [FA1] = [GRfs1]; => FQuo;
	[GRfs0] = [FA0]; [GRfs1] = [FA1];
	[FA0] = [GRfs0]; [FA1] = [GRfs1]; => FCos;
	[GRfs0] = [FA0]; [GRfs1] = [FA1];
	[FS0] = [GRfht]; => FLoadF32;
	[FB0] = [FA0]; [FB1] = [FA1];
	[FA0] = [GRfs0]; [FA1] = [GRfs1]; => FMul;
	=> FStoreF32; [GRfy] = [FS0];
	A = GRscache; A + [GRfd2]; [A] = [GRfy];
	A = GRsstamp; A + [GRfd2]; [A] = [GRprofileid];
	-> GR rh profile ready;
    "GR rh profile cached"
	A = GRscache; A + [GRfd2]; [GRfy] = [A];
    "GR rh profile ready"

	( if (y >= 0) -- check sign bit of float32 )
	A = [GRfy]; A & 80000000h;
	? A != 0 -> GR rh znext;

	( y += surfacemap[200*z + x] )
	A = [GRfrz]; A '* 200; A + [GRfrx]; [GRptr] = A;
	A = RPSM; A + [GRptr]; [MBptr] = A; A + nw; C = [A]; C & 255;
	[MBval] = C; [GRsval] = C;
	[FS0] = [GRfy]; => FLoadF32;
	[FB0] = [FA0]; [FB1] = [FA1];
	[FI] = [GRsval]; => IntToF; => FAdd; => FStoreF32;
	[GRfy] = [FS0];

	( canyon mirror or hmax clamp )
	A = [GRfcanyon];
	? A = 0 -> GR rh clamp;

	( if (y > 127) y = 254 - y -- compute y - 127.0, check sign )
	[FS0] = [GRfy]; => FLoadF32;
	[FB0] = [GRK127L]; [FB1] = [GRK127H]; => FCmp;
	? [FI] <= 0 -> GR rh store;
	( y > 127: canyon mirror )
	[FS0] = [GRfy]; => FLoadF32;
	[FB0] = [FA0]; [FB1] = [FA1];
	[FA0] = [GRK254L]; [FA1] = [GRK254H]; => FSub; => FStoreF32;
	[GRfy] = [FS0];
	-> GR rh store;

"GR rh clamp"
	( if (y > hmax) y = hmax -- compute y - hmax, check sign )
	[FS0] = [GRfhmt]; => FLoadF32;
	[FB0] = [FA0]; [FB1] = [FA1];
	[FS0] = [GRfy]; => FLoadF32; => FCmp;
	? [FI] <= 0 -> GR rh store;
	[GRfy] = [GRfhmt];			( y = hmax )

"GR rh store"
	( surfacemap[idx] = (unsigned char)y -- chop )
	[FS0] = [GRfy]; => FLoadF32; => FToIntChop;
	[GRfti] = [FI];
	A = [GRfti]; A & 0FFh;
	[MBval] = A;
	C = A; A = RPSM; A + [GRptr]; [MBptr] = A; A + nw; [A] = C;

"GR rh znext"
	[GRfrz]+;
	A = [GRfrz];
	B = [SUcyi]; B + [SUia];	( cz + r )
	? A '< B -> GR rh zloop;
	[GRfrx]+;
	-> GR rh xloop;

"GR rh done"
	end;

( srf_darkline -- NOCTIS-1.CPP:1589-1603                                )

( Input: [SUia] = align, [GRdxi] = length,                             )

( A random walk that halves bytes along its path.                      )

"GR srf darkline"
	( fx = random(align), fz = random(align) )
	C = [SUia]; => SU rnd; [GRfrx] = C;
	C = [SUia]; => SU rnd; [GRfrz] = C;

	? [GRn] '<= 0 -> GR sd done;
	( fx += random(3) + x_trend )
	C = 3; => SU rnd;

( std_crater -- NOCTIS-1.CPP:1561-1587                                  )
( ==================================================================== )
( Input: [SUcxi]=cx, [SUcyi]=cz, [SUia]=r (signed int)                 )
(        [GRfht]=h_factor (float32), [GRfhmt]=h_raiser (float32)      )
(        [GRfscl]=lim_h (int), [GRfsca]=align (int, 200 or 256)       )
(        map base in [GRpksrc] (RPSM for smap, RPBG for txtr)         )
( h_raiser=1.0: pow is identity. Other positive profiles use the x87 )
( fyl2x/frndint/f2xm1/fscale sequence, preserving the live float32 y. )

"GR std crater"
	[GRprofileid]+; [GRcraterid] = [GRprofileid];

	( h = (float)r * h_factor -- fild r; fmul h_factor; fstp dword h )
	[FI] = [SUia]; => IntToF;
	[FB0] = [FA0]; [FB1] = [FA1];
	[FS0] = [GRfht]; => FLoadF32; => FMul; => FStoreF32;
	[GRfsch] = [FS0];

	( r = abs(r) )
	A = [SUia];
	? A >= 0 -> GR sc absok;
	B = 0; B - A; A = B;
"GR sc absok"
	[SUia] = A;
	C = A; A '* C; [GRfr2] = A;

	( fr = (float)r -- fild abs(r); fstp dword fr )
	[FI] = [SUia]; => IntToF; => FStoreF32;
	[GRfscr] = [FS0];

	( Clip the source's half-open crater rectangle once.  Skipped out-of-map
	  iterations have no side effects or RNG calls, so this is identical to
	  the original per-pixel bounds checks. )
	A = [SUia];
	B = [SUcxi];
	[GRfrlo] = B; [GRfrlo] - A;
	C = B; C + A;
	[GRfrhi] = C;
	A = [GRfrlo]; ? A >= 0 -> GR sc xlo ready; [GRfrlo] = 0;
    "GR sc xlo ready"
	A = [GRfrhi]; ? A '<= [GRfsca] -> GR sc xhi ready; [GRfrhi] = [GRfsca];
    "GR sc xhi ready"
	A = [SUia]; B = [SUcyi];
	[GRsczlo] = B; [GRsczlo] - A;
	C = B; C + A; [GRsczhi] = C;
	A = [GRsczlo]; ? A >= 0 -> GR sc zlo ready; [GRsczlo] = 0;
    "GR sc zlo ready"
	A = [GRsczhi]; ? A '<= [GRfsca] -> GR sc zhi ready; [GRsczhi] = [GRfsca];
    "GR sc zhi ready"
	A = [GRfrlo]; ? A '>= [GRfrhi] -> GR sc done;
	A = [GRsczlo]; ? A '>= [GRsczhi] -> GR sc done;
	[GRfrx] = [GRfrlo];

"GR sc xloop"
	A = [GRfrx];
	? A '>= [GRfrhi] -> GR sc done;

	( The integer radial key and the map pointer are both recurrence values
	  down the z-major inner loop: d2 advances by 2*dz+1 and the address
	  advances by align. )
	B = [GRfrx]; B - [SUcxi]; [GRdxi] = B;
	A = B; A '* B; [GRfd2] = A;
	[GRfrz] = [GRsczlo];
	C = [GRfrz]; C - [SUcyi]; [GRdzi] = C;
	A = C; A '* C; D = [GRfd2]; D + A; [GRfd2] = D;
	A = C; A + C; A+; [GRfd2delta] = A;
	A = [GRfrz]; A '* [GRfsca]; A + [GRfrx]; A + [GRscmap]; [GRscaddr] = A;

"GR sc zloop"

	  and the source accepts precisely sqrt(dx*dx+dz*dz) <= r.  It also gives
	  the radial-profile cache its stable key before any x87 work. )
	A = [GRfd2];

	A = GRsstamp; A + [GRfd2]; A = [A];
	? A != [GRprofileid] -> GR sc profile miss;
	A = GRscache; A + [GRfd2]; [GRfy] = [A];
	-> GR sc profile ready;

	  then squares and adds them exactly before fsqrt.  GRfd2 is that same
	  exact integer sum, so loading it directly removes redundant float
	  conversions and arithmetic without changing fsqrt's input or result. )
	[FI] = [GRfd2]; => IntToF; => FSqrt;
	=> FStoreF32; [GRfd] = [FS0];

	( d <= fr: compute y = sin(M_PI * d / fr) * h )
	[FS0] = [GRfd]; => FLoadF32;
	A = [FA0]; [XU0] = A; A = [FA1]; [XU1] = A; => XFromF64;
	A = [XS]; [YS] = A; A = [XE]; [YE] = A;
	A = [XMH]; [YMH] = A; A = [XML]; [YML] = A;
	[FS0] = [GRfscr]; => FLoadF32;
	A = [FA0]; [XU0] = A; A = [FA1]; [XU1] = A; => XFromF64;
	( Swap the two extended operands: X=d, Y=fr, retaining d/fr live
	  through the divide and pi multiply until the required sin spill. )
	A = [XS]; B = [YS]; [XS] = B; [YS] = A;
	A = [XE]; B = [YE]; [XE] = B; [YE] = A;
	A = [XMH]; B = [YMH]; [XMH] = B; [YMH] = A;
	A = [XML]; B = [YML]; [XML] = B; [YML] = A;

	A = [XS]; [YS] = A; A = [XE]; [YE] = A;
	A = [XMH]; [YMH] = A; A = [XML]; [YML] = A;
	[FA0] = [GRKPIL]; [FA1] = [GRKPIH];
	A = [FA0]; [XU0] = A; A = [FA1]; [XU1] = A; => XFromF64;
	A = [XS]; B = [YS]; [XS] = B; [YS] = A;
	A = [XE]; B = [YE]; [XE] = B; [YE] = A;
	A = [XMH]; B = [YMH]; [XMH] = B; [YMH] = A;
	A = [XML]; B = [YML]; [XML] = B; [YML] = A;
	=> XMulCore; => XToF64; => FSin;
	[GRfs0] = [FA0]; [GRfs1] = [FA1];
	[FS0] = [GRfsch]; => FLoadF32;
	[FB0] = [FA0]; [FB1] = [FA1];
	[FA0] = [GRfs0]; [FA1] = [GRfs1]; => FMul;
	=> FStoreF32; [GRfy] = [FS0];

	( y = pow(y,h_raiser).  The private positive-power helper preserves the
	  live extended boundary through the one historical final F32 spill. )
	A = [GRfhmt]; A & 0FFFFFFFFh; ? A = 3F800000h -> GR sc power done;
	A = [GRfy]; A & 7FFFFFFFh; ? A = 0 -> GR sc power done;
	[FS0] = [GRfhmt]; => FLoadF32;
	A = [FA0]; [XU0] = A; A = [FA1]; [XU1] = A; => XFromF64; => XCopyXtoY;
	[FS0] = [GRfy]; => FLoadF32;
	A = [FA0]; [XU0] = A; A = [FA1]; [XU1] = A; => XFromF64;
	=> XPowPositive; => XToF32; [GRfy] = [FS0];
"GR sc power done"
	A = GRscache; A + [GRfd2]; [A] = [GRfy];
	A = GRsstamp; A + [GRfd2]; [A] = [GRprofileid];

"GR sc profile ready"

	( y += map[align*z + x].  The clipped rectangle proves this direct
	  one-byte load has the exact semantics of MEM get byte. )
	A = nw; A + [GRscaddr]; C = [A]; C & 255; [GRsval] = C;
	[FS0] = [GRfy]; => FLoadF32;
	[FB0] = [FA0]; [FB1] = [FA1];
	[FI] = [GRsval]; => IntToF; => FAdd; => FStoreF32;
	[GRfy] = [FS0];

	A = [GRfy]; A & 80000000h;
	? A = 0 -> GR sc clamphi;
	[GRfy] = 0;

"GR sc clamphi"
	( if (y > lim_h) y = lim_h -- compute y - lim_h, check sign )
	[FI] = [GRfscl]; => IntToF;
	[FB0] = [FA0]; [FB1] = [FA1];
	[FS0] = [GRfy]; => FLoadF32; => FCmp;
	? [FI] <= 0 -> GR sc store;
	( y >= lim_h: clamp to lim_h )
	[FI] = [GRfscl]; => IntToF; => FStoreF32;

	[FS0] = [GRfy]; => FLoadF32; => FToIntChop;
	[GRfti] = [FI];
	A = [GRfti]; A & 0FFh;
