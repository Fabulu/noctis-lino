( rockyground -- NOCTIS-1.CPP:1545-1559                                  )

      ( GR rockyground -- fills p_surfacemap with random(roughness), runs
	smoothterrain(rounding), then adjusts by level.

	Input:  [GRrough], [GRround], [GRlevel]
	Output: p_surfacemap[0..39999] modified				)

"GR rockyground"

	( phase 1: fill with random(roughness) )
	[GRptr] = 0;
"GR rg fill"
	C = [GRrough];
	=> SU rnd;
	C & 0FFh;
	[MBval] = C;
	A = RPSM; A + [GRptr]; [MBptr] = A;
	=> MEM put byte;
	[GRptr]+;
	? [GRptr] '< 40000 -> GR rg fill;

	( phase 2: smooth )
	C = [GRround];
	=> GR smoothterrain;

	( phase 3: level adjust )
	( Every level applies the source value >= abs(level) cutoff before the
	  signed adjustment. This matters for positive levels too: zero-valued
	  cells stay at zero rather than being lifted to one. )
	A = [GRlevel];
	? A >= 0 -> GR rg absok;
	B = 0; B - A; A = B;
"GR rg absok"
	[GRabs] = A;

	[GRptr] = 0;
"GR rg level"
	A = RPSM; A + [GRptr]; [MBptr] = A;
	=> MEM get byte;
	A = [MBval];
	? A '>= [GRabs] -> GR rg add;
	( else set to 0 )
	C = 0;
	[MBval] = C;
	A = RPSM; A + [GRptr]; [MBptr] = A;
	=> MEM put byte;
	-> GR rg next;

"GR rg add"
	A = [MBval]; A + [GRlevel];
	? A '<= 127 -> GR rg store;
	A = 127;
"GR rg store"
	C = A; C & 0FFh;
	[MBval] = C;
	A = RPSM; A + [GRptr]; [MBptr] = A;
	=> MEM put byte;

"GR rg next"
	[GRptr]+;
	? [GRptr] '< 40000 -> GR rg level;
	end;

( smoothterrain -- NOCTIS-1.CPP:1530-1543                                )
( ==================================================================== )

      ( GR smoothterrain -- 2x2 box average, `rounding` passes.

	  while (rounding) {
	      for (ptr = 0; ptr < 39799; ptr++) {
		  n  = p_surfacemap[ptr];
		  n += p_surfacemap[ptr + 1];
		  n += p_surfacemap[ptr + 200];
		  n += p_surfacemap[ptr + 201];
		  p_surfacemap[ptr] = n >> 2;
	      }
	      rounding--;
	  }

	Input: C = passes.  No output regs.				)

"GR smoothterrain"
	[GRsmp] = C;
"GR sm outer"
	? [GRsmp] '<= 0 -> GR sm end;
	[GRptr] = 0;
"GR sm loop"
	? [GRptr] '>= 39799 -> GR sm nextpass;

	( n = smap[ptr] )
	A = RPSM; A + [GRptr]; [MBptr] = A;
	=> MEM get byte;
	A = [MBval]; [GRn] = A;

	( n += smap[ptr+1] )
	A = RPSM; A + [GRptr]; A + 1; [MBptr] = A;
	=> MEM get byte;
	A = [GRn]; A + [MBval]; [GRn] = A;

	( n += smap[ptr+200] )
	A = 200; A + [GRptr]; A + RPSM; [MBptr] = A;
	=> MEM get byte;
	A = [GRn]; A + [MBval]; [GRn] = A;

	( n += smap[ptr+201] )
	A = 201; A + [GRptr]; A + RPSM; [MBptr] = A;
	=> MEM get byte;
	A = [GRn]; A + [MBval]; [GRn] = A;

	( smap[ptr] = n >> 2 )
	A = [GRn]; A > 2; A & 0FFh;
	[MBval] = A;
	A = RPSM; A + [GRptr]; [MBptr] = A;
	=> MEM put byte;

	[GRptr]+;
	-> GR sm loop;

"GR sm nextpass"
	[GRsmp]-;
	-> GR sm outer;
"GR sm end"
	end;

( round_hill -- NOCTIS-1.CPP:1494-1528                                  )
( ==================================================================== )
( Input: [SUcxi] = cx, [SUcyi] = cz, [SUia] = r (unsigned int)        )
(        [GRfht] = h (float32 bits), [GRfhmt] = hmax (float32 bits)   )
(        [GRfcanyon] = allowcanyons (0 or 1)                          )
(                                                                     )
( The x87 ML fragments keep intermediates on the stack and only store  )
( to float32 where the C code stores, matching Borland's codegen.     )

"GR round hill"

	( v = (float)r / M_PI_2 -- fild r; fdiv M_PI_2; fstp dword v )
	{
	    DB 87 <dSUia   mtp bytesperunit>	(fild dword [edi+SUia*4])
	    DC B7 <dGRKPI2L mtp bytesperunit>	(fdiv qword [edi+GRKPI2L*4])
	    D9 9F <dGRfv   mtp bytesperunit>	(fstp dword [edi+GRfv*4])
	}

	( Borland compares the int16 x/z locals against cx+r/cz+r as
	  uint16 because source r is unsigned.  A wrapped negative start
	  therefore skips the whole loop instead of clipping at zero. )
	A = [SUia]; A & 0FFFFh; [SUia] = A;	( uint16 r )
	B = [SUcxi];			( cx )
	[GRfrlo] = B; [GRfrlo] - A; C = [GRfrlo]; C & 0FFFFh; [GRfrlo] = C;
	C = B; C + A; C & 0FFFFh;
	[GRfrhi] = C;			( cx + r )
	[GRfrx] = [GRfrlo];

"GR rh xloop"
	A = [GRfrx];
	? A '>= [GRfrhi] -> GR rh done;

	( z loop bounds )
	A = [SUia];
	B = [SUcyi];
	D = B; D - A; D & 0FFFFh; [GRfrz] = D;

"GR rh zloop"
	A = [GRfrx];
	? A < 0 -> GR rh znext;
	? A >= 200 -> GR rh znext;
	A = [GRfrz];
	? A < 0 -> GR rh znext;
	? A >= 200 -> GR rh znext;

	( dx = (float)(x - cx) )
	B = [GRfrx]; B - [SUcxi]; [GRdxi] = B;
	{
	    DB 87 <dGRdxi mtp bytesperunit>	(fild dword [edi+GRdxi*4])
	    D9 9F <dGRfdx mtp bytesperunit>	(fstp dword [edi+GRfdx*4])
	}

	( dz = (float)(z - cz) )
	B = [GRfrz]; B - [SUcyi]; [GRdzi] = B;
	{
	    DB 87 <dGRdzi mtp bytesperunit>	(fild dword [edi+GRdzi*4])
	    D9 9F <dGRfdz mtp bytesperunit>	(fstp dword [edi+GRfdz*4])
	}

	( d = sqrt(dx*dx + dz*dz)
	  Borland passes the expression to the far sqrt(double) helper through
	  an eight-byte stack argument.  Preserve that binary64 spill before
	  fsqrt instead of carrying the sum at x87 extended precision. )
	{
	    D9 87 <dGRfdx mtp bytesperunit>	(fld dword [edi+GRfdx*4])
	    D8 8F <dGRfdx mtp bytesperunit>	(fmul dword [edi+GRfdx*4])
	    D9 87 <dGRfdz mtp bytesperunit>	(fld dword [edi+GRfdz*4])
	    D8 8F <dGRfdz mtp bytesperunit>	(fmul dword [edi+GRfdz*4])
	    DE C1				(faddp st1,st0)
	    DD 9F <dGRfs0 mtp bytesperunit>	(fstp qword [edi+GRfs0*4])
	    DD 87 <dGRfs0 mtp bytesperunit>	(fld qword [edi+GRfs0*4])
	    D9 FA				(fsqrt)
	    D9 9F <dGRfd mtp bytesperunit>	(fstp dword [edi+GRfd*4])
	}

	( y = cos(d / v) * h -- cos(double) has the same qword argument spill )
	{
	    D9 87 <dGRfd  mtp bytesperunit>	(fld dword [edi+GRfd*4])
	    D8 B7 <dGRfv  mtp bytesperunit>	(fdiv dword [edi+GRfv*4])
	    DD 9F <dGRfs0 mtp bytesperunit>	(fstp qword [edi+GRfs0*4])
	    DD 87 <dGRfs0 mtp bytesperunit>	(fld qword [edi+GRfs0*4])
	    D9 FF				(fcos)
	    D8 8F <dGRfht mtp bytesperunit>	(fmul dword [edi+GRfht*4])
	    D9 9F <dGRfy  mtp bytesperunit>	(fstp dword [edi+GRfy*4])
	}

	( if (y >= 0) -- check sign bit of float32 )
	A = [GRfy]; A & 80000000h;
	? A != 0 -> GR rh znext;

	( y += surfacemap[200*z + x] )
	A = [GRfrz]; A '* 200; A + [GRfrx]; [GRptr] = A;
	A = RPSM; A + [GRptr]; [MBptr] = A;
	=> MEM get byte;
	[GRsval] = [MBval];
	{
	    D9 87 <dGRfy   mtp bytesperunit>	(fld dword [edi+GRfy*4])
	    DB 87 <dGRsval mtp bytesperunit>	(fild dword [edi+GRsval*4])
	    DE C1				(faddp st1,st0)
	    D9 9F <dGRfy   mtp bytesperunit>	(fstp dword [edi+GRfy*4])
	}

	( canyon mirror or hmax clamp )
	A = [GRfcanyon];
	? A = 0 -> GR rh clamp;

	( if (y > 127) y = 254 - y -- compute y - 127.0, check sign )
	{
	    D9 87 <dGRfy mtp bytesperunit>	(fld dword [edi+GRfy*4])
	    DC A7 <dGRK127L mtp bytesperunit>	(fsub qword [edi+GRK127L*4])
	    DD 9F <dGRfs0 mtp bytesperunit>	(fstp qword [edi+GRfs0*4])
	}
	A = [GRfs1]; A & 80000000h;
	? A != 0 -> GR rh store;		( y - 127 < 0 → y <= 127, skip )
	( y > 127: canyon mirror )
	{
	    D9 87 <dGRfy mtp bytesperunit>	(fld dword [edi+GRfy*4])
	    DC AF <dGRK254L mtp bytesperunit>	(fsubr qword [edi+GRK254L*4])
	    D9 9F <dGRfy  mtp bytesperunit>	(fstp dword [edi+GRfy*4])
	}
	-> GR rh store;

"GR rh clamp"
	( if (y > hmax) y = hmax -- compute y - hmax, check sign )
	{
	    D9 87 <dGRfy mtp bytesperunit>	(fld dword [edi+GRfy*4])
	    D8 A7 <dGRfhmt mtp bytesperunit>	(fsub dword [edi+GRfhmt*4])
	    DD 9F <dGRfs0 mtp bytesperunit>	(fstp qword [edi+GRfs0*4])
	}
	A = [GRfs1]; A & 80000000h;
	? A != 0 -> GR rh store;		( y - hmax < 0 → skip )
	[GRfy] = [GRfhmt];			( y = hmax )

"GR rh store"
	( surfacemap[idx] = (unsigned char)y -- chop )
	{
	    D9 87 <dGRfy mtp bytesperunit>	(fld dword [edi+GRfy*4])
	    D9 AF <dGRcwc mtp bytesperunit>	(fldcw [edi+GRcwc*4])
	    DB 9F <dGRfti mtp bytesperunit>	(fistp dword [edi+GRfti*4])
	    D9 AF <dGRcwn mtp bytesperunit>	(fldcw [edi+GRcwn*4])
	}
	A = [GRfti]; A & 0FFh;
	[MBval] = A;
	A = RPSM; A + [GRptr]; [MBptr] = A;
	=> MEM put byte;

"GR rh znext"
	[GRfrz]+;
	A = [GRfrz];
	B = [SUcyi]; B + [SUia];	( cz + r )
	? A '< B -> GR rh zloop;
	[GRfrx]+;
	-> GR rh xloop;

"GR rh done"
	end;

( ==================================================================== )
( srf_darkline -- NOCTIS-1.CPP:1589-1603                                )
( ==================================================================== )
( Input: [SUia] = align, [GRdxi] = length,                             )
(        [GRfrlo] = x_trend, [GRfrhi] = z_trend                        )
( A random walk that halves bytes along its path.                      )

"GR srf darkline"
	( fx = random(align), fz = random(align) )
	C = [SUia]; => SU rnd; [GRfrx] = C;
	C = [SUia]; => SU rnd; [GRfrz] = C;
	[GRabs] = [SUia]; [GRabs] '* [SUia];	( mapsize = align*align )

"GR sd loop"
	? [GRn] '<= 0 -> GR sd done;
	( fx += random(3) + x_trend )
	C = 3; => SU rnd;
	A = [GRfrx]; A + C; A + [GRfrlo]; [GRfrx] = A;
	( fz += random(3) + z_trend )
	C = 3; => SU rnd;
	A = [GRfrz]; A + C; A + [GRfrhi]; [GRfrz] = A;
	( location = align * fz + fx )
	A = [SUia]; A '* [GRfrz]; A + [GRfrx]; A & 0FFFFh; [GRptr] = A;
	( if (location > 0 && location < mapsize) map[location] >>= 1 )
	? [GRptr] '<= 0 -> GR sd next;
	? [GRptr] '>= [GRabs] -> GR sd next;
	A = RPSM; A + [GRptr]; [MBptr] = A;
	=> MEM get byte;
	A = [MBval]; A > 1; A & 0FFh;
	[MBval] = A;
	A = RPSM; A + [GRptr]; [MBptr] = A;
	=> MEM put byte;
"GR sd next"
	[GRn]-;
	-> GR sd loop;
"GR sd done"
	end;

( ==================================================================== )
( std_crater -- NOCTIS-1.CPP:1561-1587                                  )
( ==================================================================== )
( Input: [SUcxi]=cx, [SUcyi]=cz, [SUia]=r (signed int)                 )
(        [GRfht]=h_factor (float32), [GRfhmt]=h_raiser (float32)      )
(        [GRfscl]=lim_h (int), [GRfsca]=align (int, 200 or 256)       )
(        map base in [GRpksrc] (RPSM for smap, RPBG for txtr)         )
( h_raiser=1.0: pow is identity. Other positive profiles use the x87 )
( fyl2x/frndint/f2xm1/fscale sequence, preserving the live float32 y. )

"GR std crater"
	[GRcraterid]+;

	( h = (float)r * h_factor -- fild r; fmul h_factor; fstp dword h )
	{
	    DB 87 <dSUia mtp bytesperunit>	(fild dword [edi+SUia*4])
	    D8 8F <dGRfht mtp bytesperunit>	(fmul dword [edi+GRfht*4])
	    D9 9F <dGRfsch mtp bytesperunit>	(fstp dword [edi+GRfsch*4])
	}

	( r = abs(r) )
	A = [SUia];
	? A >= 0 -> GR sc absok;
	B = 0; B - A; A = B;
"GR sc absok"
	[SUia] = A;
	C = A; A '* C; [GRfr2] = A;

	( fr = (float)r -- fild abs(r); fstp dword fr )
	{
	    DB 87 <dSUia mtp bytesperunit>	(fild dword [edi+SUia*4])
	    D9 9F <dGRfscr mtp bytesperunit>	(fstp dword [edi+GRfscr*4])
	}

	( Clip the source's half-open crater rectangle once.  Skipped out-of-map
	  iterations have no side effects or RNG calls, so this is identical to
	  the original per-pixel bounds checks. )
	A = [SUia];
	B = [SUcxi];
	[GRfrlo] = B; [GRfrlo] - A;
	C = B; C + A;
	[GRfrhi] = C;
	A = [GRfrlo]; ? A >= 0 -> GR sc xlo ready; [GRfrlo] = 0;
    "GR sc xlo ready"
	A = [GRfrhi]; ? A '<= [GRfsca] -> GR sc xhi ready; [GRfrhi] = [GRfsca];
    "GR sc xhi ready"
	A = [SUia]; B = [SUcyi];
	[GRsczlo] = B; [GRsczlo] - A;
	C = B; C + A; [GRsczhi] = C;
	A = [GRsczlo]; ? A >= 0 -> GR sc zlo ready; [GRsczlo] = 0;
    "GR sc zlo ready"
	A = [GRsczhi]; ? A '<= [GRfsca] -> GR sc zhi ready; [GRsczhi] = [GRfsca];
    "GR sc zhi ready"
	A = [GRfrlo]; ? A '>= [GRfrhi] -> GR sc done;
	A = [GRsczlo]; ? A '>= [GRsczhi] -> GR sc done;
	[GRfrx] = [GRfrlo];

"GR sc xloop"
	A = [GRfrx];
	? A '>= [GRfrhi] -> GR sc done;

	( The integer radial key and the map pointer are both recurrence values
	  down the z-major inner loop: d2 advances by 2*dz+1 and the address
	  advances by align. )
	B = [GRfrx]; B - [SUcxi]; [GRdxi] = B;
	A = B; A '* B; [GRfd2] = A;
	[GRfrz] = [GRsczlo];
	C = [GRfrz]; C - [SUcyi]; [GRdzi] = C;
	A = C; A '* C; D = [GRfd2]; D + A; [GRfd2] = D;
	A = C; A + C; A+; [GRfd2delta] = A;
	A = [GRfrz]; A '* [GRfsca]; A + [GRfrx]; A + [GRscmap]; [GRscaddr] = A;

"GR sc zloop"
	( Integer radius rejection is exact here: dx and dz are small integers,
	  and the source accepts precisely sqrt(dx*dx+dz*dz) <= r.  It also gives
	  the radial-profile cache its stable key before any x87 work. )
	A = [GRfd2];
	? A > [GRfr2] -> GR sc znext;
	A = GRsstamp; A + [GRfd2]; A = [A];
	? A != [GRcraterid] -> GR sc profile miss;
	A = GRscache; A + [GRfd2]; [GRfy] = [A];
	-> GR sc profile ready;

"GR sc profile miss"
	( The source stores the small integer dx and dz exactly as binary32,
	  then squares and adds them exactly before fsqrt.  GRfd2 is that same
	  exact integer sum, so loading it directly removes redundant float
	  conversions and arithmetic without changing fsqrt's input or result. )
	{
	    DB 87 <dGRfd2 mtp bytesperunit>	(fild dword [edi+GRfd2*4])
	    D9 FA				(fsqrt)
	    D9 9F <dGRfd mtp bytesperunit>	(fstp dword [edi+GRfd*4])
	}

	( d <= fr: compute y = sin(M_PI * d / fr) * h )
	{
	    DD 87 <dGRKPIL mtp bytesperunit>	(fld qword [edi+GRKPIL*4])
	    D9 87 <dGRfd  mtp bytesperunit>	(fld dword [edi+GRfd*4])
	    D9 87 <dGRfscr mtp bytesperunit>	(fld dword [edi+GRfscr*4])
	    DE F9				(fdivp st1,st0 : d/fr)
	    DE C9				(fmulp st1,st0 : M_PI*(d/fr))
	    D9 FE				(fsin)
	    D8 8F <dGRfsch mtp bytesperunit>	(fmul dword [edi+GRfsch*4] : sin*h)
	    D9 9F <dGRfy  mtp bytesperunit>	(fstp dword [edi+GRfy*4])
	}

	( y = pow(y,h_raiser). Non-unit raisers are used only by positive
	  type-1/type-5 terrain craters; zero stays zero without log2(0). )
	A = [GRfhmt]; A & 0FFFFFFFFh; ? A = 3F800000h -> GR sc power done;
	A = [GRfy]; A & 7FFFFFFFh; ? A = 0 -> GR sc power done;
	{
	    D9 87 <dGRfhmt mtp bytesperunit>	(fld dword [edi+GRfhmt*4] : exponent)
	    D9 87 <dGRfy   mtp bytesperunit>	(fld dword [edi+GRfy*4] : y)
	    D9 F1				(fyl2x : t=exponent*log2(y))
	    D9 C0				(fld st0)
	    D9 FC				(frndint : integral power)
	    DC E9				(fsub st1,st0 : fractional power)
	    D9 C9				(fxch st1)
	    D9 F0				(f2xm1)
	    D9 E8				(fld1)
	    DE C1				(faddp st1,st0)
	    D9 FD				(fscale)
	    DD D9				(fstp st1)
	    D9 9F <dGRfy mtp bytesperunit>	(fstp dword [edi+GRfy*4])
	}
"GR sc power done"
	A = GRscache; A + [GRfd2]; [A] = [GRfy];
	A = GRsstamp; A + [GRfd2]; [A] = [GRcraterid];

"GR sc profile ready"

	( y += map[align*z + x].  The clipped rectangle proves this direct
	  one-byte load has the exact semantics of MEM get byte. )
	A = nw; A + [GRscaddr]; C = [A]; C & 255; [GRsval] = C;
	{
	    D9 87 <dGRfy   mtp bytesperunit>	(fld dword [edi+GRfy*4])
	    DB 87 <dGRsval mtp bytesperunit>	(fild dword [edi+GRsval*4])
	    DE C1				(faddp st1,st0)
	    D9 9F <dGRfy   mtp bytesperunit>	(fstp dword [edi+GRfy*4])
	}

	( if (y < 0) y = 0 )
	A = [GRfy]; A & 80000000h;
	? A = 0 -> GR sc clamphi;
	[GRfy] = 0;

"GR sc clamphi"
	( if (y > lim_h) y = lim_h -- compute y - lim_h, check sign )
	{
	    D9 87 <dGRfy mtp bytesperunit>	(fld dword [edi+GRfy*4])
	    DA A7 <dGRfscl mtp bytesperunit>	(fisub dword [edi+GRfscl*4])
	    DD 9F <dGRfs0 mtp bytesperunit>	(fstp qword [edi+GRfs0*4])
	}
	A = [GRfs1]; A & 80000000h;
	? A != 0 -> GR sc store;	( y - lim_h < 0 → y < lim_h, no clamp )
	( y >= lim_h: clamp to lim_h )
	{
	    DB 87 <dGRfscl mtp bytesperunit>	(fild dword [edi+GRfscl*4])
	    D9 9F <dGRfy mtp bytesperunit>	(fstp dword [edi+GRfy*4])
	}

"GR sc store"
	( map[idx] = (unsigned char)y -- chop )
	{
	    D9 87 <dGRfy mtp bytesperunit>	(fld dword [edi+GRfy*4])
	    D9 AF <dGRcwc mtp bytesperunit>	(fldcw [edi+GRcwc*4])
	    DB 9F <dGRfti mtp bytesperunit>	(fistp dword [edi+GRfti*4])
	    D9 AF <dGRcwn mtp bytesperunit>	(fldcw [edi+GRcwn*4])
	}
	A = [GRfti]; A & 0FFh;
	C = A; A = nw; A + [GRscaddr]; [A] = C;

"GR sc znext"
	A = [GRfd2]; A + [GRfd2delta]; [GRfd2] = A;
	[GRfd2delta] + 2;
	A = [GRscaddr]; A + [GRfsca]; [GRscaddr] = A;
	[GRdzi]+;
	[GRfrz]+;
	A = [GRfrz]; ? A '< [GRsczhi] -> GR sc zloop;
	[GRfrx]+;
	-> GR sc xloop;

"GR sc done"
	end;
