( rockyground — NOCTIS-1.CPP:1545-1559                                  )

      ( GR rockyground — fills p_surfacemap with random(roughness), runs
	smoothterrain(rounding), then adjusts by level.

	Input:  [GRrough], [GRround], [GRlevel]
	Output: p_surfacemap[0..39999] modified				)

"GR rockyground"

	( phase 1: fill with random(roughness) )
	[GRptr] = 0;
"GR rg fill"
	C = [GRrough];
	=> SU rnd;
	C & 0FFh;
	[MBval] = C;
	A = RPSM; A + [GRptr]; [MBptr] = A;
	=> MEM put byte;
	[GRptr]+;
	? [GRptr] '< 40000 -> GR rg fill;

	( phase 2: smooth )
	C = [GRround];
	=> GR smoothterrain;

	( phase 3: level adjust )
	( compute abs(level) )
	A = [GRlevel];
	? A '>= 0 -> GR rg absok;
	B = 0; B - A; A = B;
"GR rg absok"
	[GRabs] = A;

	[GRptr] = 0;
"GR rg level"
	A = RPSM; A + [GRptr]; [MBptr] = A;
	=> MEM get byte;
	A = [MBval];
	? A '>= [GRabs] -> GR rg add;
	( else set to 0 )
	C = 0;
	[MBval] = C;
	A = RPSM; A + [GRptr]; [MBptr] = A;
	=> MEM put byte;
	-> GR rg next;

"GR rg add"
	A = [MBval]; A + [GRlevel];
	? A '<= 127 -> GR rg store;
	A = 127;
"GR rg store"
	C = A; C & 0FFh;
	[MBval] = C;
	A = RPSM; A + [GRptr]; [MBptr] = A;
	=> MEM put byte;

"GR rg next"
	[GRptr]+;
	? [GRptr] '< 40000 -> GR rg level;
	end;

      ( GR smoothterrain — 2x2 box average, `rounding` passes.

	  while (rounding) {
	      for (ptr = 0; ptr < 39799; ptr++) {
		  n  = p_surfacemap[ptr];
		  n += p_surfacemap[ptr + 1];
		  n += p_surfacemap[ptr + 200];
		  n += p_surfacemap[ptr + 201];
		  p_surfacemap[ptr] = n >> 2;
	      }
	      rounding--;
	  }

	Input: C = passes.  No output regs.				)

"GR smoothterrain"
	[GRsmp] = C;
"GR sm outer"
	? [GRsmp] '<= 0 -> GR sm end;
	[GRptr] = 0;
"GR sm loop"
	? [GRptr] '>= 39799 -> GR sm nextpass;

	( n = smap[ptr] )
	A = RPSM; A + [GRptr]; [MBptr] = A;
	=> MEM get byte;
	A = [MBval]; [GRn] = A;

	( n += smap[ptr+1] )
	A = RPSM; A + [GRptr]; A + 1; [MBptr] = A;
	=> MEM get byte;
	A = [GRn]; A + [MBval]; [GRn] = A;

	( n += smap[ptr+200] )
	A = 200; A + [GRptr]; A + RPSM; [MBptr] = A;
	=> MEM get byte;
	A = [GRn]; A + [MBval]; [GRn] = A;

	( n += smap[ptr+201] )
	A = 201; A + [GRptr]; A + RPSM; [MBptr] = A;
	=> MEM get byte;
	A = [GRn]; A + [MBval]; [GRn] = A;

	A = [GRn]; A > 2; A & 0FFh;
	[MBval] = A;
	A = RPSM; A + [GRptr]; [MBptr] = A;
	=> MEM put byte;

	[GRptr]+;
	-> GR sm loop;

	[GRsmp]-;
	-> GR sm outer;
"GR sm end"
	end;

( round_hill — NOCTIS-1.CPP:1494-1528                                  )
( ==================================================================== )
( Input: [SUcxi] = cx, [SUcyi] = cz, [SUia] = r (unsigned int)        )
(        [GRfht] = h (float32 bits), [GRfhmt] = hmax (float32 bits)   )
(        [GRfcanyon] = allowcanyons (0 or 1)                          )
(                                                                     )
( The x87 ML fragments keep intermediates on the stack and only store  )
( to float32 where the C code stores, matching Borland's codegen.     )

"GR round hill"

	( v = (float)r / M_PI_2 — fild r; fdiv M_PI_2; fstp dword v )
	{
	    DB 87 <dSUia   mtp bytesperunit>	(fild dword [edi+SUia*4])
	    DC B7 <dGRKPI2L mtp bytesperunit>	(fdiv qword [edi+GRKPI2L*4])
	    D9 9F <dGRfv   mtp bytesperunit>	(fstp dword [edi+GRfv*4])
	}

	( loop bounds: x from cx-r to cx+r-1, z from cz-r to cz+r-1 )
	A = [SUia];			( r )
	B = [SUcxi];			( cx )
	[GRfrlo] = B; [GRfrlo] - A;	( cx - r )
	C = B; C + A;
	[GRfrhi] = C;			( cx + r )
	[GRfrx] = [GRfrlo];

"GR rh xloop"
	A = [GRfrx];
	? A '>= [GRfrhi] -> GR rh done;

	( z loop bounds )
	A = [SUia];
	B = [SUcyi];
	D = B; D - A; [GRfrz] = D;

"GR rh zloop"
	A = [GRfrx];
	? A '< 0 -> GR rh znext;
	? A '>= 200 -> GR rh znext;
	A = [GRfrz];
	? A '< 0 -> GR rh znext;
	? A '>= 200 -> GR rh znext;

	( dx = (float)(x - cx) )
	B = [GRfrx]; B - [SUcxi]; [GRdxi] = B;
	{
	    DB 87 <dGRdxi mtp bytesperunit>	(fild dword [edi+GRdxi*4])
	    D9 9F <dGRfdx mtp bytesperunit>	(fstp dword [edi+GRfdx*4])
	}

	( dz = (float)(z - cz) )
	B = [GRfrz]; B - [SUcyi]; [GRdzi] = B;
	{
	    DB 87 <dGRdzi mtp bytesperunit>	(fild dword [edi+GRdzi*4])
	    D9 9F <dGRfdz mtp bytesperunit>	(fstp dword [edi+GRfdz*4])
	}

	( d = sqrt(dx*dx + dz*dz) )
	{
	    D9 87 <dGRfdx mtp bytesperunit>	(fld dword [edi+GRfdx*4])
	    D8 8F <dGRfdx mtp bytesperunit>	(fmul dword [edi+GRfdx*4])
	    D9 87 <dGRfdz mtp bytesperunit>	(fld dword [edi+GRfdz*4])
	    D8 8F <dGRfdz mtp bytesperunit>	(fmul dword [edi+GRfdz*4])
	    DE C1				(faddp st1,st0)
	    D9 FA				(fsqrt)
	    D9 9F <dGRfd mtp bytesperunit>	(fstp dword [edi+GRfd*4])
	}

	( y = cos(d / v) * h )
	{
	    D9 87 <dGRfd  mtp bytesperunit>	(fld dword [edi+GRfd*4])
	    D8 B7 <dGRfv  mtp bytesperunit>	(fdiv dword [edi+GRfv*4])
	    D9 FF				(fcos)
	    D8 8F <dGRfht mtp bytesperunit>	(fmul dword [edi+GRfht*4])
	    D9 9F <dGRfy  mtp bytesperunit>	(fstp dword [edi+GRfy*4])
	}

	( if (y >= 0) — check sign bit of float32 )
	A = [GRfy]; A & 80000000h;
	? A != 0 -> GR rh znext;

	( y += surfacemap[200*z + x] )
	A = [GRfrz]; A '* 200; A + [GRfrx]; [GRptr] = A;
	A = RPSM; A + [GRptr]; [MBptr] = A;
	=> MEM get byte;
	[GRsval] = [MBval];
	{
	    D9 87 <dGRfy   mtp bytesperunit>	(fld dword [edi+GRfy*4])
	    DB 87 <dGRsval mtp bytesperunit>	(fild dword [edi+GRsval*4])
	    DE C1				(faddp st1,st0)
	    D9 9F <dGRfy   mtp bytesperunit>	(fstp dword [edi+GRfy*4])
	}

	( canyon mirror or hmax clamp )
	A = [GRfcanyon];
	? A = 0 -> GR rh clamp;

	( if (y > 127) y = 254 - y — compute y - 127.0, check sign )
	{
	    D9 87 <dGRfy mtp bytesperunit>	(fld dword [edi+GRfy*4])
	    DC A7 <dGRK127L mtp bytesperunit>	(fsub qword [edi+GRK127L*4])
	    DD 9F <dGRfs0 mtp bytesperunit>	(fstp qword [edi+GRfs0*4])
	}
	A = [GRfs1]; A & 80000000h;
	? A != 0 -> GR rh store;		( y - 127 < 0 → y <= 127, skip )
	( y > 127: canyon mirror )
	{
	    DC 87 <dGRK254L mtp bytesperunit>	(fld qword [edi+GRK254L*4])
	    D8 87 <dGRfy  mtp bytesperunit>	(fsub dword [edi+GRfy*4])
	    D9 9F <dGRfy  mtp bytesperunit>	(fstp dword [edi+GRfy*4])
	}
	-> GR rh store;

"GR rh clamp"
	( if (y > hmax) y = hmax — compute y - hmax, check sign )
	{
	    D9 87 <dGRfy mtp bytesperunit>	(fld dword [edi+GRfy*4])
	    D8 A7 <dGRfhmt mtp bytesperunit>	(fsub dword [edi+GRfhmt*4])
	    DD 9F <dGRfs0 mtp bytesperunit>	(fstp qword [edi+GRfs0*4])
	}
	A = [GRfs1]; A & 80000000h;
	? A != 0 -> GR rh store;		( y - hmax < 0 → skip )
	[GRfy] = [GRfhmt];			( y = hmax )

"GR rh store"
	( surfacemap[idx] = (unsigned char)y — chop )
	{
	    D9 87 <dGRfy mtp bytesperunit>	(fld dword [edi+GRfy*4])
	    D9 AF <dGRcwc mtp bytesperunit>	(fldcw [edi+GRcwc*4])
	    DB 9F <dGRfti mtp bytesperunit>	(fistp dword [edi+GRfti*4])
	    D9 AF <dGRcwn mtp bytesperunit>	(fldcw [edi+GRcwn*4])
	}
	A = [GRfti]; A & 0FFh;
	[MBval] = A;
	A = RPSM; A + [GRptr]; [MBptr] = A;
	=> MEM put byte;

"GR rh znext"
	[GRfrz]+;
	A = [GRfrz];
	B = [SUcyi]; B + [SUia];	( cz + r )
	? A '< B -> GR rh zloop;
	[GRfrx]+;
	-> GR rh xloop;

"GR rh done"
	end;
