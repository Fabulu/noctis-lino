	[VHSVok] = 1;

	  A crash during the primary write therefore leaves the previous backup;
	  a crash during this best-effort copy still leaves a valid primary. )

	[Block Pointer] = vhsvbuf; [Block Size] = 256; isocall;
	[File Name] = vhsvbackup; [File Position] = 0; [File Command] = SET SIZE;
	[File Size] = 256; isocall;
    "VHSV save done"
	end;

"VHSV load"
	[VHSVusebackup] = 0; [VHSVrecovered] = 0;
	=> VHSV load one; A = [VHSVok]; ? A != 0 -> VHSV load wrapper done;
	( A deliberately missing primary starts a clean game even if an old backup
	  was left behind. Only a present-but-invalid primary triggers recovery. )
	[File Name] = vhsvname; [File Command] = TEST; isocall;
	? [File Status] + ERROR -> VHSV load wrapper done;
	[VHSVusebackup] = 1; => VHSV load one;
	A = [VHSVok]; ? A = 0 -> VHSV load wrapper done; [VHSVrecovered] = 1;
    "VHSV load wrapper done"
	end;

"VHSV select load file"
	A = [VHSVusebackup]; ? A = 0 -> VHSV select primary;
	[File Name] = vhsvbackup; end;
    "VHSV select primary"
	[File Name] = vhsvname;
	end;
