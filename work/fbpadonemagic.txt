( ==================================================================== )
( the ZONE table - 22 zones over 11 pads                                )
( ==================================================================== )

      ( 4 units per zone: base, length, owner region id (-1 = unowned),
	role (0 TAIL, 1 SUB).  Zone index zi = 2*pad + role, so the pad
	index is zi >> 1 and the table is a pure function of the layout.

	This IS FBDUMP kind 9, emitted straight out of ztab.

	Pad p covers nw[padbase(p) .. padbase(p)+15]:
	  p = 0   nw[0..15]        both zones UNOWNED
	  p = 1   nw[16..31]       SUB owned by region 0; SUB+4 = SNOFF
	  p = 2..9                 the eight inter-region gaps
	  p = 10  nw[402180..]     TAIL owned by region 8; SUB UNOWNED

	Unowned zones carry an EMPTY allowance list, so any change there
	is a violation. )

"MEM build ztab"
	[MEMi] = 0;
    "MEM bz pad"
	=> MEM pad base;		( [MEMtmp] <- padbase([MEMi]) )
	A = [MEMi]; A + A; A '* 4; A + ztab;	( 8 units per pad )

	( TAIL: owner = pad - 2, unowned for pads 0 and 1 )
	[A plus 0] = [MEMtmp];
	[A plus 1] = ZHALF;
	[A plus 3] = ROLTAIL;
	C = [MEMi]; C - 2;
	? C >= 0 -> MEM bz t;
	C = 0; C-;
    "MEM bz t"
	[A plus 2] = C;

	( SUB: owner = pad - 1, unowned for pad 0 and pad 10 )
	C = [MEMtmp]; C + ZHALF;
	[A plus 4] = C;
	[A plus 5] = ZHALF;
	[A plus 7] = ROLSUB;
	C = [MEMi]; C - 1;
	? C < 0 -> MEM bz sneg;
	? C '> 8 -> MEM bz sneg;
	-> MEM bz s;
    "MEM bz sneg"
	C = 0; C-;
    "MEM bz s"
	[A plus 6] = C;

	[MEMi]+;
	A = [MEMi]; ? A '< NPAD -> MEM bz pad;
	end;

      ( MEM pad base : [MEMtmp] <- the nw offset of pad [MEMi].
	Written out rather than derived from rtab, because rtab has nine
	entries and there are eleven pads - deriving it from rtab is
	exactly the bug that left nw[0..31] unguarded. )

"MEM pad base"
	A = [MEMi];
	? A = 0  -> MEM pb 0;
	? A = 1  -> MEM pb 1;
	? A = 2  -> MEM pb 2;
	? A = 3  -> MEM pb 3;
	? A = 4  -> MEM pb 4;
	? A = 5  -> MEM pb 5;
	? A = 6  -> MEM pb 6;
	? A = 7  -> MEM pb 7;
	? A = 8  -> MEM pb 8;
	? A = 9  -> MEM pb 9;
	C = RADPR; C + ZADPR; [MEMtmp] = C; end;
    "MEM pb 0"
	[MEMtmp] = 0; end;
    "MEM pb 1"
	[MEMtmp] = PADU; end;
    "MEM pb 2"
	C = RNOFF; C + ZNOFF; [MEMtmp] = C; end;
    "MEM pb 3"
	C = RNGLB; C + ZNGLB; [MEMtmp] = C; end;
    "MEM pb 4"
	C = RSBG;  C + ZSBG;  [MEMtmp] = C; end;
    "MEM pb 5"
	C = RPBG;  C + ZPBG;  [MEMtmp] = C; end;
    "MEM pb 6"
	C = RPSM;  C + ZPSM;  [MEMtmp] = C; end;
    "MEM pb 7"
	C = ROBJ;  C + ZOBJ;  [MEMtmp] = C; end;
    "MEM pb 8"
	C = RPVF;  C + ZPVF;  [MEMtmp] = C; end;
    "MEM pb 9"
	C = RADPT; C + ZADPT; [MEMtmp] = C; end;

      ( MEM zone allow : [ZAmask] <- the 8-bit allowance mask of the zone
	described by [ZAowner] and [ZArole].  Bit u set means "unit u of

	Three entries, each with its citation.  Deriving the mask instead
	of storing it keeps ztab exactly equal to FBDUMP kind 9. )

	[ZAmask] = 0;

	A = [ZArole]; ? A = ROLTAIL -> MEM za tail;

	A = [ZAowner]; ? A = 8 -> MEM za sub2;

	end;

      ( MEM u16 site : the same mask, COUNTED against [SAsite].

	The counter belongs AT THE TRUNCATION POINT, not at the final
	address, for the same reason the mask does.  cirrus truncates at
	"mov bx,py / add bx,px" and only then shifts, so by the time the
	address is formed the offset can no longer exceed 16 bits and a
	counter placed there would read zero forever.  spot truncates at
	the 16-bit DI, which IS the address offset, so for spot the two
	points coincide - which is precisely why one counter site is not
	enough to distinguish them. )

"MEM u16 site"
	A = [SAsite]; A '* 3; A + wtab;
	[A plus 1]+;
	? C '<= WINDOW -> MEM u16s nw;
	[A plus 2]+;
    "MEM u16s nw"
	C & 65535;
	end;

( ==================================================================== )
( the 16-bit index mask - the class-A mechanism                         )
( ==================================================================== )

      ( MEM seg addr : [SAnw] <- SEG + (OFF AND 65535).
	in  [SAseg] segment origin, [SAoff] offset
	out [SAnw]  workspace index

	THE MASK IS TAKEN AGAINST THE SEGMENT ORIGIN, NOT THE BUFFER
	BASE.  SEG is R* minus 4 for every farmalloc'd block, so a masked
	offset of 0..3 lands on the four header units immediately below
	the buffer - which is the SUB zone's allowance, not a violation.
	Using the base here instead of the origin is sabotage
	S-SEGADDR-BASE and shows up as a containment failure on every
	low-offset case.

	The wrap COUNTERS live in "MEM u16 site", at the truncation
	point.  They are the graded artifact: delete a mask and the count
	goes to zero while the page content changes, so the check is
	breakable in two independent places. )

"MEM seg addr"
	C = [SAoff];
	=> MEM u16;
	C + [SAseg];
	[SAnw] = C;
	end;

      ( MEM seg check : the containment assertion.  [SAnw] must lie in
	the region itself, from base to base+size-1, or in that region's
	SUB zone - and those two intervals are CONTIGUOUS, because SUB is
	base-8 .. base-1.  So the window is [SAlo] .. [SAhi] minus one,
	with SAlo = base - 8 and SAhi = base + size.

	Note what this does NOT claim: the wrappable window and the buffer
	are not the same interval.  adapted's 16-bit window is
	nw[271,080..336,615], so adapted[65536..65539] - the four units
	sc_bytes = 65540 exists for - lie OUTSIDE the wrap window and are
	reachable only through the non-truncating path.                  )

"MEM seg check"
	A = [SAnw];
	? A '< [SAlo] -> MEM sc bad;
	? A '>= [SAhi] -> MEM sc bad;
	end;
    "MEM sc bad"
	[MCoob]+;
	? [MCoobat] != 0 -> MEM sc done;
	[MCoobat] = [SAnw];
    "MEM sc done"
	end;

"MEM wrap reset"
	A = wtab; C = 0; B = 0; B + NSITE;
    "MEM wr site"
	[A plus 0] = C; [A plus 1] = 0; [A plus 2] = 0;
	A + 3; C+;
	B ^ MEM wr site;
	[MCoob] = 0; [MCoobat] = 0;
	end;

	[MCui]+;
	A = [MCui]; ? A '< ZHALF -> MEM cp unit;
	[MCzi]+;
	A = [MCzi]; ? A '< NZONE -> MEM cp zone;
	end;

( ==================================================================== )
( FBDUMP v2 staging                                                     )
( ==================================================================== )

"DMP reset"
	[DPptr] = dump;
	[DPcnt] = 0;
	end;

      ( DMP rec : append one FBDUMP record.
	in  [DKind] [DTag] [DW] [DH] [DBptr] [DBcnt] [DCpms] [DTicks] )

"DMP rec"
	A = [DPptr];
	[A plus 0]  = FBMAGIC;
	[A plus 1]  = FBVER;
	[A plus 2]  = [DKind];
	[A plus 3]  = [DW];
	[A plus 4]  = [DH];
	[A plus 5]  = [DBcnt];
	[A plus 6]  = [DCpms];
	[A plus 7]  = [DTicks];
	[A plus 8]  = [DTag];
	[A plus 9]  = 0;
	[A plus 10] = 0;
	[A plus 11] = 0;
	[A plus 12] = 0;
	[A plus 13] = 0;
	[A plus 14] = 0;
	[A plus 15] = 0;
	A + DHDRU;
	C = [DPcnt]; C + DHDRU; [DPcnt] = C;

	D = [DBptr];
	B = [DBcnt];
	? B = 0 -> DMP rec tail;
    "DMP rec unit"
	C = [D]; [A] = C; A+; D+;
	B ^ DMP rec unit;
	C = [DPcnt]; C + [DBcnt]; [DPcnt] = C;
    "DMP rec tail"
	[DPptr] = A;
	end;

      ( write the whole staged stream, then trim the file to it. )

"DMP write"
	[File Name]	= fbdumpname;
	[File Position]	= 0;
	[File Command]	= WRITE;
	[Block Pointer]	= dump;
	[Block Size]	= [DPcnt];
	[Block Size]	* BYTESPERUNIT;
	isocall;

	[File Name]	= fbdumpname;
	[File Position]	= 0;
	[File Command]	= SET SIZE;
	[File Size]	= [DPcnt];
	[File Size]	* BYTESPERUNIT;
	isocall;
	end;
