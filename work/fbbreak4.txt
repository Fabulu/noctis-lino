( ==================================================================== )
( the reference clock                                                   )
( ==================================================================== )

      ( TK read wall : [TKtmp] <- milliseconds since midnight.
	READ TIME's [Milliseconds] steps by 1 ms in 200 of 256 observed
	transitions, so it is a usable reference for a multi-second
	bracket even though it is useless for a single tick. )

"TK read wall"
	[Timer Command] = READ TIME;
	isocall;
	A = [Hour];   A * 3600000;
	B = [Minute]; B * 60000;   A + B;
	B = [Second]; B * 1000;    A + B;
	A + [Milliseconds];
	[TKtmp] = A;
	end;

( ==================================================================== )
( the period                                                            )
( ==================================================================== )

"TK period"
	A = [TKcpms]; A '* 55;   [TKbase]   = A;
	A = [TKcpms]; A '* SUBN; [TKsubper] = A;
	end;

      ( TK advance : deadline += one exact period. )

"TK advance"
	A = [TKacc]; A + [TKsubper];
	B = A; B '/ SUBD;
	C = B; C '* SUBD; A - C;
	[TKacc] = A;
	A = [TKbase]; A - B;
	[TKdeadline] + A;
	end;

      ( TK seed : take the runtime's guess, and remember it. )

"TK seed"
	[TKcpms0] = [Counts Per Millisecond];
	[TKcpms]  = [Counts Per Millisecond];
	[TKmargin] = TKSPINMS;

"TK cal begin"
	[Timer Command] = READ COUNTS; isocall;
	[TKcal0c] = [Counts];
	=> TK read wall; [TKcal0w] = [TKtmp];
	end;

      ( TK cal end : recompute from the bracket, but only if the bracket
	is long enough to mean anything. )

"TK cal end"
	[Timer Command] = READ COUNTS; isocall;
	A = [Counts]; A - [TKcal0c]; [TKcalcnt] = A;
	=> TK read wall;
	A = [TKtmp]; A - [TKcal0w]; [TKcalms] = A;
	? [TKcalms] '< 500 -> TK cal keep;
	A = [TKcalcnt]; A '/ [TKcalms]; [TKcpms] = A;
	=> TK period;
    "TK cal keep"
	end;

      ( TK servo : every SERVON ticks, against TOTAL counts / TOTAL ms
	since the run started, bounded to +-1%. )

"TK servo"
	[TKsrvn]+;
	A = [TKsrvn]; ? A '< SERVON -> TK sv done;
	[TKsrvn] = 0;
	[Timer Command] = READ COUNTS; isocall;
	A = [Counts]; A - [TKsrv0c]; [TKsrvcnt] = A;
	=> TK read wall;
	A = [TKtmp]; A - [TKsrv0w]; [TKsrvms] = A;
	? [TKsrvms] '< 500 -> TK sv done;
	A = [TKsrvcnt]; A '/ [TKsrvms]; [TKsrvnew] = A;
	B = [TKcpms]; B '/ 100;
	C = [TKcpms]; C - B;
	D = [TKcpms]; D + B;
	? [TKsrvnew] '>= C -> TK sv lo;
	[TKsrvnew] = C;
    "TK sv lo"
	? [TKsrvnew] '<= D -> TK sv hi;
	[TKsrvnew] = D;
    "TK sv hi"
	[TKcpms] = [TKsrvnew];
	=> TK period;
	( record it.  cpms changing mid-run changes the period in COUNTS,
	  so a reader that converts the whole tick log with one factor
	  gets the drift wrong.  The log says exactly when it changed. )
	A = [TKsrvlogn]; ? A '>= 8 -> TK sv done;
	B = A; B + B; B + tsrvlog;
	( the FIRST tick at which the new value is in force.  The servo
	  runs at the END of a tick, after that tick's deadline was
	  already advanced with the old period, so recording [TKticks]
	  itself would mis-attribute one step by a whole period. )
	A = [TKticks]; A+;
	[B plus 0] = A;
	[B plus 1] = [TKcpms];
	[TKsrvlogn]+;
    "TK sv done"
	end;

"TK start"
	[TKacc] = 0;
	[TKticks] = 0; [TKskips] = 0; [TKsleeps] = 0;
	[TKlogn] = 0;
	A = tlog; [TKlogp] = A;
	[TKsrvn] = 0; [TKsrvlogn] = 0;
	A = tsrvlog; B = 0; B + 16;
    "TK st clr"
	[A] = 0; A+;
	B ^ TK st clr;
	( entry 0 is the cpms in force at tick 0, so the reader never has
	  to guess what the period was before the first servo update )
	[tsrvlog plus 0] = 0;
	[tsrvlog plus 1] = [TKcpms];
	[TKsrvlogn] = 1;
	[Timer Command] = READ COUNTS; isocall;
	[TKdeadline] = [Counts];
	[TKsrv0c]    = [Counts];
	=> TK read wall; [TKsrv0w] = [TKtmp];
	end;

      ( TK is late : [TKlate] <- 1 if [TKnow] has reached [TKdeadline].
	THE SIGN OF THE DIFFERENCE.  Never a compare of the two values.
	Everything else in this file is a convenience; this is the part
	that is correct at the wrap. )

"TK is late"
	? [TKnow] '< [TKdeadline] -> TK islate no;
	[TKlate] = 1;
	end;
    "TK islate no"
	[TKlate] = 0;
	end;

      ( TK skip : advance past every grid point already gone.
	Rule 2.  A frame that overran does not get to catch up. )

"TK skip"
    "TK sk loop"
	[Timer Command] = READ COUNTS; isocall;
	[TKnow] = [Counts];
	=> TK is late;
	? [TKlate] = 0 -> TK sk done;
	=> TK advance;
	[TKskips]+;
	C = [TKflags]; C | TKFSKIP; [TKflags] = C;
	-> TK sk loop;
    "TK sk done"
	end;

      ( TK wait : sleep down to [TKmargin] ms, then spin. )

"TK wait"
    "TK w loop"
	[Timer Command] = READ COUNTS; isocall;
	[TKnow] = [Counts];
	=> TK is late;
	? [TKlate] != 0 -> TK w done;
	? [TKmargin] = 0 -> TK w loop;
	A = [TKdeadline]; A - [TKnow];
	A '/ [TKcpms];
	? A '<= [TKmargin] -> TK w loop;
	A - [TKmargin];
	[Sleep Timeout] = A;
	[Process Command] = SLEEP; isocall;
	[TKsleeps]+;
	C = [TKflags]; C | TKFSLEEP; [TKflags] = C;
	-> TK w loop;
    "TK w done"
	end;

      ( TK log : three RAW units.  No statistic is computed here. )

"TK log"
	A = [TKlogn]; ? A '>= TKLOGMAX -> TK log full;
	A = [TKlogp];
	[A plus 0] = [TKnow];
	[A plus 1] = [TKdeadline];
	[A plus 2] = [TKflags];
	A + 3; [TKlogp] = A;
	[TKlogn]+;
    "TK log full"
	end;

      ( TK step : one whole tick boundary, in the order that matters.
	advance, THEN skip, THEN wait, THEN log. )

"TK step"
	[TKflags] = 0;
	=> TK advance;
	=> TK skip;
	=> TK wait;
	=> TK log;
	=> TK servo;
	[TKticks]+;
	end;

      ( TK burn : busy-wait [TKtmp] milliseconds.  Used to inject a
	deterministic hitch so that rule 2 can be TESTED rather than
	asserted, and to give the calibration bracket real work. )

"TK burn"
	[Timer Command] = READ COUNTS; isocall;
	[TKburnto] = [Counts];
	A = [TKtmp]; A '* [TKcpms];
	[TKburnto] + A;
    "TK burn loop"
	[Timer Command] = READ COUNTS; isocall;
	A = [Counts]; A - [TKburnto];
	? A < 0 -> TK burn loop;
	end;
