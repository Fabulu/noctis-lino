
      ( *** mulcheck - batch verification of the 64-bit multiply *** )

      ( Runs Mul64 over a table of test vectors and writes the raw
        results to "mulout.bin" as 32-bit units, in the order:

            vector 0: lo, hi, fold
            vector 1: lo, hi, fold
            ...

        No dialogs - the results go to disk so they can be diffed
        against arbitrary-precision arithmetic automatically.

        A L.in.oleum application can only touch files inside its own
        folder, so mulout.bin lands next to the executable.            )

"directors"

	program name = { mulcheck };
	unit = 32;

"constants"

	NVECTORS	= 8;
	RESULTSPERVEC	= 3;

"variables"

	( ---- test vectors: 8 pairs of 32-bit operands ---- )

	test vectors	= 00000000h;	no label = 00000000h;	( 0 x 0 )
	no label	= 00000001h;	no label = 00000001h;	( 1 x 1 )
	no label	= FFFFFFFFh;	no label = FFFFFFFFh;	( max x max )
	no label	= 00010000h;	no label = 00010000h;	( 2^16 x 2^16 = 2^32 exactly )
	no label	= 12345678h;	no label = 76543210h;	( the known-good vector )
	no label	= DEADBEEFh;	no label = CAFEBABEh;	( both above 2^31 )
	no label	= 7FFFFFFFh;	no label = 7FFFFFFFh;	( largest positive signed )
	no label	= 000186A0h;	no label = 00030D40h;	( 100000 x 200000, Noctis sector scale )

	vecIndex	= 0;

	result file name = { mulout.bin };

	( ---- Mul64 storage ---- )

	mulA	= 0;
	mulB	= 0;
	mulLo	= 0;
	mulHi	= 0;
	mulAlo	= 0;
	mulAhi	= 0;
	mulBlo	= 0;
	mulBhi	= 0;
	mulLL	= 0;
	mulLH	= 0;
	mulHL	= 0;
	mulHH	= 0;
	mulMid	= 0;

"workspace"

	( uninitialised output buffer - this is what "workspace" is for )

	results = NVECTORS multiplied RESULTSPERVEC;

"programme"

	[vecIndex] = 0;

    "next vector"

	( fetch the operand pair at test vectors + vecIndex*2 )

	A = test vectors;
	B = [vecIndex]; B < 1;
	A + B;
	C = [A];  [mulA] = C;
	A + 1;
	C = [A];  [mulB] = C;

	=> Mul64;			( destroys all registers )

	( store lo, hi, fold at results + vecIndex*3 )

	A = results;
	B = [vecIndex]; B '* RESULTSPERVEC;
	A + B;

	C = [mulLo];			[A] = C;
	A + 1;
	C = [mulHi];			[A] = C;
	A + 1;
	C = [mulHi]; C + [mulLo];	[A] = C;

	[vecIndex] +;
	? [vecIndex] < NVECTORS -> next vector;

	( ---- dump the whole results vector to disk ---- )

	[File Name]	= result file name;
	[File Position] = 0;
	[File Command]	= WRITE;
	[Block Pointer] = results;
	[Block Size]	= NVECTORS multiplied RESULTSPERVEC;
	[Block Size]	* BYTES PER UNIT;
	isocall;

	[File Name]	= result file name;
	[File Position] = 0;
	[File Command]	= SET SIZE;
	[File Size]	= NVECTORS multiplied RESULTSPERVEC;
	[File Size]	* BYTES PER UNIT;
	isocall;

	end;

"Mul64"
    ( Unsigned 32x32 -> 64 bit multiply. )
    ( In:  [mulA], [mulB].   Out: [mulLo], [mulHi]. )
    ( All registers destroyed. )

	A = [mulA]; A & FFFFh;	[mulAlo] = A;
	A = [mulA]; A > 16;	[mulAhi] = A;
	A = [mulB]; A & FFFFh;	[mulBlo] = A;
	A = [mulB]; A > 16;	[mulBhi] = A;

	A = [mulAlo]; A '* [mulBlo];	[mulLL] = A;
	A = [mulAlo]; A '* [mulBhi];	[mulLH] = A;
	A = [mulAhi]; A '* [mulBlo];	[mulHL] = A;
	A = [mulAhi]; A '* [mulBhi];	[mulHH] = A;

	A = [mulLL]; A > 16;
	B = [mulLH]; B & FFFFh; A + B;
	B = [mulHL]; B & FFFFh; A + B;
	[mulMid] = A;

	A = [mulLL]; A & FFFFh;
	B = [mulMid]; B < 16;
	A | B;
	[mulLo] = A;

	A = [mulHH];
	B = [mulLH]; B > 16; A + B;
	B = [mulHL]; B > 16; A + B;
	B = [mulMid]; B > 16; A + B;
	[mulHi] = A;

	end;
