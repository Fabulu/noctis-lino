      ( *** vhgui - aspect-fit presenter for the Noctis 320x200 frame ***

	iGUI owns the physical display and resizable custom window.  The game
	continues to render its exact 320x200 indexed framebuffer.  This adapter
	palette-expands one logical RGB page, lets the game compose HUD/help text
	there, then nearest-neighbour scales the completed page into both iGUI
	layers so every visible element follows the same aspect-fit transform. )

"variables"

	VHGUIaw = 0; VHGUIah = 0; VHGUIdw = 0; VHGUIdh = 0;
	VHGUIleft = 0; VHGUItop = 0;
	VHGUIx = 0; VHGUIy = 0; VHGUIsx = 0; VHGUIsy = 0;
	VHGUIsrc = 0; VHGUIdst = 0;
	VHGUIpal = 0; VHGUIrow2 = 0; VHGUIgap = 0;
	VHGUIdstp = 0; VHGUIrow2p = 0; VHGUIpublished = 0;
	VHGUIxacc = 0; VHGUIyacc = 0;
	VHGUIprevaw = 0; VHGUIprevah = 0;
	vector VHGUIregion = 0; 0; 319; 199;

"workspace"

	VHGUIframe = 64000;

"programme"

"VHGUI resize"
	A = [vector Work Area plus 2]; A - [vector Work Area plus 0]; A+; [VHGUIaw] = A;
	A = [vector Work Area plus 3]; A - [vector Work Area plus 1]; A+; [VHGUIah] = A;
	( fit width first: h=w*5/8; if too tall, fit height instead )
	[VHGUIdw] = [VHGUIaw]; A = [VHGUIdw]; A '* 5; A '/ 8; [VHGUIdh] = A;
	A = [VHGUIdh]; ? A '<= [VHGUIah] -> VHGUI fit ready;
	[VHGUIdh] = [VHGUIah]; A = [VHGUIdh]; A '* 8; A '/ 5; [VHGUIdw] = A;
    "VHGUI fit ready"
	A = [VHGUIaw]; A - [VHGUIdw]; A '/ 2; A + [vector Work Area plus 0]; [VHGUIleft] = A;
	A = [VHGUIah]; A - [VHGUIdh]; A '/ 2; A + [vector Work Area plus 1]; [VHGUItop] = A;
	( Clear bars once per geometry change, not once per game frame. )
	A = [VHGUIaw]; ? A != [VHGUIprevaw] -> VHGUI geometry changed;
	A = [VHGUIah]; ? A = [VHGUIprevah] -> VHGUI geometry ready;
    "VHGUI geometry changed"
	[VHGUIprevaw] = [VHGUIaw]; [VHGUIprevah] = [VHGUIah];
	[Destination Layer] = Backdrop Layer; [L2L Region] = vector Work Area;
	[L2L Region Color] = 0; => Clear L2L Region;
	[Destination Layer] = Primary Display; [L2L Region] = vector Work Area;
	[L2L Region Color] = 0; => Clear L2L Region;
    "VHGUI geometry ready"
	end;

"VHGUI prepare"
	=> VHGUI resize;
	( Compose one logical RGB page from the authentic indexed framebuffer.
	  HUD/help text is added to this page before presentation, so it follows
	  the same nearest-neighbour scaling as the rendered scene. )
	A = nw; A + RADPT; [VHGUIsrc] = A; [VHGUIpal] = pal; [VHGUIdst] = VHGUIframe;
	( The indexed page, presenter palette, and composition page are fixed,
	  disjoint 64,000-unit buffers.  Expand every index through the same palette
	  lookup as the Lino loop, but without 16,000 interpreter dispatches. )
	{
	    53 56 57                            (push ebx, esi, edi)
	    8B B7 <dVHGUIsrc mtp bytesperunit>
	    8D 34 B7                            (esi = indexed source)
	    8B 9F <dVHGUIpal mtp bytesperunit>
	    8D 1C 9F                            (ebx = RGB palette)
	    8B 97 <dVHGUIdst mtp bytesperunit>
	    8D 14 97                            (edx = composition page)
	    B9 00 FA 00 00                      (64,000 pixels)
	    8B 06 8B 04 83 89 02                (palette-expand one pixel)
	    83 C6 04 83 C2 04 49 75 F0          (advance and repeat)
	    5F 5E 5B                            (restore edi, esi, ebx)
	}
	[VHGUIx] = 0; [VHGUIy] = 200;
	end;

"VHGUI present"
	( The 1x path copies four composed RGB pixels per loop.  Only Backdrop is
	  written here: Update Area performs the one authoritative copy into
	  Primary Display immediately afterward. )
	[VHGUIpublished] = 0;
	A = [VHGUIdw]; ? A != 320 -> VHGUI maybe 2x;
	A = [VHGUIdh]; ? A != 200 -> VHGUI maybe 2x;
	[VHGUIy] = 0;
    "VHGUI 1x row"
	A = [VHGUIy]; A '* 320; C = A; C + VHGUIframe;
	A = [VHGUIy]; A + [VHGUItop]; A '* [Display Width];
	A + [VHGUIleft]; D = A; D + Backdrop Layer;
	[VHGUIx] = 80;
    "VHGUI 1x pixel"
	A = [C]; [D] = A;
	A = [C plus 1]; [D plus 1] = A;
	A = [C plus 2]; [D plus 2] = A;
	A = [C plus 3]; [D plus 3] = A;
	C + 4; D + 4; [VHGUIx]-;
	A = [VHGUIx]; ? A != 0 -> VHGUI 1x pixel;
	[VHGUIy]+; A = [VHGUIy]; ? A < 200 -> VHGUI 1x row;
	end;

    "VHGUI maybe 2x"
	( The default client is exactly 640x400.  Expand four logical pixels into
	  two identical pairs on two destination rows per loop, so the common path
	  loads each composed pixel once instead of running the generic scaler's
	  coordinate accumulators once per destination pixel. )
	A = [VHGUIdw]; ? A != 640 -> VHGUI scaled;
	A = [VHGUIdh]; ? A != 400 -> VHGUI scaled;
	[VHGUIsrc] = VHGUIframe;
	A = [VHGUItop]; A '* [Display Width]; A + [VHGUIleft]; A + Backdrop Layer;
	[VHGUIdst] = A; A + [Display Width]; [VHGUIrow2] = A;
	A = [VHGUItop]; A '* [Display Width]; A + [VHGUIleft]; A + Primary Display;
	[VHGUIdstp] = A; A + [Display Width]; [VHGUIrow2p] = A;
	A = [Display Width]; A + A; A - 640; A '* 4; [VHGUIgap] = A;
	( Exact native expansion of 320 source dwords into two equal pixels on
	  each of two destination rows. Publish those already-loaded pixels to both
	  synchronized iGUI layers, avoiding a later 1 MiB Backdrop reread while
	  retaining the runtime display stride and normal cursor/retrace ordering.
	  Only the fixed 640x400 fit takes this path. )
	{
	    53 56 57 55                         (push ebx, esi, edi, ebp)
	    8B B7 <dVHGUIsrc mtp bytesperunit>
	    8D 34 B7                            (esi = source unit address)
	    8B 87 <dVHGUIdst mtp bytesperunit>
	    8D 04 87                            (eax = first destination row)
	    8B AF <dVHGUIrow2 mtp bytesperunit>
	    8D 2C AF                            (ebp = second destination row)
	    8B 97 <dVHGUIdstp mtp bytesperunit>
	    8D 14 97                            (edx = first primary row)
	    8B 9F <dVHGUIrow2p mtp bytesperunit>
	    8D 1C 9F                            (ebx = second primary row)
	    FF B7 <dVHGUIgap mtp bytesperunit>  (stack: row gap)
	    68 C8 00 00 00                      (stack: 200 source rows)
	    89 C7                               (edi = first destination row)
	    B9 40 01 00 00                      (320 source pixels)
	    AD AB AB                            (load once, write first pair)
	    89 45 00 89 45 04                   (write second Backdrop pair)
	    89 02 89 42 04                      (write first Primary pair)
	    89 03 89 43 04                      (write second Primary pair)
	    83 C5 08 83 C2 08 83 C3 08          (advance non-EDI rows)
	    49 75 E1                            (next source pixel)
	    03 7C 24 04 03 6C 24 04             (advance Backdrop rows)
	    03 54 24 04 03 5C 24 04             (advance Primary rows)
	    FF 0C 24 75 C7                      (decrement rows; next row)
	    83 C4 08                            (discard row count and gap)
	    5D 5F 5E 5B                         (restore ebp, edi, esi, ebx)
	}
	[VHGUIpublished] = 1;
	[VHGUIx] = 0; [VHGUIy] = 200;
	end;
    "VHGUI scaled"
	[VHGUIy] = 0; [VHGUIsy] = 0; [VHGUIyacc] = 0;
    "VHGUI row"
	A = [VHGUIsy]; A '* 320; A + VHGUIframe; [VHGUIsrc] = A;
	A = [VHGUIy]; A + [VHGUItop]; A '* [Display Width]; A + [VHGUIleft]; A + Backdrop Layer; [VHGUIdst] = A;
	[VHGUIx] = 0; [VHGUIsx] = 0; [VHGUIxacc] = 0;
    "VHGUI pixel"
	C = [VHGUIsrc]; C + [VHGUIsx]; A = [C];
	D = [VHGUIdst]; D + [VHGUIx]; [D] = A;
	A = [VHGUIxacc]; A + 320; [VHGUIxacc] = A;
    "VHGUI x advance"
	A = [VHGUIxacc]; ? A '< [VHGUIdw] -> VHGUI x ready;
	A - [VHGUIdw]; [VHGUIxacc] = A; [VHGUIsx]+; -> VHGUI x advance;
    "VHGUI x ready"
	[VHGUIx]+; A = [VHGUIx]; ? A '< [VHGUIdw] -> VHGUI pixel;
	A = [VHGUIyacc]; A + 200; [VHGUIyacc] = A;
    "VHGUI y advance"
	A = [VHGUIyacc]; ? A '< [VHGUIdh] -> VHGUI y ready;
	A - [VHGUIdh]; [VHGUIyacc] = A; [VHGUIsy]+; -> VHGUI y advance;
    "VHGUI y ready"
	[VHGUIy]+; A = [VHGUIy]; ? A '< [VHGUIdh] -> VHGUI row;
	end;
