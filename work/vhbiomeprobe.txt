	( Low seed bits 2 select DESERT. )
	[VHGNDseed] = 149130; => VHGND generate; => VHB scan;
	[vhbout plus 0] = [VHGNDsctype]; [vhbout plus 1] = [VHGNDrockdensity];

	[File Name] = VHBname; [File Position] = 0; [File Command] = SET SIZE; [File Size] = 64; isocall;
	end;

"VHB scan"
	[VHBp] = 0; [VHBmin] = 255; [VHBmax] = 0; [VHBsum] = 0;
    "VHB scan pixel"
	A = RPSM; A + [VHBp]; [MBptr] = A; => MEM get byte; A = [MBval];
	? A '>= [VHBmin] -> VHB scan max; [VHBmin] = A;
    "VHB scan max"
	A = [MBval]; ? A '<= [VHBmax] -> VHB scan sum; [VHBmax] = A;
    "VHB scan sum"
	A = [VHBsum]; A + [MBval]; [VHBsum] = A;
	[VHBp]+; A = [VHBp]; ? A < 40000 -> VHB scan pixel;
	end;
