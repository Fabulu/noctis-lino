	----------------------------------------------------------------
	0.  THE ONE INSTRUCTION EVERYTHING TURNS ON: __ftol, at 0000:1265
	----------------------------------------------------------------

	    1265  push bp / mov bp,sp / sub sp,0Ah
	    126C  fnstcw word [bp-2]
	    1271  mov   al, [bp-1]
	    1274  or    byte [bp-1], 0Ch      ; RC <- 11 (chop).  OR only.
	    1279  fldcw word [bp-2]
	    127D  fistp qword [bp-0Ah]        ; chops whatever is in st(0)
	    1280  mov   [bp-1], al
	    1284  fldcw word [bp-2]
	    1287  mov ax,[bp-0Ah] / mov dx,[bp-8]
	    1290  retf

	It has NO parameter.  Its operand is the live x87 top of stack.
	FLOATPOLICY.md 3.3 records the cast boundary as UNSETTLED because
	"the engine has no way to truncate an unstored extended value"; the
	question of which reading is the original's is settled here, and it
	is settled the same way at every one of the eleven sites: there is
	no fstp between the last arithmetic instruction and the lcall.

	Two details a tidy port gets wrong:
	  * OR 0Ch, not AND-then-OR.  Identical under the game's 133Fh.
	  * fistp QWORD.  A 32-bit fistp stores the integer indefinite
	    80000000h, masked, for everything outside int32; this one lets
	    the low half through.  work/geocast battery 8 measures it.

	:4088  p_orb_orient[n] = (double) deg * (double) random (360);
	  61494  call  random           ; arg 360
	  61501  fild  word [bp-12h]    ; the int16 random() returned
	  61505  fmul  qword [0184]     ; * deg
	  61515  fstp  qword [bx+58C7]
	  -> ONE multiply, ONE store.  orb_orient is one of exactly 360
	     binary64 values and can be tabulated (FLOATSITES 3.1b).

	:4089  p_orb_seed[n] = 3*(n*n+1)*nearstar_ray
			     + (float) random (300*nearstar_ray) / 100;
	  61519  mov ax,si / imul si / inc ax / imul ax,ax,3   ; int16
	  61533  fild  word [bp-12h]
	  61537  fmul  dword [027F]     ; * nearstar_ray, a FLOAT operand
	  61542  fstp  XWORD [bp-1Ch]   ; <-- 80-BIT spill.  Loses nothing.
	  61546  fld   dword [108C]     ; 300.0f
	  61551  fmul  dword [027F]     ; * nearstar_ray
	  61555  lcall __ftol           ; LIVE - no store in between
	  61560  push  ax               ; random(int16)
	  61563  call  random
	  61571  fild  word [bp-12h]
	  61575  fdiv  dword [0EC0]     ; / 100.0f
	  61580  fld   XWORD [bp-1Ch]   ; reload the 80-bit temp
	  61584  faddp st(1)
	  61592  fstp  qword [bx+53C7]

	  THIS SETTLES THE LINE FLOATSITES CALLED THE MOST DANGEROUS.  The
	  `(float)` cast is not materialised: random() returns an int16,
	  fild of it is exact, and the divide by 100 happens at 64-bit
	  significand and is never narrowed to binary32.  The only spill in
	  the whole statement is an XWORD, i.e. full extended precision.
	  So the entire RHS is evaluated at PC=64 with no loss and is
	  rounded to binary64 exactly once, at the store.  A port that
	  honours the C type of the sub-expression and narrows to float is
	  WRONG; so is one that spills to a double.

	:4090  p_tilt[n]     = zrandom (10*p_orb_seed[n]) / 500;
	  61602  fld   qword [bx+53C7]
	  61607  fmul  dword [0DE6]     ; * 10.0f
	  61611  lcall __ftol           ; LIVE

	  61623  fdiv  dword [1090]     ; / 500.0f

	:4091  p_orb_tilt[n] = zrandom (10*p_orb_seed[n]) / 5000;
	  61643  fld qword [bx+53C7] / fmul dword [0DE6] / lcall __ftol
	  61659  call zrandom / fdiv dword [1094] / fstp qword [bx+5647]

	  61692  fstp  qword [bp-28h]   ; fabs is a REAL CALL, arg via memory
	  61697  lcall 0000:04FC        ; fabs, result in st(0)
	  61706  fmul  dword [0DE6]     ; * 10.0f
	  61716  fadd  qword [bx+53C7]  ; + p_orb_seed[n]   <- this order
	  61720  lcall __ftol           ; LIVE
	  61728  call  random
	  61736  fild  word [bp-12h]

	  61745  fld1

	  Note fabs is a library call through a qword, so p_orb_tilt is

	  61786  fild  word [bp-12h]

	  only DL.CPP and ST.CPP carry a copy of prepare_nearstar (both
