      ( *** vhnivgen - production NIVGEN output driver ***

	This library is linked into a derived vhgame main by tools/nivtest.py.
	It drives the real star topology, orbital surface(), landed
	build_surface(), and create_sky paths, then emits their byte-bearing
	Lino units in one deterministic file.  The Python wrapper only packs
	arguments, strips the low byte from each unit, and applies the public
	NIVGEN hash boundaries. )

"constants"

	NHINMAGIC = 1313426993;	( 0x4E494E31, NIN1 )
	NHOUTMAGIC = 1313426994;	( 0x4E494E32, NIN2 )
	NHVERSION = 1;
	NHIN = 32; NHHEAD = 32;

	NHSURFU = 64800; NHATMOU = 32400; NHPALU = 768;
	NHHEIGHTU = 40000; NHGAPU = 16; NHOBJECTU = 40000;
	NHSTEXU = 65536; NHSKYU = 64800; NHSRFPALU = 768;

	NHOFFSURF = 32;
	NHOFFATMO = 64832;
	NHOFFPAL = 97232;
	NHOFFHEIGHT = 98000;
	NHOFFGAP = 138000;
	NHOFFOBJECT = 138016;
	NHOFFSTEX = 178016;
	NHOFFSKY = 243552;
	NHOFFSRFPAL = 308352;
	NHOUT = 309120;

"variables"

	NHinname = { niv-input.bin };
	NHoutname = { niv-output.bin };
	NHi = 0; NHsrc = 0; NHunits = 0; NHoff = 0; NHtmp = 0;

"workspace"

	NHinput = 32;
	NHheader = 32;

"programme"

"VHNIV run"
	=> FEnter;
	[File Name] = NHinname; [File Position] = 0;
	[File Command] = READ; [Block Pointer] = NHinput;
	[Block Size] = NHIN; [Block Size] * BYTESPERUNIT; isocall;
	A = [NHinput]; ? A != NHINMAGIC -> VHNIV invalid;
	A = [NHinput plus 1]; ? A != NHVERSION -> VHNIV invalid;

	=> MEM build rtab; => MEM build ztab; => MEM wrap reset;
	=> PAL float init; => PGF constants; => SU fp init; => SU pal init;

	[VHGNDclockoverride] = 1; [VHGNDclocksecs] = [NHinput plus 8];
	[VHTtx] = [NHinput plus 2];
	A = 0; A - [NHinput plus 3]; [VHTty] = A;
	[VHTtz] = [NHinput plus 4];

	( extract_ap_target_infos() + prepare_nearstar(), without GUI/catalogue work )
	=> VHT extract;
	[nsx] = [VHTtx]; [nsy] = [VHTty]; [nsz] = [VHTtz]; => NsSeedFromXYZ;
	[nsseedin] = A; [nsclass] = [VHTclass]; [nsstarray] = [VHTRay];
	=> NsResetAll; => NsPrepare;

	[VHGplanet] = [NHinput plus 5];
	A = [VHGplanet]; ? A '< [nsnob] -> VHNIV body valid;
	-> VHNIV invalid;
    "VHNIV body valid"
	[VHGlandinglon] = [NHinput plus 6]; [VHGlandinglat] = [NHinput plus 7];
	[VHGNDlon] = [VHGlandinglon]; [VHGNDlat] = [VHGlandinglat];
	=> VHG select texture buffer; => VHG prepare planet; => VHG fpu clean;
	=> VHNIV orbital surface; => VHG fpu clean;

	( Preserve the public harness's orbital products before build_surface
	  reuses the same allocations. )
	A = nw; A + [VHGNDtexbase]; [NHsrc] = A;
	[NHunits] = NHSURFU; [NHoff] = NHOFFSURF; => VHNIV write;
	A = nw; A + ROBJ; [NHsrc] = A;
	[NHunits] = NHATMOU; [NHoff] = NHOFFATMO; => VHNIV write;
	[NHsrc] = sutmppal; [NHunits] = NHPALU; [NHoff] = NHOFFPAL; => VHNIV write;

	=> VHNIV landed inputs;
	=> VHNIV wipe landed;
	=> SU ledger reset; => GR float init;
	[VHGNDrockscale] = 500; [VHGNDrockpeak] = 250; [VHGNDrockdensity] = 15;
	[VHGNDwaswet] = 0; [VHGNDwavescalm] = 0; [VHGNDfrosty] = 0;
	=> VHGND build surface; => VHG fpu clean;

	A = nw; A + RPSM; [NHsrc] = A;
	[NHunits] = NHHEIGHTU; [NHoff] = NHOFFHEIGHT; => VHNIV write;
	A = nw; A + RPSM; A + 40000; [NHsrc] = A;
	[NHunits] = NHGAPU; [NHoff] = NHOFFGAP; => VHNIV write;
	A = nw; A + ROBJ; [NHsrc] = A;
	[NHunits] = NHOBJECTU; [NHoff] = NHOFFOBJECT; => VHNIV write;
	A = nw; A + [VHGNDtexbase]; [NHsrc] = A;
	[NHunits] = NHSTEXU; [NHoff] = NHOFFSTEX; => VHNIV write;

	=> VHNIV sky; => VHG fpu clean;
	A = nw; A + RSBG; [NHsrc] = A;
	[NHunits] = NHSKYU; [NHoff] = NHOFFSKY; => VHNIV write;
	[NHsrc] = srfpal6; [NHunits] = NHSRFPALU; [NHoff] = NHOFFSRFPAL; => VHNIV write;

	[NHheader] = NHOUTMAGIC; [NHheader plus 1] = NHVERSION;
	[NHheader plus 2] = 0;
	[NHheader plus 3] = [NHinput plus 2];
	[NHheader plus 4] = [NHinput plus 3];
	[NHheader plus 5] = [NHinput plus 4];
	[NHheader plus 6] = [VHGplanet]; [NHheader plus 7] = [nsnob];
	[NHheader plus 8] = [VHGptype]; [NHheader plus 9] = [VHGNDowner];
	[NHheader plus 10] = [GRgseed]; [NHheader plus 11] = [VHGNDsctype];
	[NHheader plus 12] = [GRalbedo]; [NHheader plus 13] = [NHinput plus 11];
	[NHheader plus 14] = [VHGNDclocksecs];
	A = [VHGplanet]; A < 1;
	E = nspray; E + A; [NHheader plus 15] = [E plus 0]; [NHheader plus 16] = [E plus 1];
	E = nsporbray; E + A; [NHheader plus 17] = [E plus 0]; [NHheader plus 18] = [E plus 1];
	E = nsporbseed; E + A; [NHheader plus 19] = [E plus 0]; [NHheader plus 20] = [E plus 1];
	E = nsptilt; E + A; [NHheader plus 21] = [E plus 0]; [NHheader plus 22] = [E plus 1];
	E = nsporbtlt; E + A; [NHheader plus 23] = [E plus 0]; [NHheader plus 24] = [E plus 1];
	E = nsporbecc; E + A; [NHheader plus 25] = [E plus 0]; [NHheader plus 26] = [E plus 1];
	E = nspororient; E + A; [NHheader plus 27] = [E plus 0]; [NHheader plus 28] = [E plus 1];
	[NHheader plus 29] = [VHGNDplwp];
	[NHheader plus 30] = [VHGNDseedval0]; [NHheader plus 31] = [VHGNDseedval1];
	[NHsrc] = NHheader; [NHunits] = NHHEAD; [NHoff] = 0; => VHNIV write;

	[File Name] = NHoutname; [File Position] = 0; [File Command] = SET SIZE;
	[File Size] = NHOUT; [File Size] * BYTESPERUNIT; isocall;
	[VHGNDclockoverride] = 0;
	=> FLeave;
	end;

"VHNIV orbital surface"
	( The public command calls surface() immediately after prepare_nearstar().
	  At that boundary the source's shared plx/plz scratch coordinates are still
	  zero.  Normal gameplay first positions a body and therefore deliberately
	  uses VHGND globe surface instead. )
	E = nspowner; E + [VHGplanet]; A = [E]; [VHGNDowner] = A;
	=> VHGND rotation seed;

	( surface(): exact rotation-period stream and integer seconds division. )
	[FB0] = [VHGNDseedval0]; [FB1] = [VHGNDseedval1];
	=> GeoAdd4112ChopLive; A = [FI]; => SU fast srand;
	C = 50; => SU rfr; A = C; A + 1; A '* 10; [VHGNDperiod] = A;
	C = 25; => SU rfr; A = C; A '* 10; A + [VHGNDperiod]; [VHGNDperiod] = A;
	C = 250; => SU rfr; A = C; A + [VHGNDperiod]; A + 41; [VHGNDperiod] = A;
	[SUia] = A; => SU secs rot; A = [SUti]; A % 360;
	? A >= 0 -> VHNIV orbital rotation ready; A + 360;
    "VHNIV orbital rotation ready"
	[VHGNDrotation] = A;

	( cplx_planet_viewpoint(): primaries and ordinary moons observe the
	  still-zero plx/plz scratch point from the star coordinates.  A moon of
	  a companion star instead observes its live local vector from its owner. )
	A = [VHGNDowner]; ? A < 0 -> VHNIV orbital initial viewpoint;
	E = nsptype; E + A; A = [E]; ? A != 10 -> VHNIV orbital initial viewpoint;
	=> VHGND exact viewpoint; A = [VHGNDviewpoint]; -> VHNIV orbital viewpoint ready;
    "VHNIV orbital initial viewpoint"
	[FI] = [VHTtz]; => IntToF;
	[VHGNDxx0] = [FA0]; [VHGNDxx1] = [FA1];
	[FI] = [VHTtx]; => IntToF; [FB0] = [FA0]; [FB1] = [FA1];
	[FA0] = [VHGNDxx0]; [FA1] = [VHGNDxx1]; => FAtan2;
	[FB0] = 0; [FB1] = 40668000h; => FMul;
	[FB0] = 54442D18h; [FB1] = 400921FBh; => FQuo; => FToIntNear;
	A = [FI]; A % 360; ? A >= 0 -> VHNIV orbital viewpoint ready; A + 360;
    "VHNIV orbital viewpoint ready"
	C = 89; C - A; C + [VHGNDrotation]; A = C; A % 360;
	? A >= 0 -> VHNIV orbital plwp ready; A + 360;
    "VHNIV orbital plwp ready"
	[VHGNDplwp] = A;

	=> VHG fpu clean; => SU fp init; => SU pal init;
	[SUid] = [VHGplanet]; [SUtype] = [VHGptype]; [SUplwp] = [VHGNDplwp];
	[SUowner] = [VHGNDowner]; [SUnsr] = [VHTred]; [SUnsg] = [VHTgreen]; [SUnsb] = [VHTblue];
	[SUsv0] = [VHGNDseedval0]; [SUsv1] = [VHGNDseedval1]; [SUledon] = 0;
	A = [VHGNDowner]; ? A >= 0 -> VHNIV orbital moon;
	[SUcbase] = 192; [VHGNDcbase] = 192; => SU select planet; -> VHNIV orbital selected;
    "VHNIV orbital moon"
	[SUcbase] = 128; [VHGNDcbase] = 128; => SU select moon;
    "VHNIV orbital selected"
	=> SU select ovl; => SU wipe; => SU wipe ovl; => SU surface;
	end;

"VHNIV landed inputs"
	[VHGNDlocseed] = [VHGNDlat]; A = [VHGNDlocseed]; A '* [VHGNDlon]; [VHGNDlocseed] = A;
	[GRgseed] = [VHGNDseed]; [GRiptype] = [VHGptype];
	A = [NHinput plus 10]; ? A >= 0 -> VHNIV albedo ready; A = 32;
    "VHNIV albedo ready"
	[GRalbedo] = A;
	A = [VHGNDlat]; A - 60; ? A >= 0 -> VHNIV latitude absolute; A +-;
    "VHNIV latitude absolute"
	A '* 3; A '/ 2; [GRlat] = A;

	( setup_landing's type-3 global_surface_seed polar correction. )
	A = [GRiptype]; ? A != 3 -> VHNIV seed ready;
	A = [GRgseed]; A + [VHGNDlon]; => BrtlSrand;
	A = 5; => BrtlRandom; [NHtmp] = A;
	A = [GRgseed]; A % 15; A + 25; A + [NHtmp];
	C = [GRlat]; ? C <= A -> VHNIV seed ready;
	[GRgseed]+; [VHGNDseed] = [GRgseed];
    "VHNIV seed ready"

	A = [NHinput plus 9]; ? A '< 1 -> VHNIV scenario automatic;
	? A '> 4 -> VHNIV scenario automatic; [VHGNDsctype] = A;
	-> VHNIV scenario ready;
    "VHNIV scenario automatic"
	[VHGNDsctype] = 2; A = [GRiptype]; ? A != 3 -> VHNIV scenario ready;
	A = [VHGNDlocseed]; => BrtlSrand; A = 100; => BrtlRandom;
	? A '<= 5 -> VHNIV scenario local;
	A = [VHGNDplanet]; A < 1; E = nspororient; E + A;
	[FB0] = [E plus 0]; [FB1] = [E plus 1]; A = 3; => GeoSetK;
	=> GeoKMulChopLive; A = [FI]; A % 4; A + 1; [VHGNDsctype] = A;
	-> VHNIV scenario albedo;
    "VHNIV scenario local"
	A = 4; => BrtlRandom; A + 1; [VHGNDsctype] = A;
    "VHNIV scenario albedo"
	A = [GRalbedo]; ? A '>= 25 -> VHNIV scenario polar; [VHGNDsctype] = 1;
    "VHNIV scenario polar"
	A = [GRlat]; ? A '<= 75 -> VHNIV scenario near polar; [VHGNDsctype] = 4;
    "VHNIV scenario near polar"
	A = [GRlat]; ? A '<= 60 -> VHNIV scenario ready;
	A = 3; => BrtlRandom; ? A = 0 -> VHNIV scenario ready; [VHGNDsctype] = 4;
    "VHNIV scenario ready"
	[GRsctype] = [VHGNDsctype];
	end;

"VHNIV sky"
	( The public surftex command invokes create_sky() directly after a flat
	  memset.  It does not run planetary_main's later 120-row horizon pass and
	  it retains the generator globals' source defaults rather than gameplay's
	  sampled landing filters. )
	[GRSKptype] = [GRiptype]; [GRSKsctype] = [VHGNDsctype];
	[GRSKatmosphere] = 1; A = [GRiptype];
	? A = 1 -> VHNIV sky airless; ? A = 4 -> VHNIV sky airless;
	? A = 7 -> VHNIV sky airless; -> VHNIV sky atmosphere ready;
    "VHNIV sky airless"
	[GRSKatmosphere] = 0;
    "VHNIV sky atmosphere ready"
	[GRSKnightzone] = [NHinput plus 11]; [GRSKtarget] = [VHGplanet];
	[GRSKowner] = [VHGNDowner]; [GRSKstarclass] = [nsclass];
	[GRSKseed] = [GRgseed]; [GRSKalbedo] = [GRalbedo]; [GRSKrainy] = 0;
	[GRSKsrf] = 63; [GRSKsgf] = 63; [GRSKsbf] = 63;
	[GRSKgrf] = 63; [GRSKggf] = 63; [GRSKgbf] = 63;
	[GRSKbrightness] = 32;
	A = [GRiptype]; ? A = 2 -> VHNIV sky type2;
	? A = 1 -> VHNIV sky dark; ? A = 4 -> VHNIV sky dark;
	? A = 7 -> VHNIV sky dark; -> VHNIV sky brightness ready;
    "VHNIV sky type2"
	A = [GRSKnightzone]; A '* 31; C = 63; C - A; [GRSKbrightness] = C;
	-> VHNIV sky brightness ready;
    "VHNIV sky dark"
	[GRSKbrightness] = 0;
    "VHNIV sky brightness ready"
	[GRSKdsd1] = 447A0000h; [GRSKexposure] = 0; [GRSKlat] = [VHGNDlat];
	[GRSKquadwords] = 16000;
	A = nw; A + RSBG; [VHGNDskyfill] = A; [VHGNDskycount] = 64864;
	C = [GRSKbrightness]; C & 255;
    "VHNIV fill sky"
	A = [VHGNDskyfill]; [A] = C; [VHGNDskyfill]+;
	[VHGNDskycount]-; A = [VHGNDskycount]; ? A != 0 -> VHNIV fill sky;
	=> SU ledger reset; => GRSK create;
	end;

"VHNIV wipe landed"
	[NHi] = 0;
    "VHNIV wipe height"
	A = nw; A + RPSM; A + [NHi]; [A] = 0;
	[NHi]+; A = [NHi]; ? A '< 40000 -> VHNIV wipe height;
	[NHi] = 0;
    "VHNIV wipe objects"
	A = nw; A + ROBJ; A + [NHi]; [A] = 0;
	[NHi]+; A = [NHi]; ? A '< 40000 -> VHNIV wipe objects;
	[NHi] = 0;
    "VHNIV wipe texture"
	A = nw; A + [VHGNDtexbase]; A + [NHi]; [A] = 16;
	[NHi]+; A = [NHi]; ? A '< 65535 -> VHNIV wipe texture;
	A = nw; A + [VHGNDtexbase]; A + 65535; [A] = 0;

	( Replay the public harness's sixteen-byte far-heap gap. )
	A = nw; A + RPSM; A + 40000; [NHsrc] = A; [NHi] = 0;
    "VHNIV gap copy"
	A = NHinput; A + 16; A + [NHi]; C = [A];
	A = [NHsrc]; A + [NHi]; [A] = C;
	[NHi]+; A = [NHi]; ? A '< 16 -> VHNIV gap copy;
	end;

"VHNIV write"
	[File Name] = NHoutname; A = [NHoff]; A '* BYTESPERUNIT; [File Position] = A;
	[File Command] = WRITE; [Block Pointer] = [NHsrc];
	[Block Size] = [NHunits]; [Block Size] * BYTESPERUNIT; isocall;
	end;

"VHNIV invalid"
	[VHGNDclockoverride] = 0;
	=> FLeave;
	end;
