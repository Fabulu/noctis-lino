      ( Noctis IV Galactic Guide database.

	GUIDE.BIN begins with its total byte size, followed by 84-byte records:
	a binary64 catalogue identity and a zero-padded 76-byte message.  Keep
	room for CAST additions while rejecting malformed or unbounded files. )

"constants"

	VHGDBHDR = 4;
	VHGDBREC = 84;
	VHGDBMAX = 8388608;

"variables"

	VHGDBfile = { GUIDE.BIN };
	VHGDBloaded = 0; VHGDBbytes = 0; VHGDBrecs = 0; VHGDBstatus = 0;
	VHGDBid0 = 0; VHGDBid1 = 0; VHGDBmsgptr = 0; VHGDBmsglen = 0;
	VHGDBptr = 0; VHGDBi = 0; VHGDBshift = 0; VHGDBrecordno = 0; VHGDBpos = 0;
	VHGDBcleanin = 0; VHGDBcleanout = 0; VHGDBcleanword = 0;
	VHGDBcleantotal = 0; VHGDBcleanremoved = 0; VHGDBcleanboundary = 0;

"workspace"

	vhguidedata = 2097152;

"programme"

"VHGDB load"
	[VHGDBloaded] = 0; [VHGDBbytes] = 0; [VHGDBrecs] = 0;
	[File Name] = VHGDBfile; [File Command] = TEST; isocall;
	? [File Status] + ERROR -> VHGDB load done;
	A = [File Size]; ? A < VHGDBHDR -> VHGDB load done;
	? A > VHGDBMAX -> VHGDB load done; [VHGDBbytes] = A;
	A - VHGDBHDR; C = A; A '% VHGDBREC; ? A != 0 -> VHGDB load bad;
	C / VHGDBREC; [VHGDBrecs] = C;
	[File Name] = VHGDBfile; [File Position] = 0; [File Command] = READ;
	[Block Pointer] = vhguidedata; [Block Size] = [VHGDBbytes]; isocall;
	A = [Block Size]; ? A != [VHGDBbytes] -> VHGDB load bad;
	A = [vhguidedata]; ? A < VHGDBHDR -> VHGDB load bad;
	? A > [VHGDBbytes] -> VHGDB load bad;
	A - VHGDBHDR; A '% VHGDBREC; ? A != 0 -> VHGDB load bad;
	[VHGDBloaded] = 1; -> VHGDB load done;
    "VHGDB load bad"
	[VHGDBbytes] = 0; [VHGDBrecs] = 0;
    "VHGDB load done"
	end;

      ( Append one source-format guide record. The header is the consolidated
	upstream boundary and deliberately remains unchanged for local CAST data. )
"VHGDB add"
	[VHGDBstatus] = 0; A = [VHGDBloaded]; ? A = 0 -> VHGDB add done;
	A = [VHGDBmsglen]; ? A = 0 -> VHGDB add done; ? A > 76 -> VHGDB add done;
	A = [VHGDBbytes]; A + VHGDBREC; ? A > VHGDBMAX -> VHGDB add done;
	A = [VHGDBbytes]; A / 4; A + vhguidedata; [VHGDBptr] = A;
	[A] = [VHGDBid0]; [A plus 1] = [VHGDBid1];
	[A plus 2] = 0; [A plus 3] = 0; [A plus 4] = 0; [A plus 5] = 0;
	[A plus 6] = 0; [A plus 7] = 0; [A plus 8] = 0; [A plus 9] = 0;
	[A plus 10] = 0; [A plus 11] = 0; [A plus 12] = 0; [A plus 13] = 0;
	[A plus 14] = 0; [A plus 15] = 0; [A plus 16] = 0; [A plus 17] = 0;
	[A plus 18] = 0; [A plus 19] = 0; [A plus 20] = 0;
	[VHGDBi] = 0;
    "VHGDB pack message"
	A = [VHGDBi]; ? A >= [VHGDBmsglen] -> VHGDB write record;
	C = [VHGDBmsgptr]; C + A; B = [C]; ? B != 95 -> VHGDB pack character ready; B = 32;
    "VHGDB pack character ready"
	C = A; C / 4; E = [VHGDBptr]; E + 2; E + C;
	A & 3; A '* 8; B < A; A = [E]; A | B; [E] = A;
	[VHGDBi]+; -> VHGDB pack message;
    "VHGDB write record"
	[File Name] = VHGDBfile; [File Position] = [VHGDBbytes]; [File Command] = WRITE;
	[Block Pointer] = [VHGDBptr]; [Block Size] = VHGDBREC; isocall;
	? [File Status] + ERROR -> VHGDB add done;
	A = [Block Size]; ? A != VHGDBREC -> VHGDB add done;
	A = [VHGDBbytes]; A + VHGDBREC; [VHGDBbytes] = A;
	[File Name] = VHGDBfile; [File Position] = 0; [File Command] = SET SIZE;
	[File Size] = [VHGDBbytes]; isocall;
	? [File Status] + ERROR -> VHGDB add done;
	[VHGDBrecs]+; [VHGDBstatus] = 1;
    "VHGDB add done"
	end;

      ( Replace one locally contributed source-format record.  REP may read
	consolidated records but the original module protects every byte before
	the header boundary from correction.  Status 2 reports that distinction. )
"VHGDB replace"
	[VHGDBstatus] = 0; A = [VHGDBloaded]; ? A = 0 -> VHGDB replace done;
	A = [VHGDBrecordno]; ? A < 0 -> VHGDB replace done;
	? A >= [VHGDBrecs] -> VHGDB replace done;
	A = [VHGDBmsglen]; ? A = 0 -> VHGDB replace done; ? A > 76 -> VHGDB replace done;
	A = [VHGDBrecordno]; A '* VHGDBREC; A + VHGDBHDR; [VHGDBpos] = A;
	? A >= [vhguidedata] -> VHGDB replace local; [VHGDBstatus] = 2; -> VHGDB replace done;
    "VHGDB replace local"
	A = [VHGDBrecordno]; A '* 21; A + vhguidedata; A + 1; [VHGDBptr] = A;
	[A] = [VHGDBid0]; [A plus 1] = [VHGDBid1];
	[A plus 2] = 0; [A plus 3] = 0; [A plus 4] = 0; [A plus 5] = 0;
	[A plus 6] = 0; [A plus 7] = 0; [A plus 8] = 0; [A plus 9] = 0;
	[A plus 10] = 0; [A plus 11] = 0; [A plus 12] = 0; [A plus 13] = 0;
	[A plus 14] = 0; [A plus 15] = 0; [A plus 16] = 0; [A plus 17] = 0;
	[A plus 18] = 0; [A plus 19] = 0; [A plus 20] = 0;
	[VHGDBi] = 0;
    "VHGDB replace pack message"
	A = [VHGDBi]; ? A >= [VHGDBmsglen] -> VHGDB replace write;
	C = [VHGDBmsgptr]; C + A; B = [C]; ? B != 95 -> VHGDB replace character ready; B = 32;
    "VHGDB replace character ready"
	C = A; C / 4; E = [VHGDBptr]; E + 2; E + C;
	A & 3; A '* 8; B < A; A = [E]; A | B; [E] = A;
	[VHGDBi]+; -> VHGDB replace pack message;
    "VHGDB replace write"
	[File Name] = VHGDBfile; [File Position] = [VHGDBpos]; [File Command] = WRITE;
	[Block Pointer] = [VHGDBptr]; [Block Size] = VHGDBREC; isocall;
	? [File Status] + ERROR -> VHGDB replace done;
	A = [Block Size]; ? A != VHGDBREC -> VHGDB replace done;
	[VHGDBstatus] = 1;
    "VHGDB replace done"
	end;

      ( Mark one locally contributed record with the original eight-byte
	Removed: tombstone.  Consolidated records remain readable and protected. )
"VHGDB remove"
	[VHGDBstatus] = 0; A = [VHGDBloaded]; ? A = 0 -> VHGDB remove done;
	A = [VHGDBrecordno]; ? A < 0 -> VHGDB remove done;
	? A >= [VHGDBrecs] -> VHGDB remove done;
	A '* VHGDBREC; A + VHGDBHDR; [VHGDBpos] = A;
	? A >= [vhguidedata] -> VHGDB remove local; [VHGDBstatus] = 2; -> VHGDB remove done;
    "VHGDB remove local"
	A = [VHGDBrecordno]; A '* 21; A + vhguidedata; A + 1; [VHGDBptr] = A;
	[A] = 6F6D6552h; [A plus 1] = 3A646576h;
	[File Name] = VHGDBfile; [File Position] = [VHGDBpos]; [File Command] = WRITE;
	[Block Pointer] = [VHGDBptr]; [Block Size] = 8; isocall;
	? [File Status] + ERROR -> VHGDB remove done;
	A = [Block Size]; ? A != 8 -> VHGDB remove done; [VHGDBstatus] = 1;
    "VHGDB remove done"
	end;

      ( Compact Removed: records in memory and commit the completed bounded
	image in one write.  Only tombstones below the consolidated boundary move
	that boundary, matching CLEAN.CPP's swap-file result. )
"VHGDB clean"
	[VHGDBstatus] = 0; [VHGDBcleantotal] = [VHGDBrecs]; [VHGDBcleanremoved] = 0;
	A = [VHGDBloaded]; ? A = 0 -> VHGDB clean done;
	[VHGDBcleanboundary] = [vhguidedata]; [VHGDBcleanin] = 0; [VHGDBcleanout] = 0;
    "VHGDB clean record"
	A = [VHGDBcleanin]; ? A >= [VHGDBrecs] -> VHGDB clean write;
	A '* 21; C = vhguidedata; C + 1; C + A;
	A = [C]; ? A != 6F6D6552h -> VHGDB clean retain;
	A = [C plus 1]; ? A != 3A646576h -> VHGDB clean retain;
	[VHGDBcleanremoved]+; A = [VHGDBcleanin]; A '* VHGDBREC; A + VHGDBHDR;
	? A >= [VHGDBcleanboundary] -> VHGDB clean next;
	[VHGDBcleanboundary] - VHGDBREC; -> VHGDB clean next;
    "VHGDB clean retain"
	A = [VHGDBcleanout]; ? A = [VHGDBcleanin] -> VHGDB clean retained;
	A '* 21; E = vhguidedata; E + 1; E + A; [VHGDBcleanword] = 0;
    "VHGDB clean copy"
	A = [VHGDBcleanword]; B = [C]; [E] = B; C + 1; E + 1;
	[VHGDBcleanword]+; ? A < 20 -> VHGDB clean copy;
    "VHGDB clean retained"
	[VHGDBcleanout]+;
    "VHGDB clean next"
	[VHGDBcleanin]+; -> VHGDB clean record;
    "VHGDB clean write"
	[vhguidedata] = [VHGDBcleanboundary]; A = [VHGDBcleanout]; A '* VHGDBREC;
	A + VHGDBHDR; [VHGDBbytes] = A;
	[File Name] = VHGDBfile; [File Position] = 0; [File Command] = WRITE;
	[Block Pointer] = vhguidedata; [Block Size] = [VHGDBbytes]; isocall;
	? [File Status] + ERROR -> VHGDB clean reload;
	A = [Block Size]; ? A != [VHGDBbytes] -> VHGDB clean reload;
	[File Name] = VHGDBfile; [File Position] = 0; [File Command] = SET SIZE;
	[File Size] = [VHGDBbytes]; isocall;
	? [File Status] + ERROR -> VHGDB clean reload;
	[VHGDBrecs] = [VHGDBcleanout]; [VHGDBstatus] = 1; -> VHGDB clean done;
    "VHGDB clean reload"
	=> VHGDB load; [VHGDBstatus] = 0;
    "VHGDB clean done"
	end;
