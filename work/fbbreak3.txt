( ==================================================================== )
( float glue - four routines, so nothing else has to know the ABI       )
( ==================================================================== )

      ( PAL fa from f32 : FA <- (double)the binary32 in [PFbits]. )

"PAL fa from f32"
	A = [PFbits]; [FS0] = A;
	=> FLoadF32;
	end;

      ( PAL fb from f32 : FB <- (double)the binary32 in [PFbits],
	leaving FA untouched.  FLoadF32 lands in FA, so FA is saved
	across it - getting this wrong silently destroys the accumulator
	in the middle of a shade. )

"PAL fb from f32"
	A = [FA0]; [PFsa0] = A;
	A = [FA1]; [PFsa1] = A;
	A = [PFbits]; [FS0] = A;
	=> FLoadF32;
	A = [FA0]; [FB0] = A;
	A = [FA1]; [FB1] = A;
	A = [PFsa0]; [FA0] = A;
	A = [PFsa1]; [FA1] = A;
	end;

      ( PAL fa to fb : FB <- FA, no FPU involved. )

"PAL fa to fb"
	A = [FA0]; [FB0] = A;
	A = [FA1]; [FB1] = A;
	end;

      ( PAL f32 make : [PFbits] <- (float)((double)[PFnum] / (double)[PFden]).
	The one place integers become floats.  [PFden] = 1 gives a plain
	integer-valued float. )

"PAL f32 make"
	[FI] = [PFden]; => IntToF; => PAL fa to fb;
	[FI] = [PFnum]; => IntToF;
	=> FQuo;
	=> FStoreF32;
	[PFbits] = [FS0];
	end;

	float path shows up here instead of hiding inside a clamp. )

	[PFnum] = 64; [PFden] = 1; => PAL f32 make; [SH6432] = [PFbits];

( ==================================================================== )
( tavola_colori - copy, filter in place, upload from ZERO               )
( ==================================================================== )

"PAL tavola"
	( --- the filters are signed chars widened to 16-bit unsigned --- )
	=> PAL filt check;

	( --- step 1: copy, unless the source aliases the destination --- )
	? [PVself] != 0 -> PAL tv filter;
	D = [PVsrc];
	A = [PVfirst]; A * 3; A + pal6;
	B = [PVn]; B * 3;
	? B = 0 -> PAL tv filter;
    "PAL tv copy"
	C = [D]; C & 255; [A] = C; A+; D+;
	B ^ PAL tv copy;

	( --- step 2: filter in place --- )
    "PAL tv filter"
	A = [PVfirst]; A * 3; A + pal6;
	B = 0; B + [PVn];
	? B = 0 -> PAL tv up;
    "PAL tv loop"
	C = [A]; C & 255; C '* [PVfr]; C & 65535; C '/ 63;
	? C '< 64 -> PAL tv r; C = 63;
    "PAL tv r"
	[A] = C; A+;
	C = [A]; C & 255; C '* [PVfg]; C & 65535; C '/ 63;
	? C '< 64 -> PAL tv g; C = 63;
    "PAL tv g"
	[A] = C; A+;
	C = [A]; C & 255; C '* [PVfb]; C & 65535; C '/ 63;
	? C '< 64 -> PAL tv b; C = 63;
    "PAL tv b"
	[A] = C; A+;
	B ^ PAL tv loop;

	( --- step 3: upload colours 0 .. first+n-1 --- )
    "PAL tv up"
	A = [PVfirst]; A + [PVn]; [PUn] = A;
	=> PAL upload;
	end;

      ( PAL filt check : sign-extend each filter from char, widen to the
	16-bit unsigned the original's arithmetic actually uses, and
	record whether it was inside the 0..127 the callers assume.
	NOCTIS-1.CPP:3934 passes random(64)+64 = 64..127, which fits -
	but the assumption is asserted here, not trusted. )

"PAL filt check"
	C = [PVfr]; => MEM sx8;
	? C >= 0 -> PAL fc r;
	A = [PVrange]; A | 1; [PVrange] = A;
    "PAL fc r"
	? C <= 127 -> PAL fc r2;
	A = [PVrange]; A | 1; [PVrange] = A;
    "PAL fc r2"
	C & 65535; [PVfr] = C;

	C = [PVfg]; => MEM sx8;
	? C >= 0 -> PAL fc g;
	A = [PVrange]; A | 2; [PVrange] = A;
    "PAL fc g"
	? C <= 127 -> PAL fc g2;
	A = [PVrange]; A | 2; [PVrange] = A;
    "PAL fc g2"
	C & 65535; [PVfg] = C;

	C = [PVfb]; => MEM sx8;
	? C >= 0 -> PAL fc b;
	A = [PVrange]; A | 4; [PVrange] = A;
    "PAL fc b"
	? C <= 127 -> PAL fc b2;
	A = [PVrange]; A | 4; [PVrange] = A;
    "PAL fc b2"
	C & 65535; [PVfb] = C;
	end;

      ( PAL upload : the first 3*[PUn] components of pal6 go to
	curpal6, and that
	span of the LUT.  ALWAYS from zero.  This is the DAC write. )

"PAL upload"
	A = pal6; D = curpal6;
	B = [PUn]; B * 3;
	? B = 0 -> PAL up lut;
    "PAL up loop"
	C = [A]; [D] = C; A+; D+;
	B ^ PAL up loop;
    "PAL up lut"
	[PLfirst] = 0; [PLn] = [PUn];
	=> PAL lut;
	end;

( ==================================================================== )
( the LUT - the ONE place six bits become eight                         )
( ==================================================================== )

"PAL lut"
	B = 0; B + [PLn];
	? B = 0 -> PAL lut done;
	D = [PLfirst]; D * 3; D + curpal6;
	E = [PLfirst]; E + pal;
    "PAL lut loop"
	C = [D];	  C & 63; C * 4; C < 16;
	A = [D plus 1];   A & 63; A * 4; A < 8;   C + A;
	A = [D plus 2];   A & 63; A * 4;          C + A;
	[E] = C;
	D + 3; E+;
	B ^ PAL lut loop;
    "PAL lut done"
	end;

( ==================================================================== )
( shade - NOCTIS-0.CPP:1151, truncating, binary32 accumulators          )
( ==================================================================== )

      ( PAL shade : writes pal6[3*[SHfirst] ..] for [SHn] colours,
	ramping (SHsr,SHsg,SHsb) towards (SHer,SHeg,SHeb).
	All six are binary32 BIT PATTERNS, so any float the original
	could pass can be expressed. )

"PAL shade"
	( k = 1.00 / (float)n, stored to a float )
	[PFnum] = 1; [PFden] = [SHn]; => PAL f32 make; [SHk] = [PFbits];

	[shcur plus 0] = [SHsr];
	[PFbits] = [SHsr]; => PAL fa from f32;
	[PFbits] = [SHer]; => PAL fb from f32;
	=> PAL delta;  [shdel plus 0] = [PFbits];

	[shcur plus 1] = [SHsg];
	[PFbits] = [SHsg]; => PAL fa from f32;
	[PFbits] = [SHeg]; => PAL fb from f32;
	=> PAL delta;  [shdel plus 1] = [PFbits];

	[shcur plus 2] = [SHsb];
	[PFbits] = [SHsb]; => PAL fa from f32;
	[PFbits] = [SHeb]; => PAL fb from f32;
	=> PAL delta;  [shdel plus 2] = [PFbits];

	( first_color *= 3 )
	A = [SHfirst]; A * 3; A + pal6; [SHdst] = A;

	B = 0; B + [SHn];
	? B = 0 -> PAL sh done;
    "PAL sh loop"
	A = shcur;     [SHcurp] = A;
	A = shdel;     [SHdelp] = A;
	=> PAL shade comp;
	A = [SHdst]; A+; [SHdst] = A;
	A = shcur; A+; [SHcurp] = A;
	A = shdel; A+; [SHdelp] = A;
	=> PAL shade comp;
	A = [SHdst]; A+; [SHdst] = A;
	A = shcur; A + 2; [SHcurp] = A;
	A = shdel; A + 2; [SHdelp] = A;
	=> PAL shade comp;
	A = [SHdst]; A+; [SHdst] = A;
	B ^ PAL sh loop;
    "PAL sh done"
	end;

      ( PAL delta : [PFbits] <- (float)((FA - FB) * k), where FA holds
	start and FB holds finish.  The original computes
	(finish - start) * k, so the subtraction is reversed here and
	FSub is FA - FB: FB is loaded with finish and FA with start, then
	the two are swapped.  Written out rather than clever. )

"PAL delta"
	A = [FA0]; [PFsa0] = A;		( start, low  )
	A = [FA1]; [PFsa1] = A;		( start, high )
	A = [FB0]; [FA0] = A;		( FA <- finish )
	A = [FB1]; [FA1] = A;
	A = [PFsa0]; [FB0] = A;		( FB <- start  )
	A = [PFsa1]; [FB1] = A;
	=> FSub;			( FA = finish - start )
	[PFbits] = [SHk]; => PAL fb from f32;
	=> FMul;			( FA = (finish-start) * k )
	=> F32Narrow;			( stored to a float )
	=> FStoreF32;
	[PFbits] = [FS0];
	end;

      ( PAL shade comp : one component.  The clamp is the ORIGINAL's,
	including its inverted else branch, and the in-range store is a
	CHOP.  Then the running value advances and is narrowed. )

"PAL shade comp"
	A = [SHcurp]; [PFbits] = [A];
	=> PAL fa from f32;

	[PFbits] = [SHz32]; => PAL fb from f32;
	=> FCmp;
	A = [FI]; ? A < 0 -> PAL sc under;

	[PFbits] = [SH6432]; => PAL fb from f32;
	=> FCmp;
	A = [FI]; ? A < 0 -> PAL sc inrange;

	( >= 64, so the else branch, and start > 0 is true )
	[SHb] = 63; -> PAL sc store;

    "PAL sc under"
	( < 0, so the else branch, and start > 0 is false )
	[SHb] = 0; -> PAL sc store;

    "PAL sc inrange"
	=> FToIntNear;
	[SHb] = [FI];

    "PAL sc store"
	A = [SHdst]; C = [SHb]; C & 255; [A] = C;

	( start += delta, then narrowed because start is a float )
	A = [SHcurp]; [PFbits] = [A];
	=> PAL fa from f32;
	A = [SHdelp]; [PFbits] = [A];
	=> PAL fb from f32;
	=> FAdd;
	=> F32Narrow;
	=> FStoreF32;
	A = [SHcurp]; C = [FS0]; [A] = C;
	end;
