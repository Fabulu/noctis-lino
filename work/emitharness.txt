      ( *** emitharness - headless Noctis-IV-Plus engine-output emitter.

	Reproduces the output of the Noctis-IV-Plus reference engines
	(tests/harness/NIVTEST.CPP and tests/nivgen/main.rs) so the port's
	generation can be diffed byte-for-byte against both.

	Command line (shared by both references):
	    selftest
	    system  -x X -y Y -z Z
	    scan    -x X -y Y -z Z
	    planet  -x X -y Y -z Z -p INDEX
	    planet-all -x X -y Y -z Z
	    sector  -x X -y Y -z Z -p INDEX -lon L -lat B [-sc TYPE]
	    surftex -x X -y Y -z Z -p INDEX -lon L -lat B [-sc TYPE]
	    (all)    -o FILE   append text output to FILE

	Text output is written to the file named by -o (default: the 8.3
	truncated harness name).  Binary dumps use the C -dump DIR contract.

	The hash lines and binary dumps are what the parity rules compare:
	the four hash families at exact byte lengths (surface_map 46080,
	atmo_overlay 32400, palette64 192, heightmap 40000, objectchart
	40000, surf_texture 65024, sky_texture 46080), FNV-1a and CRC-32 in
	the reference's exact form (see work/emit.txt).

	Current coverage: the selftest command's hash-vector section (all
	four reference vectors) and the integer random() section.  The
	float sections (flandom) and the system/planet/sector/surftex
	generation are wired as later steps.

	Build:   powershell -File lino_build.ps1 -Src work\emitharness.txt
	Check:   python noctis-harness/niv_check.py \
	             selftest-harness work\emitself.out
        )

"libraries"

	fbmem;
	brtl;
	emit;
	emitcli;
	emitout;

"directors"

	program name = { emitharness };
	unit = 32;

"variables"

	result file name = { emitself.out };
	V0 = { };
	V1 = { a };
	V2 = { abc };
	CMDSELF = { selftest };
	outfile = { emitself.out };
	ismatch = 0;
	EHrand	= 0;
	EHix	= 0;

"workspace"

	vfsbuf = 64;	( the long selftest sentence )

"programme"

	=> EMCL run;
	=> EMO init;

	( dispatch on the command word )
	[EMOname] = result file name;
	A = CMDSELF; => EMCL cmdis; ? A = 1 -> EH selftest;
	( unknown / unimplemented command: just exit cleanly )
	-> EH done;

    "EH selftest"
	=> EH selftest hash;
	=> EH selftest rand;
	-> EH done;

	[EHix] = 0;
    "EH rand loop"
	=> BrtlRand;
	[EHrand] = A;		( save before the label emission clobbers A )
	A = 32; => EMO ch; A = 32; => EMO ch;
	A = 114; => EMO ch; A = 97; => EMO ch; A = 110; => EMO ch; A = 100; => EMO ch;
	A = 61; => EMO ch;

	=> EMO crlf;
	[EHix] + 1;
	A = [EHix]; ? A '< 8 -> EH rand loop;

	=> EMO flush;
	end;

    "EH done"
	=> EMO flush;
	end;
