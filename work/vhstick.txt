      ( *** vhstick - Stardrifter 3-D frame lines ***

	Uses the already-delivered projector nucleus for the two endpoints,
	then draws the visible segment into adapted.  This is the solid
	Stick mode used by the clean-start cupola frame.  The additive flare
	mode is added with the alogena layer. )

"variables"

	VHSx0 = 0; VHSy0 = 0; VHSz0 = 0;
	VHSx1 = 0; VHSy1 = 0; VHSz1 = 0;
	VHSpx0 = 0; VHSpy0 = 0; VHSpx1 = 0; VHSpy1 = 0;
	VHSdx = 0; VHSdy = 0; VHSsx = 0; VHSsy = 0;
	VHSerr = 0; VHSe2 = 0;
	VHSdrawn = 0; VHSflare = 0; VHSphase = 0;
	VHScolor = 62;
	VHSnearbase = 0;

"programme"

"VH stick3d"
	( NOCTIS-0.CPP stick3d() owns a fixed 200-unit near plane, separate
	  from poly3d's live plane.  Preserve the caller's plane while the
	  segment is rotated, clipped, and projected. )
	[PGFi] = FSUNEG; => PGF a; => FStoreF32; [VHSnearbase] = [FS0];
	[PGFi] = FSUNEG; [PGFt] = 43480000h; => PGF setf32;
	( load the two float endpoints )
	[PGFi] = FSINX;     [PGFt] = [VHSx0]; => PGF setf32;
	[PGFi] = FSINY;     [PGFt] = [VHSy0]; => PGF setf32;
	[PGFi] = FSINZ;     [PGFt] = [VHSz0]; => PGF setf32;
	A = FSINX; A+; [PGFi] = A; [PGFt] = [VHSx1]; => PGF setf32;
	A = FSINY; A+; [PGFi] = A; [PGFt] = [VHSy1]; => PGF setf32;
	A = FSINZ; A+; [PGFi] = A; [PGFt] = [VHSz1]; => PGF setf32;

	[PJnrv] = 2; [PJmode] = 0;
	=> PJ rotate;
	A = [PJdoflag]; ? A = 0 -> VHS done;
	? A = 2 -> VHS zload;
	( The source clips a segment with one endpoint behind stick_uneg rather
	  than discarding it.  PJ zemit performs the same float interpolation;
	  unlike the polygon walk, this line path emits exactly one intersection
	  and one surviving endpoint. )
	A = [rwf plus 0]; ? A != 0 -> VHS first visible;
	[PJbx] = 0; [PJvv] = 1; -> VHS zemit;
    "VHS first visible"
	[PJbx] = 1; [PJvv] = 0;
    "VHS zemit"
	[PJdi] = 0; => PJ zemit;
	A = [PJvv]; A + FSRXF; [PGFi] = A; A = FSUX; A+; [PGFj] = A; => PGF mov;
	A = [PJvv]; A + FSRYF; [PGFi] = A; A = FSUY; A+; [PGFj] = A; => PGF mov;
	A = [PJvv]; A + FSRZF; [PGFi] = A; A = FSUZ; A+; [PGFj] = A; => PGF mov;
	[PJvr2] = 2; -> VHS project;
    "VHS zload"
	=> PJ zload;
    "VHS project"
	=> PJ project3d;
	[VHSpx0] = [mp plus 0]; [VHSpy0] = [mp plus 1];
	[VHSpx1] = [mp plus 2]; [VHSpy1] = [mp plus 3];

	( Stick() swaps non-vertical endpoints whenever xa-xp is negative.  This
	  is visually essential for flare 0: each left-to-right 00/3E pair is
	  partly overwritten by the following sample, producing the original
	  dark reflective beads.  Reversed traversal leaves every 3E neighbour
	  untouched and incorrectly turns the framework into solid white lines. )
	A = [VHSpx1]; ? A >= [VHSpx0] -> VHS endpoints ordered;
	C = [VHSpx0]; [VHSpx0] = [VHSpx1]; [VHSpx1] = C;
	C = [VHSpy0]; [VHSpy0] = [VHSpy1]; [VHSpy1] = C;
    "VHS endpoints ordered"

	( reject segments wholly outside one edge, then clamp the endpoints;
	   cupola's clean visible lines are already inside these bounds )
	A = [VHSpx0]; ? A >= PGLBX -> VHS not left; A = [VHSpx1]; ? A < PGLBX -> VHS done;
    "VHS not left"
	A = [VHSpx0]; ? A <= PGUBX -> VHS not right; A = [VHSpx1]; ? A > PGUBX -> VHS done;
    "VHS not right"
	A = [VHSpy0]; ? A >= PGLBY -> VHS not top; A = [VHSpy1]; ? A < PGLBY -> VHS done;
    "VHS not top"
	A = [VHSpy0]; ? A <= PGUBY -> VHS not bottom; A = [VHSpy1]; ? A > PGUBY -> VHS done;
    "VHS not bottom"

	A = [VHSpx0]; ? A >= PGLBX -> VHS x0 hi; [VHSpx0] = PGLBX;
    "VHS x0 hi"
	A = [VHSpx0]; ? A <= PGUBX -> VHS x0 ok; [VHSpx0] = PGUBX;
    "VHS x0 ok"
	A = [VHSpx1]; ? A >= PGLBX -> VHS x1 hi; [VHSpx1] = PGLBX;
    "VHS x1 hi"
	A = [VHSpx1]; ? A <= PGUBX -> VHS x1 ok; [VHSpx1] = PGUBX;
    "VHS x1 ok"
	A = [VHSpy0]; ? A >= PGLBY -> VHS y0 hi; [VHSpy0] = PGLBY;
    "VHS y0 hi"
	A = [VHSpy0]; ? A <= PGUBY -> VHS y0 ok; [VHSpy0] = PGUBY;
    "VHS y0 ok"
	A = [VHSpy1]; ? A >= PGLBY -> VHS y1 hi; [VHSpy1] = PGLBY;
    "VHS y1 hi"
	A = [VHSpy1]; ? A <= PGUBY -> VHS y1 ok; [VHSpy1] = PGUBY;
    "VHS y1 ok"

	( Integer Bresenham supplies the projected samples.  Source flare 0 writes
	  the literal little-endian word 0x3E00 at every sample: a dark structural
	  point followed by its bright reflective neighbour.  Flare 1 advances
	  through the full segment but paints one point out of two, brightening the
	  indexed destination without changing its palette band. )
	A = [VHSpx1]; A - [VHSpx0]; [VHSsx] = 1;
	? A >= 0 -> VHS dx positive;
	A = 0; A - 1; [VHSsx] = A;
	A = 0; A - [VHSpx1]; A + [VHSpx0];
    "VHS dx positive"
	[VHSdx] = A;
	A = [VHSpy1]; A - [VHSpy0]; [VHSsy] = 1;
	? A >= 0 -> VHS dy positive;
	C = A; A = 0; A - 1; [VHSsy] = A; [VHSdy] = C; -> VHS dy ready;
    "VHS dy positive"
	C = 0; C - A; [VHSdy] = C;
    "VHS dy ready"
	A = [VHSdx]; A + [VHSdy]; [VHSerr] = A;
	[VHSphase] = 0;

"VHS line loop"
    ( The projected line is already clipped to the fixed 320x200 RADPT
      viewport above.  Address it directly instead of paying the generic
      segmented-memory SP get/put path for every Bresenham sample. )
    A = [VHSpy0]; A '* 320; A + [VHSpx0]; A + RADPT; A + nw; D = A;
    A = [VHSflare]; ? A = 1 -> VHS luminous point;
    [D] = 0;
    C = [VHScolor]; C & 255; [D plus 1] = C; -> VHS point done;
"VHS luminous point"
    A = [VHSphase]; A & 1; ? A != 0 -> VHS point done;
    A = [D]; A & 255; C = A; C & 63; C + 8;
    ? C <= 62 -> VHS luminous level ready; C = 62;
"VHS luminous level ready"
    A & 192; C | A; [D] = C;
    "VHS point done"
	[VHSphase]+;
	[VHSdrawn]+;
	A = [VHSpx0]; ? A != [VHSpx1] -> VHS line step;
	A = [VHSpy0]; ? A = [VHSpy1] -> VHS done;
    "VHS line step"
	A = [VHSerr]; A + A; [VHSe2] = A;
	A = [VHSe2]; ? A < [VHSdy] -> VHS skip x;
	A = [VHSerr]; A + [VHSdy]; [VHSerr] = A;
	A = [VHSpx0]; A + [VHSsx]; [VHSpx0] = A;
    "VHS skip x"
	A = [VHSe2]; ? A > [VHSdx] -> VHS skip y;
	A = [VHSerr]; A + [VHSdx]; [VHSerr] = A;
	A = [VHSpy0]; A + [VHSsy]; [VHSpy0] = A;
    "VHS skip y"
	-> VHS line loop;

    "VHS done"
	[PGFi] = FSUNEG; [PGFt] = [VHSnearbase]; => PGF setf32;
	end;
