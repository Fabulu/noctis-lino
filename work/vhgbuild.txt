      ( *** game-vh - live walkable Stardrifter ***

	Clean-source position (0,0,-500), live look controls, and movement
	inside the real VEHICLE + cupola geometry.  Arrow keys look, WASD
	moves, ESC exits.  This is the direct interactive integration of the
	Wave-10 renderer; the remaining ship consoles/light and wider game
	state are added here rather than in another standalone harness. )

	[MgT00] = [FA0]; [MgT01] = [FA1];
	=> MGD reset; [MgNframes] = 1; => MG vimana loop;

	end;

"VHG render"
	A = [VHGmode]; ? A != 0 -> VHG render surface;
	[SPpreg] = RGADP; [SPpn] = NPIX; [SPval] = 0; => SP fill page;
	=> PGF constants;
	( The source renderer's 100-unit plane is safe for exterior views, but
	  from inside the hull a fast turn can put every subdivided leaf of a
	  nearby wall/cupola behind it for one frame.  Keep the shared renderer
	  exact and narrow only this live interior view to 50 units. )
	[PGFi] = FSUNEG; [PGFt] = VHGNEAR; => PGF setf32;
	[VHSdx0] = [MgDzatX0]; [VHSdx1] = [MgDzatX1];
	[VHSdy0] = [MgDzatY0]; [VHSdy1] = [MgDzatY1];
	[VHSdz0] = [MgDzatZ0]; [VHSdz1] = [MgDzatZ1];
	[VHSalpha] = [VHGalpha];
	A = [VHGbeta]; A + 180; ? A <= 180 -> VHG external beta ok; A - 360;
    "VHG external beta ok"
	[VHSbeta] = A; => VHS stars;
	[VHTphase] = [VHGframe]; => VHT render;

	( Source vehicle() always submits both cupolas.  Their wide dome can
	  remain partly visible far outside a centre-angle reject, so clipping
	  belongs to the polygon/line paths rather than a camera shortcut. )
	( lower cupola: vehicle() camera offset z+3100,y-550 )
	[VHCroof] = 0;
	[VHVcamxi] = [VHGx]; A = [VHGy]; A - 550; [VHVcamyi] = A;
	A = [VHGz]; A + 3100; [VHVcamzi] = A;
	[VHValpha] = [VHGalpha]; [VHVbeta] = [VHGbeta]; => VH set view;
	[VHCyor] = 0; => VH polycupola; => VH cupola grid;

	( The local roof facets must begin moving before the rising camera meets
	  the dome.  The original changes to its roof rendering during the lift;
	  in this discrete live loop we arm it at the middle ascent boundary so
	  the player passes through the aperture instead of appearing on top of
	  an intact cupola. )
	A = [VHGonroof]; ? A != 0 -> VHG upper after hull;
	A = [VHGlifter]; ? A >= 0 -> VHG upper before hull now;
	A = [VHGy]; ? A >= 0FFFFFEBBh -> VHG upper before hull now;
	-> VHG upper after hull;
    "VHG upper before hull now"
	[VHCroof] = 0;
	[VHVcamyi] = [VHGy]; => VH set view;
	[VHCyor] = 1; => VH polycupola; => VH cupola grid;
	-> VHG upper before hull done;
    "VHG upper after hull"
    "VHG upper before hull done"

	( hull at the player's clean/live camera )
	[VHVcamxi] = [VHGx]; [VHVcamyi] = [VHGy]; [VHVcamzi] = [VHGz];
	=> VH set view;
	=> VH draw mode2 cache;

	A = [VHGonroof]; ? A != 0 -> VHG draw roof aperture;
	A = [VHGlifter]; ? A >= 0 -> VHG interior details;
	A = [VHGy]; ? A >= 0FFFFFEBBh -> VHG interior details;
    "VHG draw roof aperture"
	[VHCroof] = 1;
	[VHVcamxi] = [VHGx]; [VHVcamyi] = [VHGy];
	A = [VHGz]; A + 3100; [VHVcamzi] = A;
	[VHValpha] = [VHGalpha]; [VHVbeta] = [VHGbeta]; => VH set view;
	[VHCyor] = 1; => VH polycupola; => VH cupola grid;
	end;

    "VHG interior details"

	( three right-wall console faces from vehicle() )
	[VHPcamx] = [VHGx]; [VHPcamy] = [VHGy]; [VHPcamz] = [VHGz];
	[VHPalpha] = [VHGalpha]; [VHPbeta] = [VHGbeta];
	[VHPactive] = [VHGconsole]; [VHPselected] = [VHGplanet]; [VHPsysphase] = [VHGframe]; => VH screens;

	( interior halogen fixture: vehicle() camera x-3395,y+480,z+200 )
	A = [VHGx]; A - 3395; [VHVcamxi] = A;
	A = [VHGy]; A + 480; [VHVcamyi] = A;
	A = [VHGz]; A + 200; [VHVcamzi] = A;
	=> VH set view; => VH alogena;
	[VHPfcsstatus] = [MgStatus];
	A = [MgApreached]; ? A = 0 -> VHG FCS status ready; [VHPfcsstatus] = 6;
    "VHG FCS status ready"
	=> VH HUD FCS;
	end;

    "VHG render surface"

"VHG input"
	A = [Client Owns Text Cursor]; ? A = NO -> VHG input done;
	[Console Command] = GET CONSOLE INPUT; isocall;
	[VHGascii] = [Console Input];
	A = [KEY ESCAPE]; ? A = ON -> VHG quit;
	A = [VHGmode]; ? A != 0 -> VHG surface input;
	A = [VHGconsole]; ? A != 0 -> VHG console input;
	A = [VHGascii]; ? A != 13 -> VHG normal input;
	[VHGconsole] = 1; -> VHG input done;
    "VHG console input"
	A = [VHGascii]; ? A = 0 -> VHG input done;
	[VHPkey] = A; => VH GOES key;
	A = [VHPsubmitted]; ? A = 0 -> VHG input done;
	=> VHG command; [VHGconsole] = 0; -> VHG input done;
    "VHG normal input"
	A = [VHGascii]; ? A = 49 -> VHG select planet 1; ? A = 50 -> VHG select planet 2;
	? A = 51 -> VHG select planet 3; -> VHG selection done;
    "VHG select planet 1"
	[VHGplanet] = 0; -> VHG selection done;
    "VHG select planet 2"
	A = [nsnop]; ? A '< 2 -> VHG selection done; [VHGplanet] = 1; -> VHG selection done;
    "VHG select planet 3"
	A = [nsnop]; ? A '< 3 -> VHG selection done; [VHGplanet] = 2;
    "VHG selection done"
	( L lands on the first planet after calibration. )
	A = [VHGascii]; ? A = 76 -> VHG request landing; ? A = 108 -> VHG request landing;
	-> VHG look input;
    "VHG request landing"
	A = [MgApreached]; ? A = 0 -> VHG look input;
	=> VHG land; -> VHG input done;
    "VHG look input"
	A = [KEY LEFT]; ? A = OFF -> VHG no left; A = [VHGbeta]; A + 2; [VHGbeta] = A;
    "VHG no left"
	A = [KEY RIGHT]; ? A = OFF -> VHG no right; A = [VHGbeta]; A - 2; [VHGbeta] = A;
    "VHG no right"
	A = [KEY UP]; ? A = OFF -> VHG no up; A = [VHGalpha]; A - 2; [VHGalpha] = A;
    "VHG no up"
	A = [KEY DOWN]; ? A = OFF -> VHG no down; A = [VHGalpha]; A + 2; [VHGalpha] = A;
    "VHG no down"
	A = [VHGalpha]; ? A <= 44 -> VHG alpha high ok; [VHGalpha] = 44;
    "VHG alpha high ok"
	A = [VHGalpha]; ? A >= VHGN44 -> VHG alpha low ok; [VHGalpha] = VHGN44;
    "VHG alpha low ok"
	A = [VHGbeta]; ? A <= 180 -> VHG beta high ok; A = [VHGbeta]; A - 360; [VHGbeta] = A;
    "VHG beta high ok"
	A = [VHGbeta]; ? A >= VHGN180 -> VHG beta low ok; A = [VHGbeta]; A + 360; [VHGbeta] = A;
    "VHG beta low ok"
	( Refresh the movement basis after input changed the view angles. )
	[VHVcamxi] = [VHGx]; [VHVcamyi] = [VHGy]; [VHVcamzi] = [VHGz];
	[VHValpha] = [VHGalpha]; [VHVbeta] = [VHGbeta]; => VH set view;

	A = [VHGlifter]; ? A != 0 -> VHG input done;
	A = [KEY W]; ? A = OFF -> VHG no W; [VHGstepv] = VHGSTEP; => VHG forward;
    "VHG no W"
	A = [KEY S]; ? A = OFF -> VHG no S; A = 0; A - VHGSTEP; [VHGstepv] = A; => VHG forward;
    "VHG no S"
	A = [KEY A]; ? A = OFF -> VHG no A; A = 0; A - VHGSTEP; [VHGstepv] = A; => VHG strafe;
    "VHG no A"
	A = [KEY D]; ? A = OFF -> VHG input done; [VHGstepv] = VHGSTEP; => VHG strafe;
    "VHG input done"
	=> VHG clamp position;
	=> VHG platform;
	end;

    "VHG surface input"
	( R returns to the Stardrifter; the same standard WASD/arrows remain
	  active while walking. )
	A = [VHGascii]; ? A = 82 -> VHG request return; ? A = 114 -> VHG request return;
	-> VHG surface look;
    "VHG request return"
	=> VHG return ship; -> VHG input done;
    "VHG surface look"
	A = [KEY LEFT]; ? A = OFF -> VHG surface no left; A = [VHGbeta]; A + 2; [VHGbeta] = A;
    "VHG surface no left"
	A = [KEY RIGHT]; ? A = OFF -> VHG surface no right; A = [VHGbeta]; A - 2; [VHGbeta] = A;
    "VHG surface no right"
	A = [KEY UP]; ? A = OFF -> VHG surface no up; A = [VHGalpha]; A - 2; [VHGalpha] = A;
    "VHG surface no up"
	A = [KEY DOWN]; ? A = OFF -> VHG surface no down; A = [VHGalpha]; A + 2; [VHGalpha] = A;
    "VHG surface no down"
	A = [VHGalpha]; ? A <= 44 -> VHG surface alpha high; [VHGalpha] = 44;
    "VHG surface alpha high"
	A = [VHGalpha]; ? A >= VHGN44 -> VHG surface alpha low; [VHGalpha] = VHGN44;
    "VHG surface alpha low"
	A = [VHGbeta]; ? A <= 180 -> VHG surface beta high; A - 360; [VHGbeta] = A;
    "VHG surface beta high"
	A = [VHGbeta]; ? A >= VHGN180 -> VHG surface beta low; A + 360; [VHGbeta] = A;
    "VHG surface beta low"
	[VHVcamxi] = [VHGx]; [VHVcamyi] = [VHGy]; [VHVcamzi] = [VHGz];
	[VHValpha] = [VHGalpha]; [VHVbeta] = [VHGbeta]; => VH set view;
	A = [KEY W]; ? A = OFF -> VHG surface no W; [VHGstepv] = VHGNDSTEP; => VHG forward;
    "VHG surface no W"
	A = [KEY S]; ? A = OFF -> VHG surface no S; A = 0; A - VHGNDSTEP; [VHGstepv] = A; => VHG forward;
    "VHG surface no S"
	A = [KEY A]; ? A = OFF -> VHG surface no A; A = 0; A - VHGNDSTEP; [VHGstepv] = A; => VHG strafe;
    "VHG surface no A"
	A = [KEY D]; ? A = OFF -> VHG surface moved; [VHGstepv] = VHGNDSTEP; => VHG strafe;
    "VHG surface moved"
	=> VHG surface clamp;
	[VHGNDcamx] = [VHGx]; [VHGNDcamz] = [VHGz]; => VHGND eye height;
	[VHGy] = [VHGNDheight];
	end;
    "VHG quit"
	[VHGesc] = 1;
	end;

"VHG forward"
	[FI] = [VHGstepv]; => IntToF; [PGFi] = FSW0; => PGF sa;
	[PGFi] = FSTSB; => PGF a; [PGFi] = FSW0; => PGF mul;
	[PGFi] = FSTCA; => PGF mul; => FToIntNear;
	A = [VHGx]; A - [FI]; [VHGx] = A;
	[PGFi] = FSTCB; => PGF a; [PGFi] = FSW0; => PGF mul;
	[PGFi] = FSTCA; => PGF mul; => FToIntNear;
	A = [VHGz]; A + [FI]; [VHGz] = A;
	end;

"VHG strafe"
	[FI] = [VHGstepv]; => IntToF; [PGFi] = FSW0; => PGF sa;
	[PGFi] = FSTCB; => PGF a; [PGFi] = FSW0; => PGF mul;
	[PGFi] = FSTCA; => PGF mul; => FToIntNear;
	A = [VHGx]; A + [FI]; [VHGx] = A;
	[PGFi] = FSTSB; => PGF a; [PGFi] = FSW0; => PGF mul;
	[PGFi] = FSTCA; => PGF mul; => FToIntNear;
	A = [VHGz]; A + [FI]; [VHGz] = A;
	end;

"VHG clamp position"
	( vehicle main-loop interior bounds, NOCTIS.CPP:2507-2510 )
	A = [VHGx]; ? A >= 0FFFFF3E4h -> VHG x low ok; [VHGx] = 0FFFFF3E4h;
    "VHG x low ok"
	A = [VHGx]; ? A <= 3100 -> VHG x high ok; [VHGx] = 3100;
    "VHG x high ok"
	A = [VHGz]; ? A <= 0FFFFFED4h -> VHG z near ok; [VHGz] = 0FFFFFED4h;
    "VHG z near ok"
	A = [VHGz]; ? A >= 0FFFFE958h -> VHG z far ok; [VHGz] = 0FFFFE958h;
    "VHG z far ok"
	end;

"VHG platform"
	( The lift platform is centred at x=0,z=-3100 with radius 1100. )
	[VHGdx] = [VHGx]; A = [VHGz]; A + 3100; [VHGdz] = A;
	A = [VHGdx]; A '* [VHGdx]; C = A;
	A = [VHGdz]; A '* [VHGdz]; A + C; [VHGdist] = A;
	? A < 1210000 -> VHG platform inside;
	[VHGridearmed] = 1; end;
    "VHG platform inside"
	A = [VHGridearmed]; ? A = 0 -> VHG platform done;
	A = [VHGlifter]; ? A != 0 -> VHG platform done;
	A = [VHGy]; ? A = 0 -> VHG platform go roof;
	? A = 0FFFFFD12h -> VHG platform go inside;
	-> VHG platform done;
    "VHG platform go roof"
	A = 0; A - 100; [VHGlifter] = A; [VHGridearmed] = 0; end;
    "VHG platform go inside"
	[VHGlifter] = 75; [VHGridearmed] = 0;
    "VHG platform done"
	end;

"VHG lift tick"
	A = [VHGlifter]; ? A = 0 -> VHG lift roof flag;
