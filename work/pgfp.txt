      ( *** pgfp - the named float slots of TDPOLYGS, and the two rules
	    that make the port's arithmetic the original's ***

	RULE 1.  THE CHAIN IS CARRIED AT BINARY64, NOT AT LINO'S NATIVE
	BINARY32.  Wave 3 shipped a binary64 register file (fp/fpabi
	declares FA0/FA1 as the halves of a qword; fp/fpx87 implements
	every op as fld qword / op qword / fstp qword under FEnter's
	133Fh).  lino's own "=" floats are 24-bit mantissa; the engine's
	are 53.  The original's are 64.  Carrying at 53 instead of 64
	costs nothing measurable on polymap's span limits; carrying at 24
	costs 3.6e-3 of them.  So every value the original leaves LIVE in
	st(0) lives here in a binary64 slot.

	RULE 2.  A VALUE THE ORIGINAL STORES TO A "float" VARIABLE IS
	NARROWED, AND ONLY THEN.  fp/fpconv's FStoreF32 / FLoadF32 are
	that narrowing.  The four asymmetry sites - fst zz/xx/z2/yy, fst
	rzf[si], fst zk, fst video_x0/video_y0 - are exactly the places
	where BOTH happen: the wide value goes to the immediate consumer
	and the narrow slot to the later reload.  The port takes both
	sides because the original takes both sides.

	RULE 3.  EVERY float->int IS FToIntNear, NEVER a bare "=," AND
	NEVER FToIntChop.  All 16 conversion sites in this wave are
	hand-written fistp/fist in the original, and the rounding control
	stays at 00 there, so they round to NEAREST EVEN.  fp/fpconv's
	house rule forbids emitting a bare "=,"; the ambient control word
	at lino entry is 0E7Fh, RC=chop, so a bare "=," would chop.
	FToIntNear runs under FEnter's 133Fh and rounds correctly.  The
	control word is READ BACK at the census so a missing FEnter
	bracket is visible rather than silent.

	SLOTS.  L.in.oleum variables cannot be indexed, so the named
	floats of TDPOLYGS live as adjacent PAIRS in the workspace "fw",
	slot k occupying fw[2k] and fw[2k+1].  "PGF a", "PGF b", "PGF sa"
	move a slot to FA, to FB, and back.                              )

"programme"

( ==================================================================== )
( slot movement                                                         )
( ==================================================================== )

"PGF a"					( FA <- fw[[PGFi]] )
	A = [PGFi]; A + A; A + fw;
	[FA0] = [A]; [FA1] = [A plus 1];
	end;

"PGF b"					( FB <- fw[[PGFi]] )
	A = [PGFi]; A + A; A + fw;
	[FB0] = [A]; [FB1] = [A plus 1];
	end;

"PGF sa"				( fw[[PGFi]] <- FA )
	A = [PGFi]; A + A; A + fw;
	[A] = [FA0]; [A plus 1] = [FA1];
	end;

"PGF mov"				( fw[[PGFj]] <- fw[[PGFi]] )
	A = [PGFi]; A + A; A + fw;
	C = [PGFj]; C + C; C + fw;
	[C] = [A]; [C plus 1] = [A plus 1];
	end;

      ( PGF ldf32 : FA <- the binary32 whose bit pattern is [PGFt].
	Always exact - every binary32 is a binary64. )

"PGF ldf32"
	[FS0] = [PGFt];
	=> FLoadF32;
	end;

      ( PGF ldf64 : FA <- the binary64 whose halves are [PGFt] (low) and
	[PGFu] (high).  A C double literal is not a float literal and
	must not be routed through the binary32 path. )

"PGF ldf64"
	[FA0] = [PGFt]; [FA1] = [PGFu];
	end;

      ( PGF stf32 : [PGFt] <- (float)FA, the bit image, WITHOUT widening
	back.  This is a C store to a "float" variable and nothing else. )

"PGF stf32"
	=> FStoreF32;
	[PGFt] = [FS0];
	end;

      ( PGF setf32 : fw[[PGFi]] <- (double) the binary32 pattern [PGFt]. )

"PGF setf32"
	=> PGF ldf32;
	=> PGF sa;
	end;

      ( PGF narrow : fw[[PGFi]] <- (double)(float) fw[[PGFi]].
	The port's model of "the original stored this to a float". )

"PGF narrow"
	=> PGF a;
	=> F32Narrow;
	=> PGF sa;
	end;

( ==================================================================== )
( arithmetic on slots.  FA is the accumulator throughout.               )
( ==================================================================== )

"PGF add"				( FA += fw[[PGFi]] )
	=> PGF b; => FAdd; end;

"PGF sub"				( FA -= fw[[PGFi]] )
	=> PGF b; => FSub; end;

"PGF mul"				( FA *= fw[[PGFi]] )
	=> PGF b; => FMul; end;

"PGF quo"				( FA /= fw[[PGFi]] )
	=> PGF b; => FQuo; end;

      ( PGF rsub : FA <- fw[[PGFi]] - FA.  fsubr, and the original uses
	it: "fsubr dword ptr rzf[si]" in every near-plane branch. )

"PGF rsub"
	[FT0] = [FA0]; [FT1] = [FA1];
	=> PGF a;
	[FB0] = [FT0]; [FB1] = [FT1];
	=> FSub;
	end;

      ( PGF rquo : FA <- fw[[PGFi]] / FA.  fdivr; the span setup's
	"fdivr _uno" is 1/_z and nothing else. )

"PGF rquo"
	[FT0] = [FA0]; [FT1] = [FA1];
	=> PGF a;
	[FB0] = [FT0]; [FB1] = [FT1];
	=> FQuo;
	end;

      ( PGF int : [FI] <- fistp(FA), NEAREST EVEN, under the live
	control word.  Rule 3.  Never a bare "=,". )

"PGF int"
	=> FToIntNear;
	end;

      ( PGF fromint : FA <- (double)[FI].  fild of a dword is exact for
	every int32, so the subtraction MUST be done in integers and
	converted once - "fild x2 / fisub x1" - not converted twice and
	subtracted in floats.  That is what keeps kx right when a blown-up
	near-plane interpolation puts mp[] above 2^24. )

"PGF fromint"
	=> IntToF;
	end;
