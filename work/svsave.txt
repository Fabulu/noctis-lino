( ==================================================================== )
( byte-level access subroutines for the CURRENT.BIN buffer              )
( ==================================================================== )

      ( sv get byte: read one byte at offset svboff from curbuf.
	Result in A.  Preserves svboff, clobbers B and E.		    )

"sv get byte"
	E = curbuf;
	A = [svboff]; A > 2; E + A;
	A = [svboff]; A & 3; A < 3; B = A;
	A = [E]; A > B; A & FFh;
	end;

      ( sv get word: read 16-bit LE at svboff from curbuf.
	Result in A.  Preserves svboff.
	Clobbers B, E, svb0, svb1, svtmp.				    )

"sv get word"
	[svtmp] = [svboff];
	=> sv get byte;
	[svb0] = A;
	A = [svboff]; A + 1; [svboff] = A;
	=> sv get byte;
	[svb1] = A;
	[svboff] = [svtmp];
	A = [svb1]; A < 8; A | [svb0];
	end;

      ( sv get dword: read 32-bit LE at svboff from curbuf.
	Result in A and svdw.  Preserves svboff.
	Clobbers B, E, svb0..svb3, svtmp.				    )

"sv get dword"
	[svtmp] = [svboff];
	=> sv get byte; [svb0] = A;
	A = [svboff]; A + 1; [svboff] = A;
	=> sv get byte; [svb1] = A;
	A = [svboff]; A + 1; [svboff] = A;
	=> sv get byte; [svb2] = A;
	A = [svboff]; A + 1; [svboff] = A;
	=> sv get byte; [svb3] = A;
	[svboff] = [svtmp];
	A = [svb1]; A < 8; A | [svb0]; B = A;
	A = [svb2]; A < 16; A | B; B = A;
	A = [svb3]; A < 24; A | B;
	[svdw] = A;
	end;
