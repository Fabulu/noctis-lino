      ( *** svstarmap - STARMAP.BIN lookup/append/tombstone, namespace SV ***

	Ports search_id_code (NOCTIS-0.CPP:4002), starmap append
	(NOCTIS.CPP:1753-1772), and tombstone removal (NOCTIS.CPP:1728-1732).

	SEARCH: double id_codes are compared as signed 96-bit integers via
	trunc(id * 1e15), with the acceptance window |diff| < 1e10.
	APPEND: read-modify-write the whole file in a workspace buffer.
	TOMBSTONE: overwrite bytes 0..7 of the record with "Removed:".

	REQUIRES THE EXTENDED TOOLCHAIN for star-percent (*%'): use
	compiler114m.exe and Cpu i386m.					    )

"directors"

	program name = { svstarmap };
	unit = 32;

"constants"

	SZCFG		= 256;
	SZOUT		= 128;
	SZTST		= 1024;

	SVSMMAGIC	= 56534D53h;

	RECSIZE		= 32;
	HDRSIZE		= 4;

	WIN0		= 540BE400h;
	WIN1		= 2;

	E15LO		= A4C68000h;
	E15HI		= 38D7Eh;
	KZERO		= 104;
	EMAX		= 1068;

	TOMB0		= 6F6D6552h;
	TOMB1		= 3A646576h;

	ASCS		= 53h;
	ASCP		= 50h;
	ASC20		= 20h;

"variables"

	tstname		= { STARMAP.TST };
	cfgname		= { SVSM.CFG };
	outname		= { SVSM.OUT };

	svfsize		= 0;
	svnrec		= 0;
	svflags		= 0;

	nsrch		= 0;
	isrch		= 0;
	srtype		= 0;
	reci		= 0;

	srfound		= 0;
	srpos		= 0;

	dlo	= 0; dhi	= 0; drej	= 0;
	de	= 0; dsn	= 0; dmh	= 0; dml	= 0;
	dk	= 0; dq	= 0; dr	= 0; drc	= 0;
	ua0 = 0; ua1 = 0; ub0 = 0; ub1 = 0;
	uc0 = 0; uc1 = 0; ud0 = 0; ud1 = 0;
	u0 = 0; u1 = 0; u2 = 0; u3 = 0; cy1 = 0;

	sg0	= 0; sg1	= 0; sg2	= 0;
	g0	= 0; g1	= 0; g2	= 0;

	gapt0	= 0; gapt1	= 0; gapt2	= 0;
	gapc0	= 0; gapc1	= 0; gapc2	= 0;
	inwin	= 0;

	svdptr	= 0;
	cfgptr	= 0;

"workspace"

	tstbuf		= SZTST;
	cfgbuf		= SZCFG;
	outbuf		= SZOUT;
	du		= 8;

"programme"

( ==================================================================== )
( read config and test starmap                                           )
( ==================================================================== )

	[File Name]	= cfgname;
	[File Command]	= TEST;
	isocall;
      ? [File Status] + ERROR -> search start;
	[svfsize] = [File Size];

	[File Name]	= cfgname;
	[File Command]	= READ;
	[File Position]	= 0;
	[Block Pointer]	= cfgbuf;
	[Block Size]	= [svfsize];
	isocall;
      ? [File Status] + ERROR -> search start;

	E = cfgbuf;
	A = [E];
      ? A != SVSMMAGIC -> search start;
	[nsrch] = [E plus 1];
	E = cfgbuf; E + 2; [cfgptr] = E;

	[File Name]	= tstname;
	[File Command]	= TEST;
	isocall;
      ? [File Status] + ERROR -> search start;
	[svfsize] = [File Size];

	[File Name]	= tstname;
	[File Command]	= READ;
	[File Position]	= 0;
	[Block Pointer]	= tstbuf;
	[Block Size]	= [svfsize];
	isocall;
      ? [File Status] + ERROR -> search start;
      ? [Block Size] != [svfsize] -> search start;

	A = [svfsize]; A - 4; A / RECSIZE; [svnrec] = A;

( ==================================================================== )
( search loop                                                            )
( ==================================================================== )

    "search start"

	D = outbuf;
	A = SVSMMAGIC;		[D] = A; D + 1;
	A = [nsrch];		[D] = A; D + 1;

	[isrch] = 0;

    "search loop"
      ? [isrch] >= [nsrch] -> search done;

	[svdptr] = D;

	E = [cfgptr];
	A = [E];		[dlo] = A;
	A = [E plus 1];		[dhi] = A;
	A = [E plus 2];		[srtype] = A;

	=> Decode;
	[sg0] = [g0]; [sg1] = [g1]; [sg2] = [g2];

	[srfound] = 0;
	A = 0; A - 1; [srpos] = A;
	[reci] = 0;

    "rec loop"
      ? [reci] >= [svnrec] -> rec done;

	E = tstbuf; E + 1;
	A = [reci]; A * 8; E + A;

	A = [E plus 7]; A > 8; A & FFh;
      ? A != [srtype] -> rec next;

	A = [E];		[dlo] = A;
	A = [E plus 1];		[dhi] = A;
	=> Decode;

	[gapt0] = [g0]; [gapt1] = [g1]; [gapt2] = [g2];
	[gapc0] = [sg0]; [gapc1] = [sg1]; [gapc2] = [sg2];
	=> Gap96;
	=> WinTest;

      ? [inwin] = 0 -> rec next;

	A = [reci]; A * RECSIZE; A + HDRSIZE; [srpos] = A;
	[srfound] = 1;
	-> rec done;

    "rec next"
	A = [reci]; A + 1; [reci] = A;
	-> rec loop;

    "rec done"
	D = [svdptr];
	A = [srfound];		[D] = A; D + 1;
	A = [srpos];		[D] = A; D + 1;

	A = [cfgptr]; A + 3; [cfgptr] = A;
	A = [isrch]; A + 1; [isrch] = A;
	-> search loop;

    "search done"

( ==================================================================== )
( append: add record 42.0 to tstbuf and rewrite file                     )
( ==================================================================== )

	( Build record at tstbuf[1 + svnrec*8]: id=42.0, name="PORTTEST", S99 )

	E = tstbuf; E + 1;
	A = [svnrec]; A * 8; E + A;

	A = 0;			[E] = A; E + 1;
	A = 40450000h;		[E] = A; E + 1;
	A = 54524F50h;		[E] = A; E + 1;
	A = 54534554h;		[E] = A; E + 1;
	A = 20202020h;		[E] = A; E + 1;
	A = 20202020h;		[E] = A; E + 1;
	A = 20202020h;		[E] = A; E + 1;
	A = 39395320h;		[E] = A;

	( update nrec and filesize )

	A = [svnrec]; A + 1; [svnrec] = A;
	A = [svnrec]; A * RECSIZE; A + HDRSIZE; [svfsize] = A;

	( rewrite file )

	[File Name]	= tstname;
	[File Command]	= WRITE;
	[File Position]	= 0;
	[Block Pointer]	= tstbuf;
	[Block Size]	= [svfsize];
	isocall;

	[File Name]	= tstname;
	[File Command]	= SET SIZE;
	[File Size]	= [svfsize];
	isocall;

	A = 1;		( append status: success )
	[D] = A; D + 1;

( ==================================================================== )
( tombstone: overwrite the FIRST record with "Removed:" in tstbuf       )
( ==================================================================== )

	( Tombstone record 0 (the first record) to prove tombstone logic. )
	( The appended record (42.0) stays intact for separate verification. )

	E = tstbuf; E + 1;

	A = TOMB0;		[E] = A;
	A = TOMB1;		[E plus 1] = A;

	( rewrite file )

	[File Name]	= tstname;
	[File Command]	= WRITE;
	[File Position]	= 0;
	[Block Pointer]	= tstbuf;
	[Block Size]	= [svfsize];
	isocall;

	[File Name]	= tstname;
	[File Command]	= SET SIZE;
	[File Size]	= [svfsize];
	isocall;

	A = 1;		( tombstone status: success )
	[D] = A; D + 1;

( ==================================================================== )
( write output                                                           )
( ==================================================================== )

	A = D; A - outbuf;
	[svfsize] = A;

	[File Name]	= outname;
	[File Command]	= WRITE;
	[File Position]	= 0;
	[Block Pointer]	= outbuf;
	[Block Size]	= [svfsize];
	[Block Size]	* BYTESPERUNIT;
	isocall;

	[File Name]	= outname;
	[File Command]	= SET SIZE;
	[File Size]	= [svfsize];
	[File Size]	* BYTESPERUNIT;
	isocall;

	end;

( ==================================================================== )
( Decode / Neg96 / Gap96 / WinTest -- copied from starmap_read/find       )
( ==================================================================== )

"Neg96"
	A = [g0]; ! A; A + 1; [g0] = A;
	C = 0;
      ? A != 0 -> neg96 l1;
	C = 1;
    "neg96 l1"
	A = [g1]; ! A; A + C; [g1] = A;
	D = 0;
      ? C = 0 -> neg96 l2;
      ? A != 0 -> neg96 l2;
	D = 1;
    "neg96 l2"
	A = [g2]; ! A; A + D; [g2] = A;
	end;

"Decode"
	[drej] = 0;
	[g0] = 0; [g1] = 0; [g2] = 0;

	A = [dhi]; A > 20; A & 7FFh;	[de] = A;
      ? [de] = 0	-> dec reject zero;
      ? [de] = 7FFh	-> dec reject nan;
      ? [de] >= EMAX	-> dec reject big;

	A = [dhi]; A > 31;			[dsn] = A;
	A = [dhi]; A & FFFFFh; A | 100000h;	[dmh] = A;
	A = [dlo];				[dml] = A;

	A = 1075; A - [de];			[dk] = A;
      ? [dk] >= KZERO -> dec done;

	A = [dml]; B = E15LO; A *%' B;	[ua0] = A; [ua1] = B;
	A = [dml]; B = E15HI; A *%' B;	[ub0] = A; [ub1] = B;
	A = [dmh]; B = E15LO; A *%' B;	[uc0] = A; [uc1] = B;
	A = [dmh]; B = E15HI; A *%' B;	[ud0] = A; [ud1] = B;

	A = [ua0];				[u0] = A;

	A = [ua1]; B = [ub0]; A + B; E = 0;
      ? A '>= B -> dec c1a;
	E = 1;
    "dec c1a"
	B = [uc0]; A + B;
      ? A '>= B -> dec c1b;
	E + 1;
    "dec c1b"
	[u1] = A;
	[cy1] = E;

	A = [ub1]; B = [uc1]; A + B; D = 0;
      ? A '>= B -> dec c2a;
	D = 1;
    "dec c2a"
	B = [ud0]; A + B;
      ? A '>= B -> dec c2b;
	D + 1;
    "dec c2b"
	B = [cy1]; A + B;
      ? A '>= B -> dec c2c;
	D + 1;
    "dec c2c"
	[u2] = A;

	A = [ud1]; A + D;			[u3] = A;

	E = du;
	A = [u0]; [E]        = A;
	A = [u1]; [E plus 1] = A;
	A = [u2]; [E plus 2] = A;
	A = [u3]; [E plus 3] = A;
	A = 0;
	[E plus 4] = A; [E plus 5] = A; [E plus 6] = A; [E plus 7] = A;

	A = [dk]; A > 5;	[dq] = A;
	A = [dk]; A & 31;	[dr] = A;

	E = du; E + [dq];
      ? [dr] = 0 -> dec aligned;

	A = 32; A - [dr];	[drc] = A;

	A = [E];          A > [dr]; B = [E plus 1]; B < [drc]; A | B; [g0] = A;
	A = [E plus 1];   A > [dr]; B = [E plus 2]; B < [drc]; A | B; [g1] = A;
	A = [E plus 2];   A > [dr]; B = [E plus 3]; B < [drc]; A | B; [g2] = A;
	-> dec sign;

    "dec aligned"
	A = [E];	  [g0] = A;
	A = [E plus 1];	  [g1] = A;
	A = [E plus 2];	  [g2] = A;

    "dec sign"
      ? [dsn] = 0 -> dec done;
	=> Neg96;

    "dec done"
	end;

    "dec reject zero"
	[drej] = 1; end;
    "dec reject nan"
	[drej] = 2; end;
    "dec reject big"
	[drej] = 3; end;

"Gap96"
	A = [gapt0]; B = [gapc0];
	C = 0;
      ? A '>= B -> gap b0;
	C = 1;
    "gap b0"
	A - B;			[g0] = A;

	A = [gapt1]; B = [gapc1];
	D = 0;
      ? A '>= B -> gap b1;
	D = 1;
    "gap b1"
	A - B;
      ? A '>= C -> gap b2;
	D + 1;
    "gap b2"
	A - C;			[g1] = A;

	A = [gapt2]; B = [gapc2];
	A - B; A - D;		[g2] = A;

	A = [g2];
      ? A >= 0 -> gap done;
	=> Neg96;

    "gap done"
	end;

"WinTest"
	[inwin] = 0;
	A = [g2];
      ? A != 0 -> wintest done;
	A = [g1];
      ? A '> WIN1 -> wintest done;
      ? A '< WIN1 -> wintest yes;
	A = [g0];
      ? A '>= WIN0 -> wintest done;
    "wintest yes"
	[inwin] = 1;
    "wintest done"
	end;
