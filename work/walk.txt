      ( *** walk - the Wave 7b ground renderer (B2) ***

	Programme 1 of this file: HPOINT, the bilinear height query
	(NOCTIS-1.CPP:63-93).  Self-contained: it brings its own corpus
	reader and dump writer so it can be graded in isolation, and
	links ONLY the Wave 3 float engine (fp/fpabi..fp/fpconv).  No
	fbmem, no pgmem: the workspace and the byte/IO primitives live
	here, so this file owns a disjoint namespace and cannot collide
	with B1's grnd.txt or Wave 6a's pg*.txt when the full renderer
	is linked later.

	WHAT IS GRADED (tier 2 of WAVE7B_PLAN): the binary32 of py,
	case for case, against grv_ref.c (a separate C transliteration
	on the hardware x87) and grv_spec.py (an exact-rational model).
	The branch test `icx+icz < 16384` is integer and deterministic,
	so hpoint is three-way BYTE-EXACT with zero float tolerance -
	this is the foundation fragment/hpoint feeds fragment from.

	CORPUS: one signed decimal integer per token, six per case:
	    px pz s1 s2 s3 s4
	where s1..s4 are the surfacemap bytes at cpos, cpos+1,
	cpos+201, cpos+200 (h1,h2,h3,h4).  cpos = (pz>>14)*200 + (px>>14).
	row/col pinned to 0..198 so all four corners stay in-bounds.

	OUTPUT: one int32 per case = py's binary32 bit pattern, written
	to grv-out.bin (lino writes units little-endian, matching the C
	side's memcpy+fwrite).

	FLOAT MODEL: the chain runs at extended precision (PC=64, RC=-
	nearest, set by FEnter's 133Fh).  Narrowing to float32 happens
	ONLY at the two stores to py, via F32Narrow - exactly the C
	`float py = ...` / `py += ...` semantics.  h1..h4, icx, icz are
	integer->float conversions of values that fit in 24 bits, so
	they are carried as exact integer doubles (IntToF) and never
	narrowed.  qid = 1/16384 is assembled as its binary64 bit
	pattern (3F100000h:00000000h) so no decimal parser is shared
	with the other side.                                         )

( ==================================================================== )
( W run - load corpus, decode, loop cases, write dump                    )
( ==================================================================== )

"W run"
	=> FEnter;
	=> W load;
	A = [Wcbad]; ? A != 0 -> W run io;
	=> W tokenise;
	[Wcti] = 0;
	[Wdcnt] = 0;
	[Wn] = 0;

    "W run case"
	( need at least an opcode token )
	A = [Wcti]; A + 1; ? A '> [Wctokn] -> W run done;
	=> W next; [Wop] = C;
	A = [Wop]; ? A = WOPHP -> W do hp;
	A = [Wop]; ? A = WOPFR -> W do frag;
	A = [Wop]; ? A = WOPFW -> W do fwd;
	A = [Wop]; ? A = WOPCAV -> W do cav;
	-> W run done;

    "W do hp"
	A = [Wcti]; A + 6; ? A '> [Wctokn] -> W run done;
	=> W next; [HPpx] = C;
	=> W next; [HPpz] = C;
	=> W next; [Ws1] = C;
	=> W next; [Ws2] = C;
	=> W next; [Ws3] = C;
	=> W next; [Ws4] = C;
	=> W place surf;
	=> HPOINT;
	C = WOPHP; => W emit; C = [HPpy]; => W emit;
	C = 0; => W emit; C = 0; => W emit; C = 0; => W emit; C = 0; => W emit;
	C = 0; => W emit; C = 0; => W emit; C = 0; => W emit;
	[Wn]+;
	-> W run case;

    "W do cav"
	( 3 tokens: alfa beta dpp -- binary32 bit patterns )
	A = [Wcti]; A + 3; ? A '> [Wctokn] -> W run done;
	=> W next; [CAValfa] = C;
	=> W next; [CAVbeta] = C;
	=> W next; [CAVdpp] = C;
	=> CAV;
	C = WOPCAV; => W emit;
	C = [CAVpcosbeta]; => W emit;
	C = [CAVpsinbeta]; => W emit;
	C = [CAVtcosbeta]; => W emit;
	C = [CAVtsinbeta]; => W emit;
	C = [CAVpcosalfa]; => W emit;
	C = [CAVpsinalfa]; => W emit;
	C = [CAVtcosalfa]; => W emit;
	C = [CAVtsinalfa]; => W emit;
	[Wn]+;
	-> W run case;

    "W do frag"
	( 12 tokens: x z posx posz s1 s2 s3 s4 shd ssh seed branch )
	A = [Wcti]; A + 12; ? A '> [Wctokn] -> W run done;
	=> W next; [FRx] = C;
	=> W next; [FRz] = C;
	=> W next; [FRposx] = C;
	=> W next; [FRposz] = C;
	=> W next; [Fs1] = C;
	=> W next; [Fs2] = C;
	=> W next; [Fs3] = C;
	=> W next; [Fs4] = C;
	=> W next; [FRshd] = C;
	=> W next; [FRssh] = C;
	=> W next; [FRseed] = C;
	=> W next; [FRbranch] = C;
	=> W frag place;
	=> FRAGMENT;
	C = WOPFR; => W emit;
	C = [FRdepth]; => W emit;
	C = [FRvy10]; => W emit; C = [FRvy11]; => W emit; C = [FRvy12]; => W emit;
	C = [FRvy20]; => W emit; C = [FRvy21]; => W emit; C = [FRvy22]; => W emit;
	C = [FRc1]; => W emit;
	[Wn]+;
	-> W run case;

    "W do fwd"
	( 6 tokens: delta sbn cbn calf px pz -- all binary32 bit patterns )
	A = [Wcti]; A + 6; ? A '> [Wctokn] -> W run done;
	=> W next; [PFdelta] = C;
	=> W next; [PFsbn] = C;
	=> W next; [PFcbn] = C;
	=> W next; [PFcalfa] = C;
	=> W next; [PFpx] = C;
	=> W next; [PFpz] = C;
	=> P FORWARD;
	C = WOPFW; => W emit;
	C = [PFnpx]; => W emit;
	C = [PFnpz]; => W emit;
	C = 0; => W emit; C = 0; => W emit; C = 0; => W emit; C = 0; => W emit;
	C = 0; => W emit; C = 0; => W emit;
	[Wn]+;
	-> W run case;

    "W run done"
	=> W write;
	=> FLeave;
	end;

    "W run io"
	end;

( ==================================================================== )
( W place surf - write the four corner bytes into wsurfv at cpos offsets  )
( ==================================================================== )

"W place surf"
	( cpos = (pz>>14)*200 + (px>>14) ; m200[k]=k*200 inlined )
	A = [HPpz]; A > 14; A '* 200; [HProw] = A;
	A = [HPpx]; A > 14; [HPcol] = A;
	A = [HProw]; A + [HPcol]; [HPcpos] = A;
	( s1 -> wsurfv[cpos]  (h1) )
	A = wsurfv; A + [HPcpos]; C = [Ws1]; C & 255; [A] = C;
	( s2 -> wsurfv[cpos+1]  (h2) )
	A = [HPcpos]; A + 1; A + wsurfv; C = [Ws2]; C & 255; [A] = C;
	( s3 -> wsurfv[cpos+201]  (h3) )
	A = [HPcpos]; A + 201; A + wsurfv; C = [Ws3]; C & 255; [A] = C;
	( s4 -> wsurfv[cpos+200]  (h4) )
	A = [HPcpos]; A + 200; A + wsurfv; C = [Ws4]; C & 255; [A] = C;
	end;

( ==================================================================== )
( W emit - append C to wdumpv[[Wdcnt]], [Wdcnt]+                         )
( ==================================================================== )

"W emit"
	A = wdumpv; A + [Wdcnt]; [A] = C;
	[Wdcnt]+;
	end;

( ==================================================================== )
( W frag place - write the 5 corner bytes into wsurfv for fragment       )
(   h1 = x + z*200;  s1->h1  s2->h1+1  s3->h1+201  s4->h1+200            )
(   ssh -> h1 + sh_delta                                                 )
( ==================================================================== )

"W frag place"
	A = [FRz]; A '* 200; A + [FRx]; [FRh1] = A;
	A = wsurfv; A + [FRh1];            C = [Fs1]; C & 255; [A] = C;
	A = [FRh1]; A + 1;   A + wsurfv;   C = [Fs2]; C & 255; [A] = C;
	A = [FRh1]; A + 201; A + wsurfv;   C = [Fs3]; C & 255; [A] = C;
	A = [FRh1]; A + 200; A + wsurfv;   C = [Fs4]; C & 255; [A] = C;
	A = [FRh1]; A + [FRshd]; A + wsurfv; C = [FRssh]; C & 255; [A] = C;
	end;

( ==================================================================== )
( FRAGMENT - NOCTIS-1.CPP:1028-1142 (the ground vertex + depth + c1 path))
(   grades depth (the exact-required chop #1), the six vy corner heights, )
(   and c1 (both sh_delta and diffuse branches).  vx/vz are trivial       )
(   integer shifts and are not emitted.                                   )
( ==================================================================== )

"FRAGMENT"
	( vx1[0]=x<<14, vx1[1]=(x+1)<<14, vz1[0]=z<<14, vz1[2]=(z+1)<<14 )
	A = [FRx]; A < 14; [FRvx0] = A;
	A = [FRx]; A + 1; A < 14; [FRvx1] = A;
	A = [FRz]; A < 14; [FRvz0] = A;
	A = [FRz]; A + 1; A < 14; [FRvz2] = A;

	( dx = pos_x - 0.5*(vx1[0]+vx1[1])  [float] )
	[FI] = [FRvx0]; => IntToF; [FT0]=[FA0]; [FT1]=[FA1];
	[FI] = [FRvx1]; => IntToF;
	[FB0]=[FT0]; [FB1]=[FT1]; => FAdd;		( FA = vx0+vx1 )
	[FB0] = WHALFLO; [FB1] = WHALFHI; => FMul;	( FA = sum*0.5 )
	[FT0]=[FA0]; [FT1]=[FA1];			( FT = half-sum )
	[FS0] = [FRposx]; => FLoadF32;			( FA = (double)pos_x )
	[FB0]=[FT0]; [FB1]=[FT1]; => FSub;		( FA = pos_x - half-sum )
	=> F32Narrow; => FStoreF32; [FRdx] = [FS0];

	( dz = pos_z - 0.5*(vz1[0]+vz1[2]) )
	[FI] = [FRvz0]; => IntToF; [FT0]=[FA0]; [FT1]=[FA1];
	[FI] = [FRvz2]; => IntToF;
	[FB0]=[FT0]; [FB1]=[FT1]; => FAdd;
	[FB0] = WHALFLO; [FB1] = WHALFHI; => FMul;
	[FT0]=[FA0]; [FT1]=[FA1];
	[FS0] = [FRposz]; => FLoadF32;
	[FB0]=[FT0]; [FB1]=[FT1]; => FSub;
	=> F32Narrow; => FStoreF32; [FRdz] = [FS0];

	( hpdep = sqrt(dx*dx + dz*dz) ; depth = ((long)hpdep) >> 14 )
	[FS0]=[FRdx]; => FLoadF32; => HP fb fa; => FMul; => HP ft fa; ( FT = dx*dx )
	[FS0]=[FRdz]; => FLoadF32; => HP fb fa; => FMul;		( FA = dz*dz )
	[FB0]=[FT0]; [FB1]=[FT1]; => FAdd;			( FA = dx*dx+dz*dz )
	=> FSqrt;
	=> F32Narrow; => FStoreF32; [FRhpdep] = [FS0];
	[FS0]=[FRhpdep]; => FLoadF32; => FToIntChop;		( FI = (long)hpdep )
	A = [FI]; A > 14; [FRdepth] = A;			( depth = (long)hpdep >> 14 )

	( cull: if depth > 64 the tile is offscreen; still emit, depth carries it )
	( depth -= 1 ; if depth < 0 depth = 0  [happens before c1's depth>>1]
	  the test is sign-bit driven: '>=' is UNSIGNED in lino, so a negative
	  depth would read as a huge positive and skip the clamp. )
	A = [FRdepth]; A - 1; [FRdepth] = A;
	A = [FRdepth]; A > 31;			( 0 if depth>=0, nonzero if <0 )
	? A = 0 -> FR depth ok; [FRdepth] = 0;
    "FR depth ok"

	( vy corner heights = -(surf<<11) ; surf at h1, h1+1, h1+200, h1+201
	  vy1[0]=h1, vy1[1]=h1+1, vy1[2]=h1+200(h4), vy2[0]=h1+1(h2),
	  vy2[1]=h1+201(h3), vy2[2]=h1+200(h4)                            )
	A = wsurfv; A + [FRh1];             C = [A]; C & 255; C < WFRSHF;
	A = 0; A - C; [FRvy10] = A;
	A = [FRh1]; A + 1; A + wsurfv;      C = [A]; C & 255; C < WFRSHF;
	A = 0; A - C; [FRvy11] = A;
	A = [FRh1]; A + 200; A + wsurfv;    C = [A]; C & 255; C < WFRSHF;
	A = 0; A - C; [FRvy12] = A;
	( vy20 = vy11 (h2), vy22 = vy12 (h4) )
	[FRvy20] = [FRvy11]; [FRvy22] = [FRvy12];
	A = [FRh1]; A + 201; A + wsurfv;    C = [A]; C & 255; C < WFRSHF;
	A = 0; A - C; [FRvy21] = A;

	( c1 shading: sh_delta slope vs diffuse fast_random )
	A = [FRbranch]; ? A = 0 -> FR c1 diffuse;
	( sh_delta branch: c1 = surf[h1] - surf[h1+sh_delta] )
	A = wsurfv; A + [FRh1]; C = [A]; C & 255; [FRtmp] = C;
	A = [FRh1]; A + [FRshd]; A + wsurfv; C = [A]; C & 255;
	A = [FRtmp]; A - C; [FRc1] = A;
	-> FR c1 merge;
    "FR c1 diffuse"
	( fast_srand(h1 + seed) ; c1 = 8 + fast_random(7) )
	A = [FRh1]; A + [FRseed]; [FRfrs] = A;		( arg to fast_srand )
	=> FAST SRAND;
	[FRmask] = 7; => FAST RANDOM;			( FReax = fast_random(7) )
	A = 8; A + [FReax]; [FRc1] = A;
    "FR c1 merge"
	( if c1 < 0 c1 = 0 ; c1 += depth >> 1 ; if c1 > 32 c1 = 32.
	  All signed: lino's '-comparisons are UNSIGNED, so the sign tests are
	  done via arithmetic shift-right-31 (0 iff >=0). )
	A = [FRc1]; A > 31; ? A = 0 -> FR c1 pos; [FRc1] = 0;
    "FR c1 pos"
	A = [FRdepth]; A > 1; [FRtmp] = A;
	A = [FRc1]; A + [FRtmp]; [FRc1] = A;
	( if c1 > 32 c1 = 32  <=>  if (c1 - 33) >= 0  <=>  ((c1-33)>>31) == 0 )
	A = [FRc1]; A - 33; A > 31; ? A != 0 -> FR c1 cap; [FRc1] = 32;
    "FR c1 cap"
	end;

( ==================================================================== )
( P FORWARD - NOCTIS-0.CPP:1388-1392  walking physics                    )
(   pos_x -= delta * opt_tsinbeta * opt_tcosalfa                        )
(   pos_z += delta * opt_tcosbeta * opt_tcosalfa                        )
(   The multiply chain runs at extended precision; the only narrowing    )
(   is the store to pos_x / pos_z (float).  delta and the angle tables   )
(   are the binary32 inputs (pinned).                                    )
( ==================================================================== )

"P FORWARD"
	( prod_x = delta * sinbeta * cosalfa ; pos_x -= prod_x.
	  FLoadF32 always loads to FA, so each operand is staged to FB by loading
	  it to FA then copying FA->FB, saving the running product to FT first. )
	[FS0] = [PFsbn];    => FLoadF32; => HP fb fa;	( FB = sinbeta )
	[FS0] = [PFdelta];  => FLoadF32;			( FA = delta )
	=> FMul;						( FA = delta*sinbeta )
	[FT0]=[FA0]; [FT1]=[FA1];				( FT = delta*sinbeta )
	[FS0] = [PFcalfa];  => FLoadF32; => HP fb fa;	( FB = cosalfa )
	[FA0]=[FT0]; [FA1]=[FT1];				( FA = delta*sinbeta )
	=> FMul;						( FA = prod_x )
	[FT0]=[FA0]; [FT1]=[FA1];				( FT = prod_x )
	[FS0] = [PFpx];     => FLoadF32;			( FA = pos_x )
	[FB0]=[FT0]; [FB1]=[FT1];				( FB = prod_x )
	=> FSub;						( FA = pos_x - prod_x )
	=> F32Narrow; => FStoreF32; [PFnpx] = [FS0];

	( prod_z = delta * cosbeta * cosalfa ; pos_z += prod_z )
	[FS0] = [PFcbn];    => FLoadF32; => HP fb fa;	( FB = cosbeta )
	[FS0] = [PFdelta];  => FLoadF32;			( FA = delta )
	=> FMul;
	[FT0]=[FA0]; [FT1]=[FA1];
	[FS0] = [PFcalfa];  => FLoadF32; => HP fb fa;	( FB = cosalfa )
	[FA0]=[FT0]; [FA1]=[FT1];
	=> FMul;						( FA = prod_z )
	[FT0]=[FA0]; [FT1]=[FA1];
	[FS0] = [PFpz];     => FLoadF32;			( FA = pos_z )
	[FB0]=[FT0]; [FB1]=[FT1];				( FB = prod_z )
	=> FAdd;						( FA = pos_z + prod_z )
	=> F32Narrow; => FStoreF32; [PFnpz] = [FS0];
	end;

( ==================================================================== )
( CAV - change_angle_of_view (NOCTIS-0.CPP:1285, TDPOLYGS.H:365).        )
(   opt_pcosbeta = cos(b*deg)*dpp ; opt_psinbeta = sin(b*deg)*dpp        )
(   opt_tcosbeta = cos(b*deg)        ; opt_tsinbeta = sin(b*deg)         )
(   (and the same for alfa)                                              )
(   sin/cos via fsin/fcos under FEnter; the float32 store (fstp dword)   )
(   is the only narrowing.  gamma == 0 so its table is constant.         )
( ==================================================================== )

"CAV"
	( ---- beta tables ---- )
	[FS0] = [CAVbeta]; => FLoadF32;
	[FB0] = CDEGLO; [FB1] = CDEGHI; => FMul;
	=> FSin;
	=> FStoreF32; [CAVtsinbeta] = [FS0];
	[FT0]=[FA0]; [FT1]=[FA1];
	[FS0] = [CAVdpp]; => FLoadF32; => HP fb fa;
	[FA0]=[FT0]; [FA1]=[FT1]; => FMul;
	=> FStoreF32; [CAVpsinbeta] = [FS0];

	[FS0] = [CAVbeta]; => FLoadF32;
	[FB0] = CDEGLO; [FB1] = CDEGHI; => FMul;
	=> FCos;
	=> FStoreF32; [CAVtcosbeta] = [FS0];
	[FT0]=[FA0]; [FT1]=[FA1];
	[FS0] = [CAVdpp]; => FLoadF32; => HP fb fa;
	[FA0]=[FT0]; [FA1]=[FT1]; => FMul;
	=> FStoreF32; [CAVpcosbeta] = [FS0];

	( ---- alfa tables ---- )
	[FS0] = [CAValfa]; => FLoadF32;
	[FB0] = CDEGLO; [FB1] = CDEGHI; => FMul;
	=> FSin;
	=> FStoreF32; [CAVtsinalfa] = [FS0];
	[FT0]=[FA0]; [FT1]=[FA1];
	[FS0] = [CAVdpp]; => FLoadF32; => HP fb fa;
	[FA0]=[FT0]; [FA1]=[FT1]; => FMul;
	=> FStoreF32; [CAVpsinalfa] = [FS0];

	[FS0] = [CAValfa]; => FLoadF32;
	[FB0] = CDEGLO; [FB1] = CDEGHI; => FMul;
	=> FCos;
	=> FStoreF32; [CAVtcosalfa] = [FS0];
	[FT0]=[FA0]; [FT1]=[FA1];
	[FS0] = [CAVdpp]; => FLoadF32; => HP fb fa;
	[FA0]=[FT0]; [FA1]=[FT1]; => FMul;
	=> FStoreF32; [CAVpcosalfa] = [FS0];
	end;

( ==================================================================== )
( FAST SRAND / FAST RANDOM - NOCTIS-0.CPP:1075-1107                      )
(   fast_srand: flat_rnd_seed = (s & 0xFFFF0000) | ((s & 0xFFFF) | 3)    )
(   fast_random(mask): p=s*s (u64); eax=lo, edx=hi; al fold; s += eax    )
( ==================================================================== )

"FAST SRAND"
	A = [FRfrs]; A & 65535; A | 3; [FRalo] = A;
	A = [FRfrs]; A > 16; A < 16; [FRahi] = A;
	A = [FRahi]; A | [FRalo]; [FRfrs] = A;
	end;

"FAST RANDOM"
	( p = s*s unsigned 64-bit via 16x16 partials )
	[FRa] = [FRfrs]; [FRb] = [FRfrs]; => FR MUL64;
	( eax = FRlo, edx = FRhi ; al = (eax + edx) & 255 ; eax = (eax&FFFFFF00)|al )
	A = [FRlo]; [FReax] = A;
	A = [FRlo]; A & 255; [FRtmp] = A;
	A = [FRhi]; A & 255; A + [FRtmp]; A & 255; [FRtmp] = A;
	A = [FReax]; A & 0FFFFFF00h; A | [FRtmp]; [FReax] = A;
	( flat_rnd_seed += eax )
	A = [FRfrs]; A + [FReax]; [FRfrs] = A;
	( return eax & mask )
	A = [FReax]; A & [FRmask]; [FReax] = A;
	end;

( ==================================================================== )
( FR MUL64 - [FRhi]:[FRlo] <- unsigned [FRa] * [FRb], 32x32 -> 64.        )
(   16x16 partial products; high halves taken via (s>>16)&0xFFFF so the  )
(   decomposition is UNSIGNED even when bit 31 is set.                   )
( ==================================================================== )

"FR MUL64"
	A = [FRa]; A > 16; A & 65535; [FRahi] = A;
	A = [FRa]; A & 65535;        [FRalo] = A;
	A = [FRb]; A > 16; A & 65535; [FRbhi] = A;
	A = [FRb]; A & 65535;        [FRblo] = A;
	A = [FRalo]; A '* [FRblo]; [FRll] = A;
	A = [FRalo]; A '* [FRbhi]; [FRlh] = A;
	A = [FRahi]; A '* [FRblo]; [FRhl] = A;
	A = [FRahi]; A '* [FRbhi]; [FRhh] = A;
	( mid = (ll>>16) + (lh&FFFF) + (hl&FFFF) )
	A = [FRll]; A > 16;           [FRmid] = A;
	A = [FRlh]; A & 65535; A + [FRmid]; [FRmid] = A;
	A = [FRhl]; A & 65535; A + [FRmid]; [FRmid] = A;
	( lo = (ll & FFFF) | (mid << 16) )
	A = [FRll]; A & 65535; [FRlo] = A;
	A = [FRmid]; A < 16; A | [FRlo]; [FRlo] = A;
	( hi = hh + (lh>>16) + (hl>>16) + (mid>>16) )
	A = [FRhh]; A + 0; [FRhi] = A;
	A = [FRlh]; A > 16; A + [FRhi]; [FRhi] = A;
	A = [FRhl]; A > 16; A + [FRhi]; [FRhi] = A;
	A = [FRmid]; A > 16; A + [FRhi]; [FRhi] = A;
	end;

( ==================================================================== )
( HPOINT - NOCTIS-1.CPP:63-93                                            )
(   py is float; the two stores to py narrow to float32.  Everything     )
(   else runs at extended precision.  h1..h4/icx/icz are exact integer  )
(   doubles.                                                             )
( ==================================================================== )

"HPOINT"
	( h1..h4 integer values = -(s<<11) ; surf byte is at wsurfv[cpos+...]
	  but we already know s1..s4 from the caller's placement, so read
	  them back the same way hpoint reads p_surfacemap.  Faithful: this
	  exercises the cpos indexing. )
	A = wsurfv; A + [HPcpos]; C = [A]; C & 255; C < 11;
	A = 0; A - C; [HPh1i] = A;
	A = [HPcpos]; A + 1; A + wsurfv; C = [A]; C & 255; C < 11;
	A = 0; A - C; [HPh2i] = A;
	A = [HPcpos]; A + 201; A + wsurfv; C = [A]; C & 255; C < 11;
	A = 0; A - C; [HPh3i] = A;
	A = [HPcpos]; A + 200; A + wsurfv; C = [A]; C & 255; C < 11;
	A = 0; A - C; [HPh4i] = A;

	( icx = px & 16383 ; icz = pz & 16383 )
	A = [HPpx]; A & 16383; [HPicx] = A;
	A = [HPpz]; A & 16383; [HPicz] = A;

	( branch: icx + icz < 16384  [integer, exact] )
	A = [HPicx]; A + [HPicz]; ? A '< W16384 -> HP upper;
	=> HPOINT lower;
	-> HP done;
    "HP upper"
	=> HPOINT upper;
	-> HP done;
    "HP done"
	end;

( --- upper triangle: py = h1 + (h2-h1)*(icx*qid); py += (h4-h1)*(icz*qid) --- )

"HPOINT upper"
	( FT <- icx * qid )
	[FI] = [HPicx]; => IntToF;
	[FB0] = WQIDLO; [FB1] = WQIDHI; => FMul;
	[FT0] = [FA0]; [FT1] = [FA1];

	( FA <- h2 - h1 )
	[FI] = [HPh1i]; => IntToF; => HP fb fa;   ( FB <- h1 )
	[FI] = [HPh2i]; => IntToF;               ( FA <- h2 )
	=> FSub;                                  ( FA <- h2 - h1 )

	( FA <- (h2-h1) * (icx*qid) )
	[FB0] = [FT0]; [FB1] = [FT1]; => FMul;

	( FA <- h1 + that )
	[FT0] = [FA0]; [FT1] = [FA1];             ( FT <- product )
	[FI] = [HPh1i]; => IntToF; => HP fb fa;   ( FB <- h1 )
	[FA0] = [FT0]; [FA1] = [FT1]; => FAdd;    ( FA <- product + h1 )

	( py = (float) FA ; store its bits )
	=> F32Narrow; => FStoreF32; [HPpy] = [FS0];

	( FT <- icz * qid )
	[FI] = [HPicz]; => IntToF;
	[FB0] = WQIDLO; [FB1] = WQIDHI; => FMul;
	[FT0] = [FA0]; [FT1] = [FA1];

	( FA <- h4 - h1 )
	[FI] = [HPh1i]; => IntToF; => HP fb fa;
	[FI] = [HPh4i]; => IntToF; => FSub;

	( FA <- (h4-h1) * (icz*qid) )
	[FB0] = [FT0]; [FB1] = [FT1]; => FMul;
	[FT0] = [FA0]; [FT1] = [FA1];             ( FT <- product )

	( FA <- (float)py + product ; py reloaded from its narrowed bits )
	[FS0] = [HPpy]; => FLoadF32; => HP fb fa; ( FB <- (double)py )
	[FA0] = [FT0]; [FA1] = [FT1]; => FAdd;

	=> F32Narrow; => FStoreF32; [HPpy] = [FS0];
	end;

( --- lower triangle: py = h3 + (h4-h3)*((16384-icx)*qid);
                      py += (h2-h3)*((16384-icz)*qid)               --- )

"HPOINT lower"
	( FT <- (16384 - icx) * qid )
	A = W16384; A - [HPicx]; [FI] = A; => IntToF;
	[FB0] = WQIDLO; [FB1] = WQIDHI; => FMul;
	[FT0] = [FA0]; [FT1] = [FA1];

	( FA <- h4 - h3 )
	[FI] = [HPh3i]; => IntToF; => HP fb fa;
	[FI] = [HPh4i]; => IntToF; => FSub;
	[FB0] = [FT0]; [FB1] = [FT1]; => FMul;
	[FT0] = [FA0]; [FT1] = [FA1];

	( FA <- h3 + product )
	[FI] = [HPh3i]; => IntToF; => HP fb fa;
	[FA0] = [FT0]; [FA1] = [FT1]; => FAdd;

	=> F32Narrow; => FStoreF32; [HPpy] = [FS0];

	( FT <- (16384 - icz) * qid )
	A = W16384; A - [HPicz]; [FI] = A; => IntToF;
	[FB0] = WQIDLO; [FB1] = WQIDHI; => FMul;
	[FT0] = [FA0]; [FT1] = [FA1];

	( FA <- h2 - h3 )
	[FI] = [HPh3i]; => IntToF; => HP fb fa;
	[FI] = [HPh2i]; => IntToF; => FSub;
	[FB0] = [FT0]; [FB1] = [FT1]; => FMul;
	[FT0] = [FA0]; [FT1] = [FA1];

	( FA <- (float)py + product )
	[FS0] = [HPpy]; => FLoadF32; => HP fb fa;
	[FA0] = [FT0]; [FA1] = [FT1]; => FAdd;

	=> F32Narrow; => FStoreF32; [HPpy] = [FS0];
	end;

"W getch"
	A = [Wcpos];
	? A '>= [Wcbytes] -> W getch eof;
	C = A; C > 2; D = wcorpv; D + C;
	C = [D];
	A = [Wcpos]; A & 3; A '* 8;
	C > A; C & 255;
	[Wch] = C;
	end;
    "W getch eof"
	A = 0; A - 1; [Wch] = A;
	end;

"W tokenise"
	[Wcpos] = 0; [Wctokn] = 0;
    "W tk scan"
	=> W getch;
	A = [Wch];
	? A < 0 -> W tk done;
	? A = 35 -> W tk comment;		( '#' )
	? A = 45 -> W tk number;		( '-' )
	? A '< 48 -> W tk skip;
	? A '> 57 -> W tk skip;
	-> W tk number;
    "W tk skip"
	[Wcpos]+;
	-> W tk scan;
    "W tk comment"
	[Wcpos]+;
	=> W getch;
	A = [Wch];
	? A < 0 -> W tk done;
	? A != 10 -> W tk comment;
	[Wcpos]+;
	-> W tk scan;
    "W tk number"
	[Wcneg] = 0; [Wcacc] = 0;
	A = [Wch];
	? A != 45 -> W tk digits;
	[Wcneg] = 1;
	[Wcpos]+;
    "W tk digits"
	=> W getch;
	A = [Wch];
	? A '< 48 -> W tk emit;
	? A '> 57 -> W tk emit;
	C = [Wcacc]; C '* 10; A - 48; C + A; [Wcacc] = C;
	[Wcpos]+;
	-> W tk digits;
    "W tk emit"
	A = [Wcneg]; ? A = 0 -> W tk pos;
	C = [Wcacc]; C+-; [Wcacc] = C;
    "W tk pos"
	A = [Wctokn];
	? A '>= WTOKMAX -> W tk done;
	C = wtokv; C + A; [C] = [Wcacc];
	[Wctokn]+;
	-> W tk scan;
    "W tk done"
	end;

( W next - C <- wtokv[[Wcti]], [Wcti]+.  Past end yields 0. )

"W next"

	C = wtokv; C + A; C = [C];
	[Wcti]+;

    "W next eof"

	end;
