( the pvfile arena, LINOBUF alias 9                                     )
( ==================================================================== )

      ( pvfile_x[h] = (float far *)(pvfile + datatop) where datatop has
	advanced by 1*npolygs - a float array at a byte offset that is
	not a multiple of 4 whenever npolygs mod 4 is not 0.  Unit
	addressing cannot express that, so the arena is RE-LAID OUT with
	every sub-array unit-aligned.  Observationally equivalent only
	if nothing outside loadpv/unloadpv reads pvfile as raw bytes.

	What is checked here is the thing the re-layout could break: that
	16 handles worth of unit-aligned sub-arrays still fit in the
	20480 units the original arena had.  Per polygon the original
	stores 50 bytes: 4 float triples (48) plus 1 class byte plus 1
	texture byte.  Unit-aligned that is 50 units per polygon, so the
	arena holds the same 409 polygons it always did, and the ONLY
	thing that changed is that a sub-array boundary is now a unit
	boundary.  Aligning changes no size, which is the point. )

"FB pvfile check"
	A = ZPVF; A '/ 50;		( polygons the arena holds )
	B = A; B '* 50;
	C = ZPVF; C - B;		( slack, units )
	? C '< 50 -> FB pv fits;
	[FBpvok] = 0;
	end;
    "FB pv fits"
	[FBpvok] = A;
	end;

      ( [Counts] wraps about every 477 s, so the difference between a
	correct predicate and an unsigned timestamp compare is invisible
	for eight minutes and then collapses a run of ticks to nothing.
	Waiting eight minutes is not a test.  Driving [TKnow] and
	[TKdeadline] directly is: the anchors sweep all 2^32 with an odd
	stride, seven offsets each, and the expected answer is known by
	construction. )

"FB wrap battery"
	[FBwrapf] = 0; [FBwrapn] = 0;

	( the six anchors that matter, first and by name )
	[FBanch] = 0;			=> FB wrap anchor;
	[FBanch] = 1;			=> FB wrap anchor;
	[FBanch] = 2147483647;		=> FB wrap anchor;
	[FBanch] = 2147483648;		=> FB wrap anchor;
	[FBanch] = 2147483649;		=> FB wrap anchor;
	[FBanch] = 4294967295;		=> FB wrap anchor;

	[FBanch] = 0; [FBi] = 0;
    "FB wa loop"
	=> FB wrap anchor;
	[FBanch] + ANCHSTR;
	[FBi]+;
	A = [FBi]; ? A '< NANCH -> FB wa loop;
	end;

"FB wrap anchor"
	[FBoff] = 0; [FBoff] - 3;
    "FB wo loop"
	[TKnow] = [FBanch];
	A = [FBanch]; A + [FBoff]; [TKdeadline] = A;
	=> TK is late;
	[FBwrapn]+;
	( expected: late exactly when the offset is zero or negative )
	? [FBoff] > 0 -> FB wo early;
	A = [TKlate]; A - 1; ? A = 0 -> FB wo next;
	[FBwrapf]+;
	-> FB wo next;
    "FB wo early"
	A = [TKlate]; ? A = 0 -> FB wo next;
	[FBwrapf]+;
    "FB wo next"
	[FBoff]+;
	A = [FBoff]; ? A <= 3 -> FB wo loop;
	end;

( the soak                                                             )
( ==================================================================== )

"FB soak"
	=> TK start;
	[FBesc] = 0; [FBkey] = 0;
	[FBi] = 0;
    "FB sk loop"
	=> FB frame;

	( a deterministic hitch, so skip-to-grid is TESTED not asserted )
	A = HITCHMOD; ? A = 0 -> FB sk nohitch;
	A = [FBi]; A+; A % HITCHMOD;
	? A != 0 -> FB sk nohitch;
	[TKtmp] = HITCHMS; => TK burn;
    "FB sk nohitch"

	=> TK step;

	( input.  There is no window-close event; ESC is the only exit.
	  Tested against the OFF tag because that is the documented idiom
	  (14_LUCK_table_lookup.txt:99-112 does the same).  MEASURED, so
	  that the next reader does not have to wonder: the raw value of
	  [KEY ESCAPE] with the key up is 0 on this runtime, and it is
	  dumped as luck_raw rather than assumed. )
	A = [KEY ESCAPE]; [FBraw] = A;
	? A = OFF -> FB sk noesc;
	[FBesc] = 1;
    "FB sk noesc"
	[Console Command] = GET CONSOLE INPUT; isocall;
	A = [Console Input];
	? A = 0 -> FB sk nokey;
	[FBkey] = A;
    "FB sk nokey"
	? [FBesc] != 0 -> FB sk done;
	[FBi]+;
	A = [FBi]; ? A '< NTICK -> FB sk loop;
    "FB sk done"
	end;

( the dump                                                             )
( ==================================================================== )

"FB dump"
	=> DMP reset;
	[DCpms]  = [TKcpms];
	[DTicks] = [TKticks];

	( 0 : the layout, straight out of the table the code uses )
	[DKind] = KLAY; [DW] = 0; [DH] = 0;
	A = rtab; [DBptr] = A; [DBcnt] = 36; => DMP rec;

	( 1 : the CLEAN canary check )
	[DKind] = KCAN;
	A = ctclean; [DBptr] = A; [DBcnt] = 18; => DMP rec;

	( 2 : self-checks )
	=> FB build self;
	[DKind] = KSELF;
	A = self; [DBptr] = A; [DBcnt] = NSELF; => DMP rec;

	( 3 : pal6, the master )
	[DKind] = KPAL6;
	A = pal6; [DBptr] = A; [DBcnt] = 768; => DMP rec;

	( 4 : curpal6, what was uploaded - the DAC )
	[DKind] = KPAL6;
	A = curpal6; [DBptr] = A; [DBcnt] = 768; => DMP rec;

	( 5 : the LUT )
	[DKind] = KLUT;
	A = pal; [DBptr] = A; [DBcnt] = 256; => DMP rec;

	( 6 : adapted, the hidden page - what snapshot() writes )
	[DKind] = KPAGE; [DW] = 320; [DH] = 200;
	A = nw; A + RADPT; [DBptr] = A; [DBcnt] = NPIX; => DMP rec;

	( 7 : adaptor, the visible page - what DOSBox captures )
	[DKind] = KPAGE;
	A = nw; A + RADPR; [DBptr] = A; [DBcnt] = NPIX; => DMP rec;

	( 8 : raw frame costs )
	[DKind] = KFRM; [DW] = 0; [DH] = 0;
	A = fsamp; [DBptr] = A; C = NFRAME; C + C; [DBcnt] = C; => DMP rec;

	( 9 : the RAW tick log )
	[DKind] = KTICK;
	A = tlog; [DBptr] = A; C = [TKlogn]; C * 3; [DBcnt] = C; => DMP rec;

	=> DMP write;
	end;

	C = [D]; [A] = C; A+; D+;
	B ^ FB bs srv;

	C = [D]; [A] = C; A+; D+;
	B ^ FB bs loop;
	end;
