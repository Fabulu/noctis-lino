      ( *** sucase - surface()'s prologue, its ten-way switch, the
	    renormalisation, the merge, the terminator and the retouches ***

	NOCTIS-0.CPP:4766-5152.  Everything below the switch runs for
	every type, and three of the four post-switch stages are
	type-conditional in ways that are easy to get subtly wrong, so
	they are written out in source order rather than regrouped.

	THE PROLOGUE'S TWO CROSS-SEEDING BRIDGES.  The whole function is
	seeded twice from the same double and the two seedings are NOT the
	same arithmetic:

	    fast_srand (seedval + 4112);   :4784  <- the +4112 is added in
	                                             DOUBLE, before the
	                                             truncation
	    fast_srand (seedval * 10);     :4811
	    seed = fast_random (0xFFFF);   :4812  <- FAST feeds BRTL
	    srand (seed);                  :4817
	    ...
	    srand (seed);                  :4844  <- the same value again

	Three fast draws happen between them, for the rotation period, and
	they are the reason a wrong seedval shows up as a wrong rtperiod
	long before it shows up as a wrong pixel.

	THE SECOND srand IS UNGRADED, DELIBERATELY.  Nothing between the
	two calls draws from the brtl stream and Borland's srand is
	idempotent, so deleting the second call cannot change any output.
	It is implemented because the source has it.  No check in this
	wave claims to catch its removal, and the sabotage that deletes it
	is expected to be caught by NOTHING - that measurement is the
	evidence, not a pass.

	plwp IS AN INPUT HERE, NOT A RESULT.  :4798 computes it as
	89 - cplx_planet_viewpoint(id) + rotation, and
	cplx_planet_viewpoint is Wave 8.  The corpus supplies plwp
	directly (recon C recovers it from the capture's band as
	term_start - 35 mod 360).  So the terminator's SHAPE is predicted
	and graded; its POSITION is supplied, and the exit report says so.

	THE SWITCH'S TWO SECRET SHARERS.  Cases 2, 3, 5 and 6 read the
	global "secs", so their maps are functions of WHEN the frame ran.
	Cases 2 and 3 additionally consume the brtl stream from inside the
	switch - switch(random(2)) at :4889 chooses which fast draws
	happen next, which is the one coupling that runs FAST-from-BRTL
	rather than the other way round.

	COUNTING RULE, applied everywhere below: a ranged_fast_random
	ALWAYS draws, and the operands of + and - are evaluated LEFT to
	RIGHT, so in

	    g  = 26 + ranged_fast_random(3) - ranged_fast_random(5);
	    gr = ranged_fast_random (cr/2) + cr/2 + 2;

	the rfr(3) precedes the rfr(5), and in the second line the single
	draw precedes both constants.  Function ARGUMENTS go the other
	way - right to left - which matters at exactly two sites, the
	contrast and randoface calls in case 5.                          )

      ( SU map hash / SU ovl hash - FNV-1a over the live buffers.  These
	are DERIVED from what the programme did; nothing writes an
	expected value on both sides. )

"SU map hash"
	A = nw; A + [SUpbase];
	D = FNVOFF;
	B = NPIXM;
    "SU mh u"
	C = [A]; C & 0FFh;
	D # C; D '* FNVPR;

	B ^ SU mh u;

	end;

	B = NOVLM;

	B ^ SU oh u;
	[SUoh] = D;
	end;

	buffers, which costs 97,200 byte reads and is therefore done at
	phase boundaries but not inside the hot loops. )

	C = 5; => SU rfr; A = C; A + 5; [SUr] = A;
	=> SU nssmooth;

	C = 50; => SU rfr; A = C; A + 10; [SUr] = A;
	C = 20; => SU rfr; A = C; A + 5;  [SUg] = A;
	C = 2;  => SU rfr; A = C; A + 1;  [SUb] = A;
	[SUc] = 0;
    "SU c7 fr"
	A = [SUc]; ? A >= [SUr] -> SU c7 frend;
	C = 360; => SU rfr; [SUcxi] = C;
	C = 180; => SU rfr; [SUcyi] = C;
	C = 300; => SU rfr; [SUgr] = C;
	=> SU fracture;
	=> SU led it;
	[SUc]+;
	-> SU c7 fr;
    "SU c7 frend"

	C = 2; => SU rfr; A = C;
	? A = 0 -> SU c7 nols;
	=> SU lssmooth;
    "SU c7 nols"
	C = 10; => SU rfr; A = C; A + 1; [SPt] = A;
	[SPu] = 1;
	=> SU randoface;
	C = 2; => SU rfr; A = C;
	? A = 0 -> SU c7 noneg;
	=> SU negate;
    "SU c7 noneg"
	end;

( ==================================================================== )
( case 9 - substellar object: flat, then smoothed six times             )
( ==================================================================== )

      ( No draws at all inside the switch.  pclear fills the SURFACE
	buffer - recon C's capture settles that against niv-lr, whose
	type 9 writes the offscreen video page instead: the DOS binary's
	p_background for a type 9 comes back fully populated, min 15 max
	62, and its objectschart holds exactly 32,400 non-zero bytes. )

"SU case 9"
	[SUval] = 1Fh;
	=> SU pclear;
	A = nw; A + [SUobase];
	B = NOVLM;
    "SU c9 u"
	[A] = 1Fh; A+;
	B ^ SU c9 u;
	end;

( ==================================================================== )
( the tail - everything after the switch                                )
( ==================================================================== )

"SU tail"
	[SUledh] = 1; [SUph] = PHSWEND; [SUphtag] = [SUr];
	=> SU led;

	( renormalisation to 00..1F, felisian and martian only )
	A = [SUtype];
	? A = 3 -> SU tl half;
	? A = 5 -> SU tl half;
	-> SU tl nohalf;
    "SU tl half"
	A = nw; A + [SUpbase];
	B = NPIXM;
    "SU tl hu"
	C = [A]; C & 0FFh; C > 1;
	[A] = C; A+;
	B ^ SU tl hu;
    "SU tl nohalf"

	( felisian final retouch )
	A = [SUtype];
	? A != 3 -> SU tl no3;
	C = 2; => SU rfr; A = C;
	? A = 0 -> SU tl 3ss;
	=> SU lssmooth;
	-> SU tl no3;
    "SU tl 3ss"
	=> SU ssmooth;
    "SU tl no3"

	[SUledh] = 1; [SUph] = PHNORM; [SUphtag] = 0;
	=> SU led;

	( the merge: two surface bytes per overlay byte, each clamped )
	[SCi] = 0; [SCj] = 0;
    "SU tl mg"
	[SUidx] = [SCi];
	=> SU odget;
	[SCt] = [SUval];

	A = nw; A + [SUpbase]; A + [SCj];
	C = [A]; C & 0FFh; C + [SCt]; C & 0FFh;
	? C '<= 3Eh -> SU tl mg1;
	C = 3Eh;
    "SU tl mg1"
	[A] = C;
	C = [A plus 1]; C & 0FFh; C + [SCt]; C & 0FFh;
	? C '<= 3Eh -> SU tl mg2;
	C = 3Eh;
    "SU tl mg2"
	[A plus 1] = C;

	A = [SCj]; A + 2; [SCj] = A;
	[SCi]+;
	A = [SCi]; ? A '< NOVLM -> SU tl mg;

	[SUledh] = 1; [SUph] = PHMERGE; [SUphtag] = 0;
	=> SU led;

	( venusian pre-terminator retouch - a BRTL draw, the only one
	  outside the switch apart from the palette's eighteen )
	[SUknot1] = 0;
	A = [SUtype];
	? A != 2 -> SU tl no2a;
	C = 3; => SU rnd; A = C;
	? A != 0 -> SU tl no2a;
	=> SU psmooth grays;
	[SUknot1] = 1;
    "SU tl no2a"

	( the day/night terminator )
	A = [SUplwp]; A + 35; [SUts] = A;
	A = [SUts]; ? A < 360 -> SU tl ts;
	A - 360; [SUts] = A;
    "SU tl ts"
	A = [SUts]; A + 130; [SUte] = A;
	A = [SUte]; ? A < 360 -> SU tl te;
	A - 360; [SUte] = A;
    "SU tl te"

	( "add di, plwp / add di, 35" is NOT reduced modulo 360, so the
	  strip's own start is plwp+35 and not term_start.  At plwp = 359
	  it begins 394 bytes in - one row and a bit - and the top-left
	  corner of the band lies on panorama row 1.  130 + 230 = 360, so
	  the strip is vertical and not diagonal, and cx = 179 leaves row
	  179 alone. )
	A = 4; A + [SUplwp]; A + 35; A & 0FFFFh; [SUoff] = A;
	[SCi] = 0;
    "SU tl row"
	[SCj] = 0;
    "SU tl col"
	=> SU get;
	A = [SUval]; A > 2; [SUval] = A;
	=> SU put;
	A = [SUoff]; A + 1; A & 0FFFFh; [SUoff] = A;
	[SCj]+;
	A = [SCj]; ? A '< 130 -> SU tl col;
	A = [SUoff]; A + 230; A & 0FFFFh; [SUoff] = A;
	[SCi]+;
	A = [SCi]; ? A '< 179 -> SU tl row;

	[SUledh] = 1; [SUph] = PHTERM; [SUphtag] = [SUts];
	=> SU led;

	( post-terminator retouches )
	A = [SUtype];
	? A != 2 -> SU tl no2b;
	A = [SUknot1]; ? A = 0 -> SU tl 2many;
	=> SU ssmooth;
	-> SU tl no2b;
    "SU tl 2many"
	C = 5; => SU rfr; A = C; A + 3; [SUr] = A;
	=> SU nssmooth;
    "SU tl no2b"

	A = [SUtype];
	? A != 6 -> SU tl no6;
	[SUc] = 0;
    "SU tl 6u"
	A = [SUc]; ? A >= 3 -> SU tl no6;
	C = 2; => SU rfr; A = C;
	? A = 0 -> SU tl 6n;
	=> SU ssmooth;
    "SU tl 6n"
	[SUc]+;
	-> SU tl 6u;
    "SU tl no6"

	A = [SUtype];
	? A != 9 -> SU tl no9;
	[SUc] = 0;
    "SU tl 9u"
	A = [SUc]; ? A >= 6 -> SU tl no9;
	=> SU ssmooth;
	[SUc]+;
	-> SU tl 9u;
    "SU tl no9"

	[SUledh] = 1; [SUph] = PHPOST; [SUphtag] = 0;
	=> SU led;
	end;

( surface() itself                                                      )
( ==================================================================== )

      ( "if (type == 10) return;" at :4776 is UNREACHABLE from the two
	live call sites - :5341 skips a type 10 with a goto before it can
	get here - and so is the "colorbase == 255" early return at
	:5153, because the only callers pass 128 or 192 and TGTPVIEW.CPP
	is not in NOCTIS.MAK.  Both are implemented and neither is
	claimed. )

"SU surface"
	A = [SUtype]; ? A = 10 -> SU sf out;
	=> SU prologue;
	=> SU switch;
	=> SU tail;
	A = [SUcbase]; ? A = 255 -> SU sf out;
	=> SU palette;
    "SU sf out"
	[SUledh] = 1; [SUph] = PHDONE; [SUphtag] = [SUtype];
	=> SU led;
	end;
