( ==================================================================== )
( NV nearstar - the full nearstar driver for the CLI coords.            )
( Sets nsclass, nsseedin, nsstarray, rgb/spin, then NsPrepare.          )

"NV nearstar"
	[nsx] = [NRX]; [nsy] = [NRY]; [nsz] = [NRZ];
	[FJ0] = [nsx]; [FJ1] = [nsy]; [FJ2] = [nsz];
	=> NsIdentityD;
	[nsid0] = [FA0]; [nsid1] = [FA1];
	[FJ0] = [nsx]; [FJ1] = [nsy]; [FJ2] = [nsz];
	=> NsIdentChop16;
	A = [FI]; => NsSrand;
	A = 12; => NsRandom;
	[nsclass] = A;

	( ray variance draw on the same stream )
	E = NVrayvar; E + [nsclass]; A = [E]; => NsRandom;
	[NVv] = A;

	( star rgb from CLASS_RGB[class*3..] )
	E = NVrgb; E + [nsclass]; E + [nsclass]; E + [nsclass];
	A = [E]; [NVr] = A;
	E + 1; A = [E]; [NVg] = A;
	E + 1; A = [E]; [NVb] = A;

	( spin: class 11 -> random(30)+1 ; 7 -> random(12)+1 ; 2 -> random(4)+1 )
	[NVspin] = 0;
	A = [nsclass]; ? A = 11 -> NV spin30;
	? A = 7 -> NV spin12;
	? A = 2 -> NV spin4;
	-> NV spin done;
    "NV spin30"
	A = 30; => NsRandom; A + 1; [NVspin] = A; -> NV spin done;
    "NV spin12"
	A = 12; => NsRandom; A + 1; [NVspin] = A; -> NV spin done;
    "NV spin4"
	A = 4; => NsRandom; A + 1; [NVspin] = A;
    "NV spin done"

	( the seed.  srand((x%10000*y)%10000*z%10000), engine y )
	[nsx] = [NRX]; [nsy] = [NRY]; [nsz] = [NRZ];
	=> NsSeedFromXYZ;
	[nsseedin] = A;

	( star ray = (variance + CLASS_RAY[class]) / 1000, stored as f32 )
	A = [NVv]; E = NVray; E + [nsclass]; B = [E]; A + B;
	[FI] = A; => IntToF;
	[FB0] = [FA0]; [FB1] = [FA1];
	[FI] = 1000; => IntToF;
	[FC0] = [FB0]; [FC1] = [FB1];
	[FB0] = [FA0]; [FB1] = [FA1];
	[FA0] = [FC0]; [FA1] = [FC1];
	=> FQuo;
	=> FStoreF32; [nsstarray] = [FS0];

	=> NsStarnop;
	=> NsResetAll;
	=> NsPrepare;
	end;

( ==================================================================== )
( NV dump surface - write surface.raw and palette.raw to [NVdump].      )
( The File WRITE emits one byte per 32-bit unit (4 bytes per unit), so   )
( callers that need the raw byte stream must keep every 4th byte.        )

"NV dump surface"
	A = [NVdump]; ? A = 0 -> NV dump out;

	( DIR + /surface.raw -> NVdumpbuf, write RPBG[0..64800] )
	[NVdi] = 0;
	A = [NVdump]; => NV dump cat;
	A = NVDUMPNAME; => NV dump cat;
	A = 0; => NV dump term;
	[File Name] = NVdumpbuf;
	[File Position] = 0;
	[File Command] = WRITE;
	A = nw; A + RPBG; [Block Pointer] = A;
	[Block Size] = 64800;
	[Block Size] * BYTESPERUNIT;
	isocall;

	( DIR + /palette.raw -> NVdumpbuf, write sutmppal[0..768] )
	[NVdi] = 0;
	A = [NVdump]; => NV dump cat;
	A = NVPALNAME; => NV dump cat;
	A = 0; => NV dump term;
	[File Name] = NVdumpbuf;
	[File Position] = 0;
	[File Command] = WRITE;
	A = sutmppal; [Block Pointer] = A;
	[Block Size] = 768;
	[Block Size] * BYTESPERUNIT;
	isocall;

	( clear the dump dir so a second body does not re-dump )
	[NVdump] = 0;
    "NV dump out"
	end;

( NV dump cat - append the null-terminated unit string at A to           )
( NVdumpbuf at [NVdi].  Leaves [NVdi] past the last unit copied.         )

"NV dump cat"
    "NV dump cat ch"
	B = [A]; ? B = 0 -> NV dump cat done;
	E = NVdumpbuf; E + [NVdi]; [E] = B;
	[NVdi] + 1; A+;
	-> NV dump cat ch;
    "NV dump cat done"
	end;

( NV dump term - store the terminator unit in A at NVdumpbuf[NVdi].     )

"NV dump term"
	E = NVdumpbuf; E + [NVdi]; [E] = A;
	end;
