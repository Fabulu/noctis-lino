
      ( *** newmul - exercising the new split-multiply instruction *** )

      ( This is the first program written against the extended toolchain. It
        uses instructions that do not exist in stock L.in.oleum:

            a *%  b     signed   32x32 -> 64,  a = low, b = high
            a *%' b     unsigned 32x32 -> 64,  a = low, b = high

        mirroring the existing /% and /%' split-divide pair.

        Operands: 12345678h and FFFFFFFFh. The second is -1 when signed, so
        the two forms must disagree in the high word - signed gives FFFFFFFF,
        unsigned gives 12345677. If they came back equal, the distinction
        would not be real.

        Requires the extended CPU pack; the stock pack has no such patterns
        and the stock compiler does not know the symbols.                   )

"directors"

	program name = { newmul };
	unit = 32;

"variables"

	x	= 0;
	y	= 0;

	r0 = 0; r1 = 0; r2 = 0; r3 = 0; r4 = 0; r5 = 0;

	result file name = { newmul.bin };

"workspace"

	results = 6;

"programme"

	( ---- signed, register x register ---- )

	A = 12345678h;
	B = FFFFFFFFh;
	A *% B;
	[r0] = A;		( low  )
	[r1] = B;		( high )

	( ---- unsigned, register x register ---- )

	A = 12345678h;
	B = FFFFFFFFh;
	A *%' B;
	[r2] = A;
	[r3] = B;

	( ---- signed, direct x direct: both operands in memory ---- )

	[x] = 12345678h;
	[y] = FFFFFFFFh;
	[x] *% [y];
	A = [x]; [r4] = A;
	A = [y]; [r5] = A;

	( ---- dump ---- )

	A = results;
	C = [r0]; [A] = C;
	A + 1; C = [r1]; [A] = C;
	A + 1; C = [r2]; [A] = C;
	A + 1; C = [r3]; [A] = C;
	A + 1; C = [r4]; [A] = C;
	A + 1; C = [r5]; [A] = C;

	[File Name]	= result file name;
	[File Position] = 0;
	[File Command]	= WRITE;
	[Block Pointer] = results;
	[Block Size]	= 6;
	[Block Size]	* BYTES PER UNIT;
	isocall;

	[File Name]	= result file name;
	[File Position] = 0;
	[File Command]	= SET SIZE;
	[File Size]	= 6;
	[File Size]	* BYTES PER UNIT;
	isocall;

	end;
