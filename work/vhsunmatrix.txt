      ( Cross-class indexed-page oracle for NIV+'s Stardrifter sun path.

	The first twelve inputs are real generated stars, one per class, centred
	at roughly thirty stellar radii.  The thirteenth is the exact opening
	Stardrifter star and 200-unit capture distance used by the production
	gallery.  The programme writes thirteen consecutive 64,000-byte pages for
	each native checkpoint so they can be compared directly with Borland
	captures.  Expected combined SHA-256:

	white  DBD099FE42D0ED8C3CF2B14827B8FAD0BE38830AA41E63C3B382C3EB0BDF4B71
	flare  109113B8DB1415FDEA137A5495ECD9B15AB307FBC57C392C812FC353AEF5FA89

	[vhsmclass] = 0;
    "VHSM class ready"
	A = vhsmrays; A + [vhsmindex]; [vhsmray] = [A];
	A = vhsmdlos; A + [vhsmindex]; [vhsmd0] = [A];
	A = vhsmdhis; A + [vhsmindex]; [vhsmd1] = [A];
	[PGFt] = 0; [PGFu] = 0; => PGF ldf64; [PGFi] = SFXX; => PGF sa;
	[PGFi] = SFYY; => PGF sa;
	[FA0] = [vhsmd0]; [FA1] = [vhsmd1]; [FI] = 1; => IntToF;
	[FB0] = [FA0]; [FB1] = [FA1]; [FA0] = [vhsmd0]; [FA1] = [vhsmd1]; => FSub;
	[PGFi] = SFZZ; => PGF sa;
	[PGFt] = [vhsmray]; => PGF ldf32; [FI] = 3; => IntToF;
	[FB0] = [FA0]; [FB1] = [FA1]; [PGFt] = [vhsmray]; => PGF ldf32; => FMul;
	=> F32Narrow; [WHmag] = [FS0]; [WHfgm] = 3E99999Ah;
	[WHshape] = 2; [WHsun] = 0; [WHdstreg] = RGADP; => SP white;
	[vhsmfile] = vhsmwname; => VHSM dump page;

	A = [vhsmclass]; ? A = 5 -> VHSM flare done;
	? A = 6 -> VHSM flare done; ? A = 10 -> VHSM flare done;
	[VHFray] = [vhsmray];
	[FS0] = [vhsmray]; => FLoadF32;
	[VHFspaceray0] = [FA0]; [VHFspaceray1] = [FA1];
	[VHFdist0] = [vhsmd0]; [VHFdist1] = [vhsmd1]; [VHFghost] = 1;
	=> VH space flare;
    "VHSM flare done"
	[vhsmfile] = vhsmfname; => VHSM dump page;
	=> VHT smooth grays;
	[vhsmfile] = vhsmsname; => VHSM dump page;
	=> VHT mask page;
	[vhsmfile] = vhsmoname; => VHSM dump page;
	A = [vhsmpos]; A + 64000; [vhsmpos] = A;
	[vhsmindex]+; A = [vhsmindex]; ? A < 13 -> VHSM fixture;

	[vhsmfile] = vhsmwname; => VHSM size file;
	[vhsmfile] = vhsmfname; => VHSM size file;
	[vhsmfile] = vhsmsname; => VHSM size file;
	[vhsmfile] = vhsmoname; => VHSM size file;
	=> FLeave;
	end;

	end;
