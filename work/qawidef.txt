
      ( *** galaxy - Noctis IV's star-position hash, in L.in.oleum *** )

      ( Every sector of space is 100000 units on a side and holds at most
        one star, whose position is a pure hash of the sector's integer
        coordinates. There is no star table: the Feltyrion galaxy IS this
        function. 78 billion stars out of about forty lines of arithmetic.

        Ported from noctis-iv-lr src/noctis-0.cpp sky(), which is itself a
        translation of the original DOS code's hand-encoded 386 opcodes.

        The load-bearing detail is that the multiply is SIGNED. Sector
        coordinates go negative either side of the galactic centre, and an
        unsigned product yields a different high word - which would build a
        perfectly plausible galaxy that simply is not this one. So the
        fragment below is IMUL (F7 EB), not MUL (F7 E3).

        Sweeps k = -3..+3 on each axis, 343 sectors, and writes 5 units per
        sector to galaxy.bin: temp_x, temp_y, temp_z, netpos, flags.
        Cutoffs are recorded as flag bits rather than skipping the sector,
        so the arithmetic is exercised everywhere.                        )

"directors"

	program name = { qawidef };
	unit = 32;

"constants"

	SECTORSIZE	= 100000;
	CUTOFF		= 50000;
	SPAN		= 21;		( k = -3..+3 )
	KOFFSET 	= 10;
	PERSECTOR	= 5;

"variables"

	ix	= 0;
	iy	= 0;
	iz	= 0;
	outIdx	= 0;

	sectx	= 0;
	secty	= 0;
	sectz	= 0;

	sumxz	= 0;
	tempx	= 0;
	tempy	= 0;
	tempz	= 0;
	accum	= 0;
	idk	= 0;
	netpos	= 0;
	flags	= 0;

	fmA	= 0;
	fmB	= 0;
	fmResult = 0;

	result file name = { qawidef.bin };

"workspace"

	results = SPAN multiplied SPAN multiplied SPAN multiplied PERSECTOR;

"programme"

	[outIdx] = 0;
	[ix] = 0;

    "loop x"
	[iy] = 0;

    "loop y"
	[iz] = 0;

    "loop z"

	( sect = (i - 3) * 100000, so sectors straddle the origin )

	A = [ix]; A - KOFFSET; A * SECTORSIZE;	[sectx] = A;
	A = [iy]; A - KOFFSET; A * SECTORSIZE;	[secty] = A;
	A = [iz]; A - KOFFSET; A * SECTORSIZE;	[sectz] = A;

	[flags] = 0;

	( sum_xz = sect_x + sect_z )

	A = [sectx]; A + [sectz];		[sumxz] = A;

	( temp_x = (sum_xz & 1FFFF) + sect_x )

	A = [sumxz]; A & 1FFFFh; A + [sectx];	[tempx] = A;
      ? [tempx] != CUTOFF -> gx no cutoff x;
	A = [flags]; A | 1;			[flags] = A;
    "gx no cutoff x"
	A = [tempx]; A - CUTOFF;		[tempx] = A;

	( accum = foldmul(temp_x, sum_xz) )

	A = [tempx]; [fmA] = A;
	A = [sumxz]; [fmB] = A;
	=> FoldMul;
	A = [fmResult];				[accum] = A;

	( idk = sum_xz + accum )

	A = [sumxz]; A + [accum];		[idk] = A;

	( temp_y = (accum & 1FFFF) + sect_y )

	A = [accum]; A & 1FFFFh; A + [secty];	[tempy] = A;
      ? [tempy] != CUTOFF -> gx no cutoff y;
	A = [flags]; A | 2;			[flags] = A;
    "gx no cutoff y"
	A = [tempy]; A - CUTOFF;		[tempy] = A;

	( accum = foldmul(temp_y, idk) )

	A = [tempy]; [fmA] = A;
	A = [idk];   [fmB] = A;
	=> FoldMul;
	A = [fmResult];				[accum] = A;

	( temp_z = (accum & 1FFFF) + sect_z )

	A = [accum]; A & 1FFFFh; A + [sectz];	[tempz] = A;
      ? [tempz] != CUTOFF -> gx no cutoff z;
	A = [flags]; A | 4;			[flags] = A;
    "gx no cutoff z"
	A = [tempz]; A - CUTOFF;		[tempz] = A;

	( netpos = temp_x + temp_y + temp_z )

	A = [tempx]; A + [tempy]; A + [tempz];	[netpos] = A;

	( store the five results for this sector )

	A = results; A + [outIdx];
	C = [tempx];  [A] = C;
	A + 1; C = [tempy];  [A] = C;
	A + 1; C = [tempz];  [A] = C;
	A + 1; C = [netpos]; [A] = C;
	A + 1; C = [flags];  [A] = C;
	A = [outIdx]; A + PERSECTOR;		[outIdx] = A;

	( advance the three nested loops )

	A = [iz]; A + 1; [iz] = A;
      ? [iz] < SPAN -> loop z;
	A = [iy]; A + 1; [iy] = A;
      ? [iy] < SPAN -> loop y;
	A = [ix]; A + 1; [ix] = A;
      ? [ix] < SPAN -> loop x;

	( dump everything to disk )

	[File Name]	= result file name;
	[File Position] = 0;
	[File Command]	= WRITE;
	[Block Pointer] = results;
	[Block Size]	= SPAN multiplied SPAN multiplied SPAN multiplied PERSECTOR;
	[Block Size]	* BYTES PER UNIT;
	isocall;

	[File Name]	= result file name;
	[File Position] = 0;
	[File Command]	= SET SIZE;
	[File Size]	= SPAN multiplied SPAN multiplied SPAN multiplied PERSECTOR;
	[File Size]	* BYTES PER UNIT;
	isocall;

	end;

"FoldMul"
    ( Signed 32x32 -> 64 multiply, high half folded into the low. )
    ( This is the original's imul followed by "edx += eax".        )
    ( In: [fmA], [fmB].  Out: [fmResult].  Destroys A, B, D.       )

	A = [fmA];
	B = [fmB];
	{
	    F7 EB	(imul ebx  ->  edx:eax = eax * ebx, SIGNED)
	}
	A + D;		( fold the high half into the low: edx += eax )
	[fmResult] = A;

	end;
