      ( *** vhspace - live source-derived Feltyrion galaxy stars ***

	A production library for the Stardrifter game.  Noctis does not keep
	a star catalogue: sky() hashes every 100000-unit sector into at most
	one star.  This is the original signed 32x32->64 folded hash and the
	normal nine-sector viewing cube.

	Inputs before VHS stars:
	  VHSdx0/1, VHSdy0/1, VHSdz0/1  live dzat binary64 halves
	  VHSalpha, VHSbeta              player's integer look angles
	  VHSsurface                     0 for sky(405Ch), 1 for sky(003Eh)

	The cached values are global integer star coordinates.  Projection
	first subtracts the live binary64 dzat value and only then narrows to
	float, matching sky()'s fild/sub/fst sequence.                         )

"constants"

	VHSNORMAL = 9;
	VHSAMPLIFIED = 14;
	VHSMAXCOUNT = 2744;
	VHSENTRY = 4;
	VHSSECTOR = 100000;
	VHSNORMALHALF = 450000;
	VHSAMPHALF = 700000;
	VHSCUTOFF = 50000;

"variables"

	VHSdx0 = 0; VHSdx1 = 0;
	VHSdy0 = 0; VHSdy1 = 0;
	VHSdz0 = 0; VHSdz1 = 0;
	VHSalpha = 0; VHSbeta = 0;
	VHSsurface = 0;
	VHSamp = 0; VHSvisible = VHSNORMAL; VHScount = 729; VHShalfspan = VHSNORMALHALF;

	VHSready = 0; VHSbuiltvisible = 0;
	VHSbasex = 0; VHSbasey = 0; VHSbasez = 0;
	VHSnewx = 0; VHSnewy = 0; VHSnewz = 0;
	VHSix = 0; VHSiy = 0; VHSiz = 0;
	VHSsectx = 0; VHSsecty = 0; VHSsectz = 0;
	VHSstar = 0; VHSstarptr = 0;

	VHSsumxz = 0; VHStempx = 0; VHStempy = 0; VHStempz = 0;
	VHSaccum = 0; VHSidk = 0;
	VHSfmA = 0; VHSfmB = 0; VHSfmResult = 0;

	VHSaxis0 = 0; VHSaxis1 = 0; VHSbaseout = 0; VHSfwbase = 0;
	VHSnum0 = 0; VHSnum1 = 0;
	VHSdepth = 0; VHScolour = 0; VHSscreeny = 0;
	VHSdrawready = 0; VHSdrawcount = 0; VHSdrawi = 0; VHSdrawptr = 0;
	VHSdrawalpha = 0; VHSdrawbeta = 0; VHSdrawsurface = 0;
	VHSdrawdx0 = 0; VHSdrawdx1 = 0; VHSdrawdy0 = 0; VHSdrawdy1 = 0;
	VHSdrawdz0 = 0; VHSdrawdz1 = 0;
	VHSfadebase = 0; VHSfadei = 0;

"workspace"

	vhsstarcache = 10976;		( 2744 * x,y,z,present )
	vhsscreencache = 5488;		( at most 2744 visible offset,color pairs )

"programme"

      ( ---------------------------------------------------------------- )
      ( Public entry: refresh sector cache when its origin changes, then  )
      ( project the exact stars through the live player view.              )
      ( ---------------------------------------------------------------- )

"VHS stars"
	[VHSvisible] = VHSNORMAL; [VHScount] = 729; [VHShalfspan] = VHSNORMALHALF;
	A = [VHSamp]; ? A = 0 -> VHS amplification ready;
	[VHSvisible] = VHSAMPLIFIED; [VHScount] = VHSMAXCOUNT; [VHShalfspan] = VHSAMPHALF;
    "VHS amplification ready"
	[VHSaxis0] = [VHSdx0]; [VHSaxis1] = [VHSdx1]; => VHS sector base;
	[VHSnewx] = [VHSbaseout];
	[VHSaxis0] = [VHSdy0]; [VHSaxis1] = [VHSdy1]; => VHS sector base;
	[VHSnewy] = [VHSbaseout];
	[VHSaxis0] = [VHSdz0]; [VHSaxis1] = [VHSdz1]; => VHS sector base;
	[VHSnewz] = [VHSbaseout];

	A = [VHSready]; ? A = 0 -> VHS refresh;
	A = [VHSvisible]; ? A != [VHSbuiltvisible] -> VHS refresh;
	A = [VHSnewx]; ? A != [VHSbasex] -> VHS refresh;
	A = [VHSnewy]; ? A != [VHSbasey] -> VHS refresh;
	A = [VHSnewz]; ? A != [VHSbasez] -> VHS refresh;
	-> VHS cache ready;
    "VHS refresh"
	[VHSbasex] = [VHSnewx]; [VHSbasey] = [VHSnewy]; [VHSbasez] = [VHSnewz];
	=> VHS build cache; [VHSready] = 1; [VHSbuiltvisible] = [VHSvisible]; [VHSdrawready] = 0;
    "VHS cache ready"
	( Cache the resulting visible candidates only while every projection input
	  is bit-identical.  The prior four-degree/eight-frame heuristic made the
	  galaxy jump during a smooth turn and delayed sub-sector parallax.  Exact
	  keys retain the static-view win without changing a moving source frame. )
	A = [VHSdrawready]; ? A = 0 -> VHS rebuild draw cache;
	A = [VHSalpha]; ? A != [VHSdrawalpha] -> VHS rebuild draw cache;
	A = [VHSbeta]; ? A != [VHSdrawbeta] -> VHS rebuild draw cache;
	A = [VHSsurface]; ? A != [VHSdrawsurface] -> VHS rebuild draw cache;
	A = [VHSdx0]; ? A != [VHSdrawdx0] -> VHS rebuild draw cache;
	A = [VHSdx1]; ? A != [VHSdrawdx1] -> VHS rebuild draw cache;
	A = [VHSdy0]; ? A != [VHSdrawdy0] -> VHS rebuild draw cache;
	A = [VHSdy1]; ? A != [VHSdrawdy1] -> VHS rebuild draw cache;
	A = [VHSdz0]; ? A != [VHSdrawdz0] -> VHS rebuild draw cache;
	A = [VHSdz1]; ? A != [VHSdrawdz1] -> VHS rebuild draw cache;
	=> VHS replay draw cache; end;

    "VHS rebuild draw cache"
	[VHSdrawalpha] = [VHSalpha]; [VHSdrawbeta] = [VHSbeta];
	[VHSdrawsurface] = [VHSsurface];
	[VHSdrawdx0] = [VHSdx0]; [VHSdrawdx1] = [VHSdx1];
	[VHSdrawdy0] = [VHSdy0]; [VHSdrawdy1] = [VHSdy1];
	[VHSdrawdz0] = [VHSdz0]; [VHSdrawdz1] = [VHSdz1];
	[VHSdrawcount] = 0;
	A = fw; [VHSfwbase] = A;

	[VHVcamxi] = 0; [VHVcamyi] = 0; [VHVcamzi] = 0;
	[VHValpha] = [VHSalpha]; [VHVbeta] = [VHSbeta]; => VH set view;
	=> VHS rebuild draw cache common;
	end;

"VHS rebuild draw cache common"
	-> VHS rebuild draw cache source;

"VHS rebuild draw cache source"
	[VHSstar] = 0;
    "VHS draw loop"
	A = [VHSstar]; A '* VHSENTRY; A + vhsstarcache; [VHSstarptr] = A;
	A = [A plus 3]; ? A = 0 -> VHS draw next;

	=> VHS relative xyz native;

	=> VHS rotate depth native;
	A = [VHSdepth]; ? A < 10000 -> VHS draw next;
	C = [VHSamp]; ? C != 0 -> VHS amplified depth; A > 13; -> VHS depth ready;
    "VHS amplified depth"
	A > 14;
    "VHS depth ready"
	? A > 63 -> VHS draw next;
	C = 63; C - A; [VHScolour] = C;

	=> VHS project native;
	A = [GCx]; ? A <= 10 -> VHS draw next; ? A >= 310 -> VHS draw next;
	C = [GCy]; C - 2; [VHSscreeny] = C;
	? C <= 10 -> VHS draw next; ? C >= 190 -> VHS draw next;
	C '* 320; C + A; C + 4; [SPoff] = C;
	A = [VHSdrawcount]; A '* 2; A + vhsscreencache; [VHSdrawptr] = A;
	[A plus 0] = [SPoff]; [A plus 1] = [VHScolour]; [VHSdrawcount]+;
	( adapted is based at DOS segment offset four.  The orbital sky(405Ch)
	  accepts the 64..92 space band, except colour 68.  The landed sky(003Eh)
	  runs before mask_pixels and accepts the unmasked 0..62 sky band. )
	[SPreg] = RGADP; => SP get;
	C = [SPval]; A = [VHSsurface]; ? A != 0 -> VHS draw surface gate;
	? C = 68 -> VHS draw next; ? C < 64 -> VHS draw next; ? C > 92 -> VHS draw next;
	-> VHS draw gate ready;
    "VHS draw surface gate"
	? C > 62 -> VHS draw next;
    "VHS draw gate ready"
	D = C; D & 192; C & 63; C + [VHScolour]; ? C <= 63 -> VHS draw colour ready; C = 63;
    "VHS draw colour ready"
	C | D; [SPval] = C; [SPreg] = RGADP; => SP put;
    "VHS draw next"
	[VHSstar]+; A = [VHSstar]; ? A < [VHScount] -> VHS draw loop;
	[VHSdrawready] = 1;
	end;

"VHS replay draw cache"
	[VHSdrawi] = 0; A = [VHSdrawcount]; ? A = 0 -> VHS replay done;
    "VHS replay loop"
	A = [VHSdrawi]; A '* 2; A + vhsscreencache;
	[SPoff] = [A plus 0];
	[SPreg] = RGADP; => SP get; C = [SPval];
	A = [VHSsurface]; ? A != 0 -> VHS replay surface gate;
	? C = 68 -> VHS replay next; ? C < 64 -> VHS replay next; ? C > 92 -> VHS replay next;
	-> VHS replay gate ready;
    "VHS replay surface gate"
	? C > 62 -> VHS replay next;
    "VHS replay gate ready"
	D = C; D & 192; C & 63;
	A = [VHSdrawi]; A '* 2; A + vhsscreencache;
	C + [A plus 1]; ? C <= 63 -> VHS replay colour ready; C = 63;
    "VHS replay colour ready"
	C | D; [SPoff] = [A plus 0]; [SPval] = C; [SPreg] = RGADP; => SP put;
    "VHS replay next"
	[VHSdrawi]+; A = [VHSdrawi]; ? A < [VHSdrawcount] -> VHS replay loop;
    "VHS replay done"
	end;

"VHS fade"
	( pfade(adapted,180,8) with live QUADWORDS=14560 begins at the
	  segment-derived adapted offset 2876 and processes 57,920 pixels. )
	[SPreg] = RGADP; => SP base; C + 2876; [VHSfadebase] = C; [VHSfadei] = 0;
    "VHS fade pixel"
	A = nw; A + [VHSfadebase]; A + [VHSfadei]; C = [A]; C & 63;
	? C >= 8 -> VHS fade subtract; C = 0; -> VHS fade store;
    "VHS fade subtract"
	C - 8;
    "VHS fade store"
	[A] = C; [VHSfadei]+; A = [VHSfadei]; ? A < 57920 -> VHS fade pixel;
	end;

      ( sect = chop((dzat - 450000) / 100000) * 100000 )
"VHS sector base"
	[FI] = [VHShalfspan]; => IntToF;
	[FB0] = [FA0]; [FB1] = [FA1];
	[FA0] = [VHSaxis0]; [FA1] = [VHSaxis1]; => FSub;
	[VHSnum0] = [FA0]; [VHSnum1] = [FA1];
	[FA0] = [VHSnum0]; [FA1] = [VHSnum1];
	[FB0] = [MgskA]; [FB1] = [MgskB]; => FQuo; => FToIntChop;
	A = [FI]; A '* VHSSECTOR; [VHSbaseout] = A;
	end;

"VHS build cache"
	[VHSstar] = 0; [VHSix] = 0;
    "VHS build x"
	[VHSiy] = 0;
    "VHS build y"
	[VHSiz] = 0;
    "VHS build z"
	A = [VHSstar]; A '* VHSENTRY; A + vhsstarcache; [VHSstarptr] = A;
	[A plus 3] = 0;
	A = [VHSix]; A '* VHSSECTOR; A + [VHSbasex]; [VHSsectx] = A;
	A = [VHSiy]; A '* VHSSECTOR; A + [VHSbasey]; [VHSsecty] = A;
	A = [VHSiz]; A '* VHSSECTOR; A + [VHSbasez]; [VHSsectz] = A;

	A = [VHSsectx]; A + [VHSsectz]; [VHSsumxz] = A;
	A & 1FFFFh; A + [VHSsectx];
	? A = VHSCUTOFF -> VHS absent;
	A - VHSCUTOFF; [VHStempx] = A;
	[VHSfmA] = A; [VHSfmB] = [VHSsumxz]; => VHS foldmul;
	[VHSaccum] = [VHSfmResult];
	A = [VHSsumxz]; A + [VHSaccum]; [VHSidk] = A;

	A = [VHSaccum]; A & 1FFFFh; A + [VHSsecty];
	? A = VHSCUTOFF -> VHS absent;
	A - VHSCUTOFF; [VHStempy] = A;
	[VHSfmA] = A; [VHSfmB] = [VHSidk]; => VHS foldmul;
	[VHSaccum] = [VHSfmResult];

	A = [VHSaccum]; A & 1FFFFh; A + [VHSsectz];
	? A = VHSCUTOFF -> VHS absent;
	A - VHSCUTOFF; [VHStempz] = A;

	A = [VHSstarptr]; C = [VHStempx]; [A plus 0] = C;
	C = [VHStempy]; [A plus 1] = C;
	C = [VHStempz]; [A plus 2] = C;
	[A plus 3] = 1;
    "VHS absent"
	[VHSstar]+;
	[VHSiz]+; A = [VHSiz]; ? A < [VHSvisible] -> VHS build z;
	[VHSiy]+; A = [VHSiy]; ? A < [VHSvisible] -> VHS build y;
	[VHSix]+; A = [VHSix]; ? A < [VHSvisible] -> VHS build x;
	end;

"VHS relative xyz native"
	( Exact fused form of the three former helper chains.  Each coordinate is
	  converted to binary64, subtracted from the live binary64 dzat with the
	  scalar result spill retained, then narrowed once to binary32 and widened
	  into the corresponding PG input slot. )
	{
	    55                                      (save Lino X/ebp)
	    8B AF <dVHSstarptr mtp bytesperunit>     (ebp = star-cache unit index)
	    8D 2C AF                                 (ebp = runtime star record)
	    8B B7 <dVHSfwbase mtp bytesperunit>      (esi = fw unit index)
	    8D 34 B7                                 (esi = runtime fw)

	    DB 45 00                                 (fild x)
	    DD 9F <dFA0 mtp bytesperunit>            (binary64 IntToF spill)
	    DD 87 <dFA0 mtp bytesperunit>
	    DC A7 <dVHSdx0 mtp bytesperunit>         (subtract dzat x)
	    DD 9F <dFA0 mtp bytesperunit>            (binary64 scalar spill)
	    DD 87 <dFA0 mtp bytesperunit>
	    D9 9F <dFS0 mtp bytesperunit>            (narrow to binary32)
	    D9 87 <dFS0 mtp bytesperunit>
	    DD 9E E0 07 00 00                        (widen into FSINX, 252*8)

	    DB 45 04                                 (fild y)
	    DD 9F <dFA0 mtp bytesperunit>
	    DD 87 <dFA0 mtp bytesperunit>
	    DC A7 <dVHSdy0 mtp bytesperunit>
	    DD 9F <dFA0 mtp bytesperunit>
	    DD 87 <dFA0 mtp bytesperunit>
	    D9 9F <dFS0 mtp bytesperunit>
	    D9 87 <dFS0 mtp bytesperunit>
	    DD 9E 00 08 00 00                        (FSINY, 256*8)

	    DB 45 08                                 (fild z)
	    DD 9F <dFA0 mtp bytesperunit>
	    DD 87 <dFA0 mtp bytesperunit>
	    DC A7 <dVHSdz0 mtp bytesperunit>
	    DD 9F <dFA0 mtp bytesperunit>
	    DD 87 <dFA0 mtp bytesperunit>
	    D9 9F <dFS0 mtp bytesperunit>
	    D9 87 <dFS0 mtp bytesperunit>
	    DD 9E 20 08 00 00                        (FSINZ, 260*8)
	    5D                                      (restore ebp)
	}
	end;

"VHS rotate depth native"
	( Star-specific polymap nucleus.  The operation order and every binary64
	  scalar spill/binary32 named-variable narrowing match PJ rotate, but the
	  generic vertex loop, slot shuttles and near-plane bookkeeping are absent.
	  The narrowed camera-relative inputs are already based at a zero camera. )
	{
	    8B B7 <dVHSfwbase mtp bytesperunit>
	    8D 34 B7                                 (esi = runtime fw)

	    ( rxf = z*sinbeta + x*cosbeta )
	    DD 86 20 08 00 00                        (FSINZ)
	    DC 8E B8 06 00 00                        (* FSTSB)
	    DD 9F <dFA0 mtp bytesperunit>
	    DD 87 <dFA0 mtp bytesperunit>
	    DD 9E C0 07 00 00                        (FSW0)
	    DD 86 E0 07 00 00                        (FSINX)
	    DC 8E B0 06 00 00                        (* FSTCB)
	    DD 9F <dFA0 mtp bytesperunit>
	    DD 87 <dFA0 mtp bytesperunit>
	    DC 86 C0 07 00 00                        (+ FSW0)
	    DD 9F <dFA0 mtp bytesperunit>
	    DD 87 <dFA0 mtp bytesperunit>
	    D9 9F <dFS0 mtp bytesperunit>
	    D9 87 <dFS0 mtp bytesperunit>
	    DD 9E 00 01 00 00                        (FSRXF)

	    ( z2 = z*cosbeta - x*sinbeta )
	    DD 86 20 08 00 00
	    DC 8E B0 06 00 00
	    DD 9F <dFA0 mtp bytesperunit>
	    DD 87 <dFA0 mtp bytesperunit>
	    DD 9E C0 07 00 00
	    DD 86 E0 07 00 00
	    DC 8E B8 06 00 00
	    DD 9F <dFA0 mtp bytesperunit>
	    DD 86 C0 07 00 00                        (FSW0)
	    DC A7 <dFA0 mtp bytesperunit>            (- second product)
	    DD 9F <dFA0 mtp bytesperunit>
	    DD 87 <dFA0 mtp bytesperunit>
	    D9 9F <dFS0 mtp bytesperunit>
	    D9 87 <dFS0 mtp bytesperunit>
	    DD 9E 88 07 00 00                        (FSZ2)

	    ( rzf = z2*cosalfa + y*sinalfa )
	    DD 86 88 07 00 00
	    DC 8E D0 06 00 00                        (* FSTCA)
	    DD 9F <dFA0 mtp bytesperunit>
	    DD 87 <dFA0 mtp bytesperunit>
	    DD 9E C0 07 00 00
	    DD 86 00 08 00 00                        (FSINY)
	    DC 8E D8 06 00 00                        (* FSTSA)
	    DD 9F <dFA0 mtp bytesperunit>
	    DD 87 <dFA0 mtp bytesperunit>
	    DC 86 C0 07 00 00
	    DD 9F <dFA0 mtp bytesperunit>
	    DD 87 <dFA0 mtp bytesperunit>
	    DD 9E C8 07 00 00                        (wide FSW1)
	    DD 87 <dFA0 mtp bytesperunit>
	    D9 9F <dFS0 mtp bytesperunit>
	    D9 87 <dFS0 mtp bytesperunit>
	    DD 9E 40 01 00 00                        (narrow FSRZF)

	    ( ryf = y*cosalfa - z2*sinalfa )
	    DD 86 00 08 00 00
	    DC 8E D0 06 00 00
	    DD 9F <dFA0 mtp bytesperunit>
	    DD 87 <dFA0 mtp bytesperunit>
	    DD 9E C0 07 00 00
	    DD 86 88 07 00 00
	    DC 8E D8 06 00 00
	    DD 9F <dFA0 mtp bytesperunit>
	    DD 86 C0 07 00 00
	    DC A7 <dFA0 mtp bytesperunit>
	    DD 9F <dFA0 mtp bytesperunit>
	    DD 87 <dFA0 mtp bytesperunit>
	    D9 9F <dFS0 mtp bytesperunit>
	    D9 87 <dFS0 mtp bytesperunit>
	    DD 9E 20 01 00 00                        (FSRYF)

	    DD 86 40 01 00 00                        (narrow rzf)
	    DB 9F <dVHSdepth mtp bytesperunit>       (nearest integer depth)
	}
	end;

"VHS project native"
	( Exact getcoords projection for the already-front star. )
	{
	    8B B7 <dVHSfwbase mtp bytesperunit>
	    8D 34 B7
	    DD 86 C8 00 00 00                        (FSDPP)
	    DC B6 40 01 00 00                        (/ FSRZF)
	    DD 9F <dFA0 mtp bytesperunit>
	    DD 87 <dFA0 mtp bytesperunit>
	    DD 9E D8 07 00 00                        (FSW3)

	    DD 87 <dFA0 mtp bytesperunit>
	    DC 8E 00 01 00 00                        (* FSRXF)
	    DD 9F <dFA0 mtp bytesperunit>
	    DD 87 <dFA0 mtp bytesperunit>
	    DC 86 98 00 00 00                        (+ FSXC)
	    DD 9F <dFA0 mtp bytesperunit>
	    DD 87 <dFA0 mtp bytesperunit>
	    DB 9F <dGCx mtp bytesperunit>

	    DD 86 D8 07 00 00                        (FSW3)
	    DC 8E 20 01 00 00                        (* FSRYF)
	    DD 9F <dFA0 mtp bytesperunit>
	    DD 87 <dFA0 mtp bytesperunit>
	    DC 86 A0 00 00 00                        (+ FSYC)
	    DD 9F <dFA0 mtp bytesperunit>
	    DD 87 <dFA0 mtp bytesperunit>
	    DB 9F <dGCy mtp bytesperunit>
	}
	end;

"VHS foldmul"
	A = [VHSfmA]; B = [VHSfmB];
	{ F7 EB }		( signed imul ebx -> edx:eax )
	A + D; [VHSfmResult] = A;
	end;
