      ( *** spmem - Wave 6b region access, asset unpacking, the 4-unit
	    float view, the corpus tokeniser and the SPDUMP writer ***

	ADDITIVE ONLY.  work/fbmem.txt is LINKED, never edited and never
	forked; every base, segment origin and mask primitive comes from
	there.  Nothing in this file restates a layout number.

	THE ADDRESS CONVENTION, stated once and obeyed everywhere.

	Wave 6b's four DOS routines reach their buffers like this, and in
	EVERY one of them farmalloc's offset of 4 is either carried in the
	index or supplied by the instruction's displacement - never both,
	and never neither:

	  globe      les ax,tapestry / add start,ax / mov bx,start
	             -> SOURCE index carries the +4;  fs:[bx]
	             les ax,target                (offset DISCARDED)
	             -> DEST supplies it: es:[di+4] in every gman
	  glowinglobe les ax,target               (offset DISCARDED)
	             -> es:[di+4]
	  background les ax,target / add screenshift,ax
	             -> DEST index carries the +4;  es:[di], es:[di+4], ...
	             les bx,background / mov bx,es (offset DISCARDED)
	             mov bp,start / add bp,4
	             -> SOURCE index carries the +4;  fs:[bp]
	  surface    les di,p_background / add di,plwp / add di,35
	             -> es:[di], the +4 arriving inside DI from les

	So every one of them collapses to  base + index  in the flat
	workspace, and there is NO "+4" to add on top.  Adding one is the
	SCRATCHOFF-class defect Wave 6a measured at 23 moved pages.
	niv-lr's commented-out /*+4*/ at noctis-0.cpp:2280 is a bug in
	LR's model, where the offset was never included to begin with.

	THE ROUTINES BELOW THEREFORE TAKE A DOS SEGMENT OFFSET - one that
	already includes the 4 - and mask it against the region's SEGMENT
	ORIGIN, which is base-4.  Consequence: DOS offset 4 IS region
	byte 0 and there is no four-byte skew anywhere.

	background()'s DI is the first GAME call site of fbmem's class-A
	16-bit mask.  At the shipped screenshift of -643 the smallest
	paint word already exceeds 65535 and folds; without the mask the
	write lands 65,536 units past the page.  BUFFERMODEL 10.6 is an
	XFAIL because "the class-A mask has no game call site" - it has
	one now, and it is masked at the SEGMENT ORIGIN, not at the buffer
	base.  The two differ only when the truncated offset lands in
	0..3; that four-value boundary is the falsifier.

	PVFILE STAYS ONE BYTE PER UNIT.  BUFFERMODEL 5 alias 9 directs a
	re-layout of the pvfile arena so every float sub-array is unit
	aligned.  That is OVERRULED here: the arena obeys Decision 1 like
	every other region and the float sub-arrays are read and written
	through the 4-unit assemble/disassemble pair below, exactly as
	alias 7 already does for the STARMAP doubles.  Every DOS byte
	offset then transfers verbatim, and the 4n-into-n colour overrun
	lands exactly where DOS put it - inside pv_mid_x, which is zeroed
	before any read - so it is faithfully dead rather than
	conditionally dead.  docs-notes owner: record this overrule.

	FILENAME TRAP: underscores in lino string literals become spaces.
	Every name here uses hyphens or a dot.                            )

	A + 4;
	[A plus 0] = RSBG;  [A plus 1] = SSBG;  [A plus 2] = ZSBG;  [A plus 3] = RGSBG;
	A + 4;
	[A plus 0] = RPBG;  [A plus 1] = SPBG;  [A plus 2] = ZPBG;  [A plus 3] = RGPBG;
	A + 4;
	[A plus 0] = RPVF;  [A plus 1] = SPVF;  [A plus 2] = ZPVF;  [A plus 3] = RGPVF;
	A + 4;
	[A plus 0] = RADPT; [A plus 1] = SADPT; [A plus 2] = ZADPT; [A plus 3] = RGADP;
	end;

      ( SP base : C <- base of region [SPreg]. )

"SP base"
	A = [SPreg]; A '* 4; A + rgt;
	C = [A plus 0];
	end;

      ( SP seg : C <- segment origin of region [SPreg]. )

"SP seg"
	A = [SPreg]; A '* 4; A + rgt;
	C = [A plus 1];
	end;

( ==================================================================== )
( the ONE truncation point                                              )
( ==================================================================== )

      ( SP put : the byte at DOS offset [SPoff] of region [SPreg]
	<- [SPval] AND 255.

	nw = SEG(reg) + ([SPoff] AND 65535), fbmem's class-A mechanism,
	called at the point where the DOS code truncates DI.  The wrap is
	COUNTED in [CSwr] so a build with the mask removed shows a zero
	counter as well as a changed page - two independent falsifiers
	for one defect. )

"SP put"
	A = [SPoff];
	? A '<= WINDOW -> SP put nowrap;
	[CSwr]+;
    "SP put nowrap"
	=> SP seg;
	[SAseg] = C;
	[SAoff] = [SPoff];
	=> MEM seg addr;
	A = nw; A + [SAnw];
	C = [SPval]; C & 255;
	[A] = C;
	end;

      ( SP get : [SPval] <- the byte at DOS offset [SPoff], unsigned. )

"SP get"
	A = [SPoff];
	? A '<= WINDOW -> SP get nowrap;
	[CSwr]+;
    "SP get nowrap"
	=> SP seg;
	[SAseg] = C;
	[SAoff] = [SPoff];
	=> MEM seg addr;
	A = nw; A + [SAnw];
	C = [A]; C & 255;
	[SPval] = C;
	end;

      ( SP dput / SP dget : DIRECT region-relative access, base + index.

	FIXTURE and OBSERVER routines only - laying a pre-state, reading
	an asset back, packing a page.  They deliberately do NOT truncate,
	because a fixture is not a DOS store.  The distinction is the
	point: every rasteriser store goes through "SP put". )

"SP dput"
	=> SP base;
	A = nw; A + C; A + [SPidx];
	C = [SPval]; C & 255;
	[A] = C;
	end;

"SP dget"
	=> SP base;
	A = nw; A + C; A + [SPidx];
	C = [A]; C & 255;
	[SPval] = C;
	end;

      ( SP sx8 : C <- sign extension of C's low 8 bits.  cbw, which is
	what globe and glowinglobe do to BOTH map bytes on the draw path.
	The SKIP byte is read with "xor ah,ah" instead and must NOT come
	through here. )

"SP sx8"
	C & 255;
	A = C; A & 128;
	? A = 0 -> SP sx8 done;
	C - 256;
    "SP sx8 done"
	end;

      ( SP s16 : C <- sign extension of C's low 16 bits.  Borland large
	model int. )

"SP s16"
	C & 65535;
	A = C; A & 32768;
	? A = 0 -> SP s16 done;
	C - 65536;
    "SP s16 done"
	end;

( ==================================================================== )
( the 4-unit binary32 view - the alias-9 overrule made mechanical       )
( ==================================================================== )

      ( SP getf32 : [SPf] <- the little-endian binary32 held in the four
	byte units at region [SPreg], index [SPidx]. )

"SP getf32"
	=> SP base;
	A = nw; A + C; A + [SPidx];
	C = [A plus 3]; C & 255; C < 8;
	D = [A plus 2]; D & 255; C + D; C < 8;
	D = [A plus 1]; D & 255; C + D; C < 8;
	D = [A plus 0]; D & 255; C + D;
	[SPf] = C;
	end;

      ( SP putf32 : the four byte units at region [SPreg], index [SPidx]
	<- the little-endian binary32 pattern [SPf]. )

"SP putf32"
	=> SP base;
	A = nw; A + C; A + [SPidx];
	C = [SPf]; C & 255;         [A plus 0] = C;
	C = [SPf]; C > 8;  C & 255; [A plus 1] = C;
	C = [SPf]; C > 16; C & 255; [A plus 2] = C;
	C = [SPf]; C > 24; C & 255; [A plus 3] = C;
	end;

      ( SP getu16 / SP putu16 : the same for a little-endian uint16.
	pv_dep_i is an int far * and its entries are Borland 16-bit ints. )

"SP getu16"
	=> SP base;
	A = nw; A + C; A + [SPidx];
	C = [A plus 1]; C & 255; C < 8;
	D = [A plus 0]; D & 255; C + D;
	[SPf] = C;
	end;

"SP putu16"
	=> SP base;
	A = nw; A + C; A + [SPidx];
	C = [SPf]; C & 255;        [A plus 0] = C;
	C = [SPf]; C > 8; C & 255; [A plus 1] = C;
	end;

( ==================================================================== )
( loading the OPERAND register without disturbing the accumulator       )
( ==================================================================== )

      ( pgfp's "PGF ldf32" and "PGF ldf64" both land in FA, so the
	sequence

	    ... FA now holds a partial result ...
	    [PGFt] = k; => PGF ldf64; [FB0] = [FA0]; [FB1] = [FA1];
	    => FAdd;

	computes k + k and throws the partial result away.  It reads as
	"put k in the operand register" and is not.  This was a live
	defect in this port: whiteglobe's "mag = mag_factor*100 + 1.5"
	came out as 1.5 + 1.5 = 3, which is a plausible-looking radius and
	produced a plausible-looking disc.  It was caught because the
	emitted row and column counts are DERIVED from the loops rather
	than asserted - 3 rows where the geometry says 8.

	These two put the constant where it belongs and leave FA alone. )

"SP fb64"				( FB <- binary64 [PGFt],[PGFu] )
	[FB0] = [PGFt]; [FB1] = [PGFu];
	end;

"SP fb32"				( FB <- (double) binary32 [PGFt] )
	[FT0] = [FA0]; [FT1] = [FA1];
	=> PGF ldf32;
	[FB0] = [FA0]; [FB1] = [FA1];
	[FA0] = [FT0]; [FA1] = [FT1];
	end;

( ==================================================================== )
( asset loading - disk bytes to one byte per unit                       )
( ==================================================================== )

      ( SP unpack : stage[] holds [SPfbytes] bytes packed four to a unit;
	spread them one per unit into region [SPreg] from index 0. )

"SP unpack"
	=> SP base;
	[SPtmp] = C;
	[SPMi] = 0;
    "SP up byte"
	A = [SPMi]; C = A; C > 2; D = stage; D + C;
	C = [D];
	A = [SPMi]; A & 3; A '* 8;
	C > A; C & 255;
	A = nw; A + [SPtmp]; A + [SPMi];
	[A] = C;
	[SPMi]+;
	A = [SPMi]; ? A '< [SPfbytes] -> SP up byte;
	end;

      ( SP read asset : read the file named by [File Name] into stage[]
	and spread it into region [SPreg].  [SPfbytes] is the length the
	FILE reported, never a constant - a truncated or absent asset is
	then visible as a wrong length rather than as a wrong page. )

"SP read asset"
	[File Position]	= 0;
	[File Command]	= READ;
	[Block Pointer]	= stage;
	[Block Size]	= STAGEMAX;
	[Block Size]	* BYTESPERUNIT;
	isocall;
	A = [Block Size];
	[SPfbytes] = A;
	? A '> 0 -> SP ra ok;
	end;
    "SP ra ok"
	=> SP unpack;
	end;

      ( SP load maps : GLOBES.MAP into n_globes_map, OFFSETS.MAP into
	n_offsets_map.  The lengths are CHECKED against the header
	constants and a mismatch is recorded in [SPload], not ignored. )

"SP load maps"
	[SPload] = 0;

	[File Name] = spglbname;
	[SPreg] = RGGLB;
	=> SP read asset;
	A = [SPfbytes]; ? A = GLBYTES -> SP lm g ok;
	C = [SPload]; C | 1; [SPload] = C;
    "SP lm g ok"

	[File Name] = spoffname;
	[SPreg] = RGOFF;
	=> SP read asset;
	A = [SPfbytes]; ? A = OMBYTES -> SP lm o ok;
	C = [SPload]; C | 2; [SPload] = C;
    "SP lm o ok"
	end;

      ( SP census page : count non-zero and 255 bytes over the first
	[SPpn] bytes of region [SPpreg].  Counted AFTER the fact, from
	the page, so it cannot agree with a rasteriser that never ran. )

"SP census page"
	[SPCSnz] = 0; [SPCS255] = 0;
	[SPreg] = [SPpreg];
	=> SP base;
	[SPtmp] = C;
	[SPMi] = 0;
    "SP cen unit"
	A = nw; A + [SPtmp]; A + [SPMi];
	C = [A]; C & 255;
	? C = 0 -> SP cen next;
	[SPCSnz]+;
	? C != 255 -> SP cen next;
	[SPCS255]+;
    "SP cen next"
	[SPMi]+;
	A = [SPMi]; ? A '< [SPpn] -> SP cen unit;
	end;

    "SP pk unit"
	A = nw; A + [SPtmp]; A + [SPMi];
	D = [A plus 0]; D & 255;
	C = [A plus 1]; C & 255; C < 8;  D | C;
	C = [A plus 2]; C & 255; C < 16; D | C;
	C = [A plus 3]; C & 255; C < 24; D | C;
	A = sppack; A + [SPMj];
	[A] = D;
	[SPMi] + 4;
	[SPMj]+;
	A = [SPMi]; ? A '< [SPpn] -> SP pk unit;
	end;

      ( SP fill page : region [SPpreg] bytes 0..[SPpn]-1 <- [SPval]. )

"SP fill page"
	[SPreg] = [SPpreg];
	=> SP base;
	[SPtmp] = C;
	C = [SPval]; C & 255; [SPMk] = C;
	[SPMi] = 0;
    "SP fp unit"
	A = nw; A + [SPtmp]; A + [SPMi];
	[A] = [SPMk];
	[SPMi]+;
	A = [SPMi]; ? A '< [SPpn] -> SP fp unit;
	end;

	A = [SPMi]; ? A '< [SPpn] -> SP rp unit;
	end;

      ( SDMP surround : the eight units the PAGE record cannot see, and
	the only place the two candidate class-A rules disagree.

	adapted's DOS segment runs from its base MINUS FOUR - Borland's
	far heap gave the block an offset of 4 - so a 16-bit index that
	folds to 0..3 lands on the four far-heap header units BELOW the
	buffer.  Masking at the buffer base instead sends the same write
	to base + 65532, four units above the visible page.  Neither
	address is inside the 64,000 pixels the PAGE record carries, so
	without this record the two rules produce identical dumps and the
	sabotage that tells them apart reads as NOT CAUGHT.

	Emitted every case, so the "class-A mask has no game call site"
	XFAIL in BUFFERMODEL 10.6 can be closed with a number rather than
	an argument:

	    0..3   nw[SADPT+0 .. SADPT+3]   the segment's offsets 0..3
	    4..7   adapted[65536 .. 65539]  the four units sc_bytes exists for
	    8      non-zero units in adapted[64000..65535]
	    9      the first such index, or -1                          )

"SDMP surround"
	[SPreg] = RGADP;
	=> SP seg;
	D = nw; D + C;
	A = sphdr;
	[A plus 0] = [D plus 0];
	[A plus 1] = [D plus 1];
	[A plus 2] = [D plus 2];
	[A plus 3] = [D plus 3];
	[SPreg] = RGADP;
	=> SP base;
	[SPtmp] = C;
	D = nw; D + C; D + 65536;
	A = sphdr;
	[A plus 4] = [D plus 0];
	[A plus 5] = [D plus 1];
	[A plus 6] = [D plus 2];
	[A plus 7] = [D plus 3];
	[SPMi] = NPIX;
	[SPMj] = 0;
	[SPMk] = 0; [SPMk]-;
    "SDMP sr u"
	A = nw; A + [SPtmp]; A + [SPMi];
	C = [A];
	? C = 0 -> SDMP sr n;
	[SPMj]+;
	A = [SPMk]; ? A >= 0 -> SDMP sr n;
	[SPMk] = [SPMi];
    "SDMP sr n"
	[SPMi]+;
	A = [SPMi]; ? A '< 65536 -> SDMP sr u;
	A = sphdr;
	[A plus 8]  = [SPMj];
	[A plus 9]  = [SPMk];
	[A plus 10] = 0; [A plus 11] = 0;
	[A plus 12] = 0; [A plus 13] = 0; [A plus 14] = 0; [A plus 15] = 0;
	[SKind] = K31; [STag] = 2;
	[SW] = 16; [SH] = 1;
	[SBptr] = sphdr; [SBcnt] = 16;
	=> SDMP rec;
	end;

( the corpus tokeniser                                                  )
( ==================================================================== )

      ( A flat stream of signed decimal integers.  '#' starts a comment
	that runs to end of line.  Anything else is a separator.  A
	decimal integer is the ONLY lexeme, so no shared decimal float
	parser - which would be a shared divergence - exists anywhere:
	float inputs are given as the signed decimal value of their
	binary32 BIT PATTERN.                                            )

"SP CR load"
	[SPCRbad] = 0;
	[File Name]	= spcorpname;
	[File Command]	= READ;
	[File Position] = 0;
	[Block Pointer] = scorpus;
	[Block Size]	= SPCORPUSMAX;
	[Block Size]	* BYTESPERUNIT;
	isocall;
	A = [Block Size];
	[SPCRbytes] = A;
	? A '> 0 -> SP CR load ok;
	[SPCRbad] = 1;
    "SP CR load ok"
	end;

"SP CR getch"
	A = [SPCRpos];
	? A '>= [SPCRbytes] -> SP CR getch eof;
	C = A; C > 2; D = scorpus; D + C;
	C = [D];
	A = [SPCRpos]; A & 3; A '* 8;
	C > A; C & 255;
	[SPCRch] = C;
	end;
    "SP CR getch eof"
	A = 0; A - 1; [SPCRch] = A;
	end;

"SP CR tokenise"
	[SPCRpos] = 0; [SPCRtokn] = 0;
    "SP CR tk scan"
	=> SP CR getch;
	A = [SPCRch];
	? A < 0 -> SP CR tk done;
	? A = 35 -> SP CR tk comment;		( '#' )
	? A = 45 -> SP CR tk number;		( '-' )
	? A '< 48 -> SP CR tk skip;
	? A '> 57 -> SP CR tk skip;
	-> SP CR tk number;
    "SP CR tk skip"
	[SPCRpos]+;
	-> SP CR tk scan;
    "SP CR tk comment"
	[SPCRpos]+;
	=> SP CR getch;
	A = [SPCRch];
	? A < 0 -> SP CR tk done;
	? A != 10 -> SP CR tk comment;
	[SPCRpos]+;
	-> SP CR tk scan;
    "SP CR tk number"
	[SPCRneg] = 0; [SPCRacc] = 0;
	A = [SPCRch];
	? A != 45 -> SP CR tk digits;
	[SPCRneg] = 1;
	[SPCRpos]+;
    "SP CR tk digits"
	=> SP CR getch;
	A = [SPCRch];
	? A '< 48 -> SP CR tk emit;
	? A '> 57 -> SP CR tk emit;
	C = [SPCRacc]; C '* 10; A - 48; C + A; [SPCRacc] = C;
	[SPCRpos]+;
	-> SP CR tk digits;
    "SP CR tk emit"
	A = [SPCRneg]; ? A = 0 -> SP CR tk pos;
	C = [SPCRacc]; C+-; [SPCRacc] = C;
    "SP CR tk pos"
	A = [SPCRtokn];
	? A '>= SPTOKMAX -> SP CR tk done;
	C = stok; C + A; [C] = [SPCRacc];
	[SPCRtokn]+;
	-> SP CR tk scan;
    "SP CR tk done"
	end;

      ( SP CR next : C <- tok[[SPCRti]], [SPCRti]+.  Reading past the end
	yields SPOPEND, so a truncated corpus terminates rather than
	looping. )

"SP CR next"
	A = [SPCRti];
	? A '>= [SPCRtokn] -> SP CR next eof;
	C = stok; C + A; C = [C];
	[SPCRti]+;
	end;
    "SP CR next eof"
	C = SPOPEND;
	end;
