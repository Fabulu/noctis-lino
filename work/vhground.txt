	  real build_surface switch arms: type 8 milky/quartz and type 4 rocky.

	[VHGNDptr] = 0;
    "VHGND wipe objects"
	A = ROBJ; A + [VHGNDptr]; [MBptr] = A; [MBval] = 0; => MEM put byte;

	[VHGNDptr]+; A = [VHGNDptr]; ? A '< 65535 -> VHGND wipe texture;
	A = [VHGNDtexbase]; A + 65535; [MBptr] = A; [MBval] = 0; => MEM put byte;

	=> SU ledger reset; => GR float init;
	[VHGNDrockscale] = 500; [VHGNDrockpeak] = 250; [VHGNDrockdensity] = 15;
	A = [VHGNDlat]; A - 60; ? A >= 0 -> VHGND latitude absolute; A +-;
    "VHGND latitude absolute"
	A '* 3; A '/ 2; [GRlat] = A;
	[VHGNDsctype] = 0; A = [GRiptype]; ? A != 3 -> VHGND scenario ready;
	( NIV+ planetary_main: most sites derive their biome from orbital
	  orientation; a five-percent branch is locally random.  Low sampled
	  albedo forces ocean and high latitude forces ice. )
	A = [VHGNDlocseed]; A & 0FFFFh; => SU srand;
	C = 100; => SU rnd; A = C; ? A '<= 5 -> VHGND scenario local;
	A = [VHGNDplanet]; A < 1; E = nspororient; E + A;
	[FB0] = [E plus 0]; [FB1] = [E plus 1]; A = 3; => GeoSetK;
	=> GeoKMulChopLive; A = [FI]; A % 4; A + 1; [VHGNDsctype] = A;
	-> VHGND scenario albedo;
    "VHGND scenario local"
	C = 4; => SU rnd; C + 1; [VHGNDsctype] = C;
    "VHGND scenario albedo"
	A = [GRalbedo]; ? A '>= 25 -> VHGND scenario polar; [VHGNDsctype] = 1;
    "VHGND scenario polar"
	A = [GRlat]; ? A '<= 75 -> VHGND scenario near polar; [VHGNDsctype] = 4;
    "VHGND scenario near polar"
	A = [GRlat]; ? A '<= 60 -> VHGND scenario rain; C = 3; => SU rnd;
	? C = 0 -> VHGND scenario rain; [VHGNDsctype] = 4;
    "VHGND scenario rain"
	( Dry and icy biomes reduce the sampled cloud load using the same BRTL
	  stream that selected the scenario. )
	A = [VHGNDsctype]; ? A = 3 -> VHGND scenario desert rain;
	? A = 4 -> VHGND scenario icy rain; -> VHGND scenario ready;
    "VHGND scenario desert rain"
	C = 4; => SU rnd; C + 1; -> VHGND scenario scale rain;
    "VHGND scenario icy rain"
	C = 3; => SU rnd; C + 2;
    "VHGND scenario scale rain"

	[FB0] = [FA0]; [FB1] = [FA1]; [FS0] = [VHGNDcloudrain]; => FLoadF32;
	=> FQuo; => F32Narrow; => FStoreF32; [VHGNDcloudrain] = [FS0];
    "VHGND scenario ready"

	=> GR prologue;
	( build_surface resets both random streams from landing_pt_lat*landing_pt_lon

	A = [VHGNDlocseed]; => SU fast srand;

	[VHGNDgenn]-; A = [VHGNDgenn]; ? A != 0 -> VHGND type8 hill loop;

"VHGND tile shade"
	( NOCTIS-1.CPP:1125-1150, diffuse-light arm (sh_delta == 0).
	  Each tile gets the original deterministic 8..15 base shade from
	  fast_random(7), followed by the source distance haze.  The live
	  landing fixtures are close to a perpendicular sun, so this is the
	  active source path and selects surface_palette's ground band 0..63. )
	A = [VHGNDh1]; A + [VHGNDseed]; => SU fast srand;
	[SUfmask] = 7; => SU frnd;
	C + 8; [VHGNDshade] = C;

	( hpdep = sqrt((pos_x-tile_centre_x)^2 +
	               (pos_z-tile_centre_z)^2), then depth=(hpdep>>14)-1. )
	A = [VHGNDx]; A < 14; C = [VHGNDlodstep]; C < 13; A + C;
	C = [VHGNDcamx]; C - A;
	[VHGNDdx] = C; [FI] = C; => IntToF;
	[FB0] = [FA0]; [FB1] = [FA1]; => FMul;
	[FT0] = [FA0]; [FT1] = [FA1];
	A = [VHGNDz]; A < 14; C = [VHGNDlodstep]; C < 13; A + C;
	C = [VHGNDcamz]; C - A;
	[VHGNDdz] = C; [FI] = C; => IntToF;
	[FB0] = [FA0]; [FB1] = [FA1]; => FMul;
	[FB0] = [FT0]; [FB1] = [FT1]; => FAdd; => FSqrt; => FToIntChop;
	A = [FI]; A > 14; A - 1; ? A >= 0 -> VHGND tile depth ready; A = 0;
    "VHGND tile depth ready"
	[VHGNDdepth] = A; A > 1; C = [VHGNDshade]; C + A;
	? C '<= 32 -> VHGND tile shade ready; C = 32;
    "VHGND tile shade ready"
	[VHGNDshade] = C;
	end;
