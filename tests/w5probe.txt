      ( *** w5probe - the Wave 5 buffer-model / framebuffer regression probe ***

	OWNED BY tests/.  test_wave5.py copies this file, the three
	libraries under test (work/fbmem.txt, work/fbpal.txt,
	work/fbtick.txt) and the Wave 3 float engine into
	tests/gen/w5/ on every run, builds it there, and grades every
	number it dumps against tests/w5spec.py - a Python model written
	from NOCTIS-D.H, NOCTIS.CPP and NOCTIS-0.CPP, never from this
	file and never from the lino libraries.

	THIS PROBE IS NOT work/fbshell.txt.  It links the same three
	libraries but its own shell, so that the shell's own defects
	cannot be mistaken for the model's, and so that the present path
	can be written the way BUFFERMODEL.md rules it must be: the
	expand is READ-ONLY on the index page and the background colour
	cycle is a separate pass over s_background, which is the buffer
	NOCTIS.CPP:3779-3784 actually cycles.

	NO DISPLAY IS OPENED HERE.  There are no display directors, so
	this probe never puts a window on the user's desktop and never
	calls RETRACE.  The on-screen present is a separate, deliberately
	tiny probe (w5disp.txt) and the test header says so.

	Everything this probe writes is one FBDUMP v1 stream (LINOBUF
	section 6) staged in "twdump" - the probe's own staging vector,
	not fbmem's 140,000-unit one, because three 64,000-unit pages do
	not fit there.  The record format is still fbmem's "DMP rec", so
	the format is the library's and only the buffer is local.

	FILENAME TRAP: underscores in lino string literals become spaces.
	The output name comes from fbmem's [fbdumpname] = { fb-out.bin }
	and the harness renames it afterwards.                            )

( battery 1 - byte semantics, and the proof that one-per-unit is real   )
( ==================================================================== )

"TW byte battery"
	[TWbchk] = 0; [TWonek] = 0;

	( a char store wraps: 300 -> 44 )
	[MBptr] = 5000; [MBval] = 300; => MEM put byte;
	[MBptr] = 5000; => MEM get byte;
	A = [MBval]; A - 44; ? A = 0 -> TW bb1;
	A = [TWbchk]; A | 1; [TWbchk] = A;
    "TW bb1"
	( and -1 -> 255 )
	[MBptr] = 5001; A = 0; A-; [MBval] = A; => MEM put byte;
	[MBptr] = 5001; => MEM get byte;
	A = [MBval]; A - 255; ? A = 0 -> TW bb2;
	A = [TWbchk]; A | 2; [TWbchk] = A;
    "TW bb2"
	( ...and the truncation must happen in the STORE, not only in the
	  load.  Reading the unit back through "MEM get byte" masks with
	  255 as well, so a store that forgot to truncate is invisible
	  that way and sabotage 1 walked straight through the first
	  version of this battery.  Read the RAW unit. )
	A = nw; A + 5000; C = [A]; C - 44;
	? C = 0 -> TW bb2a;
	A = [TWbchk]; A | 256; [TWbchk] = A;
    "TW bb2a"
	A = nw; A + 5001; C = [A]; C - 255;
	? C = 0 -> TW bb2b;
	A = [TWbchk]; A | 512; [TWbchk] = A;
    "TW bb2b"
	( a byte store must not disturb the NEXT unit )
	A = nw; A + 5002; [A plus 1] = 305419896;
	[MBptr] = 5002; [MBval] = 200; => MEM put byte;
	A = nw; A + 5002; C = [A plus 1]; C - 305419896;
	? C = 0 -> TW bb3;
	A = [TWbchk]; A | 4; [TWbchk] = A;
    "TW bb3"

	( ONE ITEM PER UNIT.  Eight byte offsets, four of them inside a
	  single would-be packed unit and four at the very top of the
	  workspace, each given a distinct value and each read back.  A
	  store that divided the offset by four collides 0,1,2,3 into one
	  unit and this is where that shows. )
	[MBptr] = 0;      [MBval] = 11;  => MEM put byte;
	[MBptr] = 1;      [MBval] = 22;  => MEM put byte;
	[MBptr] = 2;      [MBval] = 33;  => MEM put byte;
	[MBptr] = 3;      [MBval] = 44;  => MEM put byte;
	[MBptr] = 402192; [MBval] = 55;  => MEM put byte;
	[MBptr] = 402193; [MBval] = 66;  => MEM put byte;
	[MBptr] = 402194; [MBval] = 77;  => MEM put byte;
	[MBptr] = 402195; [MBval] = 88;  => MEM put byte;

	[MBptr] = 0;      => MEM get byte; A = [MBval]; A - 11; ? A != 0 -> TW ok bad;
	[MBptr] = 1;      => MEM get byte; A = [MBval]; A - 22; ? A != 0 -> TW ok bad;
	[MBptr] = 2;      => MEM get byte; A = [MBval]; A - 33; ? A != 0 -> TW ok bad;
	[MBptr] = 3;      => MEM get byte; A = [MBval]; A - 44; ? A != 0 -> TW ok bad;
	[MBptr] = 402192; => MEM get byte; A = [MBval]; A - 55; ? A != 0 -> TW ok bad;
	[MBptr] = 402193; => MEM get byte; A = [MBval]; A - 66; ? A != 0 -> TW ok bad;
	[MBptr] = 402194; => MEM get byte; A = [MBval]; A - 77; ? A != 0 -> TW ok bad;
	[MBptr] = 402195; => MEM get byte; A = [MBval]; A - 88; ? A != 0 -> TW ok bad;
	-> TW bb4;
    "TW ok bad"
	[TWonek] = 1;
    "TW bb4"
	( put the four low units back to zero: they sit in the low pad and
	  the low-pad coverage probe writes there next. )
	[MBptr] = 0; [MBval] = 0; => MEM put byte;
	[MBptr] = 1; [MBval] = 0; => MEM put byte;
	[MBptr] = 2; [MBval] = 0; => MEM put byte;
	[MBptr] = 3; [MBval] = 0; => MEM put byte;
	end;

( battery 3 - the alias register, exercised in BOTH directions          )
( ==================================================================== )

      ( Each alias is written through one name and read through the
	other.  Nothing here asserts that an alias exists; every value is
	recomputed by w5spec.py from the layout and the fill rule. )

"TW alias battery"
	( --- A1: objectschart <-> ruinschart, BUFFERMODEL alias 3 ---------
	  Same byte, different views.  131 = nr_of_objects 3, obj2 2.
	  ruinschart[h1] = AF1 is a WHOLE-BYTE store of 0x40, so it
	  clobbers the object slot - which is the observable behaviour
	  that makes splitting the two charts illegal. )
	[MBptr] = 0; [MBptr] + ROBJ; [MBptr] + 12345; [MBval] = 131; => MEM put byte;
	[MBptr] = 0; [MBptr] + ROBJ; [MBptr] + 12346; [MBval] = 255; => MEM put byte;
	[MBptr] = 0; [MBptr] + ROBJ; [MBptr] + 12345; => MEM get byte;
	[MQval] = [MBval]; [MQfld] = 0; => MEM quad get; [twself plus 10] = [MQout];
	( now the ruins writer, at the SAME index )
	[MBptr] = 0; [MBptr] + ROBJ; [MBptr] + 12345; [MBval] = 64; => MEM put byte;
	[MBptr] = 0; [MBptr] + ROBJ; [MBptr] + 12345; => MEM get byte;
	[twself plus 9] = [MBval];
	[MQval] = [MBval]; [MQfld] = 3; => MEM quad get; [twself plus 11] = [MQout];
	[MBptr] = 0; [MBptr] + ROBJ; [MBptr] + 12346; => MEM get byte;
	[twself plus 12] = [MBval];

	( --- A2: n_globes_map <-> the sea/horizon texture, alias 1 ----
	  Write a globe byte at the texel the texture unit will fetch,
	  then fetch it through the 16-bit texel address:
	      texel = ((V>>8) AND 255)*256 + ((U>>8) AND 255)
	  U = 0x2A00, V = 0x1300 -> texel 19*256 + 42 = 4906. )
	[MBptr] = 0; [MBptr] + RNGLB; [MBptr] + 4906; [MBval] = 200; => MEM put byte;
	=> TW texel;
	[MBptr] = 0; [MBptr] + RNGLB; [MBptr] + [TWq]; => MEM get byte;
	[twself plus 13] = [MBval];

	( digimap2 is SPLIT, so it is a vector of its own and holds one
	  packed uint32 per unit rather than four byte units.  Written
	  BEFORE the sea gradient runs, so that surviving it means
	  something. )
	[digimap2 plus 7] = 305419896;		( 0x12345678 )

	( the sea gradient destroys globes.map: sea[i] = (i >> 8) AND 63 )
	=> TW sea fill;
	[MBptr] = 0; [MBptr] + RNGLB; [MBptr] + 1000; => MEM get byte;
	[twself plus 14] = [MBval];
	=> TW texel;
	[MBptr] = 0; [MBptr] + RNGLB; [MBptr] + [TWq]; => MEM get byte;
	[twself plus 15] = [MBval];

	( --- A3: digimap2 is SPLIT, BUFFERMODEL alias 2 ------------------
	  In DOS the 32x36 pilot font lives at &n_globes_map[gl_bytes] and
	  the sea fill destroys it.  Split, it survives - and the unit
	  that used to hold it is destroyed exactly as before.  Both
	  halves of that are recorded. )
	[twself plus 16] = [digimap2 plus 7];
	[MBptr] = 0; [MBptr] + RNGLB; [MBptr] + 22586; [MBptr] + 28; => MEM get byte;
	[twself plus 17] = [MBval];

	( --- A4: p_background BECOMES s_background at runtime, alias 5.
	  The base is an offset VARIABLE; the swap is one assignment. )
	[TWpbg] = 0; [TWpbg] + RPBG;
	[MBptr] = [TWpbg]; [MBptr] + 50; [MBval] = 7; => MEM put byte;
	[TWpbg] = 0; [TWpbg] + RSBG;
	[MBptr] = [TWpbg]; [MBptr] + 50; [MBval] = 9; => MEM put byte;
	[MBptr] = 0; [MBptr] + RPBG; [MBptr] + 50; => MEM get byte;
	[twself plus 18] = [MBval];
	[MBptr] = 0; [MBptr] + RSBG; [MBptr] + 50; => MEM get byte;
	[twself plus 19] = [MBval];

	( --- A5: txtr re-based onto p_surfacemap and p_surfacemap+2064,
	  alias 6.  NOCTIS.CPP:1010.  A byte slide is a unit slide. )
	[TWtxtr] = 0; [TWtxtr] + RPSM;
	[MBptr] = [TWtxtr]; [MBptr] + 100; [MBval] = 5; => MEM put byte;
	[TWtxtr] = 0; [TWtxtr] + RPSM; [TWtxtr] + 2064;
	[MBptr] = [TWtxtr]; [MBptr] + 100; [MBval] = 99; => MEM put byte;
	[MBptr] = 0; [MBptr] + RPSM; [MBptr] + 100; => MEM get byte;
	[twself plus 20] = [MBval];
	[MBptr] = 0; [MBptr] + RPSM; [MBptr] + 2164; => MEM get byte;
	[twself plus 21] = [MBval];
	end;

      ( TW texel : [TWq] <- ((V>>8) AND 255)*256 + ((U>>8) AND 255) for
	U = 0x2A00, V = 0x1300.  TDPOLYGS.H:2817-2821. )

"TW texel"
	C = 4864;			( 0x1300 )
	C > 8; C & 255; C < 8;
	[TWt1] = 10752;			( 0x2A00 )
	[TWt1] > 8; [TWt1] & 255;
	C + [TWt1];
	[TWq] = C;
	end;

      ( TW sea fill : sea[i] = (i >> 8) AND 63 over the whole 32768. )

"TW sea fill"
	A = nw; A + RNGLB; C = 0; B = 0; B + TWSEAN;
    "TW sf loop"
	[TWt1] = C; [TWt1] > 8; [TWt1] & 63;
	[A] = [TWt1]; A+; C+;
	B ^ TW sf loop;
	end;

( ==================================================================== )
( battery 6 - the pinned palette state                                  )
( ==================================================================== )

      ( A sequence chosen so that every documented trap is exercised at
	least once and the CLEAN build itself carries the evidence:
	  P2  a real tavola_colori from range8088, filter 64 on red so
	      that the "> 63" clamp fires exactly once
	  P3  shade with a non-dyadic k (n = 24)
	  P4  shade with DESCENDING ramps, so the chop is visible
	  P5  shade whose running value leaves 0..64 at BOTH ends, which
	      is the original's inverted clamp
	  P6  the SELF-COPY form, uploading 0..191
	  P7  a shade that rewrites 192..255 and uploads NOTHING, so
	      curpal6 and pal6 differ in one dump - upload-from-zero
	  P8  a filter above 127, which is a NEGATIVE signed char )

"TW palette pinned"
	=> PAL zero;
	=> PAL range;

	( P2 )

      ( TW shade i : shade() with integer-valued float arguments held in
	[TWt1],[TWt2],[TWi] (start) and [TWj],[TWp],[TWq] (finish). )

"TW shade i"
	[PFden] = 1;
	[PFnum] = [TWt1]; => PAL f32 make; [SHsr] = [PFbits];
	[PFnum] = [TWt2]; => PAL f32 make; [SHsg] = [PFbits];
	[PFnum] = [TWi];  => PAL f32 make; [SHsb] = [PFbits];
	[PFnum] = [TWj];  => PAL f32 make; [SHer] = [PFbits];
	[PFnum] = [TWp];  => PAL f32 make; [SHeg] = [PFbits];
	[PFnum] = [TWq];  => PAL f32 make; [SHeb] = [PFbits];

( ==================================================================== )
( battery 7 - the period, as pure deterministic arithmetic              )
( ==================================================================== )

      ( TK advance driven from SYNTHETIC counts-per-millisecond values,
	so the exact rational recurrence is graded with no clock in it at
	all.  Four values spanning the 0.43% spread recon C measured
	across launches. )

"TW adv battery"
	A = twadv; [TWadvp] = A;
	[TWt1] = 8984; => TW adv one;
	[TWt1] = 8999; => TW adv one;
	[TWt1] = 9000; => TW adv one;
	[TWt1] = 9023; => TW adv one;
	end;

"TW adv one"
	[TKcpms] = [TWt1]; => TK period;
	[TKacc] = 0; [TKdeadline] = 0;
	( the counter is a VARIABLE, not B: "TK advance" clobbers A, B and
	  C, so a register loop counter around a call runs away.  This is
	  the same trap fbshell's own comment on [FBy] describes. )
	[TWj] = 0;
    "TW ao loop"
	=> TK advance;
	A = [TWadvp]; C = [TKdeadline]; [A] = C; A+; [TWadvp] = A;
	[TWj]+;
	A = [TWj]; ? A '< TWADVN -> TW ao loop;
	end;

( ==================================================================== )
( battery 8 - the servo, driven across the 32-bit counter wrap          )
( ==================================================================== )

      ( [Counts] is 32 bits and wraps every 2^32/cpms seconds - about
	477 s at 9000 counts/ms.  A servo that brackets against the START
	of the run therefore reads a WRAPPED difference the moment the run
	passes that point.  Waiting eight minutes is not a test; setting
	the origin is.  Six elapsed times, three of them past the wrap.

	Six units per case: elapsed ms, srvcnt, srvms, cpms in, srvnew,
	cpms out. )

"TW srv battery"
	A = twsrv; [TWsrvp] = A;
	[TWelap] = 60000;  => TW srv one;
	[TWelap] = 300000; => TW srv one;
	[TWelap] = 450000; => TW srv one;
	[TWelap] = 600000; => TW srv one;
	[TWelap] = 900000; => TW srv one;
	[TWelap] = 1200000; => TW srv one;
	end;

"TW srv one"
	[TKcpms] = 8999; => TK period;
	[Timer Command] = READ COUNTS; isocall;
	A = [Counts];
	C = [TWelap]; C '* 8999;	( wraps mod 2^32, exactly as the counter does )
	A - C; [TKsrv0c] = A;
	=> TK read wall;
	A = [TKtmp]; A - [TWelap]; [TKsrv0w] = A;
	[TKsrvn] = 0; [TKsrvn] + SERVON; [TKsrvn] - 1;
	=> TK servo;
	A = [TWsrvp];
	[A plus 0] = [TWelap];
	[A plus 1] = [TKsrvcnt];
	[A plus 2] = [TKsrvms];
	[A plus 3] = 8999;
	[A plus 4] = [TKsrvnew];
	[A plus 5] = [TKcpms];
	A + 6; [TWsrvp] = A;
	end;

( ==================================================================== )
( battery 9 - the wait predicate, across the wrap and the sign boundary )
( ==================================================================== )

"TW wrap battery"
	[TWwrapf] = 0; [TWwrapn] = 0;

	[TWank] = 0;			=> TW wrap anchor;
	[TWank] = 1;			=> TW wrap anchor;
	[TWank] = 2147483647;		=> TW wrap anchor;
	[TWank] = 2147483648;		=> TW wrap anchor;
	[TWank] = 2147483649;		=> TW wrap anchor;
	[TWank] = 4294967295;		=> TW wrap anchor;

	[TWank] = 0; [TWi] = 0;
    "TW wa loop"
	=> TW wrap anchor;
	[TWank] + 65537;		( odd stride: the anchors sweep all 2^32 )
	[TWi]+;
	A = [TWi]; ? A '< TWANCH -> TW wa loop;
	end;

"TW wrap anchor"
	[TWoff] = 0; [TWoff] - 3;
    "TW wo loop"
	[TKnow] = [TWank];
	A = [TWank]; A + [TWoff]; [TKdeadline] = A;
	=> TK is late;
	[TWwrapn]+;
	( late exactly when the deadline is at or behind the anchor )
	? [TWoff] > 0 -> TW wo early;
	A = [TKlate]; A - 1; ? A = 0 -> TW wo next;
	[TWwrapf]+;
	-> TW wo next;
    "TW wo early"
	A = [TKlate]; ? A = 0 -> TW wo next;
	[TWwrapf]+;
    "TW wo next"
	[TWoff]+;
	A = [TWoff]; ? A <= 3 -> TW wo loop;
	end;

( ==================================================================== )
( the pages                                                             )
( ==================================================================== )

"TW riga build"
	A = twriga; C = 0; B = 0; B + 200;
    "TW rg loop"
	[A] = C; A+; C + 320;
	B ^ TW rg loop;
	end;

"TW page build"
	( pclear(adapted, 0) over the FULL SEGMENT - 65540, not 64000 )
	A = nw; A + RADPT; B = 0; B + ZADPT; C = 0;
    "TW pb c1"
	[A] = C; A+;
	B ^ TW pb c1;
	A = nw; A + RADPR; B = 0; B + ZADPR; C = 0;
    "TW pb c2"
	[A] = C; A+;
	B ^ TW pb c2;

	=> TW draw;

	( tinta and escrescenze, TDPOLYGS.H.  Under farmalloc offset == 4
	  they are VISIBLE PIXELS: row 199, columns 316 and 317.  niv-lr
	  relocated them to 64000, outside the page; BUFFERMODEL alias 8 keeps
	  them where the original put them. )
	A = nw; A + RADPT; A + 63996;
	[A plus 0] = 211;
	[A plus 1] = 212;

	( pcopy(adaptor, adapted) : index -> index, both pages real )
	D = nw; D + RADPT; E = nw; E + RADPR; A = 0; A + NPIX;
    "TW pb copy"
	C = [D]; [E] = C; D+; E+;
	A ^ TW pb copy;

	( vanilla's type-9 substellar case writes the VISIBLE page.  Do
	  not optimise adaptor away. )
	A = nw; A + RADPR; A + [twriga plus 100]; A + 128;
	B = 0; B + 64; C = 255;
    "TW pb poke"
	[A] = C; A+;
	B ^ TW pb poke;
	end;

      ( TW draw : adapted[riga[y]+x] = (((x*y)>>4) + 3x + 5y) AND 255.
	Two-dimensional and not symmetric in x and y, so a transposed or
	off-by-one scanline table shows up rather than cancelling. )

"TW draw"
	[TWy] = 0;
    "TW dr y"
	A = [TWy]; A + twriga; A = [A];
	D = nw; D + RADPT; D + A;
	E = 0;
    "TW dr x"
	C = E; C * [TWy]; C > 4;
	[TWt1] = E; [TWt1] * 3; C + [TWt1];
	[TWt2] = [TWy]; [TWt2] * 5; C + [TWt2];
	C & 255;
	[D] = C; D+; E+;
	? E '< 320 -> TW dr x;
	[TWy]+;
	A = [TWy]; ? A '< 200 -> TW dr y;
	end;

( ==================================================================== )
( the present path - READ-ONLY on the index page                        )
( ==================================================================== )

      ( BUFFERMODEL.md section 7.  The expand reads adaptor and writes
	fb and touches nothing else.  LINOBUF section 5.4 fused Noctis's
	background colour cycle into this loop; that is the wrong buffer
	at the wrong rate - NOCTIS.CPP:3779-3784 cycles s_background, an
	offscreen map, in the sky-palette path, not the visible page and
	not once per present.  The cycle is "TW sky cycle" below and it
	is a separate pass.

	Unrolled four ways.  Not pre-biased: recon B measured the bias at
	0.0001 ms and it would destroy the byte semantics. )

"TW expand"
	D = nw; D + RADPR; E = fb; A = 0; A + NQUAD;
    "TW ex loop"
	C = [D];	C + pal; C = [C]; [E] = C;
	C = [D plus 1];	C + pal; C = [C]; [E plus 1] = C;
	C = [D plus 2];	C + pal; C = [C]; [E plus 2] = C;
	C = [D plus 3];	C + pal; C = [C]; [E plus 3] = C;
	D + 4; E + 4;
	A ^ TW ex loop;
	end;

      ( TW present : expand, then hand the buffer to the platform.

	NOT CALLED in the default build - there are no display directors,
	so no window is created and nothing is presented.  test_wave5.py
	generates ONE extra variant, w5disp, which adds the directors and
	one call to this routine, so that the on-screen path is exercised
	exactly once per suite run instead of once per probe.

	The costs are logged RAW, one count per present, and the test
	recomputes them.  They are a measurement, not a pass criterion:
	the same binary measured 1.47 / 2.60 / 4.78 ms p50 for a frame on
	three different runs of this machine.  Only a catastrophic
	regression - a present costing a whole tick - is graded. )

"TW present"
	[Display Origin] = fb;
	[Display Live Region] = WHOLE DISPLAY;
	A = twfrm; [TWadvp] = A;
	[TWj] = 0;
    "TW pr loop"
	[Timer Command] = READ COUNTS; isocall;
	[TWw0] = [Counts];
	=> TW expand;
	[Display Command] = RETRACE; isocall;
	[Timer Command] = READ COUNTS; isocall;
	A = [Counts]; A - [TWw0];
	C = [TWadvp]; [C] = A; C+; [TWadvp] = C;
	[TWj]+;
	A = [TWj]; ? A '< TWPRESN -> TW pr loop;
	[Display Live Region] = VOID REGION;
	[twself plus 63] = [TWj];
	[twself plus 64] = [Display Status];
	[twself plus 65] = [Display Physical Width];
	[twself plus 66] = [Display Physical Height];
	end;

      ( TW sky cycle : NOCTIS.CPP:3779-3784, verbatim.
	    ig = (s_background[ir]+1) % 64;
	    ib = (s_background[ir] >> 6) << 6;
	    s_background[ir] = ig + ib;
	over all 64800 entries.  Seeded deterministically first, because
	the residency touch left the buffer full of noise. )

"TW sky cycle"
	A = nw; A + RSBG; B = 0; B + ZSBG; C = 0;
    "TW sc zero"
	[A] = C; A+;
	B ^ TW sc zero;
	A = nw; A + RSBG; C = 0; B = 0; B + 256;
    "TW sc seed"
	[A] = C; A+; C+;
	B ^ TW sc seed;

	A = nw; A + RSBG; B = 0; B + ZSBG;
    "TW sc loop"
	C = [A]; C & 255;
	[TWt1] = C; [TWt1] & 192;
	C+; C & 63; C + [TWt1];
	[A] = C; A+;
	B ^ TW sc loop;

	A = nw; A + RSBG; D = twsky; B = 0; B + TWSKYN;
    "TW sc samp"
	C = [A]; [D] = C; A+; D+;
	B ^ TW sc samp;
	end;

( ==================================================================== )
( the soak                                                             )
( ==================================================================== )

"TW soak"
	=> TK start;
	[TWi] = 0;
    "TW sk loop"
	=> TW page build;
	=> TW expand;

	( a deterministic hitch, so skip-to-grid is TESTED not asserted )
	A = 0; A + TWHITCH; ? A = 0 -> TW sk nohitch;
	A = [TWi]; A+; A % TWHITCH;
	? A != 0 -> TW sk nohitch;
	[TKtmp] = 0; [TKtmp] + TWHITMS; => TK burn;
    "TW sk nohitch"

	=> TK step;
	[TWi]+;
	A = [TWi]; ? A '< TWNTICK -> TW sk loop;
	end;
