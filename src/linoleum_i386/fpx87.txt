      ( *** fpx87 - staged native i386 x87 scalar backend ***

        This target-specific library is copied over work/fp/fpx87.txt only in an
        isolated win32/i386m build stage.  It must never enter the portable work
        closure: programme-period byte blocks are emitted without a target guard.

        The public scalar ABI is identical to the portable backend.  Every entry
        preserves A/B/C/D/E and leaves the x87 stack balanced.  Operations store
        binary64 results to FA, matching the original scalar boundary; generation
        expressions continue to use the portable fpchainssoft backend. )

"programme"

"FLoad"
        {
            50                                  (push eax)
            8B 87 <dFB0 mtp bytesperunit>        (mov eax,[edi+FB0*4])
            89 87 <dFA0 mtp bytesperunit>        (mov [edi+FA0*4],eax)
            8B 87 <dFB1 mtp bytesperunit>        (mov eax,[edi+FB1*4])
            89 87 <dFA1 mtp bytesperunit>        (mov [edi+FA1*4],eax)
            58                                  (pop eax)
        }
        end;

"FAdd"
        {
            DD 87 <dFA0 mtp bytesperunit>        (fld  qword [edi+FA0*4])
            DC 87 <dFB0 mtp bytesperunit>        (fadd qword [edi+FB0*4])
            DD 9F <dFA0 mtp bytesperunit>        (fstp qword [edi+FA0*4])
        }
        end;

"FSub"
        {
            DD 87 <dFA0 mtp bytesperunit>        (fld  qword [edi+FA0*4])
            DC A7 <dFB0 mtp bytesperunit>        (fsub qword [edi+FB0*4])
            DD 9F <dFA0 mtp bytesperunit>        (fstp qword [edi+FA0*4])
        }
        end;

"FMul"
        {
            DD 87 <dFA0 mtp bytesperunit>        (fld  qword [edi+FA0*4])
            DC 8F <dFB0 mtp bytesperunit>        (fmul qword [edi+FB0*4])
            DD 9F <dFA0 mtp bytesperunit>        (fstp qword [edi+FA0*4])
        }
        end;

"FQuo"
        {
            DD 87 <dFA0 mtp bytesperunit>        (fld  qword [edi+FA0*4])
            DC B7 <dFB0 mtp bytesperunit>        (fdiv qword [edi+FB0*4])
            DD 9F <dFA0 mtp bytesperunit>        (fstp qword [edi+FA0*4])
        }
        end;

"FSqrt"
        {
            DD 87 <dFA0 mtp bytesperunit>        (fld  qword [edi+FA0*4])
            D9 FA                               (fsqrt)
            DD 9F <dFA0 mtp bytesperunit>        (fstp qword [edi+FA0*4])
        }
        end;

"FNeg"
        {
            81 B7 <dFA1 mtp bytesperunit> 00 00 00 80   (xor dword [edi+FA1*4],80000000h)
        }
        end;

"FAbs"
        {
            81 A7 <dFA1 mtp bytesperunit> FF FF FF 7F   (and dword [edi+FA1*4],7FFFFFFFh)
        }
        end;

"FSin"
        {
            DD 87 <dFA0 mtp bytesperunit>        (fld  qword [edi+FA0*4])
            D9 FE                               (fsin)
            DD 9F <dFA0 mtp bytesperunit>        (fstp qword [edi+FA0*4])
        }
        end;

"FCos"
        {
            DD 87 <dFA0 mtp bytesperunit>        (fld  qword [edi+FA0*4])
            D9 FF                               (fcos)
            DD 9F <dFA0 mtp bytesperunit>        (fstp qword [edi+FA0*4])
        }
        end;

"FAtan2"
        {
            DD 87 <dFA0 mtp bytesperunit>        (fld  qword [edi+FA0*4])
            DD 87 <dFB0 mtp bytesperunit>        (fld  qword [edi+FB0*4])
            D9 F3                               (fpatan)
            DD 9F <dFA0 mtp bytesperunit>        (fstp qword [edi+FA0*4])
        }
        end;

"FCmp"
        ( Preserve eax explicitly, then decode x87 C2/C3/C0 without routing
          through Lino registers.  Unordered keeps the portable sticky flag. )
        {
            50                                  (push eax)
            DD 87 <dFB0 mtp bytesperunit>        (fld qword [edi+FB0*4])
            DD 87 <dFA0 mtp bytesperunit>        (fld qword [edi+FA0*4])
            DE D9                               (fcompp)
            9B                                  (fwait)
            DF E0                               (fnstsw ax)
            F6 C4 04                            (test ah,04h: C2 unordered)
            75 2E                               (jnz unordered)
            F6 C4 40                            (test ah,40h: C3 equal)
            75 11                               (jnz equal)
            F6 C4 01                            (test ah,01h: C0 less)
            75 18                               (jnz less)
            C7 87 <dFI mtp bytesperunit> 01 00 00 00
            EB 29                               (jmp done)
                                                (equal:)
            C7 87 <dFI mtp bytesperunit> 00 00 00 00
            EB 1D                               (jmp done)
                                                (less:)
            C7 87 <dFI mtp bytesperunit> FF FF FF FF
            EB 11                               (jmp done)
                                                (unordered:)
            83 8F <dFFLG mtp bytesperunit> 01   (or FFLG,FFLGUNORD)
            C7 87 <dFI mtp bytesperunit> 02 00 00 00
                                                (done:)
            58                                  (pop eax)
        }
        end;
